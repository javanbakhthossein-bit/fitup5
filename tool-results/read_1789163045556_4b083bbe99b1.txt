    28→    28→/** پارس لیست ذخیره‌شده به‌صورت JSON یا CSV */
    29→    29→function safeParseList(raw: string | null | undefined): string[] {
    30→    30→  if (!raw) return [];
    31→    31→  const t = raw.trim();
    32→    32→  if (!t) return [];
    33→    33→  try {
    34→    34→    const p = JSON.parse(t);
    35→    35→    if (Array.isArray(p)) return p.map((x) => String(x));
    36→    36→    if (typeof p === "string") return p.split(",").map((s) => s.trim()).filter(Boolean);
    37→    37→    return [];
    38→    38→  } catch {
    39→    39→    return t.split(",").map((s) => s.trim()).filter(Boolean);
    40→    40→  }
    41→    41→}

   319→   319→ * گارد «already_has_fresh_plan» دور زده می‌شود (چون برنامهٔ فعال وجود دارد

   333→   333→
   334→   334→  const profile = await db.onboardingProfile.findUnique({ where: { userId } });
   335→   335→  if (!profile) {
   336→   336→    return { started: false, reason: "no_profile" };

   338→   338→
   339→   339→  // پیش‌نیازها باید تعیین تکلیف شده باشند

   342→   342→    return { started: false, reason: "prerequisites_incomplete", blockingReason: prereqCheck.blockingReason };
   343→   343→  }
   344→   344→
   345→   345→  // ─── 🩹 v45 (باگ بحرانی مالک + نشتی اعتبار AI) — دو گارد سخت ───
   346→   346→  // ① سقف روزانه: حداکثر ۵ شروع تولید در ۲۴ ساعت برای هر کاربر — هر منبعی
   347→   347→  // (پرداخت، تحلیل ویدیو/خون، watchdog، سوئیپ، ادمین) از این سقف عبور
   348→   348→  // نمی‌کند؛ حلقه‌های بی‌پایان دیگر نمی‌توانند اعتبار AI را بسوزانند.
   349→   349→  // ② گارد برنامهٔ تازه: اگر آخرین درخواست ready است و برنامهٔ فعالِ
   350→   350→  // ساخته‌شده بعد از آن وجود دارد، تولید دوباره بی‌معنی است (قبلاً reason
   351→   351→  // «already_has_fresh_plan» در interface بود ولی هرگز برگردانده نمی‌شد و

   363→   363→  }
   364→   364→

   366→   366→  // (پنجره ۲۰ دقیقه‌ای هم‌راستا با بازیابی درخواست‌های گیرکرده — M3)
   367→   367→  const latestReq = await db.programRequest.findFirst({

   377→   377→      where: {
   378→   378→        userId,
   379→   379→        active: true,
   380→   380→        createdAt: { gte: latestReq.updatedAt },

   394→   394→  // تولیدِ دوباره AI (هزینه دوبرابر + برنامه تکراری) شروع می‌شد.
   395→   395→  // حالا خودِ به‌روزرسانی شرطی است: فقط اگر درخواست در حال generatingِ تازه

   403→   403→          { status: { not: "generating" } },

   408→   408→        status: "generating",

   415→   415→      return { started: false, reason: "already_generating" };
   416→   416→    }

   418→   418→  } else {
   419→   419→    // اولین درخواست — create می‌کند، سپس برای امنیت دوباره چک می‌کنیم که

   422→   422→      data: {
   423→   423→        userId,
   424→   424→        plan: effectivePlan,
   425→   425→        billingPeriod: "monthly",
   426→   426→        status: "generating",
   427→   427→        attempts: 1,
   428→   428→        lastAttemptAt: new Date(),
   429→   429→      },
   430→   430→    });

   432→   432→      where: {
   433→   433→        userId,
   434→   434→        status: "generating",
   435→   435→        id: { not: created.id },
   436→   436→        updatedAt: { gt: stuckWindowAgo },
   437→   437→      },

   457→   457→    }

   465→   465→    const startedAt = Date.now();

   468→   468→      const [planData, extras] = await Promise.all([
   469→   469→        buildOnboardingData(userId),
   470→   470→        buildGenerationExtras(userId),

   496→   496→      // v38-fix: موفقیتِ «این چرخه» با نتیجهٔ راندها سنجیده می‌شود نه وجود

   500→   500→      let mealOk = mRes.status === "fulfilled";

   537→   537→        // همین پیام هر ۳۰ دقیقه برای مالک تکرار شد.

   581→   581→            where: { id: reqId },
   582→   582→            data: { status: "failed", lastError: errMsg },
   583→   583→          });
   584→   584→        }

   597→   597→  })();
   598→   598→
   599→   599→  return { started: true };

   602→   602→/** پیام فارسی استاندارد «در حال آماده‌سازی» — برای پاسخ API‌ها */
   603→   603→export const PROGRAM_PREPARING_MESSAGE =

   608→   608→ * تولیدِ هنوز-در-حال-اجرا اشتباهی «یتیم» تشخیص داده نشود و تولید موازی دوباره

   610→   610→ * v38: زنجیره حالا فال‌بک مدل دوم هم دارد + راند دومِ سمتِ ناموفق →
   611→   611→ * بدترین حالت: زنجیرهٔ ۱۸دقیقه‌ای + راند دوم تا ۱۸ دقیقه ≈ ۳۶ دقیقه

   615→   615→
   616→   616→/**
   617→   617→ * Watchdog خودترمیم — بازیابی درخواست‌های «generating» گیرکرده (یتیم‌شده).
   618→   618→ *
   619→   619→ * چرا: تولید برنامه fire-and-forget است. اگر پروسه سرور وسط تولید restart شود
   620→   620→ * (dev-mode memory restart، crash، دیپلوی)، پروسه پس‌زمینه می‌میرد و
   621→   621→ * ProgramRequest برای همیشه «generating» می‌ماند.
   622→   622→ *
   623→   623→ * رفتار:
   624→   624→ *  - درخواست گیرکرده (>۲۰ دقیقه بدون به‌روزرسانی) پیدا می‌شود
   625→   625→ *  - اگر برنامه‌ای بعد از شروع همین چرخه تولید ذخیره شده باشد → ready
   626→   626→ *    (تولید کامل شده ولی آپدیت status به‌خاطر مرگ پروسه نرسیده)
   627→   627→ *  - اگر برنامه‌ای نیست و autoRetryCount < 1:
   628→   628→ *      → یک‌بار به‌صورت خودکار تولید دوباره شروع می‌شود (بدون دخالت کاربر)
   629→   629→ *  - وگرنه: failed (کاربر از تب برنامه‌ها دکمه retry را می‌زند)
   630→   630→ *
   631→   631→ * این تابع از endpointهایی صدا زده می‌شود که فرانت‌اند موقع generating به آن‌ها
   632→   632→ * poll می‌کند (coach/plan GET و program-history GET) — یعنی watchdog همان‌جا
   633→   633→ * فعال می‌شود که کاربر منتظر برنامه است.
   634→   634→ */
   635→   635→export async function recoverStuckGenerations(userId: string): Promise<void> {
   636→   636→  try {
   637→   637→    const stuckWindowAgo = new Date(Date.now() - STUCK_GENERATION_WINDOW_MS);
   638→   638→    const stuck = await db.programRequest.findFirst({
   639→   639→      where: { userId, status: "generating", updatedAt: { lt: stuckWindowAgo } },
   640→   640→      orderBy: { createdAt: "desc" },
   641→   641→    });
   642→   642→    if (!stuck) return;
   643→   643→
   644→   644→    // آیا برنامه‌ای در این مدت ذخیره شده؟ (تولید ممکن است کامل شده باشد ولی
   645→   645→    // آپدیت status به‌خاطر مرگ پروسه نرسیده باشد)
   646→   646→    // ─── M2: فقط برنامه‌ای که «بعد از شروع همین چرخه تولید» ساخته شده دلیل
   647→   647→    // آماده‌بودن است — updatedAt همان لحظه‌ای است که وضعیت generating شده؛
   648→   648→    // برنامه دوره قبلی نباید تولید گیرکردهِ کاربر تمدیدی را ready کند.
   649→   649→    const hasPlan = await db.workoutPlan.findFirst({
   650→   650→      where: { userId, createdAt: { gte: stuck.updatedAt } },
   651→   651→      orderBy: { createdAt: "desc" },
   652→   652→      select: { id: true },
   653→   653→    });
   654→   654→    if (hasPlan) {
   655→   655→      // برنامه موجود است — وضعیت را ready کن (نه failed!)
   656→   656→      await db.programRequest.update({
   657→   657→        where: { id: stuck.id },
   658→   658→        data: { status: "ready" },
   659→   659→      });
   660→   660→      // v37: نوتیف همزمان (قبلاً فقط پیامک می‌رفت — درخواست مالک: پیامک ↔ نوتیف جفت باشند)
   661→   661→      // v63 — dedupe عنوان‌محورِ ۲۴ساعته (هم‌راستا با مسیرهای دیگر) + متنِ یکسان با
   662→   662→      // مسیر تولید عادی. قبلاً متنِ متفاوت بود و dedupeِ (عنوان+متن) آن را رد نمی‌کرد
   663→   663→      // → کاربر هم نوتیف «ساخته شد» و هم «آماده است» می‌گرفت (گزارش نوتیف ۵تایی).
   664→   664→      const alreadyNotified = await db.notification.findFirst({
   665→   665→        where: {
   666→   666→          userId,
   667→   667→          type: "achievement",
   668→   668→          title: "برنامه شما آماده شد! 🎯",
   669→   669→          createdAt: { gte: new Date(Date.now() - 24 * 60 * 60 * 1000) },
   670→   670→        },
   671→   671→        select: { id: true },
   672→   672→      });
   673→   673→      if (!alreadyNotified) {
   674→   674→        await createNotification(
   675→   675→          userId,
   676→   676→          "achievement",
   677→   677→          "برنامه شما آماده شد! 🎯",
   678→   678→          "برنامه تمرینی و غذایی شخصی‌سازی‌شده شما توسط فیتاپ هوشمند ساخته شد. از بخش «تمرینات» و «تغذیه» مشاهده کنید.",
   679→   679→          "?tab=programs"
   680→   680→        );
   681→   681→      }
   682→   682→      // v32: پیامک «برنامه آماده شد» (قالب 663678)
   683→   683→      // v66: دداپ بر چرخهٔ بازیابی‌شده (stuck.id) — دورهٔ بعدی هم پیامک می‌گیرد
   684→   684→      notifyProgramReadySms(userId, stuck.id);
   685→   685→      console.log("[watchdog] recovered orphaned generating → ready (plan exists):", stuck.id);
   686→   686→      return;
   687→   687→    }
   688→   688→
   689→   689→    // برنامه‌ای نیست — اگر تابه‌حال خودترمیم نشده، یک‌بار دوباره شروع کن
   690→   690→    if (stuck.autoRetryCount < 1) {
   691→   691→      await db.programRequest.update({
   692→   692→        where: { id: stuck.id },
   693→   693→        data: { status: "failed", autoRetryCount: { increment: 1 } },
   694→   694→      });
   695→   695→      console.log("[watchdog] auto-retrying orphaned generation:", stuck.id);
   696→   696→      const result = await startProgramGenerationInBackground(userId);
   697→   697→      if (result.started) {
   698→   698→        console.log("[watchdog] auto-retry started for:", userId);
   699→   699→      } else {
   700→   700→        console.log("[watchdog] auto-retry not started:", result.reason);
   701→   701→      }
   702→   702→      return;
   703→   703→    }
   704→   704→
   705→   705→    // قبلاً یک‌بار خودترمیم شده و باز گیر کرده — failed بماند (کاربر دستی retry می‌کند)
   706→   706→    await db.programRequest.update({
   707→   707→      where: { id: stuck.id },
   708→   708→      data: { status: "failed" },
   709→   709→    });
   710→   710→    console.log("[watchdog] marked stuck generation as failed (retry budget exhausted):", stuck.id);
   711→   711→  } catch (e) {
   712→   712→    // watchdog هرگز نباید مسیر caller را بشکند
   713→   713→    console.error("[watchdog] recoverStuckGenerations error:", e);
   714→   714→  }
   715→   715→}

   717→   717→/**
   718→   718→ * v38 — سوئیپ خودکار درخواست‌های «ناموفق» (درخواست مالک: «به هیچ وجه نباید
   719→   719→ * کاربر بدون برنامه بماند»).
   720→   720→ *
   721→   721→ * قبلاً درخواست failed برای همیشه failed می‌ماند مگر کاربر/ادمین دستی retry
   722→   722→ * می‌زد. حالا کرون رفتاری (هر ۳۰ دقیقه) این سوئیپ را صدا می‌زند:
   723→   723→ *   - درخواست‌های failed با attempts < FAILED_AUTO_RETRY_MAX (۳)
   724→   724→ *   - که حداقل FAILED_RETRY_GAP_MS (۱۵ دقیقه) از آخرین تلاش گذشته باشد
   725→   725→ *     (ضد حلقهٔ سریع روی خطای دائمی)
   726→   726→ * دوباره تولید شروع می‌شود (startProgramGenerationInBackground خودش claim
   727→   727→ * اتمیک انجام می‌دهد؛ درخواست failed قابل claim است).
   728→   728→ *
   729→   729→ * خروجی: تعداد درخواست‌هایی که دوباره صف شدند (برای summary کرون).
   730→   730→ */
   731→   731→const FAILED_AUTO_RETRY_MAX = 3;
   732→   732→const FAILED_RETRY_GAP_MS = 15 * 60 * 1000;
   733→   733→
   734→   734→// ─── 🩹 v46 — بودجهٔ روزانهٔ سوئیپ در سطح کل سیستم (لایهٔ آخر ضد نشتی AI) ───
   735→   735→// حتی با همهٔ گاردهای v45، اگر روزی دیتای خراب جدید یا مسیر دورزدنی پیدا شود،
   736→   736→// این بودجه در-حافظه حداکثر سوئیپ‌های مجاز در شبانه‌روز را محدود می‌کند.
   737→   737→// هر تولید برنامهٔ کامل ~۵۰هزار توکن گران است؛ سقف ۶ در ۲۴h یعنی بدترین-case
   738→   738→// نشتی از مسیر سوئیپ ≈ ۳۰۰K توکن/روز — محدود و قابل‌قبول، نه ابدی.
   739→   739→const SWEEP_DAILY_REQUEUE_MAX = 6;
   740→   740→const sweepBudget = { date: "", count: 0 };
   741→   741→
   742→   742→function tryConsumeSweepBudget(): boolean {
   743→   743→  const today = new Date().toISOString().slice(0, 10);
   744→   744→  if (sweepBudget.date !== today) {
   745→   745→    sweepBudget.date = today;
   746→   746→    sweepBudget.count = 0;
   747→   747→  }
   748→   748→  if (sweepBudget.count >= SWEEP_DAILY_REQUEUE_MAX) {
   749→   749→    console.warn(
   750→   750→      `[failed-recovery] daily sweep budget exhausted (${SWEEP_DAILY_REQUEUE_MAX}/24h) — skipping requeue`
   751→   751→    );
   752→   752→    return false;
   753→   753→  }
   754→   754→  sweepBudget.count++;
   755→   755→  return true;
   756→   756→}

   758→   758→export async function recoverFailedGenerations(limit = 3): Promise<number> {
   759→   759→  try {
   760→   760→    const gapAgo = new Date(Date.now() - FAILED_RETRY_GAP_MS);
   761→   761→    const failedRequests = await db.programRequest.findMany({
   762→   762→      where: {
   763→   763→        status: "failed",
   764→   764→        attempts: { lt: FAILED_AUTO_RETRY_MAX },
   765→   765→        updatedAt: { lt: gapAgo },
   766→   766→      },
   767→   767→      orderBy: { updatedAt: "asc" },
   768→   768→      take: limit,
   769→   769→      select: { id: true, userId: true, plan: true, createdAt: true, updatedAt: true },
   770→   770→    });
   771→   771→
   772→   772→    let requeued = 0;
   773→   773→    for (const req of failedRequests) {
   774→   774→      // کاربر نبلاک و دارای پروفایل باشد (خطای دائمی ورودی را دور بزن)
   775→   775→      const user = await db.user.findUnique({
   776→   776→        where: { id: req.userId },
   777→   777→        select: { isBlocked: true },
   778→   778→      });
   779→   779→      if (!user || user.isBlocked) continue;
   780→   780→
   781→   781→      // ─── 🩹 v45 — ریشه باگ بحرانی مالک (نوتیف هر ۳۰ دقیقه + سوختن AI) ───
   782→   782→      // قبلاً این سوئیپ رکورد قدیمی failed را انتخاب می‌کرد ولی
   783→   783→      // startProgramGenerationInBackground روی «آخرین» درخواست کاربر عمل
   784→   784→      // می‌کرد (که معمولاً ready بود) → هر ۳۰ دقیقه: claim برنامهٔ سالم →
   785→   785→      // تولید کامل AI (۴ فراخوانی مدل گران) → نوتیف تکراری → رکورد قدیمی
   786→   786→      // همچنان failed/attempts=0 می‌ماند → حلقهٔ ابدی.
   787→   787→      // حالا:
   788→   788→      //   ① فقط اگر همین رکورد «آخرین» درخواست کاربر باشد صف‌مجدد می‌شود؛
   789→   789→      //   ② رکوردهای قدیمی‌تر terminal می‌شوند (attempts=MAX) تا دیگر
   790→   790→      //      هرگز انتخاب نشوند — دیتای خرابِ موجود هم خودترمیم می‌شود؛
   791→   791→      //   ③ اگر بعد از ساخت همین رکوردِ failed برنامهٔ فعال ذخیره شده
   792→   792→      //      باشد، کاربر برنامه را دارد → تولید تکرار ممنوع، terminal.
   793→   793→      const latest = await db.programRequest.findFirst({
   794→   794→        where: { userId: req.userId },
   795→   795→        orderBy: { createdAt: "desc" },
   796→   796→        select: { id: true },
   797→   797→      });
   798→   798→      if (!latest || latest.id !== req.id) {
   799→   799→        await db.programRequest
   800→   800→          .update({
   801→   801→            where: { id: req.id },
   802→   802→            data: {
   803→   803→              attempts: FAILED_AUTO_RETRY_MAX,
   804→   804→              lastError: "v45 sweep: superseded by a newer request — terminal",
   805→   805→            },
   806→   806→          })
   807→   807→          .catch(() => {});
   808→   808→        console.log(
   809→   809→          `[failed-recovery] ${req.id} is stale (latest=${latest?.id ?? "none"}) → terminalized, will not requeue`
   810→   810→        );
   811→   811→        continue;
   812→   812→      }
   813→   813→      const planAfterFailure = await db.workoutPlan.findFirst({
   814→   814→        where: {
   815→   815→          userId: req.userId,
   816→   816→          active: true,
   817→   817→          createdAt: { gte: req.createdAt },
   818→   818→        },
   819→   819→        select: { id: true },
   820→   820→      });
   821→   821→      if (planAfterFailure) {
   822→   822→        await db.programRequest
   823→   823→          .update({
   824→   824→            where: { id: req.id },
   825→   825→            data: {
   826→   826→              attempts: FAILED_AUTO_RETRY_MAX,
   827→   827→              status: "ready",
   828→   828→              lastError: null,
   829→   829→            },
   830→   830→          })
   831→   831→          .catch(() => {});
   832→   832→        console.log(
   833→   833→          `[failed-recovery] ${req.id} already has a plan created after it → marked ready (no duplicate AI generation)`
   834→   834→        );
   835→   835→        continue;
   836→   836→      }
   837→   837→
   838→   838→      const result: StartGenerationResult = tryConsumeSweepBudget()
   839→   839→        ? await startProgramGenerationInBackground(req.userId)
   840→   840→        : { started: false, reason: "daily_budget" };
   841→   841→      if (result.started) {
   842→   842→        requeued++;
   843→   843→        console.log(
   844→   844→          `[failed-recovery] requeued failed generation for user ${req.userId} (request ${req.id})`
   845→   845→        );
   846→   846→      } else {
   847→   847→        console.log(
   848→   848→          `[failed-recovery] could not requeue ${req.id}: ${result.reason}${result.blockingReason ? ` — ${result.blockingReason}` : ""}`
   849→   849→        );
   850→   850→      }
   851→   851→    }
   852→   852→    return requeued;
   853→   853→  } catch (e) {
   854→   854→    console.error("[failed-recovery] recoverFailedGenerations error:", e);
   855→   855→    return 0;
   856→   856→  }
   857→   857→}
