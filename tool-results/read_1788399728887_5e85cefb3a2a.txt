    48→/**
    49→ * محاسبه مبلغ نهایی خرید پلن — همان منطق checkout (قیمت - تخفیف - اعتبار ارتقا).
    50→ * استفاده مشترک: checkout / bazaar dynamic-price / bazaar purchase (ثبت مبلغ واقعی).
    51→ * خطا: DiscountInvalidError با پیام فارسی (کالر → 400).
    52→ */
    53→export async function computePlanFinalAmount(
    54→  userId: string,
    55→  plan: { id: string; price: number },
    56→  opts: { discountCode?: string; userDiscountCode?: string } = {}
    57→): Promise<ComputedAmount> {
    58→  let finalAmount = plan.price;
    59→  const originalAmount = plan.price;
    60→  let discountValue = 0;
    61→  let appliedCode: string | null = null;
    62→  let userDiscountCodeUsed = false;
    63→  let upgradeCredit = 0;
    64→  let isUpgrade = false;
    65→
    66→  // ─── تخفیف عمومی ───
    67→  if (opts.discountCode) {
    68→    const code = opts.discountCode.trim().toUpperCase();
    69→    const dc = await db.discountCode.findUnique({ where: { code } });
    70→    if (!dc || !dc.active) throw new DiscountInvalidError("کد تخفیف نامعتبر است.");
    71→    if (dc.validUntil && dc.validUntil < new Date())
    72→      throw new DiscountInvalidError("کد تخفیف منقضی شده است.");
    73→    if (dc.maxUses !== -1 && dc.usedCount >= dc.maxUses)
    74→      throw new DiscountInvalidError("سقف استفاده از این کد تخفیف تکمیل شده است.");
    75→    if (dc.applicablePlans !== "all") {
    76→      const allowed = dc.applicablePlans.split(",");
    77→      if (!allowed.includes(plan.id))
    78→        throw new DiscountInvalidError("این کد تخفیف برای پلن انتخاب‌شده قابل استفاده نیست.");
    79→    }
    80→    appliedCode = dc.code;
    81→    discountValue =
    82→      dc.type === "percent"
    83→        ? Math.round((originalAmount * dc.value) / 100)
    84→        : Math.min(dc.value, originalAmount);
    85→  }
    86→
    87→  // ─── کد اختصاصی کاربر (اولویت روی کد عمومی) ───
    88→  if (opts.userDiscountCode) {
    89→    const code = opts.userDiscountCode.trim().toUpperCase();
    90→    const udc = await db.userDiscountCode.findUnique({ where: { code } });
    91→    if (!udc) throw new DiscountInvalidError("کد تخفیف اختصاصی یافت نشد.");
    92→    if (udc.userId !== userId)
    93→      throw new DiscountInvalidError("این کد تخفیف متعلق به حساب شما نیست.");
    94→    if (udc.isUsed) throw new DiscountInvalidError("این کد تخفیف قبلاً استفاده شده است.");
    95→    if (udc.validUntil && udc.validUntil < new Date())
    96→      throw new DiscountInvalidError("کد تخفیف اختصاصی منقضی شده است.");
    97→    // کد اختصاصی بر کد عمومی ارجحیت دارد
    98→    if (appliedCode) {
    99→      discountValue = 0;
   100→      appliedCode = null;
   101→    }
   102→    appliedCode = udc.code;
   103→    userDiscountCodeUsed = true;
   104→    discountValue =
   105→      udc.type === "percent"
   106→        ? Math.round((originalAmount * udc.value) / 100)
   107→        : Math.min(udc.value, originalAmount);
   108→  }
   109→
   110→  // ─── اعتبار ارتقا ───
   111→  const now = new Date();
   112→  const [activeSub, pendingSubs] = await Promise.all([
   113→    db.subscription.findFirst({
   114→      where: { userId, status: "active", endDate: { gt: now } },
   115→      orderBy: { endDate: "desc" },
   116→    }),
   117→    db.subscription.findMany({
   118→      where: {
   119→        userId,
   120→        status: "pending",
   121→        OR: [{ endDate: null }, { endDate: { gt: now } }],
   122→      },
   123→      orderBy: { createdAt: "desc" },
   124→    }),
   125→  ]);
   126→  if (activeSub && activeSub.endDate && activeSub.plan !== plan.id) {
   127→    const oldPlan = await getActivePlan(activeSub.plan as Plan);
   128→    if (oldPlan) {
   129→      const daysLeft = activeSub.endDate
   130→        ? Math.ceil((activeSub.endDate.getTime() - now.getTime()) / (1000 * 60 * 60 * 24))
   131→        : 0;
   132→      if (daysLeft > 0) {
   133→        upgradeCredit = Math.round((activeSub.pricePaid / activeSub.durationDays) * daysLeft);
   134→        isUpgrade = true;
   135→      }
   136→    }
   137→  }
   138→  for (const pSub of pendingSubs) {
   139→    upgradeCredit += pSub.pricePaid;
   140→    isUpgrade = true;
   141→  }
   142→
   143→  finalAmount = Math.max(0, originalAmount - discountValue - upgradeCredit);
   144→
   145→  return {
   146→    finalAmount,
   147→    originalAmount,
   148→    discountValue,
   149→    discountCode: appliedCode,
   150→    userDiscountCodeUsed,
   151→    upgradeCredit,
   152→    isUpgrade,
   153→  };
   154→}

   191→/** نتیجه getRecoveryCutoff — firstRun یعنی تنظیم همین حالا ایجاد شد */
   192→export interface RecoveryCutoffInfo {
   193→  cutoff: Date;
   194→  /** true = این فراخوان مقدار را ایجاد کرد (legacy sweep باید اجرا شود) */
   195→  firstRun: boolean;
   196→}

   198→export async function getRecoveryCutoff(): Promise<RecoveryCutoffInfo> {
   199→  const existing = await db.siteSetting.findUnique({
   200→    where: { key: RECOVERY_CUTOFF_SETTING },
   201→  });
   202→  if (existing) {
   203→    const d = new Date(existing.value);
   204→    if (!isNaN(d.getTime())) return { cutoff: d, firstRun: false };
   205→  }
   206→
   207→  // اولین اجرا (بعد از deploy) — مقدار را «۴۵ دقیقه قبل از الان» قفل کن تا
   208→  // خریدارهای در-حال-پرداخت هم پوشش داده شوند (race-safe با کلید یکتا)
   209→  const effectiveCutoff = new Date(Date.now() - RECOVERY_CUTOFF_GRACE_MS);
   210→  try {
   211→    await db.siteSetting.create({
   212→      data: {
   213→        key: RECOVERY_CUTOFF_SETTING,
   214→        value: effectiveCutoff.toISOString(),
   215→        label: "شروع بازیابی خودکار پرداخت‌ها (پیش از این: رسیدگی دستی)",
   216→      },
   217→    });
   218→  } catch (e: any) {
   219→    if (e?.code === "P2002") {
   220→      // فراخوان موازی برنده شد — مقدارش معتبر است
   221→      const winner = await db.siteSetting.findUnique({
   222→        where: { key: RECOVERY_CUTOFF_SETTING },
   223→      });
   224→      if (winner) {
   225→        const d = new Date(winner.value);
   226→        if (!isNaN(d.getTime())) return { cutoff: d, firstRun: false };
   227→      }
   228→      return { cutoff: effectiveCutoff, firstRun: false };
   229→    }
   230→    throw e;
   231→  }
   232→
   233→  // ─── FIX (درخواست مالک ۱۴۰۵-۰۶: «همه پرداخت‌های موفق وریفای شوند») ───
   234→  // قبلاً: همه معلق‌های قدیمی‌تر از برش «کورکورانه» manual_resolved می‌شدند —
   235→  // حتی پرداخت‌هایی که کاربر واقعاً پول داده بود! حالا معلق‌های legacy با
   236→  // استعلام واقعی زرین‌پال تعیین‌تکلیف می‌شوند (resolveLegacyPendingPayments
   237→  // توسط کرون/جاروی داخلی صدا زده می‌شود):
   238→  //   • زرین‌پال پرداخت را تأیید کند (100/101) → تحویل کامل پلن/کیف پول
   239→  //   • پاسخ قطعی منفی → manual_resolved (رسیدگی دستی — پولی نداده)
   240→  //   • خطای شبکه → pending می‌ماند (جاروی بعدی دوباره)
   241→  return { cutoff: effectiveCutoff, firstRun: true };
   242→}

   244→/**
   245→ * تعیین‌تکلیف آگاهانه معلق‌های legacy (قدیمی‌تر از برش) — به‌جای بستن کورکورانه.
   246→ *
   247→ * فقط همان‌جا اجرا می‌شود که cutoff «برای اولین بار» ایجاد شده (firstRun).
   248→ * برای هر پرداخت معلق قدیمی:
   249→ *  - درگاهی + authority → zarinpalVerify (استعلام واقعی پول):
   250→ *      • ok (100/101 بدون replay) → تحویل پلن یا شارژ کیف پول → success
   251→ *      • خطای قطعی (پرداخت‌نشده) → manual_resolved («رسیدگی دستی (قدیمی)»)
   252→ *      • خطای شبکه → pending می‌ماند؛ جاروی بعدی دوباره می‌کوشد
   253→ *  - کیف‌پولی (بدون درگاه) → expired (پولی کسر نشده؛ ناگهان کسر نکن)
   254→ *
   255→ * امنیت: claim اتمیک مثل بقیه مسیرها — دو اجرای موازی یکی را انجام می‌دهند.
   256→ */
   257→export async function resolveLegacyPendingPayments(
   258→  cutoff: Date
   259→): Promise<{
   260→  scanned: number;
   261→  delivered: number;
   262→  closed: number;
   263→  results: Array<{ paymentId: string; status: string; message?: string }>;
   264→}> {
   265→  const legacy = await db.payment.findMany({
   266→    where: {
   267→      status: { in: ["pending", "verifying"] },
   268→      createdAt: { lt: cutoff },
   269→    },
   270→    orderBy: { createdAt: "asc" },
   271→    take: 200,
   272→  });
   273→  const results: Array<{ paymentId: string; status: string; message?: string }> = [];
   274→  let delivered = 0;
   275→  let closed = 0;
   276→
   277→  // ایمپورت تنبل برای جلوگیری از circular dependency (zarinpal → provider)
   278→  const { zarinpalVerify } = await import("@/lib/fitness/zarinpal");
   279→
   280→  for (let payment of legacy) {
   281→    // verifying گیرکرده → ریست اتمیک به pending قبل از claim
   282→    if (payment.status === "verifying") {
   283→      const reset = await db.payment.updateMany({
   284→        where: { id: payment.id, status: "verifying" },
   285→        data: { status: "pending" },
   286→      });
   287→      if (reset.count === 1) {
   288→        payment = { ...payment, status: "pending" };
   289→      }
   290→    }
   291→
   292→    const claim = await claimPayment(payment);
   293→    if (claim !== "claimed") {
   294→      // busy (فراخوان دیگری) / processed (تعیین‌تکلیف شده) / stuck (ریست شد)
   295→      results.push({ paymentId: payment.id, status: claim });
   296→      continue;
   297→    }
   298→
   299→    try {
   300→      // کیف‌پولی قدیمی → بسته (پولی کسر نشده — کسر ناگهانی نکن)
   301→      if (payment.paymentMethod !== "gateway" || !payment.authority) {
   302→        await markPaymentExpired(payment.id);
   303→        closed++;
   304→        results.push({ paymentId: payment.id, status: "closed-wallet-legacy" });
   305→        continue;
   306→      }
   307→
   308→      const zRes = await zarinpalVerify({
   309→        authority: payment.authority,
   310→        amount: payment.amount,
   311→      });
   312→
   313→      if (!zRes.ok) {
   314→        if (zRes.transportError) {
   315→          // شبکه — نتیجه قطعی نیست؛ pending بماند برای جاروی بعدی
   316→          await releaseClaim(payment.id);
   317→          results.push({ paymentId: payment.id, status: "network-error-pending" });
   318→          continue;
   319→        }
   320→        // زرین‌پال قطعیاً پرداختی ندارد → منقضی (پولی نداده — قابل بازیابی دستی)
   321→        await markPaymentExpired(payment.id);
   322→        closed++;
   323→        results.push({ paymentId: payment.id, status: "closed-unpaid" });
   324→        continue;
   325→      }
   326→
   327→      // ok (100/101) — چک replay
   328→      if (zRes.code === 101 || zRes.alreadyVerified) {
   329→        if (await isAuthorityUsedElsewhere(payment.authority, payment.id)) {
   330→          await markPaymentFailed(payment.id, zRes.refId ?? null);
   331→          results.push({ paymentId: payment.id, status: "failed-replay" });
   332→          continue;
   333→        }
   334→      }
   335→
   336→      const refId =
   337→        zRes.refId ?? payment.refId ?? `${Date.now()}${Math.floor(Math.random() * 10000)}`;
   338→
   339→      // شارژ کیف پول legacy → helper مشترک
   340→      if (payment.plan === "wallet_topup") {
   341→        await deliverWalletTopupPayment({
   342→          payment,
   343→          refId,
   344→          cardPan: zRes.cardPan ?? null,
   345→          cardHash: zRes.cardHash ?? null,
   346→          fee: zRes.fee ?? null,
   347→        });
   348→      } else {
   349→        await deliverPlanPayment({
   350→          userId: payment.userId,
   351→          payment,
   352→          refId,
   353→          cardPan: zRes.cardPan ?? null,
   354→          cardHash: zRes.cardHash ?? null,
   355→          fee: zRes.fee ?? null,
   356→        });
   357→      }
   358→
   359→      delivered++;
   360→      results.push({ paymentId: payment.id, status: "delivered-legacy" });
   361→      console.log(
   362→        `[payment-delivery] legacy pending ${payment.id} → تحویل شد (refId=${refId})`
   363→      );
   364→    } catch (err) {
   365→      console.error("[payment-delivery] legacy resolve error:", payment.id, err);
   366→      await releaseClaim(payment.id);
   367→      results.push({ paymentId: payment.id, status: "error-pending" });
   368→    }
   369→  }
   370→
   371→  return { scanned: legacy.length, delivered, closed, results };
   372→}

   374→/**
   375→ * تحویل شارژ کیف پول (wallet_topup) — تراکنش اتمیک idempotent.
   376→ * مشترک بین: verify route / recover route / cron sweep / legacy sweep.
   377→ * (قبلاً این منطق فقط داخل recover route بود — cron به آن دسترسی نداشت.)
   378→ */
   379→export async function deliverWalletTopupPayment(params: {
   380→  payment: PaymentRecord;
   381→  refId: string;
   382→  cardPan?: string | null;
   383→  cardHash?: string | null;
   384→  fee?: number | null;
   385→}): Promise<{ newBalance: number }> {
   386→  const { payment, refId } = params;
   387→
   388→  // ذخیره اطلاعات کارت/کارمزد
   389→  if (params.cardPan || params.cardHash || params.fee != null) {
   390→    await db.payment.update({
   391→      where: { id: payment.id },
   392→      data: {
   393→        cardPan: params.cardPan ?? null,
   394→        cardHash: params.cardHash ?? null,
   395→        fee: params.fee ?? null,
   396→      },
   397→    });
   398→  }
   399→
   400→  const newBalance = await db.$transaction(async (tx) => {
   401→    // اگر همین پرداخت هم‌زمان توسط فراخوان دیگری success شد → skip (idempotent)
   402→    const fresh = await tx.payment.findUnique({

   482→/** نتیجه عملیات claim */
   483→export type ClaimResult = "claimed" | "busy" | "stuck" | "processed";
   484→
   485→/**
   486→ * F12: claim اتمیک وضعیت pending → verifying.
   487→ * دو فراخوانی همزمان روی همان paymentId فقط یکی را عبور می‌دهد.
   488→ * - "claimed": ما claim کردیم (ادامه بده)
   489→ * - "busy": فراخوان دیگری در حال پردازش است (بعداً دوباره)
   490→ * - "stuck": claim قبلی >۱۵ دقیقه معلق مانده (پروسه قبلی مرده) → failed
   491→ *            شده تا قابل تکرار باشد (کالر باید پیام مناسب بدهد)
   492→ * - "processed": وضعیت نهایی است (success/failed/cancelled/...)
   493→ */
   494→export async function claimPayment(payment: PaymentRecord): Promise<ClaimResult> {
   495→  if (payment.status !== "pending") {
   496→    if (payment.status !== "verifying") return "processed";
   497→    const stuckFor = payment.verifiedAt ? Date.now() - payment.verifiedAt.getTime() : 0;
   498→    if (payment.verifiedAt && stuckFor > VERIFYING_STUCK_MS) {
   499→      await db.payment.update({
   500→        where: { id: payment.id },
   501→        data: { status: "failed" },
   502→      });
   503→      return "stuck";
   504→    }
   505→    return "busy";
   506→  }
   507→  const claimed = await db.payment.updateMany({
   508→    where: { id: payment.id, status: "pending" },
   509→    data: { status: "verifying", verifiedAt: new Date() },
   510→  });
   511→  return claimed.count === 0 ? "busy" : "claimed";
   512→}

   514→/** آزادسازی claim در خطاهای غیرقطعی (شبکه) — پرداخت دوباره pending می‌شود */
