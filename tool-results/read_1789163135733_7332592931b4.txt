   196→  function handleAvatarChange(e: React.ChangeEvent<HTMLInputElement>) {
   197→    const file = e.target.files?.[0];
   198→    e.target.value = ""; // ریست input تا انتخاب دوباره‌ی همان فایل هم کار کند
   199→    if (!file || !user) return;
   200→    if (!file.type.startsWith("image/")) {
   201→      toast.error("فقط فایل تصویری مجاز است.");
   202→      return;
   203→    }
   204→    if (file.size > 8 * 1024 * 1024) {
   205→      toast.error("حجم عکس بیش از ۸ مگابایت است.");
   206→      return;
   207→    }
   208→    if (avatarUploadInFlight) return; // ضد دوبار-ارسال
   209→    avatarUploadInFlight = true;
   210→    setAvatarUploading(true); // فقط نشانگر ظریف روی بج دوربین (مسدودکننده نیست)
   211→
   212→    const previousAvatarUrl = user.avatarUrl ?? null;
   213→
   214→    // عمداً await نمی‌کنیم — UI بلاک نشود؛ حتی اگر overlay unmount شود
   215→    // ادامه پیدا می‌کند (همه‌ی به‌روزرسانی‌ها از useAppStore.getState()).
   216→    void (async () => {
   217→      let objectUrl: string | null = null;
   218→      try {
   219→        const { downscaleImage } = await import("@/lib/fitness/client-image");
   220→        const resized = await downscaleImage(file);
   221→
   222→        // آپدیت خوش‌بینانه — همه‌ی مصرف‌کننده‌های store فوری عوض می‌شوند
   223→        objectUrl = URL.createObjectURL(resized);
   224→        const st0 = useAppStore.getState();
   225→        if (st0.user) st0.setUser({ ...st0.user, avatarUrl: objectUrl });
   226→        toast("در حال بروزرسانی عکس پروفایل…");
   227→
   228→        const fd = new FormData();
   229→        fd.append("image", resized);
   230→        const res = await fetch("/api/user/avatar", { method: "POST", body: fd });
   231→        const data = await res.json().catch(() => ({}));
   232→        if (!res.ok || !data.ok || !data.avatarUrl) {
   233→          throw new Error(data.error || "آپلود عکس پروفایل ناموفق بود.");
   234→        }
   235→        const serverUrl = data.avatarUrl as string;
   236→        // URL نهایی سرور — merge با آخرین state کاربر (نه closure کهنه)
   237→        const st = useAppStore.getState();
   238→        if (st.user) st.setUser({ ...st.user, avatarUrl: serverUrl });
   239→        toast.success("عکس پروفایل به‌روزرسانی شد.");
   240→      } catch (err) {
   241→        // برگشت به آواتار قبلی
   242→        const st = useAppStore.getState();
   243→        if (st.user) st.setUser({ ...st.user, avatarUrl: previousAvatarUrl });
   244→        toast.error(err instanceof Error ? err.message : "خطا در آپلود عکس پروفایل");
   245→      } finally {
   246→        if (objectUrl) {
   247→          // با تأخیر آزاد می‌شود تا <img> فرصت لود objectURL را داشته باشد
   248→          const url = objectUrl;
   249→          setTimeout(() => URL.revokeObjectURL(url), 1500);
   250→        }
   251→        avatarUploadInFlight = false;
   252→        setAvatarUploading(false);
   253→      }
   254→    })();
   255→  }

   992→/** خلاصهٔ کوتاه یک آزمایش خون برای ردیف لیست — امتیاز + تعداد نشانگرها/غیرنرمال‌ها */
   993→function bloodTestSummary(result: any): string {
   994→  if (!result || typeof result !== "object") return "نتیجهٔ تحلیل موجود است";
   995→  const parts: string[] = [];
   996→  if (typeof result.score === "number") parts.push(`امتیاز ${toPersianDigits(result.score)}`);
   997→  if (Array.isArray(result.markers) && result.markers.length > 0) {
   998→    const high = result.markers.filter((m: any) => m?.status === "high").length;
   999→    const low = result.markers.filter((m: any) => m?.status === "low").length;
  1000→    parts.push(`${toPersianDigits(result.markers.length)} نشانگر`);
  1001→    if (high > 0) parts.push(`${toPersianDigits(high)} بالا`);
  1002→    if (low > 0) parts.push(`${toPersianDigits(low)} پایین`);
  1003→  } else if (typeof result.overall === "string" && result.overall) {
  1004→    return result.overall.length > 60 ? result.overall.slice(0, 60) + "…" : result.overall;
  1005→  }
  1006→  return parts.length > 0 ? parts.join(" · ") : "نتیجهٔ تحلیل موجود است";
  1007→}

  1025→/** تبدیل ارقام فارسی/عربی به لاتین (ورودی مبلغ شارژ) */
  1026→const FA_DIGITS = "۰۱۲۳۴۵۶۷۸۹";
  1027→const AR_DIGITS = "٠١٢٣٤٥٦٧٨٩";
  1028→function toEnDigits(s: string): string {
  1029→  return s
  1030→    .replace(/[۰-۹]/g, (d) => String(FA_DIGITS.indexOf(d)))
  1031→    .replace(/[٠-٩]/g, (d) => String(AR_DIGITS.indexOf(d)));
  1032→}
  1033→
  1034→/**
  1035→ * ورودی مبلغ شارژ را به «فقط ارقام لاتین بدون جداکننده» پاک می‌کند.
  1036→ * (state همیشه digits-only است؛ کاما فقط در نمایش است)
  1037→ */
  1038→function sanitizeChargeInput(raw: string): string {
  1039→  return toEnDigits(raw).replace(/\D/g, "").replace(/^0+(?=\d)/, "");
  1040→}
  1041→
  1042→/**
  1043→ * نمایش مبلغ داخل ورودی شارژ: کاما بعد از هر ۳ رقم + ارقام فارسی.
  1044→ * مثال: 1500000 → «۱,۵۰۰,۰۰۰»
  1045→ */
  1046→function formatChargeInput(digits: string): string {
  1047→  const clean = digits.replace(/\D/g, "");
  1048→  if (!clean) return "";
  1049→  return toPersianDigits(Number(clean).toLocaleString("en-US"));
  1050→}
