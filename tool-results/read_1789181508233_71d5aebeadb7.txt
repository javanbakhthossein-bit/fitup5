   154→  function startPolling(baselineCreatedAt: string | null) {
   155→    stopPolling();
   156→    pollInFlightRef.current = false;
   157→    pollBaselineRef.current = baselineCreatedAt;
   158→    pollDeadlineRef.current = Date.now() + ANALYSIS_POLL_MAX_MS;
   159→    pollTimerRef.current = setInterval(async () => {
   160→      if (pollInFlightRef.current) return;
   161→      pollInFlightRef.current = true;
   162→      try {
   163→        const res = await fetch("/api/coach/analyze-video", { cache: "no-store" });
   164→        if (res.ok) {
   165→          const data = await res.json();
   166→          const resultCreatedAt = typeof data?.createdAt === "string" ? data.createdAt : null;
   167→          if (data?.result && resultCreatedAt !== pollBaselineRef.current) {
   168→            // نتیجهٔ جدید (createdAt متفاوت از پایه) — مستقل از ساعت کلاینت
   169→            handleAnalysisSuccess(data.result);
   170→            return;
   171→          }
   172→          if (data?.analyzing === false) {
   173→            // تحلیل پس‌زمینه تمام شد ولی نتیجهٔ جدیدی ثبت نشد → شکست
   174→            handleAnalysisFailure("تحلیل ویدیو ناتمام ماند. لطفاً دوباره تلاش کنید.");
   175→            return;
   176→          }
   177→          if (Date.now() > pollDeadlineRef.current) {
   178→            handleAnalysisFailure("تحلیل ویدیو بیش از حد انتظار طول کشید. لطفاً دوباره تلاش کنید.");
   179→            return;
   180→          }
   181→        }
   182→        // خطای گذرای شبکه/سرور → سکوت و ادامه poll تا ددلاین
   183→      } catch {
   184→        // ادامه poll
   185→      } finally {
   186→        pollInFlightRef.current = false;
   187→      }
   188→    }, ANALYSIS_POLL_MS);
   189→  }

   227→  // فایل‌های بزرگ‌تر بعد از آپلود سمت سرور فشرده می‌شوند؛ کلاینت فقط گارد سخت می‌گذارد.
   228→  const MAX_VIDEO_MB = 300;

   231→  /** حجم خوانا برای نمایش فایل انتخاب‌شده (مثلاً «۴۸.۲ مگابایت») */

   236→  function handleFile(e: React.ChangeEvent<HTMLInputElement>) {
   237→    const f = e.target.files?.[0];
   238→    if (!f) return;
   239→    if (!f.type.startsWith("video/")) {
   240→      toast.error("فقط فایل ویدیویی مجاز است");
   241→      return;
   242→    }
   243→    setAnalysisFailed(false);
   244→    // سقف نرم فنی ۳۰۰MB — ویدیوی بزرگ‌تر: «ویدیوی بسیار بزرگ است»
   245→    // (فشرده‌سازی پس از آپلود سمت سرور انجام می‌شود؛ اینجا فقط گارد سخت)
   246→    if (f.size > MAX_VIDEO_BYTES) {
   247→      const mb = Math.round(f.size / (1024 * 1024));
   248→      toast.error(`ویدیوی بسیار بزرگ است (${toPersianDigits(mb)} مگابایت) — حداکثر ${toPersianDigits(MAX_VIDEO_MB)} مگابایت`);
   249→      return;
   250→    }

   252→    setVideoUrl(URL.createObjectURL(f));
   253→    setResult(null);
   254→  }

   256→  async function analyze() {
   257→    if (!file) return;
   258→    if (limitReached) {
   259→      toast.error("سقف استفاده از این قابلیت پر شده است.");
   260→      return;
   261→    }
   262→    // گارد سخت دوباره (در برابر تغییر فایل) — سقف نرم ۳۰۰MB
   263→    if (file.size > MAX_VIDEO_BYTES) {
   264→      toast.error("ویدیوی بسیار بزرگ است — حداکثر ۳۰۰ مگابایت");
   265→      return;
   266→    }
   267→    setAnalyzing(true);
   268→    setAnalysisFailed(false);
   269→    try {
   270→      // ─── ارسال به‌صورت multipart/form-data (نه base64) ───
   271→      // base64 حجم payload را ~۱.۳۳ برابر می‌کند و باعث خطای «Unexpected token»
   272→      // در پاسخ‌های بزرگ می‌شود. FormData از حالت باینری استفاده می‌کند.
   273→      const formData = new FormData();
   274→      formData.append("video", file);
   275→      formData.append("userContext", "تحلیل فرم بدن و تکنیک حرکات ورزشی");
   276→
   277→      // v74: پایهٔ createdAt نتیجهٔ قبلی (قبل از POST) — برای تشخیص «نتیجهٔ جدید»
   278→      // در poll بدون وابستگی به ساعت کلاینت
   279→      let baselineCreatedAt: string | null = null;
   280→      try {
   281→        const pre = await fetch("/api/coach/analyze-video", { cache: "no-store" });
   282→        if (pre.ok) {
   283→          const preData = await pre.json();
   284→          baselineCreatedAt = typeof preData?.createdAt === "string" ? preData.createdAt : null;
   285→        }
   286→      } catch {
   287→        // poll با پایهٔ null هم درست کار می‌کند (نتیجهٔ جدید همیشه createdAt متفاوت دارد)
   288→      }
   289→
   290→      const res = await fetch("/api/coach/analyze-video", {
   291→        method: "POST",
   292→        body: formData,
   293→        // هدر Content-Type خودکار توسط browser برای FormData با boundary تنظیم می‌شود.
   294→        // نباید دستی set شود.
   295→      });
   296→      // ─── بررسی وضعیت قبل از JSON parse ───
   297→      // اگر سرور HTML برگرداند (مثلاً خطای 500)، res.json() خطای
   298→      // "Unexpected token" می‌دهد. ابتدا res.ok را بررسی می‌کنیم.
   299→      if (!res.ok) {
   300→        let errMsg = "خطا در تحلیل ویدیو";
   301→        try {
   302→          const errData = await res.json();
   303→          errMsg = errData?.error || errMsg;
   304→        } catch {
   305→          // اگر JSON parse ناموفق بود (مثلاً HTML خطا یا پاسخ خالی)
   306→          errMsg = `خطای سرور (${res.status}). لطفاً دوباره تلاش کنید.`;
   307→        }
   308→        throw new Error(errMsg);
   309→      }
   310→      const data = await res.json();
   311→
   312→      if (data?.status === "analyzing" || data?.started) {
   313→        // ─── v74: پاسخ فوری سرور — تحلیل در پس‌زمینه شروع شد ───
   314→        // وضعیت واقعی «در حال تحلیل» می‌ماند و GET هر ۵ ثانیه poll می‌شود
   315→        // تا نتیجه بیاید (حتی با بستن صفحه، نتیجه در DB ذخیره می‌شود).
   316→        useAppStore.setState((s) => ({
   317→          user: s.user ? { ...s.user, videoStatus: "analyzing" } : s.user,
   318→        }));
   319→        if (typeof window !== "undefined") {
   320→          window.dispatchEvent(new CustomEvent("prereq-updated"));
   321→        }
   322→        startPolling(baselineCreatedAt);
   323→        return;
   324→      }
   325→
   326→      // مسیر همگام (سازگاری با پاسخ قدیمی سرور) — نتیجه مستقیم در همان POST
   327→      handleAnalysisSuccess(data);
   328→    } catch (e) {
   329→      setAnalyzing(false);
   330→      toast.error(e instanceof Error ? e.message : "خطا در تحلیل");
   331→    }
   332→  }
