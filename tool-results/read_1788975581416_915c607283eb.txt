   186→    /**
   187→     * v1.2.6 — لینک عمیق ورودی (App Link یا اسکیم fitup:// یا اعلان) یا null:
   188→     *  ۱) extra «fitup_link» از NotificationSync — مسیر/کوئری نسبی مثل
   189→     *     "?tab=progress&section=checkup" یا "/panel" → SITE_URL + آن
   190→     *  ۲) dataِ https روی دامنهٔ خودمان (fittup.ir) → همان URL — پارامترهای
   191→     *     ?tab= ?ref= ?renewal= ?screen= ?article= دست‌نخورده می‌مانند
   192→     *  ۳) v1.2.6 — اسکیم اختصاصی fitup://open?url=<encoded> از صفحهٔ /go سایت
   193→     *     (لینک‌های هوشمند پیامک — فقط URL دامنهٔ خودمان پذیرفته می‌شود)
   194→     * (null-امن — هرگز به‌خاطر لینک خراب کرش نمی‌کند)
   195→     */
   196→    private fun deepLinkUrl(from: Intent?): String? {
   197→        if (from == null) return null
   198→        try {
   199→            val extra = from.getStringExtra("fitup_link")
   200→            if (!extra.isNullOrBlank() && (extra.startsWith("/") || extra.startsWith("?"))) {
   201→                return BuildConfig.SITE_URL.trimEnd('/') + extra
   202→            }
   203→            val data = from.data ?: return null
   204→            val scheme = data.scheme?.lowercase()
   205→            // ─── v1.2.6: fitup://open?url=<encoded> — از لینک هوشمند /go سایت ───
   206→            if (scheme == "fitup") {
   207→                val target = data.getQueryParameter("url") ?: return null
   208→                if (target.startsWith("/")) {
   209→                    // مسیر نسبی روی دامنهٔ خودمان
   210→                    return BuildConfig.SITE_URL.trimEnd('/') + target
   211→                }
   212→                val t = android.net.Uri.parse(target)
   213→                val tScheme = t.scheme?.lowercase()
   214→                val tHost = t.host?.lowercase() ?: return null
   215→                val isOurHost = tHost == "fittup.ir" || tHost.endsWith(".fittup.ir")
   216→                if (isOurHost && (tScheme == "https")) return target
   217→                return null
   218→            }
   219→            if (scheme != "https") return null
   220→            val host = data.host?.lowercase() ?: return null
   221→            // فقط دامنهٔ خودمان — همان منطق routeUrl (fittup.ir و زیردامنه‌ها)
   222→            if (host == "fittup.ir" || host.endsWith(".fittup.ir")) return data.toString()
   223→        } catch (_: Exception) {
   224→        }
   225→        return null
   226→    }

   361→        webView.setDownloadListener { url, _, contentDisposition, mimetype, _ ->
   362→            try {
   363→                // v1.2.7 — Safety-net: لینک‌های data: (PNG/PDF برنامه‌ها) را
   364→                // DownloadManager پشتیبانی نمی‌کند (فقط http/https) — از پل
   365→                // MediaStore ذخیره می‌شوند. مسیر اصلی از پل جاوااسکریپت

   369→                        mimetype.contains("pdf", true) -> "pdf"

   376→                    return@setDownloadListener

   379→                // v1.2.8 — APK (آپدیت/دانلود مستقیم) → مرورگر بیرونی؛ اپ دیگر
   380→                // مجوز نصب ندارد و APK از DownloadManager خود اپ نمی‌گذرد

   385→                    return@setDownloadListener

   430→    private fun handleExternalScheme(uri: Uri): Boolean {
   431→        try {
   432→            when (uri.scheme?.lowercase()) {
   433→                "tel" -> startActivity(Intent(Intent.ACTION_DIAL, uri))
   434→                "mailto" -> startActivity(Intent(Intent.ACTION_SENDTO, uri))
   435→                "sms" -> startActivity(Intent(Intent.ACTION_SENDTO, uri))

   447→        } catch (_: Exception) {
   448→            // اپی برای این لینک نیست — نادیده بگیر
   449→        }
   450→        return true

   575→    /** شروع دانلود فایل معمولی با DownloadManager → پوشه Download اپ (بدون مجوز نوشتن).
   576→     *  (APK از این مسیر نمی‌گذرد — v1.2.8)
   577→     *  @return true = واقعاً enqueue شد؛ false = شکست */
   578→    private fun startNativeDownload(url: String, fileName: String): Boolean {
   579→        val safeName = fileName.replace(Regex("[^A-Za-z0-9._-]"), "_").ifBlank {
   580→            "fitup-${System.currentTimeMillis()}"

   582→        try {
   583→            val request = DownloadManager.Request(Uri.parse(url))
   584→                .setTitle(safeName)
   585→                .setDescription("دانلود فایل")
   586→                .setNotificationVisibility(DownloadManager.Request.VISIBILITY_VISIBLE_NOTIFY_COMPLETED)

   600→    /** جلوی دوبار اجرای پشت‌سرهم (برادکست + onResume هم‌زمان) */
   601→    private var handlingDownloadFinish = false
   602→
   603→    /** پایان یک دانلود: فقط پاک‌سازی وضعیت pending.
   604→     *  فایل‌های معمولی نوتیف خود DownloadManager را دارند؛ APK دیگر اصلاً از
   605→     *  این مسیر نمی‌گذرد (v1.2.8 → مرورگر بیرونی). */
   606→    private fun handleDownloadFinished(id: Long) {
   607→        if (id <= 0 || id != pendingDownloadId()) return
   608→        if (handlingDownloadFinish) return
   609→        handlingDownloadFinish = true
   610→        try {
   611→            clearPendingDownload()
   612→        } finally {
   613→            handlingDownloadFinish = false
   614→        }
   615→    }
   616→
   617→    /** safety-net هر onResume: پایان دانلود فایل معمولی (برادکست گم‌شده با
   618→     *  بازسازی اکتیویتی/بستن اپ) — فقط پاک‌سازی وضعیت pending (v1.2.8). */
   619→    private fun checkPendingDownload() {
   620→        val id = pendingDownloadId()
   621→        if (id > 0) handleDownloadFinished(id)
   622→    }
   623→
   624→    /** گیرندهٔ پایان دانلود (فایل‌های معمولی) → پاک‌سازی وضعیت pending.
   625→     *  ⚠ RECEIVER_EXPORTED (نه NOT_EXPORTED): برادکست ACTION_DOWNLOAD_COMPLETE
   626→     *  «protected» است و فقط دانلودمنیجر/سیستم می‌تواند بفرستد؛ در اندروید ۱۴+
   627→     *  گیرندهٔ NOT_EXPORTED این برادکست را نمی‌گیرد (فرستنده DownloadProvider است). */
   628→    private fun setupDownloadCompleteReceiver() {
   629→        val receiver = object : BroadcastReceiver() {
   630→            override fun onReceive(context: Context?, intent: Intent?) {
   631→                if (intent?.action != DownloadManager.ACTION_DOWNLOAD_COMPLETE) return
   632→                val id = intent.getLongExtra(DownloadManager.EXTRA_DOWNLOAD_ID, -1L)
   633→                handleDownloadFinished(id)
   634→            }
   635→        }
   636→        try {
   637→            if (Build.VERSION.SDK_INT >= 33) {
   638→                registerReceiver(
   639→                    receiver,
   640→                    IntentFilter(DownloadManager.ACTION_DOWNLOAD_COMPLETE),
   641→                    Context.RECEIVER_EXPORTED
   642→                )
   643→            } else {
   644→                registerReceiver(receiver, IntentFilter(DownloadManager.ACTION_DOWNLOAD_COMPLETE))
   645→            }
   646→        } catch (_: Exception) {}
   647→    }

   649→    /** نام فایل از Content-Disposition یا URL */
   650→    private fun guessFileName(contentDisposition: String?, mimeType: String?, url: String): String {
   651→        try {
   652→            if (!contentDisposition.isNullOrBlank()) {
   653→                val m = Regex("filename\\*=UTF-8''([^;]+)|filename=\"?([^\";]+)\"?", RegexOption.IGNORE_CASE)
   654→                    .find(contentDisposition)
   655→                if (m != null) {
   656→                    val name = (m.groupValues[1].ifBlank { m.groupValues[2] })
   657→                        .replace("+", " ")
   658→                    if (name.isNotBlank()) return java.net.URLDecoder.decode(name, "UTF-8")
   659→                }
   660→            }
   661→            if (!mimeType.isNullOrBlank() && mimeType.equals("application/vnd.android.package-archive", true)) {
   662→                return "fitup-${System.currentTimeMillis()}.apk"
   663→            }
   664→            val path = URI(url).path ?: return "fitup-file"
   665→            return path.substringAfterLast('/').ifBlank { "fitup-file" }
   666→        } catch (_: Exception) {
   667→            return "fitup-file"
   668→        }
   669→    }

   671→    /* ───────────── ذخیره data URL (PNG/PDF) ───────────── */
   672→
   673→    @SuppressLint("InlinedApi")
   674→    private fun downloadDataUrl(filename: String, dataUrl: String) {
   675→        try {
   676→            val safeName = filename.ifBlank { "fitup-${System.currentTimeMillis()}" }
   677→                .replace(Regex("[^A-Za-z0-9._-\\u0600-\\u06FF ]"), "_")
   678→            val comma = dataUrl.indexOf(',')
   679→            if (comma < 0 || !dataUrl.startsWith("data:", ignoreCase = true)) {
   680→                runOnUiThread { toast("فایل قابل ذخیره نیست") }
   681→                return
   682→            }
   683→            val meta = dataUrl.substring(5, comma)
   684→            val mime = meta.substringBefore(";").ifBlank { "application/octet-stream" }
   685→            val isBase64 = meta.contains("base64", ignoreCase = true)
   686→            val bytes = if (isBase64) {
   687→                Base64.decode(dataUrl.substring(comma + 1), Base64.DEFAULT)
   688→            } else {
   689→                dataUrl.substring(comma + 1).toByteArray(Charsets.UTF_8)
   690→            }
   691→
   692→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   693→                val values = ContentValues().apply {
   694→                    put(MediaStore.Downloads.DISPLAY_NAME, safeName)
   695→                    put(MediaStore.Downloads.MIME_TYPE, mime)
   696→                    put(MediaStore.Downloads.IS_PENDING, 1)
   697→                }
   698→                val uri = contentResolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, values)
   699→                if (uri == null) {
   700→                    runOnUiThread { toast("خطا در ذخیره فایل") }
   701→                    return
   702→                }
   703→                contentResolver.openOutputStream(uri)?.use { it.write(bytes) }
   704→                values.clear()
   705→                values.put(MediaStore.Downloads.IS_PENDING, 0)
   706→                contentResolver.update(uri, values, null, null)
   707→            } else {
   708→                val dir = getExternalFilesDir(Environment.DIRECTORY_DOCUMENTS) ?: filesDir
   709→                val file = File(dir, safeName)
   710→                FileOutputStream(file).use { it.write(bytes) }
   711→            }
   712→            runOnUiThread { toast("«$safeName» ذخیره شد ✓") }
   713→        } catch (e: Exception) {
   714→            Log.e("FitUpApp", "download failed: ${e.message}")
   715→            runOnUiThread { toast("خطا در ذخیره فایل") }
   716→        }
   717→    }

   791→    private fun requestMediaRuntimePermissions(webRequest: PermissionRequest) {
   792→        pendingWebPermissionRequest?.let { if (it !== webRequest) it.deny() }
   793→        pendingWebPermissionRequest = webRequest
   794→        val perms = mutableListOf<String>()
   795→        if (webRequest.resources.contains(PermissionRequest.RESOURCE_VIDEO_CAPTURE)) {
   796→            perms.add(android.Manifest.permission.CAMERA)
   797→        }
   798→        if (webRequest.resources.contains(PermissionRequest.RESOURCE_AUDIO_CAPTURE)) {
   799→            perms.add(android.Manifest.permission.RECORD_AUDIO)
   800→        }
   801→        val alreadyGranted = perms.all {
   802→            ContextCompat.checkSelfPermission(this, it) == PackageManager.PERMISSION_GRANTED
   803→        }
   804→        if (perms.isEmpty() || alreadyGranted) {
   805→            pendingWebPermissionRequest = null
   806→            runOnUiThread { webRequest.grant(webRequest.resources) }
   807→        } else {

   828→    /** کد OTP از کلیپ‌بورد — v1.2.9: هم کدِ خالص و هم «متن کامل پیامک» کپی‌شده.
   829→     *  برای جلوگیری از درج اشتباه عدد ۴-رقمی نامرتبط (مثل سال «1404»)، استخراج
   830→     *  گروه ۴-رقمی فقط وقتی انجام می‌شود که متن شامل «فیتاپ» یا fittup باشد یا
   831→     *  خود کلیپ‌بورد دقیقاً کد ۴-رقمی باشد. */
   832→    private fun maybeDispatchClipboardOtp() {
   833→        try {
   834→            val cm = getSystemService(Context.CLIPBOARD_SERVICE) as? android.content.ClipboardManager ?: return
   835→            val text = cm.primaryClip?.getItemAt(0)?.coerceToText(this)?.toString()?.trim() ?: return
   836→            if (text == lastClipboardDispatched) return
   837→            val code = when {
   838→                Regex("^\\d{4}$").matches(text) -> text
   839→                text.contains("فیتاپ") || text.contains("fittup", ignoreCase = true) ->
   840→                    Regex("(?<!\\d)\\d{4}(?!\\d)").findAll(text).map { it.value }.lastOrNull()
   841→                else -> null
   842→            }
   843→            if (code != null) {
   844→                lastClipboardDispatched = text
   845→                dispatchOtpCode(code)
   846→            }
   847→        } catch (_: Exception) {}
   848→    }

   915→    /**
   916→     * در هر اجرا /api/app/own/latest را می‌خواند.
   917→     * مودال زیبای آپدیت را خود سایت (AppUpdateModal) نشان می‌دهد؛ این چک نیتیو
   918→     * فقط safety-net است: اگر صفحه سایت بالا نیامده باشد (خطای شبکه) و نسخهٔ
   919→     * جدید «اجباری» بود، دیالوگ نیتیو با دانلود مستقیم نشان می‌دهیم.
   920→     */
   921→    private fun checkAppVersion() {
   922→        thread {
   923→            try {
   924→                val url = URL(BuildConfig.SITE_URL.trimEnd('/') + "/api/app/own/latest")
   925→                val conn = url.openConnection() as HttpURLConnection
   926→                conn.connectTimeout = 10_000
   927→                conn.readTimeout = 10_000
   928→                conn.instanceFollowRedirects = true
   929→                val body = conn.inputStream.bufferedReader().use { it.readText() }
   930→                conn.disconnect()
   931→                val json = JSONObject(body)
   932→                if (!json.optBoolean("available", false)) return@thread
   933→                val latest = json.optInt("latestVersionCode", 1)
   934→                val forced = json.optBoolean("forceUpdate", false)
   935→                val apkUrl = BuildConfig.SITE_URL.trimEnd('/') + "/api/app/own/download"
   936→                if (BuildConfig.VERSION_CODE < latest && forced) {
   937→                    runOnUiThread {
   938→                        // فقط وقتی سایت بالا نیامده (وگرنه مودال سایت خودش را نشان می‌دهد)
   939→                        if (errorView.visibility == View.VISIBLE && !forceDialogShown) {
   940→                            forceDialogShown = true
   941→                            showForceUpdateDialog(apkUrl)
   942→                        }
   943→                    }
   944→                }
   945→            } catch (e: Exception) {
   946→                // آفلاین/خطا — نادیده بگیر؛ در اجرای بعدی دوباره تلاش می‌شود
   947→            }
   948→        }
   949→    }

  1076→        // اگر کاربر روی صفحهٔ غیر از سایت خودمان است (زرین‌پال/بانک/شاپرک)،
  1077→        // پل وب (__fitupNativeBack) آنجا وجود ندارد → 'unknown' → قبلاً مستقیم
