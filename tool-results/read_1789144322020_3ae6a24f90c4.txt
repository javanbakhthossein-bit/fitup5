    83→function fileToDataUrl(file: File): Promise<string> {
    84→  return new Promise((resolve, reject) => {
    85→    const reader = new FileReader();
    86→    reader.onload = () => resolve(reader.result as string);
    87→    reader.onerror = reject;
    88→    reader.readAsDataURL(file);
    89→  });
    90→}

   104→/** حجم فایل به‌صورت فارسی و خوانا (برای چیپ پیش‌نمایش ویدیو) */
   105→function formatFileSizeFa(bytes: number): string {
   106→  if (!bytes || bytes <= 0) return "۰";
   107→  const mb = bytes / (1024 * 1024);
   108→  if (mb >= 1) return `${toPersianDigits(mb >= 10 ? Math.round(mb) : Math.round(mb * 10) / 10)} مگابایت`;
   109→  return `${toPersianDigits(Math.max(1, Math.round(bytes / 1024)))} کیلوبایت`;
   110→}

   121→/**
   122→ * آپلود ویدیو با XMLHttpRequest — برای دریافت درصد پیشرفت واقعی آپلود
   123→ * (fetch هنوز upload progress قابل‌اعتماد ندارد). مسیر نسبی — بدون پورت.
   124→ */
   125→function uploadVideoWithProgress(
   126→  file: File,
   127→  onProgress: (pct: number) => void
   128→): Promise<{ mediaUrl: string; frames: string[] }> {
   129→  return new Promise((resolve, reject) => {
   130→    const xhr = new XMLHttpRequest();
   131→    xhr.open("POST", "/api/coach/chat/upload");
   132→    xhr.responseType = "json";
   133→    xhr.timeout = 10 * 60_000; // آپلود فایل‌های بزرگ — ۱۰ دقیقه
   134→    xhr.upload.onprogress = (e) => {
   135→      if (e.lengthComputable && e.total > 0) {
   136→        onProgress(Math.min(99, Math.round((e.loaded / e.total) * 100)));
   137→      }
   138→    };
   139→    xhr.onload = () => {
   140→      const data = xhr.response as
   141→        | { mediaUrl?: string; frames?: string[]; error?: string }
   142→        | null;
   143→      if (xhr.status >= 200 && xhr.status < 300 && data?.mediaUrl) {
   144→        onProgress(100);
   145→        resolve({
   146→          mediaUrl: data.mediaUrl,
   147→          frames: Array.isArray(data.frames) ? data.frames : [],
   148→        });
   149→      } else {
   150→        reject(
   151→          new Error(
   152→            (data && data.error) ||
   153→              "آپلود ویدیو ناموفق بود. دوباره تلاش کن."
   154→          )
   155→        );
   156→      }
   157→    };
   158→    xhr.onerror = () => reject(new Error(SERVER_UNREACHABLE_MESSAGE));
   159→    xhr.ontimeout = () =>
   160→      reject(new Error("آپلود ویدیو بیش از حد طول کشید. دوباره تلاش کن."));
   161→    const fd = new FormData();
   162→    fd.append("file", file);
   163→    xhr.send(fd);
   164→  });
   165→}
