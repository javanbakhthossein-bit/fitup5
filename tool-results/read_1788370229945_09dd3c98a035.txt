   153→  async function skipVideo() {

   156→      // پاسخ جدید: {ok, status, programStarted, message} — اگر این تصمیم

   194→  function handlePhotos(e: React.ChangeEvent<HTMLInputElement>) {
   195→    const files = Array.from(e.target.files || []).filter((f) => f.type.startsWith("image/"));
   196→    if (files.length === 0) return;
   197→    if (photos.length + files.length > 4) {
   198→      toast.error("حداکثر ۴ عکس مجاز است");
   199→      e.target.value = "";
   200→      return;
   201→    }
   202→    setPhotos((prev) => [...prev, ...files]);
   203→    e.target.value = "";
   204→  }

   206→  function handleVideo(e: React.ChangeEvent<HTMLInputElement>) {
   207→    const f = e.target.files?.[0];
   208→    if (!f) return;
   209→    if (!f.type.startsWith("video/")) {
   210→      toast.error("فقط فایل ویدیویی مجاز است");
   211→      e.target.value = "";
   212→      return;
   213→    }
   214→    // محدودیت حجم افزایش یافت — تا ۵۰ مگابایت
   215→    if (f.size > 50 * 1024 * 1024) {
   216→      toast.error("حجم ویدیو باید کمتر از ۵۰ مگابایت باشد");
   217→      e.target.value = "";
   218→      return;
   219→    }
   220→    setVideo(f);
   221→    e.target.value = "";
   222→  }

   228→  async function saveMeasurements() {
   229→    setSavingMeasurements(true);
   230→    try {
   231→      const payload: Record<string, number | undefined> = {};
   232→      const fields: { key: keyof MeasurementsForm; api: string }[] = [
   233→        { key: "waist", api: "waistMeasurement" },
   234→        { key: "neck", api: "neckMeasurement" },
   235→        { key: "hip", api: "hipMeasurement" },
   236→        { key: "chest", api: "chestMeasurement" },
   237→        { key: "arm", api: "armMeasurement" },
   238→        { key: "thigh", api: "thighMeasurement" },
   239→        { key: "shoulder", api: "shoulderMeasurement" },
   240→        { key: "calf", api: "calfMeasurement" },
   241→      ];
   242→      for (const f of fields) {
   243→        const v = measurements[f.key];
   244→        if (v && Number(v) > 0) payload[f.api] = Number(v);
   245→      }
   246→      if (Object.keys(payload).length > 0) {
   247→        await fetchJsonOrThrow<any>(
   248→          "/api/checkup/baseline-measurements",
   249→          {
   250→            method: "POST",
   251→            headers: { "Content-Type": "application/json" },
   252→            body: JSON.stringify(payload),
   253→          },
   254→          "خطا در ذخیره اندازه‌ها"
   255→        );
   256→        toast.success("اندازه‌های شما ذخیره شد ✅");
   257→      }
   258→      setMeasurementsStep("done");
   259→    } catch (e) {
   260→      toast.error(e instanceof Error ? e.message : "خطا در ذخیره اندازه‌ها");
   261→    } finally {
   262→      setSavingMeasurements(false);
   263→    }
   264→  }

   266→  async function submit() {
   267→    if (photos.length === 0) {
   268→      toast.error("حداقل یک عکس بدن ارسال کنید");
   269→      return;
   270→    }
   271→    // ویدیو اختیاری است — حتی برای پلن Ultimate — نیازی به بررسی الزامی بودن نیست
   272→    setSubmitting(true);
   273→    setSubmitError("");
   274→    setSubmitStage("uploading");
   275→    // تایمر مرحله‌بندی: تحلیل عکس بدن چند ثانیه سمت سرور طول می‌کشد؛
   276→    // مرحله «در حال ساخت برنامه» حذف شد چون تولید برنامه در پس‌زمینه انجام
   277→    // می‌شود و این درخواست دیگر برای تولید برنامه منتظر نمی‌ماند.
   278→    const stageTimers: ReturnType<typeof setTimeout>[] = [];
   279→    stageTimers.push(setTimeout(() => setSubmitStage("analyzing"), 1500));
   280→    try {
   281→      const fd = new FormData();
   282→      photos.forEach((p) => fd.append("bodyPhotos", p));
   283→      if (video) fd.append("bodyVideo", video);
   284→
   285→      // fetchJson: پاسخ HTML (صفحه خطای گیت‌وی/سرور) → خطای فارسی دوستانه
   286→      // به‌جای «Unexpected token '<'»
   287→      const { res, data } = await fetchJson<any>("/api/coach/submit-body-analysis", {
   288→        method: "POST",
   289→        body: fd,
   290→        // توجه: keepalive نگذارید — کروم بدنه‌های keepalive بزرگتر از 64KB را رد می‌کند
   291→        // و عکس/ویدیوهای واقعی هرگز به سرور نمی‌رسند.
   292→      });
   293→      if (!res.ok) throw new Error(data?.error || "خطا در ارسال مدیا");
   294→
   295→      // ─── قرارداد جدید (async): این endpoint دیگر برنامه را سینکرون نمی‌سازد ───
   296→      // پاسخ سریع برمی‌گردد: {programStarted, pendingStatus, awaitingDecision, message}
   297→      //  - programStarted=true / pendingStatus="generating" → تولید در پس‌زمینه شروع شد
   298→      //  - awaitingDecision=true / pendingStatus="pending_body_photo" → پیش‌نیاز دیگری مانده
   299→      const started =
   300→        Boolean(data?.programStarted) || data?.pendingStatus === "generating";
   301→      const blocked =
   302→        Boolean(data?.awaitingDecision) ||
   303→        data?.pendingStatus === "pending_body_photo";
   304→      const hasPlan = !started && !blocked && Boolean(data?.hasWorkoutPlan);
   305→
   306→      setSubmitStage("done");
   307→      // پاک کردن فرم
   308→      setPhotos([]);
   309→      setVideo(null);
   310→      setOpen(false);
   311→
   312→      // به‌روزرسانی state بر اساس پاسخ واقعی سرور (نه هاردکد true/ready)
   313→      setState({
   314→        needsBodyPhoto: state?.needsBodyPhoto ?? false,
   315→        canSubmitVideo: state?.canSubmitVideo ?? false,
   316→        pendingStatus: started
   317→          ? "generating"
   318→          : data?.pendingStatus ?? (hasPlan ? "ready" : null),
   319→        hasWorkoutPlan: hasPlan,
   320→        awaitingMedia: started ? false : Boolean(data?.awaitingMedia),
   321→      });
   322→
   323→      if (started) {
   324→        // تولید برنامه در پس‌زمینه شروع شد — خبر فوری و صادقانه به کاربر
   325→        toast.success(
   326→          "عکس‌های بدن شما ذخیره و تحلیل شد ✅ برنامه شما در حال آماده‌سازی است — پس از آماده‌سازی به شما اطلاع می‌دهیم."
   327→        );
   328→      } else if (blocked) {
   329→        // برنامه ساخته نشد — دلیل بلاک‌کننده سرور را نشان بده
   330→        // (مثلاً «تعیین تکلیف ویدیو یا آزمایش خون»)
   331→        toast.info(data?.message || "پیش‌نیازهای باقی‌مانده را تعیین تکلیف کنید.");
   332→      } else if (hasPlan) {
   333→        toast.success(data?.message || "برنامه شما ساخته شد! 🎯");
   334→        // بارگذاری برنامه جدید — فقط وقتی واقعاً برنامه ساخته شده
   335→        const { res: planRes, data: planData } = await fetchJson<any>(
   336→          "/api/coach/plan",
   337→          { cache: "no-store" }
   338→        );
   339→        if (planRes.ok) {
   340→          if (planData?.workout) setWorkoutPlan(planData.workout);
   341→          if (planData?.meal) setMealPlan(planData.meal);
   342→        }
   343→      } else {
   344→        toast.info(
   345→          data?.message || "عکس‌های بدن شما ذخیره و تحلیل شد. وضعیت برنامه خود را از بخش «برنامه‌ها» بررسی کنید."
   346→        );
   347→      }
   348→
   349→      // وضعیت پیش‌نیازها سمت سرور تغییر کرده — بنر پیش‌نیازها/داشبورد را refresh کن
   350→      // (مثلاً حالت «در حال آماده‌سازی» در PrerequisitesBanner ظاهر شود)
   351→      if (typeof window !== "undefined") window.dispatchEvent(new Event("prereq-updated"));
   352→    } catch (e) {
   353→      const msg = e instanceof Error ? e.message : "خطا در ارسال مدیا";
   354→      setSubmitError(msg);
   355→      setSubmitStage("error");
   356→      toast.error(msg);
   357→    } finally {
   358→      for (const t of stageTimers) clearTimeout(t);
   359→      setSubmitting(false);
   360→      // ریست مرحله بعد از ۳ ثانیه (مگر خطا)
   361→      setTimeout(() => {
   362→        setSubmitStage((s) => (s === "error" ? s : "idle"));
   363→      }, 3000);
   364→    }
   365→  }
   366→
   367→  const measurementFields: { key: keyof MeasurementsForm; label: string; required?: boolean }[] = [
   368→    { key: "waist", label: "دور کمر", required: true },
   369→    { key: "neck", label: "دور گردن", required: true },
   370→    { key: "hip", label: "دور باسن" },
   371→    { key: "chest", label: "دور سینه" },
   372→    { key: "arm", label: "دور بازو" },
   373→    { key: "thigh", label: "دور ران" },
   374→    { key: "shoulder", label: "دور شانه" },
   375→    { key: "calf", label: "دور ساق پا" },
   376→  ];
   377→
   378→  return (
   379→    <>
   380→      {/* بنر قدیمی حذف شد — فقط مودال آپلود عکس بدن */}
   381→      <AnimatePresence>
   382→        {open && (
   383→          <motion.div
   384→            initial={{ opacity: 0 }}
   385→            animate={{ opacity: 1 }}
   386→            exit={{ opacity: 0 }}
   387→            className="fixed inset-0 z-50 flex items-end sm:items-center justify-center bg-black/50 backdrop-blur-sm p-0 sm:p-4"
   388→            onClick={() => !submitting && !savingMeasurements && setOpen(false)}
   389→          >
   390→            <motion.div
   391→              ref={modalRef}
   392→              role="dialog"
   393→              aria-modal="true"
   394→              aria-label={measurementsStep === "done" ? "ارسال عکس بدن" : "اندازه‌های بدنی"}
   395→              tabIndex={-1}
   396→              initial={{ y: 60, opacity: 0 }}
   397→              animate={{ y: 0, opacity: 1 }}
   398→              exit={{ y: 60, opacity: 0 }}
   399→              transition={{ type: "spring", stiffness: 280, damping: 28 }}
   400→              onClick={(e) => e.stopPropagation()}
   401→              className="bg-white w-full sm:max-w-md sm:rounded-3xl rounded-t-3xl max-h-[90vh] overflow-y-auto custom-scrollbar focus:outline-none"
   402→              dir="rtl"
   403→            >
   404→              {/* Header */}
   405→              <div className="sticky top-0 bg-white border-b border-orange-100 p-4 flex items-center justify-between z-10">
   406→                <div className="flex items-center gap-2">
   407→                  {measurementsStep === "done" ? (
   408→                    <>
   409→                      <Camera className="w-5 h-5 text-orange-500" />
   410→                      <h3 className="font-bold text-slate-900">
   411→                        {state.canSubmitVideo ? "ارسال عکس بدن" : "ارسال عکس بدن"}
   412→                      </h3>
   413→                    </>
   414→                  ) : (
   415→                    <>
   416→                      <Ruler className="w-5 h-5 text-orange-500" />

   953→ * می‌شود و در مرحله «done» فقط خبر «در حال آماده‌سازی» را می‌دهیم.
   954→ */
   955→function SubmitStageDisplay({
   956→  stage,
   957→  photoCount,
   958→}: {
   959→  stage: "idle" | "uploading" | "analyzing" | "done" | "error";
   960→  photoCount: number;
   961→}) {
   962→  const stages: { key: typeof stage; label: string; desc: string }[] = [

   975→
   976→  return (
   977→    <div className="space-y-1.5">
   978→      {stages.map((s, i) => {
   979→        const isDone = stage === "done" || i < activeIdx;
