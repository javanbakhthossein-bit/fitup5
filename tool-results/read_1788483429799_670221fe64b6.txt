    54→ *
    55→ * پوسته اندرویدی (WebView) پنل کاربری فیتاپ — دقیقاً همان ساختار خود سایت:
    56→ *  - شروع با صفحه OTP (?screen=auth) → آنبوردینگ → پنل ورزشکار

    63→ *  سایت قبل از هر دیالوگ سیستمی، مودال انیمه‌دار خودش را نشان می‌دهد

   297→     * true  → بیرون از WebView هندل شد (مرورگر/دیالر)
   298→     * false → داخل WebView بارگذاری شود

   300→    private fun routeUrl(url: Uri): Boolean {
   301→        val host = url.host?.lowercase() ?: return false
   302→        val siteHost = try { URI(BuildConfig.SITE_URL).host?.lowercase() } catch (_: Exception) { null }
   303→
   304→        // سایت خودمان — همیشه داخل WebView
   305→        if (host == "fittup.ir" || host.endsWith(".fittup.ir")) return false
   306→        if (siteHost != null && (host == siteHost || host.endsWith(".$siteHost"))) return false
   307→
   308→        // درگاه پرداخت سایت (زرین‌پال) و بانک‌ها (شاپراک) — داخل WebView تا کاربر
   309→        // بعد از پرداخت به پنل برگردد (payment_verify در همان WebView اجرا می‌شود)
   310→        if (host == "zarinpal.com" || host.endsWith(".zarinpal.com")) return false
   311→        if (host == "zarin.link" || host.endsWith(".zarin.link")) return false
   312→        if (host == "shaparak.ir" || host.endsWith(".shaparak.ir")) return false
   313→
   314→        // شبکه‌های اجتماعی/استورها/سایت‌های دیگر → مرورگر بیرونی
   315→        openExternal(url)
   316→        return true
   317→    }

   319→    /** اسکیم‌های غیر وب — tel/mailto/intent و غیره */
   320→    private fun handleExternalScheme(uri: Uri): Boolean {
   321→        try {
   322→            when (uri.scheme?.lowercase()) {
   323→                "tel" -> startActivity(Intent(Intent.ACTION_DIAL, uri))
   324→                "mailto" -> startActivity(Intent(Intent.ACTION_SENDTO, uri))
   325→                "sms" -> startActivity(Intent(Intent.ACTION_SENDTO, uri))
   326→                "intent" -> {
   327→                    val intent = Intent.parseUri(uri.toString(), Intent.URI_INTENT_SCHEME)
   328→                    startActivity(intent)
   329→                }
   330→                else -> startActivity(Intent(Intent.ACTION_VIEW, uri))
   331→            }
   332→        } catch (_: Exception) {
   333→            // اپی برای این لینک نیست — نادیده بگیر
   334→        }
   335→        return true
   336→    }

   362→    /** helper سمت سایت — تشخیص تمیز محیط اپ اختصاصی */
   363→    private fun injectBridgeHelper() {
   364→        val js = """
   365→            (function() {
   366→              if (window.__fitupOwnAppInjected) return;
   367→              window.__fitupOwnAppInjected = true;
   368→              // آیا داخل اپ اختصاصی فیتاپ هستیم؟
   369→              window.isFitUpOwnApp = function() {
   370→                try { return !!(window.FitUpNative && window.FitUpNative.isOwnApp && window.FitUpNative.isOwnApp()); }
   371→                catch (e) { return false; }
   372→              };
   373→              // کوکی pwa_standalone — سرور برای URL «/» مستقیم صفحهٔ auth را
   374→              // رندر می‌کند (نه لندینگ) — همان رفتار وب‌اپ. اپ همیشه با
   375→              // ?screen=auth شروع می‌شود؛ این کوکی فقط حالت رفرش/بعد از خروج
   376→              // را هم درست نگه می‌دارد.
   377→              try {
   378→                if (document.cookie.indexOf('pwa_standalone=1') === -1) {
   379→                  document.cookie = 'pwa_standalone=1; path=/; max-age=31536000; samesite=lax';
   380→                }
   381→              } catch (e) {}
   382→            })();
   383→        """.trimIndent()
   384→        runOnUiThread {
   385→            webView.evaluateJavascript(js, null)
   386→        }
   387→    }

   391→    private fun setupFileChooser() {
   392→        fileChooserLauncher = registerForActivityResult(ActivityResultContracts.StartActivityForResult()) { result ->
   393→            val callback = filePathCallback ?: return@registerForActivityResult
   394→            filePathCallback = null
   395→            val uris: Array<Uri>? = if (result.resultCode == RESULT_OK) {
   396→                val data = result.data
   397→                val clip = data?.clipData
   398→                if (clip != null) {
   399→                    Array(clip.itemCount) { i -> clip.getItemAt(i).uri }
   400→                } else {
   401→                    data?.data?.let { arrayOf(it) }
   402→                }
   403→            } else null
   404→            callback.onReceiveValue(uris ?: arrayOf())
   405→        }
   406→    }

   422→    /** شروع دانلود با DownloadManager → پوشه Download اپ (بدون مجوز نوشتن) */
   423→    private fun startNativeDownload(url: String, fileName: String, isApk: Boolean) {
   424→        val safeName = fileName.replace(Regex("[^A-Za-z0-9._-]"), "_").ifBlank {
   425→            "fitup-${System.currentTimeMillis()}" + if (isApk) ".apk" else ""
   426→        }
   427→        val request = DownloadManager.Request(Uri.parse(url))
   428→            .setTitle(if (isApk) "نسخه جدید فیتاپ" else safeName)
   429→            .setDescription(if (isApk) "دانلود مستقیم از فیتاپ" else "دانلود فایل")

   447→    /** گیرندهٔ پایان دانلود → اگر APK بود، دیالوگ نصب */
   448→    private fun setupDownloadCompleteReceiver() {
   449→        val receiver = object : BroadcastReceiver() {
   450→            override fun onReceive(context: Context?, intent: Intent?) {
   451→                if (intent?.action != DownloadManager.ACTION_DOWNLOAD_COMPLETE) return
   452→                try {
   453→                    val id = intent.getLongExtra(DownloadManager.EXTRA_DOWNLOAD_ID, -1L)
   454→                    if (id != lastDownloadId) return
   455→                    val file = lastApkFile
   456→                    if (file == null || !file.name.endsWith(".apk", true)) return
   457→                    if (!file.exists() || file.length() < 10_000) {
   458→                        // فایل ناقص/خراب — دانلود منیجر خودش خطا نشان داده
   459→                        lastApkFile = null
   460→                        return
   461→                    }
   462→                    showInstallDialog(file)
   463→                } catch (_: Exception) {}
   464→            }
   465→        }
   466→        try {
   467→            if (Build.VERSION.SDK_INT >= 33) {
   468→                registerReceiver(receiver, IntentFilter(DownloadManager.ACTION_DOWNLOAD_COMPLETE), Context.RECEIVER_NOT_EXPORTED)
   469→            } else {
   470→                registerReceiver(receiver, IntentFilter(DownloadManager.ACTION_DOWNLOAD_COMPLETE))
   471→            }
   472→        } catch (_: Exception) {}
   473→    }

   523→    /** نام فایل از Content-Disposition یا URL */
   524→    private fun guessFileName(contentDisposition: String?, mimeType: String?, url: String): String {
   525→        try {
   526→            if (!contentDisposition.isNullOrBlank()) {
   527→                val m = Regex("filename\\*=UTF-8''([^;]+)|filename=\"?([^\";]+)\"?", RegexOption.IGNORE_CASE)
   528→                    .find(contentDisposition)
   529→                if (m != null) {
   530→                    val name = (m.groupValues[1].ifBlank { m.groupValues[2] })
   531→                        .replace("+", " ")
   532→                    if (name.isNotBlank()) return java.net.URLDecoder.decode(name, "UTF-8")
   533→                }
   534→            }
   535→            if (!mimeType.isNullOrBlank() && mimeType.equals("application/vnd.android.package-archive", true)) {
   536→                return "fitup-${System.currentTimeMillis()}.apk"
   537→            }
   538→            val path = URI(url).path ?: return "fitup-file"
   539→            return path.substringAfterLast('/').ifBlank { "fitup-file" }
   540→        } catch (_: Exception) {
   541→            return "fitup-file"
   542→        }
   543→    }

   545→    /* ───────────── ذخیره data URL (PNG/PDF) ───────────── */
   546→
   547→    @SuppressLint("InlinedApi")
   548→    private fun downloadDataUrl(filename: String, dataUrl: String) {
   549→        try {
   550→            val safeName = filename.ifBlank { "fitup-${System.currentTimeMillis()}" }
   551→                .replace(Regex("[^A-Za-z0-9._-\\u0600-\\u06FF ]"), "_")
   552→            val comma = dataUrl.indexOf(',')
   553→            if (comma < 0 || !dataUrl.startsWith("data:", ignoreCase = true)) {
   554→                runOnUiThread { toast("فایل قابل ذخیره نیست") }
   555→                return
   556→            }
   557→            val meta = dataUrl.substring(5, comma)
   558→            val mime = meta.substringBefore(";").ifBlank { "application/octet-stream" }
   559→            val isBase64 = meta.contains("base64", ignoreCase = true)
   560→            val bytes = if (isBase64) {
   561→                Base64.decode(dataUrl.substring(comma + 1), Base64.DEFAULT)
   562→            } else {
   563→                dataUrl.substring(comma + 1).toByteArray(Charsets.UTF_8)
   564→            }
   565→
   566→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   567→                val values = ContentValues().apply {
   568→                    put(MediaStore.Downloads.DISPLAY_NAME, safeName)
   569→                    put(MediaStore.Downloads.MIME_TYPE, mime)
   570→                    put(MediaStore.Downloads.IS_PENDING, 1)
   571→                }
   572→                val uri = contentResolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, values)
   573→                if (uri == null) {
   574→                    runOnUiThread { toast("خطا در ذخیره فایل") }
   575→                    return
   576→                }
   577→                contentResolver.openOutputStream(uri)?.use { it.write(bytes) }
   578→                values.clear()
   579→                values.put(MediaStore.Downloads.IS_PENDING, 0)
   580→                contentResolver.update(uri, values, null, null)
   581→            } else {
   582→                val dir = getExternalFilesDir(Environment.DIRECTORY_DOCUMENTS) ?: filesDir
   583→                val file = File(dir, safeName)
   584→                FileOutputStream(file).use { it.write(bytes) }
   585→            }
   586→            runOnUiThread { toast("«$safeName» ذخیره شد ✓") }
   587→        } catch (e: Exception) {
   588→            Log.e("FitUpApp", "download failed: ${e.message}")
   589→            runOnUiThread { toast("خطا در ذخیره فایل") }
   590→        }
   591→    }

   665→    private fun requestMediaRuntimePermissions(webRequest: PermissionRequest) {
   666→        pendingWebPermissionRequest?.let { if (it !== webRequest) it.deny() }
   667→        pendingWebPermissionRequest = webRequest
   668→        val perms = mutableListOf<String>()
   669→        if (webRequest.resources.contains(PermissionRequest.RESOURCE_VIDEO_CAPTURE)) {
   670→            perms.add(android.Manifest.permission.CAMERA)
   671→        }
   672→        if (webRequest.resources.contains(PermissionRequest.RESOURCE_AUDIO_CAPTURE)) {
   673→            perms.add(android.Manifest.permission.RECORD_AUDIO)
   674→        }
   675→        val alreadyGranted = perms.all {
   676→            ContextCompat.checkSelfPermission(this, it) == PackageManager.PERMISSION_GRANTED
   677→        }
   678→        if (perms.isEmpty() || alreadyGranted) {
   679→            pendingWebPermissionRequest = null
   680→            runOnUiThread { webRequest.grant(webRequest.resources) }
   681→        } else {
   682→            try {
   683→                mediaPermissionLauncher.launch(perms.toTypedArray())
   684→            } catch (_: Exception) {
   685→                pendingWebPermissionRequest = null
   686→                runOnUiThread { webRequest.deny() }
   687→            }
   688→        }
   689→    }

   715→
   716→    /**
   717→     * در هر اجرا /api/app/own/latest را می‌خواند.
   718→     * مودال زیبای آپدیت را خود سایت (AppUpdateModal) نشان می‌دهد؛ این چک نیتیو
   719→     * فقط safety-net است: اگر صفحه سایت بالا نیامده باشد (خطای شبکه) و نسخهٔ
   720→     * جدید «اجباری» بود، دیالوگ نیتیو با دانلود مستقیم نشان می‌دهیم.
   721→     */
   722→    private fun checkAppVersion() {
   723→        thread {
   724→            try {
   725→                val url = URL(BuildConfig.SITE_URL.trimEnd('/') + "/api/app/own/latest")
   726→                val conn = url.openConnection() as HttpURLConnection
   727→                conn.connectTimeout = 10_000
   728→                conn.readTimeout = 10_000
   729→                conn.instanceFollowRedirects = true
   730→                val body = conn.inputStream.bufferedReader().use { it.readText() }
   731→                conn.disconnect()
   732→                val json = JSONObject(body)
   733→                if (!json.optBoolean("available", false)) return@thread
   734→                val latest = json.optInt("latestVersionCode", 1)
   735→                val forced = json.optBoolean("forceUpdate", false)
   736→                val apkUrl = BuildConfig.SITE_URL.trimEnd('/') + "/api/app/own/download"
   737→                if (BuildConfig.VERSION_CODE < latest && forced) {
   738→                    runOnUiThread {
   739→                        // فقط وقتی سایت بالا نیامده (وگرنه مودال سایت خودش را نشان می‌دهد)
   740→                        if (errorView.visibility == View.VISIBLE && !forceDialogShown) {
   741→                            forceDialogShown = true
   742→                            showForceUpdateDialog(apkUrl)
   743→                        }
   744→                    }
   745→                }
   746→            } catch (e: Exception) {
   747→                // آفلاین/خطا — نادیده بگیر؛ در اجرای بعدی دوباره تلاش می‌شود
   748→            }
   749→        }
   750→    }
