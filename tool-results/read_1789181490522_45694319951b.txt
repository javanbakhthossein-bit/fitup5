    85→interface ParsedApplySwap {
    86→  type: "exercise" | "food";
    87→  from: string;
    88→  to: string;
    89→  day?: string;
    90→  /** متن پاسخ بدون خط directive */
    91→  remainder: string;

    93→
    94→function parseApplySwapDirective(content: string): ParsedApplySwap | null {
    95→  if (!content || content.length > 12000) return null;
    96→  const idx = content.lastIndexOf("[APPLY_SWAP ");
    97→  if (idx < 0) return null;
    98→  const tail = content.slice(idx);
    99→  const m = tail.match(
   100→    /^\[APPLY_SWAP type=(exercise|food) from="([^"]*)" to="([^"]*)"(?: day="([^"]*)")?\][ \t\r\n]*$/
   101→  );
   102→  if (!m) return null;
   103→  const from = (m[2] ?? "").trim();
   104→  const to = (m[3] ?? "").trim();
   105→  if (!from || !to) return null;
   106→  return {
   107→    type: m[1] as "exercise" | "food",
   108→    from,
   109→    to,
   110→    day: (m[4] ?? "").trim() || undefined,
   111→    remainder: content.slice(0, idx).trimEnd(),
   112→  };
   113→}

   151→  if (!bytes || bytes <= 0) return "۰";
   152→  const mb = bytes / (1024 * 1024);

   166→/**
   167→ * آپلود ویدیو با XMLHttpRequest — برای دریافت درصد پیشرفت واقعی آپلود
   168→ * (fetch هنوز upload progress قابل‌اعتماد ندارد). مسیر نسبی — بدون پورت.
   169→ */
   170→function uploadVideoWithProgress(
   171→  file: File,
   172→  onProgress: (pct: number) => void
   173→): Promise<{ mediaUrl: string; frames: string[] }> {
   174→  return new Promise((resolve, reject) => {
   175→    const xhr = new XMLHttpRequest();
   176→    xhr.open("POST", "/api/coach/chat/upload");
   177→    xhr.responseType = "json";
   178→    xhr.timeout = 10 * 60_000; // آپلود فایل‌های بزرگ — ۱۰ دقیقه
   179→    xhr.upload.onprogress = (e) => {
   180→      if (e.lengthComputable && e.total > 0) {
   181→        onProgress(Math.min(99, Math.round((e.loaded / e.total) * 100)));
   182→      }
   183→    };
   184→    xhr.onload = () => {
   185→      const data = xhr.response as
   186→        | { mediaUrl?: string; frames?: string[]; error?: string }
   187→        | null;
   188→      if (xhr.status >= 200 && xhr.status < 300 && data?.mediaUrl) {
   189→        onProgress(100);
   190→        resolve({
   191→          mediaUrl: data.mediaUrl,
   192→          frames: Array.isArray(data.frames) ? data.frames : [],
   193→        });
   194→      } else {
   195→        reject(
   196→          new Error(
   197→            (data && data.error) ||
   198→              "آپلود ویدیو ناموفق بود. دوباره تلاش کن."
   199→          )
   200→        );
   201→      }
   202→    };
   203→    xhr.onerror = () => reject(new Error(SERVER_UNREACHABLE_MESSAGE));
   204→    xhr.ontimeout = () =>
   205→      reject(new Error("آپلود ویدیو بیش از حد طول کشید. دوباره تلاش کن."));
   206→    const fd = new FormData();
   207→    fd.append("file", file);
   208→    xhr.send(fd);
   209→  });
   210→}

   574→  async function send(text?: string) {
   575→    const message = (text ?? input).trim();
   576→    if ((!message && !selectedImage && !selectedVideoFile) || sending) return;
   577→    if (!canChat) {
   578→      toast.info("برای چت با فیتاپ باید پلن خود را ارتقا دهید");
   579→      setOverlay("subscription");
   580→      return;
   581→    }
   582→    setInput("");
   583→    setSending(true);
   584→
   585→    // ─── مقادیر مدیا قبل از ارسال — برای بازگرداندن در خطا ───
   586→    const imageData = selectedImage;
   587→    const videoFile = selectedVideoFile;
   588→    const videoPreviewUrl = selectedVideo; // blob URL برای پیش‌نمایش پیام موقت
   589→    const videoPoster = selectedVideoPoster;
   590→
   591→    // ساخت پیام موقت با مدیا برای نمایش فوری
   592→    // (ویدیو با blob URL + poster فوراً فریم اول را نشان می‌دهد — Task 4-a)
   593→    const tempUserMsg: ChatMessageDto = {
   594→      id: tempId("temp"),
   595→      role: "user",
   596→      content:
   597→        message ||
   598→        (selectedImage ? "📷 عکس" : videoFile ? "🎬 ویدیو" : ""),
   599→      mediaUrl: selectedImage ?? videoPreviewUrl,
   600→      mediaType: selectedImage ? "image" : videoFile ? "video" : null,
   601→      createdAt: new Date().toISOString(),
   602→    };
   603→    addChatMessage(tempUserMsg);
   604→
   605→    setSelectedImage(null);
   606→    clearVideoDraft();
   607→
   608→    try {
   609→      let videoBase64: string | undefined;
   610→      let uploadedVideoUrl: string | undefined;
   611→      let uploadedFrames: string[] | undefined;
   612→
   613→      if (videoFile) {
   614→        if (videoFile.size > VIDEO_MULTIPART_THRESHOLD) {
   615→          // ─── Task 4-a: ویدیوهای بزرگ → مسیر multipart بدون محدودیت حجم ───
   616→          // ابتدا آپلود (ذخیره + فشرده‌سازی + استخراج فریم + مصرف سهمیه)،
   617→          // سپس POST چت با videoUrl/videoFrames.
   618→          setUploadPct(0);
   619→          try {
   620→            const up = await uploadVideoWithProgress(videoFile, (p) =>
   621→              setUploadPct(p)
   622→            );
   623→            uploadedVideoUrl = up.mediaUrl;
   624→            uploadedFrames = up.frames;
   625→          } finally {
   626→            setUploadPct(null);
   627→          }
   628→        } else {
   629→          // مسیر قدیمی base64 (<۱۲MB) — دست‌نخورده
   630→          videoBase64 = await fileToDataUrl(videoFile);
   631→        }
   632→      }
   633→
   634→      // fetchJsonOrThrow: پاسخ HTML (۵۰۲ گیت‌وی / تایم‌اوت درخواست‌های بلند ویدیو)
   635→      // → پیام فارسی «ارتباط با سرور برقرار نشد...» به‌جای «Unexpected token '<'»
   636→      // v73.4 — clientId: همان tempId پیام موقت → idempotency سمت سرور (فیکس دبل‌سند)
   637→      const data = await fetchJsonOrThrow<any>(
   638→        "/api/coach/chat",
   639→        {
   640→          method: "POST",
   641→          headers: { "Content-Type": "application/json" },
   642→          body: JSON.stringify({
   643→            message,
   644→            clientId: tempUserMsg.id,
   645→            imageBase64: imageData || undefined,
   646→            videoBase64,
   647→            videoUrl: uploadedVideoUrl,
   648→            videoFrames: uploadedFrames,
   649→            isVoice: lastWasVoiceRef.current, // آیا پیام از طریق ویس بود؟
   650→          }),
   651→        },
   652→        "خطا در ارتباط با سرور"
   653→      );
   654→
   655→      // ─── v57: حباب پیشنهاد «دستیار تغذیه» — درخواست مالک ───
   656→      // تحلیل غذا در چت انجام می‌شود ولی در تاریخچهٔ غذایی ثبت نمی‌شود؛ کاربر
   657→      // بعد از اولین تحلیل عکس غذا، یک‌بار در سشن، پیشنهاد اختصاصی می‌گیرد.
   658→      // v59 — گارد storage: getItem در حالت خصوصی می‌تواند throw کند
   659→      let foodTipShown = false;
   660→      try {
   661→        foodTipShown = !!window.sessionStorage.getItem(FOOD_TIP_SESSION_FLAG);
   662→      } catch {}
   663→      const shouldPushFoodTip =
   664→        !!imageData && !videoFile && typeof window !== "undefined" && !foodTipShown;
   665→      if (shouldPushFoodTip) {
   666→        try {
   667→          window.sessionStorage.setItem(FOOD_TIP_SESSION_FLAG, "1");
   668→        } catch {}
   669→      }
   670→
   671→      // ─── v73.4 — فیکس باگ کلوژر: به‌جای ساخت آرایه از chatMessages زمان render
   672→      // (که stale می‌ماند و پیام را دوبار نمایش می‌داد)، آپدیت functional روی
   673→      // آخرین state انجام می‌شود + حذف هر ردیف تکراری سروری (idempotent replay).
   674→      setChatMessages((prev) => {
   675→        const next: ChatMessageDto[] = [];
   676→        for (const m of prev) {
   677→          if (m.id === tempUserMsg.id) continue; // پیام موقت → نسخهٔ سروری می‌آید
   678→          if (data.userMessage && m.id === data.userMessage.id) continue;
   679→          if (data.aiMessage && m.id === data.aiMessage.id) continue;
   680→          next.push(m);
   681→        }
   682→        if (data.userMessage) next.push(data.userMessage as ChatMessageDto);
   683→        if (data.aiMessage) next.push(data.aiMessage as ChatMessageDto);
   684→        if (shouldPushFoodTip) {
   685→          next.push({
   686→            id: FOOD_TIP_MESSAGE_ID,
   687→            role: "assistant",
   688→            content: "", // محتوا در FoodAssistantTipBubble رندر می‌شود
   689→            createdAt: new Date().toISOString(),
   690→          });
   691→        }
   692→        return next;
   693→      });
   694→
   695→      // blob پیش‌نمایش دیگر لازم نیست (پیام نهایی URL سروری دارد)
   696→      if (videoPreviewUrl !== selectedVideoUrlRef.current) {
   697→        revokeObjectUrl(videoPreviewUrl);
   698→      }
   699→
   700→      // به‌روزرسانی سهمیه‌ها بعد از ارسال موفق عکس/ویدیو (Task 4-a)
   701→      if (imageData || videoFile) refreshQuota();
   702→
   703→      // ─── v56/v73.4 — راهنمایی کاربر به «دستیار تغذیه» بعد از ارسال عکس غذا ───
   704→      // (تحلیل خودِ عکس حالا در پس‌زمینه انجام می‌شود و نتیجه در حباب pending می‌آید)
   705→      if (imageData && !videoFile) {
   706→        toast.info("عکس دریافت شد — تحلیل در جریان است ⏳", {
   707→          description:
   708→            "از این به بعد تحلیل عکس غذا را از «دستیار تغذیه» انجام بده تا در تاریخچهٔ غذایی‌ات ثبت شود و برنامه‌ات دقیق‌تر پیگیری بشه 🍽",
   709→          action: {
   710→            label: "رفتن به دستیار تغذیه",
   711→            onClick: () => useAppStore.getState().setMainTab("nutrition"),
   712→          },
   713→          duration: 10000,
   714→        });
   715→      }
   716→
   717→      // ─── v73.4 — دوفازی مدیا: پاسخ فوری با pending=true → poll تا تکمیل تحلیل ───
   718→      if (data.pending && data.aiMessage?.id) {
   719→        startPendingPoll(data.aiMessage.id);
   720→      }
   721→    } catch (e) {
   722→      // v73.4 — functional update (فیکس باگ کلوژر)
   723→      setChatMessages((prev) => prev.filter((m) => m.id !== tempUserMsg.id));
   724→      // ─── بازیابی متن تایپ‌شده و پیوست‌ها ───
   725→      // قبلاً پیام کاربر (متن + عکس/ویدیو) با خطای شبکه «برای همیشه» از دست
   726→      // می‌رفت و کاربر مجبور بود از اول تایپ/انتخاب کند — حالا مقادیر pre-send
   727→      // به input و پیش‌نمایش مدیا برمی‌گردند (فقط اگر کاربر در این بین چیزی
   728→      // جدید تایپ/انتخاب نکرده باشد تا کار تازه‌اش دوباره‌کاری نشود).
   729→      if (message) setInput((prev) => prev || message);
   730→      if (imageData) setSelectedImage((prev) => prev ?? imageData);
   731→      if (videoFile) {
   732→        videoSelectionSeqRef.current += 1;
   733→        selectedVideoUrlRef.current = videoPreviewUrl; // blob هنوز زنده است — revoke نشده
   734→        setSelectedVideoFile(videoFile);
   735→        setSelectedVideo(videoPreviewUrl);
   736→        setSelectedVideoPoster(videoPoster);
   737→      }
   738→      const errMsg =
   739→        e instanceof Error ? e.message : "خطا در ارتباط با سرور";
   740→      // پیام سهمیه (۴۲۹) در toast هم نمایش داده می‌شود تا فوری دیده شود
   741→      if (e instanceof ApiError && e.status === 429) {
   742→        toast.error(errMsg);
   743→      }
   744→      addChatMessage({
   745→        id: tempId("err"),
   746→        role: "assistant",
   747→        content: `⚠️ ${errMsg}`,
   748→        createdAt: new Date().toISOString(),
   749→      });
   750→      // همگام‌سازی سهمیه بعد از خطا (مثلاً ۴۲۹ — شمارنده سرور ممکن است تغییر کرده باشد)
   751→      refreshQuota();
   752→    } finally {
   753→      setSending(false);
   754→      lastWasVoiceRef.current = false; // reset برای پیام بعدی
   755→      inputRef.current?.focus();
   756→    }
   757→  }

  1515→function ApplySwapButton({ swap }: { swap: ParsedApplySwap }) {
  1516→  const [state, setState] = useState<"idle" | "applying" | "done">("idle");
  1517→
  1518→  async function handleApply() {
  1519→    if (state !== "idle") return;

  1522→      const data = await fetchJsonOrThrow<{ applied?: number; message?: string }>(

  1533→        },
  1534→        "خطا در اعمال جایگزینی"
  1535→      );
