   139→// ─────────────────────────────────────────────────────────────────────────────
   140→// Macro ring — circular progress for protein/carbs/fat (pure SVG, no deps)
   141→// ─────────────────────────────────────────────────────────────────────────────
   142→function MacroRing({
   143→  label,
   144→  grams,
   145→  calories,
   146→  color,
   147→  icon,
   148→}: {
   149→  label: string;
   150→  grams: number;
   151→  calories: number;
   152→  color: string;
   153→  icon: React.ReactNode;
   154→}) {
   155→  const size = 84;
   156→  const stroke = 7;
   157→  const radius = (size - stroke) / 2;
   158→  const circumference = 2 * Math.PI * radius;
   159→  const fillFraction = 0.78; // visual fill — purely aesthetic
   160→
   161→  return (
   162→    <div className="flex flex-col items-center gap-1.5">
   163→      <div className="relative" style={{ width: size, height: size }}>
   164→        <svg width={size} height={size} className="-rotate-90">
   165→          <circle
   166→            cx={size / 2}
   167→            cy={size / 2}
   168→            r={radius}
   169→            fill="none"
   170→            stroke="#f1f5f9"
   171→            strokeWidth={stroke}
   172→          />
   173→          <motion.circle
   174→            cx={size / 2}
   175→            cy={size / 2}
   176→            r={radius}
   177→            fill="none"
   178→            stroke={color}
   179→            strokeWidth={stroke}
   180→            strokeLinecap="round"
   181→            strokeDasharray={circumference}
   182→            initial={{ strokeDashoffset: circumference }}
   183→            animate={{ strokeDashoffset: circumference * (1 - fillFraction) }}
   184→            transition={{ duration: 1, ease: "easeOut" }}
   185→          />
   186→        </svg>
   187→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   188→          <span className="text-slate-400">{icon}</span>
   189→          <span className="text-base font-black font-stat text-slate-900 leading-none">
   190→            {toPersianDigits(grams)}
   191→          </span>
   192→          <span className="text-[9px] text-slate-400">گرم</span>
   193→        </div>
   194→      </div>
   195→      <p className="text-xs font-bold text-slate-700">{label}</p>
   196→      <p className="text-[10px] text-slate-400">{toPersianDigits(calories)} کیلوکالری</p>
   197→    </div>
   198→  );
   199→}

   520→  async function fetchAnalysisOnce(url: string): Promise<{ ok: boolean; permanentFail: boolean; json: any }> {
   521→    const controller = new AbortController();

   527→      // خطاهای دائمی — retry بی‌فایده است (پروفایل نبود / سشن نیست)
   528→      if (res.status === 400 || res.status === 401) {
   529→        return { ok: false, permanentFail: true, json };
   530→      }
   531→      return { ok: false, permanentFail: false, json };
   532→    } catch {
   533→      // abort / network — قابل‌retry
   534→      return { ok: false, permanentFail: false, json: null };
   535→    } finally {
   536→      clearTimeout(timer);
   537→    }

   552→        // جذاب نشان داده می‌شود، نه کارت خطا
   553→        if (attempt < ANALYSIS_MAX_ATTEMPTS) {
   554→          await new Promise((r) => setTimeout(r, attempt <= 4 ? 2_500 : 6_000));
   555→        }
   556→      }

   560→      setRefreshing(false);
