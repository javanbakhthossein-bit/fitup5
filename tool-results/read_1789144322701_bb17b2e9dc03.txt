    24→export function smsFirstName(name: string | null | undefined): string {
    25→  const first = String(name || "")
    26→    .replace(/[\r\n\t]/g, " ")
    27→    .trim()
    28→    .split(/\s+/)[0] || "";
    29→  return first.slice(0, 25);
    30→}

   695→export interface AbandonedCartSmsResult {
   696→  sent: boolean;
   697→  /** کدام مسیر تحویل داد (v61: فقط قالب — bulk حذف شد) */
   698→  via?: "template";
   699→  /** already_sent = دداپ قبلی (هرگز دوباره نفرست) */
   700→  skipped?: "already_sent" | "invalid_mobile";
   701→  error?: string;

   704→/** رکورد failed در SmsLog (best-effort — هرگز نتیجهٔ ارسال را تغییر نمی‌دهد) */
   705→async function recordAbandonedCartFailure(
   706→  normalizedMobile: string,
   707→  dedupeKey: string,
   708→  templateId: string,
   709→  userId: string | null | undefined,
   710→  error: string
   711→): Promise<void> {
   712→  try {
   713→    const { db } = await import("@/lib/db");
   714→    await db.smsLog.upsert({
   715→      where: { mobile_key: { mobile: normalizedMobile, key: dedupeKey } },
   716→      create: {
   717→        mobile: normalizedMobile,
   718→        key: dedupeKey,
   719→        templateId: templateId || "template",
   720→        userId: userId ?? null,
   721→        status: "failed",
   722→        error,
   723→      },
   724→      update: {
   725→        status: "failed",
   726→        error,
   727→      },
   728→    });
   729→  } catch (e) {
   730→    console.warn("[sendAbandonedCartSms] failed-record upsert error:", e);
   731→  }
   732→}
   733→
   734→/**
   735→ * ارسال پیامک ترک درگاه — فقط با قالب sms.ir (نردبان خودترمیم پارامترها).
   736→ * v61: مسیر bulk/متن آزاد حذف شد (دیریکتیو مالک + خط خدماتی که bulk را
   737→ * پشتیبانی نمی‌کند). شکست قالب = رکورد failed و تلاشِ sweep بعدی.
   738→ */
   739→export async function sendAbandonedCartSms(
   740→  opts: AbandonedCartSmsOptions
   741→): Promise<AbandonedCartSmsResult> {
   742→  const { user, dedupeKey, link, templateId } = opts;
   743→
   744→  const normalized = normalizeMobileForSmsIr(user.mobile);
   745→  if (!/^9\d{9}$/.test(normalized)) {
   746→    return { sent: false, skipped: "invalid_mobile", error: `شماره نامعتبر: ${normalized}` };
   747→  }
   748→
   749→  const name = smsFirstName(user.name);
   750→
   751→  // ─── مسیر واحد — قالب sms.ir (v61: قالبی الزامی؛ bulk حذف شد) ───
   752→  const templateIdSafe = templateId?.trim() || "";
   753→  if (!templateIdSafe) {
   754→    const noTpl =
   755→      "شناسه قالب ترک خرید تنظیم نیست (SMSIR_ABANDONED_CART_TEMPLATE_ID یا SiteSetting: abandoned_cart_sms_template_id) — پیامک قالبی ارسال نشد.";
   756→    console.warn(`[sendAbandonedCartSms] (key=${dedupeKey}) ${noTpl}`);
   757→    await recordAbandonedCartFailure(normalized, dedupeKey, "", user.id, noTpl);
   758→    return { sent: false, error: noTpl };
   759→  }
   760→
   761→  // نردبان خودترمیم پارامترهای قالب: [NAME+LINK] → [NAME] → []
   762→  const attempts: Array<Array<{ name: string; value: string }>> = [
   763→    [
   764→      { name: "NAME", value: name },
   765→      { name: "LINK", value: link },
   766→    ],
   767→    [{ name: "NAME", value: name }],
   768→    [],
   769→  ];
   770→  let last: SendTemplateSmsResult = { sent: false, error: "no attempt" };
   771→  for (const params of attempts) {
   772→    last = await sendTemplateSms({
   773→      key: dedupeKey,
   774→      templateEnvKey: undefined,
   775→      templateIdOverride: templateIdSafe,
   776→      mobile: user.mobile,
   777→      params,
   778→      userId: user.id,
   779→    });
   780→    if (last.sent) return { sent: true, via: "template" };
   781→    if (last.skipped === "already_sent" || last.skipped === "invalid_mobile") {
   782→      return {
   783→        sent: false,
   784→        skipped: last.skipped === "invalid_mobile" ? "invalid_mobile" : "already_sent",
   785→        error: last.error,
   786→      };
   787→    }
   788→    // no_template / no_name / خطای درگاه → رانگ بعدی نردبان
   789→  }
   790→
   791→  // ─── شکست نهایی قالب — رکورد failed برای دیده‌شدن؛ sweep بعدی دوباره تلاش می‌کند ───
   792→  // (sendTemplateSms خودش در هر تلاش رکورد ثبت می‌کند؛ این upsert فقط گارانتی
   793→  //  می‌کند که حتماً یک رکورد failed با آخرین خطا موجود باشد.)
