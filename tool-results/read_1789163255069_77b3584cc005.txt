    15→ * اگر داخل request handler به‌صورت سینکرون اجرا شود، درخواست از سقف تایم‌اوت
    16→ * گیت‌وی عبور می‌کند و مرورگر پاسخ HTML خطا می‌گیرد («Unexpected token '<'») —
    17→ * در حالی که سرور کار را ادامه می‌دهد و برنامه در نهایت ساخته می‌شود؛ یعنی
    18→ * کاربر هم خطا می‌بیند هم بعداً برنامه دارد (باگ گزارش‌شده).
    19→ *

    24→ *  ۳) با موفقیت: برنامه‌ها ذخیره + وضعیت ready + نوتیفیکیشن «برنامه آماده شد»

    28→/** پارس لیست ذخیره‌شده به‌صورت JSON یا CSV */
    29→function safeParseList(raw: string | null | undefined): string[] {
    30→  if (!raw) return [];
    31→  const t = raw.trim();
    32→  if (!t) return [];
    33→  try {
    34→    const p = JSON.parse(t);
    35→    if (Array.isArray(p)) return p.map((x) => String(x));
    36→    if (typeof p === "string") return p.split(",").map((s) => s.trim()).filter(Boolean);
    37→    return [];
    38→  } catch {
    39→    return t.split(",").map((s) => s.trim()).filter(Boolean);
    40→  }
    41→}

   305→/**
   306→ * شروع تولید برنامه در پس‌زمینه برای کاربر.
   307→ *
   308→ * پیش‌شرط‌ها:
   309→ *  - اشتراک فعال یا pending (پلن مؤثر)
   310→ *  - پروفایل آنبوردینگ موجود
   311→ *  - همه پیش‌نیازها تعیین تکلیف شده باشند
   312→ *  - هیچ تولید در حال اجرای تازه‌ای وجود نداشته باشد (جلوگیری از دوباره‌کاری)
   313→ *
   314→ * خروجی فوری برمی‌گرداند؛ تولید در پس‌زمینه ادامه می‌یابد.
   315→ *
   316→ * v73.2 — opts.source === "checkup": به‌روزرسانی برنامه پس از چکاپ دوره‌ای
   317→ * (درخواست مالک: «با هر چکاپ اگر نیاز بود برنامه آپدیت شود و برنامهٔ جدید
   318→ * جای قبلی بنشیند — بدون هیچ تغییری در پلن و زمان اشتراک»). در این حالت
   319→ * گارد «already_has_fresh_plan» دور زده می‌شود (چون برنامهٔ فعال وجود دارد
   320→ * و دقیقاً باید جایگزین شود) ولی سقف روزانه و claim اتمیک برقرار می‌مانند.
   321→ */
   322→export async function startProgramGenerationInBackground(
   323→  userId: string,
   324→  opts?: { source?: "checkup" }
   325→): Promise<StartGenerationResult> {
   326→  const source = opts?.source ?? null;
   327→  // پلن مؤثر (active یا pending داخل پنجره)
   328→  const dto = await buildUserDto(userId);
   329→  const effectivePlan = (dto?.planName ?? null) as Plan | null;
   330→  if (!effectivePlan) {
   331→    return { started: false, reason: "no_plan" };
   332→  }
   333→
   334→  const profile = await db.onboardingProfile.findUnique({ where: { userId } });
   335→  if (!profile) {
   336→    return { started: false, reason: "no_profile" };
   337→  }
   338→
   339→  // پیش‌نیازها باید تعیین تکلیف شده باشند
   340→  const prereqCheck = await checkPrerequisites(userId, effectivePlan);
   341→  if (!prereqCheck.canGenerateProgram) {
   342→    return { started: false, reason: "prerequisites_incomplete", blockingReason: prereqCheck.blockingReason };
   343→  }
   344→
   345→  // ─── 🩹 v45 (باگ بحرانی مالک + نشتی اعتبار AI) — دو گارد سخت ───
   346→  // ① سقف روزانه: حداکثر ۵ شروع تولید در ۲۴ ساعت برای هر کاربر — هر منبعی
   347→  // (پرداخت، تحلیل ویدیو/خون، watchdog، سوئیپ، ادمین) از این سقف عبور
   348→  // نمی‌کند؛ حلقه‌های بی‌پایان دیگر نمی‌توانند اعتبار AI را بسوزانند.
   349→  // ② گارد برنامهٔ تازه: اگر آخرین درخواست ready است و برنامهٔ فعالِ
   350→  // ساخته‌شده بعد از آن وجود دارد، تولید دوباره بی‌معنی است (قبلاً reason
   351→  // «already_has_fresh_plan» در interface بود ولی هرگز برگردانده نمی‌شد و
   352→  // همین حلقهٔ «هر ۳۰ دقیقه نوتیف برنامهٔ شما آماده شد» را ساخته بود).

   606→/**
   607→ * پنجره «تولید گیرکرده» (M3) — باید از بدترین حالت تولید بزرگ‌تر باشد تا
   608→ * تولیدِ هنوز-در-حال-اجرا اشتباهی «یتیم» تشخیص داده نشود و تولید موازی دوباره
   609→ * شروع نشود (هزینه ۲× AI).
   610→ * v38: زنجیره حالا فال‌بک مدل دوم هم دارد + راند دومِ سمتِ ناموفق →
   611→ * بدترین حالت: زنجیرهٔ ۱۸دقیقه‌ای + راند دوم تا ۱۸ دقیقه ≈ ۳۶ دقیقه
   612→ * → پنجره ۵۰ دقیقه.
   613→ */
   614→const STUCK_GENERATION_WINDOW_MS = 50 * 60 * 1000;
   615→
   616→/**
   617→ * Watchdog خودترمیم — بازیابی درخواست‌های «generating» گیرکرده (یتیم‌شده).
   618→ *
   619→ * چرا: تولید برنامه fire-and-forget است. اگر پروسه سرور وسط تولید restart شود
   620→ * (dev-mode memory restart، crash، دیپلوی)، پروسه پس‌زمینه می‌میرد و
   621→ * ProgramRequest برای همیشه «generating» می‌ماند.
   622→ *
   623→ * رفتار:
   624→ *  - درخواست گیرکرده (>۲۰ دقیقه بدون به‌روزرسانی) پیدا می‌شود
   625→ *  - اگر برنامه‌ای بعد از شروع همین چرخه تولید ذخیره شده باشد → ready
   626→ *    (تولید کامل شده ولی آپدیت status به‌خاطر مرگ پروسه نرسیده)
   627→ *  - اگر برنامه‌ای نیست و autoRetryCount < 1:
   628→ *      → یک‌بار به‌صورت خودکار تولید دوباره شروع می‌شود (بدون دخالت کاربر)
   629→ *  - وگرنه: failed (کاربر از تب برنامه‌ها دکمه retry را می‌زند)
   630→ *
   631→ * این تابع از endpointهایی صدا زده می‌شود که فرانت‌اند موقع generating به آن‌ها
   632→ * poll می‌کند (coach/plan GET و program-history GET) — یعنی watchdog همان‌جا
   633→ * فعال می‌شود که کاربر منتظر برنامه است.
   634→ */

   637→    const stuckWindowAgo = new Date(Date.now() - STUCK_GENERATION_WINDOW_MS);
   638→    const stuck = await db.programRequest.findFirst({
   639→      where: { userId, status: "generating", updatedAt: { lt: stuckWindowAgo } },
   640→      orderBy: { createdAt: "desc" },
   641→    });
   642→    if (!stuck) return;
   643→
   644→    // آیا برنامه‌ای در این مدت ذخیره شده؟ (تولید ممکن است کامل شده باشد ولی
   645→    // آپدیت status به‌خاطر مرگ پروسه نرسیده باشد)
   646→    // ─── M2: فقط برنامه‌ای که «بعد از شروع همین چرخه تولید» ساخته شده دلیل
   647→    // آماده‌بودن است — updatedAt همان لحظه‌ای است که وضعیت generating شده؛
   648→    // برنامه دوره قبلی نباید تولید گیرکردهِ کاربر تمدیدی را ready کند.
   649→    const hasPlan = await db.workoutPlan.findFirst({
   650→      where: { userId, createdAt: { gte: stuck.updatedAt } },
   651→      orderBy: { createdAt: "desc" },
   652→      select: { id: true },
   653→    });
   654→    if (hasPlan) {
   655→      // برنامه موجود است — وضعیت را ready کن (نه failed!)
   656→      await db.programRequest.update({
   657→        where: { id: stuck.id },
   658→        data: { status: "ready" },
   659→      });
   660→      // v37: نوتیف همزمان (قبلاً فقط پیامک می‌رفت — درخواست مالک: پیامک ↔ نوتیف جفت باشند)
   661→      // v63 — dedupe عنوان‌محورِ ۲۴ساعته (هم‌راستا با مسیرهای دیگر) + متنِ یکسان با
   662→      // مسیر تولید عادی. قبلاً متنِ متفاوت بود و dedupeِ (عنوان+متن) آن را رد نمی‌کرد
   663→      // → کاربر هم نوتیف «ساخته شد» و هم «آماده است» می‌گرفت (گزارش نوتیف ۵تایی).
   664→      const alreadyNotified = await db.notification.findFirst({
   665→        where: {
   666→          userId,
   667→          type: "achievement",
   668→          title: "برنامه شما آماده شد! 🎯",
   669→          createdAt: { gte: new Date(Date.now() - 24 * 60 * 60 * 1000) },
   670→        },
   671→        select: { id: true },
   672→      });
   673→      if (!alreadyNotified) {
   674→        await createNotification(
   675→          userId,
   676→          "achievement",
   677→          "برنامه شما آماده شد! 🎯",
   678→          "برنامه تمرینی و غذایی شخصی‌سازی‌شده شما توسط فیتاپ هوشمند ساخته شد. از بخش «تمرینات» و «تغذیه» مشاهده کنید.",
   679→          "?tab=programs"
   680→        );
   681→      }
   682→      // v32: پیامک «برنامه آماده شد» (قالب 663678)
   683→      // v66: دداپ بر چرخهٔ بازیابی‌شده (stuck.id) — دورهٔ بعدی هم پیامک می‌گیرد
   684→      notifyProgramReadySms(userId, stuck.id);
   685→      console.log("[watchdog] recovered orphaned generating → ready (plan exists):", stuck.id);

   690→    if (stuck.autoRetryCount < 1) {
   691→      await db.programRequest.update({
   692→        where: { id: stuck.id },
   693→        data: { status: "failed", autoRetryCount: { increment: 1 } },
   694→      });
   695→      console.log("[watchdog] auto-retrying orphaned generation:", stuck.id);
   696→      const result = await startProgramGenerationInBackground(userId);
   697→      if (result.started) {
   698→        console.log("[watchdog] auto-retry started for:", userId);
   699→      } else {
   700→        console.log("[watchdog] auto-retry not started:", result.reason);
   701→      }
   702→      return;
   703→    }
   704→
   705→    // قبلاً یک‌بار خودترمیم شده و باز گیر کرده — failed بماند (کاربر دستی retry می‌کند)
   706→    await db.programRequest.update({
   707→      where: { id: stuck.id },
   708→      data: { status: "failed" },
   709→    });
   710→    console.log("[watchdog] marked stuck generation as failed (retry budget exhausted):", stuck.id);
   711→  } catch (e) {
   712→    // watchdog هرگز نباید مسیر caller را بشکند
   713→    console.error("[watchdog] recoverStuckGenerations error:", e);
   714→  }
   715→}
   716→
   717→/**
   718→ * v38 — سوئیپ خودکار درخواست‌های «ناموفق» (درخواست مالک: «به هیچ وجه نباید
   719→ * کاربر بدون برنامه بماند»).
   720→ *
   721→ * قبلاً درخواست failed برای همیشه failed می‌ماند مگر کاربر/ادمین دستی retry
   722→ * می‌زد. حالا کرون رفتاری (هر ۳۰ دقیقه) این سوئیپ را صدا می‌زند:
   723→ *   - درخواست‌های failed با attempts < FAILED_AUTO_RETRY_MAX (۳)
   724→ *   - که حداقل FAILED_RETRY_GAP_MS (۱۵ دقیقه) از آخرین تلاش گذشته باشد
   725→ *     (ضد حلقهٔ سریع روی خطای دائمی)
   726→ * دوباره تولید شروع می‌شود (startProgramGenerationInBackground خودش claim
   727→ * اتمیک انجام می‌دهد؛ درخواست failed قابل claim است).
   728→ *
   729→ * خروجی: تعداد درخواست‌هایی که دوباره صف شدند (برای summary کرون).
   730→ */
   731→const FAILED_AUTO_RETRY_MAX = 3;
   732→const FAILED_RETRY_GAP_MS = 15 * 60 * 1000;
   733→
   734→// ─── 🩹 v46 — بودجهٔ روزانهٔ سوئیپ در سطح کل سیستم (لایهٔ آخر ضد نشتی AI) ───
   735→// حتی با همهٔ گاردهای v45، اگر روزی دیتای خراب جدید یا مسیر دورزدنی پیدا شود،
   736→// این بودجه در-حافظه حداکثر سوئیپ‌های مجاز در شبانه‌روز را محدود می‌کند.
   737→// هر تولید برنامهٔ کامل ~۵۰هزار توکن گران است؛ سقف ۶ در ۲۴h یعنی بدترین-case
   738→// نشتی از مسیر سوئیپ ≈ ۳۰۰K توکن/روز — محدود و قابل‌قبول، نه ابدی.
   739→const SWEEP_DAILY_REQUEUE_MAX = 6;
   740→const sweepBudget = { date: "", count: 0 };
   741→
   742→function tryConsumeSweepBudget(): boolean {
   743→  const today = new Date().toISOString().slice(0, 10);
   744→  if (sweepBudget.date !== today) {
   745→    sweepBudget.date = today;
   746→    sweepBudget.count = 0;
   747→  }
   748→  if (sweepBudget.count >= SWEEP_DAILY_REQUEUE_MAX) {
   749→    console.warn(
   750→      `[failed-recovery] daily sweep budget exhausted (${SWEEP_DAILY_REQUEUE_MAX}/24h) — skipping requeue`
   751→    );
   752→    return false;
   753→  }
   754→  sweepBudget.count++;
   755→  return true;
   756→}
   757→
   758→export async function recoverFailedGenerations(limit = 3): Promise<number> {
   759→  try {
   760→    const gapAgo = new Date(Date.now() - FAILED_RETRY_GAP_MS);
   761→    const failedRequests = await db.programRequest.findMany({
   762→      where: {
   763→        status: "failed",
   764→        attempts: { lt: FAILED_AUTO_RETRY_MAX },
   765→        updatedAt: { lt: gapAgo },
   766→      },
   767→      orderBy: { updatedAt: "asc" },
   768→      take: limit,
   769→      select: { id: true, userId: true, plan: true, createdAt: true, updatedAt: true },
   770→    });
   771→
   772→    let requeued = 0;
   773→    for (const req of failedRequests) {
   774→      // کاربر نبلاک و دارای پروفایل باشد (خطای دائمی ورودی را دور بزن)
   775→      const user = await db.user.findUnique({
   776→        where: { id: req.userId },
   777→        select: { isBlocked: true },
   778→      });
   779→      if (!user || user.isBlocked) continue;
   780→
   781→      // ─── 🩹 v45 — ریشه باگ بحرانی مالک (نوتیف هر ۳۰ دقیقه + سوختن AI) ───
   782→      // قبلاً این سوئیپ رکورد قدیمی failed را انتخاب می‌کرد ولی
   783→      // startProgramGenerationInBackground روی «آخرین» درخواست کاربر عمل
   784→      // می‌کرد (که معمولاً ready بود) → هر ۳۰ دقیقه: claim برنامهٔ سالم →
   785→      // تولید کامل AI (۴ فراخوانی مدل گران) → نوتیف تکراری → رکورد قدیمی
   786→      // همچنان failed/attempts=0 می‌ماند → حلقهٔ ابدی.
   787→      // حالا:
   788→      //   ① فقط اگر همین رکورد «آخرین» درخواست کاربر باشد صف‌مجدد می‌شود؛
   789→      //   ② رکوردهای قدیمی‌تر terminal می‌شوند (attempts=MAX) تا دیگر
   790→      //      هرگز انتخاب نشوند — دیتای خرابِ موجود هم خودترمیم می‌شود؛
   791→      //   ③ اگر بعد از ساخت همین رکوردِ failed برنامهٔ فعال ذخیره شده
   792→      //      باشد، کاربر برنامه را دارد → تولید تکرار ممنوع، terminal.
   793→      const latest = await db.programRequest.findFirst({
   794→        where: { userId: req.userId },
   795→        orderBy: { createdAt: "desc" },
   796→        select: { id: true },
   797→      });
   798→      if (!latest || latest.id !== req.id) {
   799→        await db.programRequest
   800→          .update({
   801→            where: { id: req.id },
   802→            data: {
   803→              attempts: FAILED_AUTO_RETRY_MAX,
   804→              lastError: "v45 sweep: superseded by a newer request — terminal",
   805→            },
   806→          })
   807→          .catch(() => {});
   808→        console.log(
   809→          `[failed-recovery] ${req.id} is stale (latest=${latest?.id ?? "none"}) → terminalized, will not requeue`
   810→        );
   811→        continue;
   812→      }
   813→      const planAfterFailure = await db.workoutPlan.findFirst({
   814→        where: {
   815→          userId: req.userId,
   816→          active: true,
   817→          createdAt: { gte: req.createdAt },
   818→        },
   819→        select: { id: true },
   820→      });
   821→      if (planAfterFailure) {
   822→        await db.programRequest
   823→          .update({
   824→            where: { id: req.id },
   825→            data: {
   826→              attempts: FAILED_AUTO_RETRY_MAX,
   827→              status: "ready",
   828→              lastError: null,
   829→            },
   830→          })
   831→          .catch(() => {});
   832→        console.log(
   833→          `[failed-recovery] ${req.id} already has a plan created after it → marked ready (no duplicate AI generation)`
   834→        );
   835→        continue;
   836→      }
   837→
   838→      const result: StartGenerationResult = tryConsumeSweepBudget()
   839→        ? await startProgramGenerationInBackground(req.userId)
   840→        : { started: false, reason: "daily_budget" };
   841→      if (result.started) {
   842→        requeued++;
   843→        console.log(
   844→          `[failed-recovery] requeued failed generation for user ${req.userId} (request ${req.id})`
   845→        );
   846→      } else {
   847→        console.log(
   848→          `[failed-recovery] could not requeue ${req.id}: ${result.reason}${result.blockingReason ? ` — ${result.blockingReason}` : ""}`
   849→        );
   850→      }
   851→    }
   852→    return requeued;
   853→  } catch (e) {
   854→    console.error("[failed-recovery] recoverFailedGenerations error:", e);
   855→    return 0;
   856→  }
   857→}
