    79→    79→// Macro ring — circular progress for protein/carbs/fat (pure SVG, no deps)
    80→    80→// ─────────────────────────────────────────────────────────────────────────────
    81→    81→function MacroRing({
    82→    82→  label,
    83→    83→  grams,
    84→    84→  calories,
    85→    85→  color,
    86→    86→  icon,
    87→    87→}: {
    88→    88→  label: string;
    89→    89→  grams: number;
    90→    90→  calories: number;
    91→    91→  color: string;
    92→    92→  icon: React.ReactNode;
    93→    93→}) {
    94→    94→  const size = 84;
    95→    95→  const stroke = 7;
    96→    96→  const radius = (size - stroke) / 2;
    97→    97→  const circumference = 2 * Math.PI * radius;
    98→    98→  // Each macro is shown as a fraction of its calorie contribution vs the
    99→    99→  // other two. This gives a sense of proportion rather than a fixed max.
   100→   100→  // We just render a fully-stroked ring with the grams label inside.
   101→   101→  const fillFraction = 0.78; // visual fill — purely aesthetic
   102→   102→
   103→   103→  return (
   104→   104→    <div className="flex flex-col items-center gap-1.5">
   105→   105→      <div className="relative" style={{ width: size, height: size }}>
   106→   106→        <svg width={size} height={size} className="-rotate-90">
   107→   107→          <circle
   108→   108→            cx={size / 2}
   109→   109→            cy={size / 2}
   110→   110→            r={radius}
   111→   111→            fill="none"
   112→   112→            stroke="#f1f5f9"
   113→   113→            strokeWidth={stroke}
   114→   114→          />
   115→   115→          <motion.circle
   116→   116→            cx={size / 2}
   117→   117→            cy={size / 2}
   118→   118→            r={radius}
   119→   119→            fill="none"
   120→   120→            stroke={color}
   121→   121→            strokeWidth={stroke}
   122→   122→            strokeLinecap="round"
   123→   123→            strokeDasharray={circumference}
   124→   124→            initial={{ strokeDashoffset: circumference }}
   125→   125→            animate={{ strokeDashoffset: circumference * (1 - fillFraction) }}
   126→   126→            transition={{ duration: 1, ease: "easeOut" }}
   127→   127→          />
   128→   128→        </svg>
   129→   129→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   130→   130→          <span className="text-slate-400">{icon}</span>
   131→   131→          <span className="text-base font-black font-stat text-slate-900 leading-none">
   132→   132→            {toPersianDigits(grams)}
   133→   133→          </span>
   134→   134→          <span className="text-[9px] text-slate-400">گرم</span>
   135→   135→        </div>
   136→   136→      </div>
   137→   137→      <p className="text-xs font-bold text-slate-700">{label}</p>
   138→   138→      <p className="text-[10px] text-slate-400">{toPersianDigits(calories)} کیلوکالری</p>
   139→   139→    </div>
   140→   140→  );
   141→   141→}
