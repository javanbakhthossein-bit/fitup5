   279→    async function fetchAuthInBackground() {
   280→      // 🩹 v45: epoch در شروع — اگر وسط fetch کاربر logout کرد، پاسخ کهنه را نپذیر
   281→      const epochAtStart = useAppStore.getState().sessionEpoch;
   282→      try {
   283→        const res = await fetch("/api/auth/me", { cache: "no-store" });
   284→        if (!res.ok) return;
   285→        if (useAppStore.getState().sessionEpoch !== epochAtStart) return;
   286→        const data = await res.json();
   287→        if (data?.user) {
   288→          if (useAppStore.getState().sessionEpoch !== epochAtStart) return;
   289→          setUser(data.user);
   290→          // ─── FIX (مسابقهٔ verify/recover — باگ «پرداخت معلق یافت نشد») ───
   291→          // اگر همین حالا از درگاه برگشته‌ایم (payment_verify=1)،
   292→          // PaymentVerifyHandler مالک verify است؛ recover پس‌زمینه نباید همزمان
   293→          // پرداخت را claim کند و صفحهٔ رسید را با ۴۰۴ مواجه کند.
   294→          if (!pv && shouldRecoverNow()) recoverPendingPayments();
   295→        }
   296→        if (data?.termsUpdateRequired) setTermsUpdateRequired(true);
   297→      } catch {}
   298→    }

   300→    async function doAuthCheck(fallbackToAuth: boolean, attempt = 0) {
   301→      try {
   302→        const res = await fetch("/api/auth/me", { cache: "no-store" });
   303→        // 🩹 v49 (باگ «تغییر VPN → رفرش → لاگ‌اوت»): خطای HTTP (۵۰۰ و…) هم مثل
   304→        // خطای شبکه گذرا است — نه «لاگین‌نبودن». قبلاً res.ok چک نمی‌شد و بدنهٔ
   305→        // خطا به‌جای user می‌خواند و کاربر به‌زور به لندینگ/auth پرت می‌شد.
   306→        if (!res.ok) throw new Error(`auth/me HTTP ${res.status}`);
   307→        const data = await res.json();
   308→        if (data?.termsUpdateRequired) setTermsUpdateRequired(true);
   309→        // v51 (مسدودسازی): کاربر مسدود → صفحهٔ زیبای «مسدود» — نه لندینگ نه لاگ‌اوت
   310→        if (data?.blocked && !data?.user) {
   311→          setUser(null);
   312→          setScreen("blocked");
   313→          return;
   314→        }
   315→        if (data?.user) {
   316→          setUser(data.user);
   317→          // ─── بازیابی خودکار پرداخت‌های معلق (باگ «در انتظار») ───
   318→          // کاربری که در درگاه پرداخت کرده ولی به سایت برنگشته/سرور نیمه‌کاره مانده:
   319→          // همینجا استعلام می‌کنیم؛ اگر پرداخت موفق بوده، پلن تحویل و کاربر به‌روز می‌شود.
   320→          // FIX (مسابقهٔ verify/recover): در بازگشت از درگاه (pv) recover صدا زده
   321→          // نمی‌شود — PaymentVerifyHandler مالک است؛ وگرنه claim همزمان باعث
   322→          // «پرداخت معلق یافت نشد» روی صفحهٔ رسید می‌شد.
   323→          if (!pv) recoverPendingPayments();
   324→          if (data.user.role === "ADMIN") {
   325→            setScreen("admin");
   326→          } else {
   327→            setScreen(data.user.onboardingDone ? "main" : "onboarding");
   328→            // کاربر با لینک ?renewal=1 آمده و وارد پنل شد → صفحه تمدید
   329→            if (wantsRenewal && data.user.onboardingDone) {
   330→              setOverlay("renewal");
   331→            }
   332→          }
   333→        } else {
   334→          // کاربر لاگین نیست (پاسخ قطعی ۲۰۰ — رفرش هم مشکلی ندارد)
   335→          if (wantsRenewal) {
   336→            // با لینک تمدید آمده — اول وارد شو؛ بعد از لاگین صفحه تمدید باز می‌شود
   337→            try { window.sessionStorage.setItem("fitap_open_renewal", "1"); } catch {}
   338→            setScreen("auth");
   339→          } else if (isStandalone && !forceLanding) {
   340→            setScreen("auth");
   341→          } else if (fallbackToAuth) {
   342→            setScreen("auth");
   343→          } else {
   344→            showLanding();
   345→          }
   346→        }
   347→      } catch {
   348→        // 🩹 v49: خطای شبکه/سرور ≠ خروج از حساب. وسط سوئیچ VPN چند ثانیه
   349→        // قطعی/timeout طبیعی است — قبلاً همینجا کاربر لاگ‌اوت دیده می‌شد.
   350→        // تا ۳ بار با فاصلهٔ فزاینده تلاش می‌کنیم (۸۰۰ms/۱۶۰۰ms/۲۴۰۰ms).
   351→        if (attempt < 2) {
   352→          window.setTimeout(() => { void doAuthCheck(fallbackToAuth, attempt + 1); }, 800 * (attempt + 1));
   353→          return;
   354→        }
   355→        // بعد از اتمام تلاش‌ها: رفتار قبلی (صفحهٔ مقصد بر اساس محیط)
   356→        if (isStandalone && !forceLanding) {
   357→          setScreen("auth");
   358→        } else if (fallbackToAuth) {
   359→          setScreen("auth");
   360→        } else {
   361→          showLanding();
   362→        }
   363→      }
   364→    }
   365→  };
