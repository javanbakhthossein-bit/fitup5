    58→/**
    59→ * پارس امن aiAnalysis (FE-M3) — JSON.parse بدون try/catch در render می‌توانست
    60→ * با رشته خراب کل ویو (و اپ) را crash کند. حالا null برمی‌گرداند و رندر گارد می‌شود.
    61→ */
    62→function safeParseAiAnalysis(value: any): any | null {
    63→  if (value == null) return null;
    64→  if (typeof value !== "string") return value;
    65→  try {
    66→    const parsed = JSON.parse(value);
    67→    return parsed ?? null;
    68→  } catch {
    69→    return null;
    70→  }
    71→}

   468→            }}

   470→        ) : (

   478→      </Card>

   481→      <ProgressGallery photos={data?.photos || []} onRefresh={load} user={user} />
   482→    </div>
   483→  );
