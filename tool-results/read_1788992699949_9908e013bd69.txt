    32→/* ─────────────────────── پروکسی برای APIهای گوگل (FIX ۴۰۳ ایران) ───────────────────────
    33→ *
    34→ * مشکل واقعی: برخی سرورها (به‌ویژه سرورهای داخل ایران) توسط گوگل روی
    35→ * searchconsole.googleapis.com بلاک می‌شوند — پاسخ، صفحه‌ی HTML
    36→ * «Error 403 (Forbidden)!!1» است بدون کد خطای JSON. کلید سرویس‌اکانت در
    37→ * این حالت سالم است (تبادل توکن oauth2 موفق بوده) و مشکل فقط IP است.
    38→ *
    39→ * راه‌حل: عبور ترافیکِ فراخوانی‌های گوگل از پروکسی:
    40→ *   GSC_PROXY_URL=http://user:pass@host:port   (یا socks)
    41→ * در .env سرور — با ری‌استارت اپ، dispatcher تعویض می‌شود.
    42→ */
    43→let cachedDispatcher: object | null | undefined;
    44→
    45→async function googleDispatch(): Promise<object | undefined> {
    46→  if (cachedDispatcher !== undefined) return cachedDispatcher ?? undefined;
    47→  const proxyUrl =
    48→    process.env.GSC_PROXY_URL ||
    49→    process.env.HTTPS_PROXY ||
    50→    process.env.HTTP_PROXY;
    51→  if (!proxyUrl) {
    52→    cachedDispatcher = null;
    53→    return undefined;
    54→  }
    55→  try {
    56→    // undici کنار پروژه نصب است؛ ProxyAgent با fetch بومی Node سازگار است.
    57→    const { ProxyAgent } = await import("undici");
    58→    cachedDispatcher = new ProxyAgent(proxyUrl) as object;
    59→    const safeHost = proxyUrl.split("@").pop() ?? proxyUrl;
    60→    console.log(`[gsc] ترافیک API گوگل از پروکسی عبور می‌کند (${safeHost})`);
    61→    return cachedDispatcher;
    62→  } catch (e) {
    63→    console.warn("[gsc] ساخت ProxyAgent ناموفق — اتصال مستقیم:", e);
    64→    cachedDispatcher = null;
    65→    return undefined;
    66→  }
    67→}

    69→/** fetch با پروکسیِ اختیاری برای دامنه‌های گوگل (oauth2 / searchconsole) */
    70→async function googleFetch(
    71→  url: string,
    72→  init: RequestInit = {}
    73→): Promise<Response> {
    74→  const dispatcher = await googleDispatch();
    75→  if (dispatcher) {
    76→    // Node fetch (undici) گزینه‌ی dispatcher را می‌فهمد — تایپ استاندارد ندارد.
    77→    const initWithDispatcher = { ...init, dispatcher } as RequestInit;
    78→    return fetch(url, initWithDispatcher);
    79→  }
    80→  return fetch(url, init);
    81→}

   199→export async function saveGscConfig(saJson: string, siteUrl: string, apiKey?: string): Promise<{ ok: boolean; error?: string }> {
   200→  const trimmed = saJson.trim();
   201→  if (trimmed) {
   202→    try {
   203→      const parsed = JSON.parse(trimmed) as ServiceAccountJson;
   204→      if (!parsed.client_email || !parsed.private_key) {
   205→        return { ok: false, error: "JSON سرویس‌اکانت ناقص است — فیلدهای client_email و private_key لازم است." };
   206→      }
   207→    } catch {
   208→      return { ok: false, error: "JSON سرویس‌اکانت نامعتبر است — کل فایل دانلودی از گوگل کلاد را کامل paste کنید." };
   209→    }
   210→  }
   211→  const site = siteUrl.trim();
   212→  // فرمت‌های مجاز: sc-domain:fittup.ir | https://fittup.ir | http://... | fittup.ir (دامنه‌ی خالی — بعداً خودکار با پراپرتی گوگل تطبیق می‌شود)
   213→  if (site && !/^(sc-domain:|https?:\/\/)(\S+\.)\S+$/i.test(site) && !/^(\S+\.)\S{2,}$/i.test(site)) {
   214→    return { ok: false, error: "آدرس سایت نامعتبر است. نمونه‌های درست: sc-domain:fittup.ir یا https://fittup.ir یا فقط fittup.ir" };
   215→  }
   216→  await setSetting(SETTING_SA, trimmed, "Google Search Console — Service Account JSON");
   217→  await setSetting(SETTING_SITE, site, "Google Search Console — Site URL");
   218→  if (apiKey !== undefined) await setSetting(SETTING_APIKEY, apiKey.trim(), "Google API Key");
   219→  // تغییر پیکربندی → کش باطل
   220→  await setSetting(SETTING_CACHE, "", "GSC cache");

   222→  // برمی‌گشت و تستِ فوری بعد از save-config با اعتبار قبلی انجام می‌شد (نتیجه گمراه‌کننده).
   223→  tokenCache = null;

   231→/* ─────────────────────── توکن OAuth2 (JWT RS256) ─────────────────────── */
   232→
   233→let tokenCache: { token: string; exp: number } | null = null;
   234→
   235→async function getAccessToken(sa: ServiceAccountJson): Promise<string> {
   236→  const now = Math.floor(Date.now() / 1000);
   237→  // کش توکن — ۵۰ دقیقه (توکن گوگل ۱ ساعت اعتبار دارد؛ margin امن)
   238→  if (tokenCache && tokenCache.exp > now + 300) return tokenCache.token;
   239→
   240→  const header = { alg: "RS256", typ: "JWT" };
   241→  const claims = {
   242→    iss: sa.client_email,
   243→    scope: "https://www.googleapis.com/auth/webmasters.readonly",
   244→    aud: "https://oauth2.googleapis.com/token",
   245→    iat: now,
   246→    exp: now + 3600,
   247→  };
   248→  const b64url = (o: object | string) =>
   249→    Buffer.from(typeof o === "string" ? o : JSON.stringify(o)).toString("base64url");
   250→  const unsigned = `${b64url(header)}.${b64url(claims)}`;
   251→  let signature: Buffer;
   252→  try {
   253→    signature = createSign("RSA-SHA256").update(unsigned).sign(sa.private_key);
   254→  } catch {
   255→    tokenCache = null;
   256→    throw new Error(
   257→      "کلید خصوصی سرویس‌اکانت نامعتبر است — کل فایل JSON دانلودی از گوگل کلاد را بدون تغییر paste کنید."
   258→    );
   259→  }
   260→  const jwt = `${unsigned}.${signature.toString("base64url")}`;
   261→
   262→  const res = await googleFetch("https://oauth2.googleapis.com/token", {
   263→    method: "POST",
   264→    headers: { "Content-Type": "application/x-www-form-urlencoded" },
   265→    body: new URLSearchParams({
   266→      grant_type: "urn:ietf:params:oauth:grant-type:jwt-bearer",
   267→      assertion: jwt,
   268→    }),
   269→    signal: AbortSignal.timeout(20_000),
   270→  });
   271→  // بدنه را اول به‌صورت متن می‌خوانیم — اگر صفحه‌ی HTML باشد (بلاک IP) JSON.parse می‌شکند
   272→  const rawBody = await res.text().catch(() => "");
   273→  let data: { access_token?: string; expires_in?: number; error_description?: string } = {};
   274→  try {
   275→    data = JSON.parse(rawBody) as typeof data;
   276→  } catch {
   277→    /* بدنه JSON نبود */
   278→  }
   279→  if (!res.ok || !data.access_token) {
   280→    tokenCache = null;
   281→    if (isHtmlErrorPage(rawBody)) {
   282→      throw new Error(googleIpBlockedMessage());
   283→    }
   284→    throw new Error(
   285→      data.error_description === "Invalid JWT Signature"
   286→        ? "کلید خصوصی این سرویس‌اکانت دیگر در گوگل معتبر نیست — در گوگل کلاد (IAM → Service Accounts → fitup-551 → Keys → Add Key → JSON) یک کلید جدید بسازید و در تب پیکربندی جایگذاری کنید."
   287→        : `احراز هویت گوگل ناموفق: ${data.error_description || res.status} — JSON سرویس‌اکانت و دسترسی آن به پراپرتی را بررسی کنید.`
   288→    );
   289→  }
   290→  tokenCache = { token: data.access_token, exp: now + (data.expires_in ?? 3600) };
   291→  return data.access_token;
   292→}

   296→const fetchTimestamps: number[] = [];
   297→function tryAcquireFetchSlot(): boolean {
   298→  const hourAgo = Date.now() - 3600_000;
   299→  while (fetchTimestamps.length && fetchTimestamps[0] < hourAgo) fetchTimestamps.shift();
   300→  if (fetchTimestamps.length >= MAX_FETCH_PER_HOUR) return false;
   301→  fetchTimestamps.push(Date.now());
   302→  return true;
   303→}

   309→/** خواندن دقیق بدنه‌ی خطای گوگل — متن خام را هم برمی‌گرداند تا هیچ ۴۰۳ «خاموشی» نداشته باشیم */
   310→async function readGoogleError(res: Response): Promise<string> {
   311→  const raw = await res.text().catch(() => "");
   312→  if (raw) {
   313→    try {
   314→      const j = JSON.parse(raw) as { error?: { message?: string; errors?: { message?: string }[] } };
   315→      const msg = j?.error?.message || j?.error?.errors?.[0]?.message;
   316→      if (msg) return msg;
   317→    } catch {
   318→      // بدنه JSON نیست — خود متن را برسان (بریده‌شده)
   319→      return raw.slice(0, 300);
   320→    }
   321→  }
   322→  return `HTTP ${res.status}`;
   323→}

   346→async function querySearchAnalytics(
   347→  token: string,
   348→  siteUrl: string,
   349→  body: Record<string, unknown>
   350→): Promise<{ rows: GscQueryRow[] }> {
   351→  const url = `${GSC_API}/${encodeURIComponent(siteUrl)}/searchAnalytics/query`;
   352→  const res = await googleFetch(url, {
   353→    method: "POST",
   354→    headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json" },
   355→    body: JSON.stringify(body),
   356→    signal: AbortSignal.timeout(30_000),
   357→  });
   358→  if (!res.ok) {
   359→    const msg = await readGoogleError(res);
   360→    if (res.status === 403) {
   361→      if (isApiDisabledError(msg)) throw new Error(apiDisabledMessage(msg));
   362→      // صفحه‌ی HTML 403 = بلاکِ IP توسط گوگل (سرورهای ایرانی) — نه مشکل کلید
   363→      if (isHtmlErrorPage(msg)) throw new Error(googleIpBlockedMessage());
   364→      throw new Error(
   365→        `دسترسی به پراپرتی «${siteUrl}» رد شد (403). علت‌ها: ۱) ایمیل سرویس‌اکانت در Search Console (Settings → Users & permissions) با دسترسی Restricted اضافه نشده، ۲) فرمت آدرس با پراپرتی یکسان نیست. جزئیات گوگل: ${msg}`
   366→      );
   367→    }
   368→    throw new Error(`خطای Search Console API (${res.status}): ${msg}`);
   369→  }
   370→  const data = (await res.json().catch(() => ({}))) as { rows?: GscQueryRow[] };
   371→  return { rows: data.rows ?? [] };
   372→}

   405→/**
   406→ * تطبیق آدرس واردشده با پراپرتی‌های واقعی گوگل — کاربر لازم نیست فرمت دقیق بداند.
   407→ * ترتیب: تطابق دقیق → نرمال‌شده (بدون اسلش/پروتکل/ بزرگ‌ و کوچک حروف) → دامنه‌ی خالی → www.
   408→ */
   409→export function resolveSiteUrl(entered: string, sites: string[]): { match: string | null; corrected: boolean } {
   410→  const e = entered.trim();
   411→  if (!e) return { match: null, corrected: false };
   412→  const lower = e.toLowerCase();
   413→  // ۱) تطابق دقیق
   414→  const exact = sites.find((s) => s.toLowerCase() === lower);
   415→  if (exact) return { match: exact, corrected: exact !== e };
   416→
   417→  const norm = (u: string) => {
   418→    let x = u.trim().toLowerCase();
   419→    if (/^https?:\/\//.test(x) && x.endsWith("/")) x = x.slice(0, -1);
   420→    return x;
   421→  };
   422→  const nE = norm(e);
   423→
   424→  // ۲) پراپرتی دامنه‌ای — فقط با همان sc-domain: خودش معنی‌الیک است
   425→  if (nE.startsWith("sc-domain:")) {
   426→    const dm = sites.find((s) => norm(s) === nE);
   427→    return { match: dm ?? null, corrected: dm !== null && dm !== e };
   428→  }
   429→
   430→  // ۳) URL-prefix نرمال‌شده (https://fittup.ir == https://fittup.ir/)
   431→  const urlMatch = sites.find((s) => !s.toLowerCase().startsWith("sc-domain:") && norm(s) === nE);
   432→  if (urlMatch) return { match: urlMatch, corrected: urlMatch !== e };
   433→
   434→  // ۴) ورودی دامنه‌ی خالی (fittup.ir بدون پروتکل) — اول پراپرتی دامنه‌ای (پوشش کامل‌تر)، بعد https
   435→  const bare = nE.replace(/^https?:\/\//, "");
   436→  const dm = sites.find((s) => s.toLowerCase() === `sc-domain:${bare}`);
   437→  if (dm) return { match: dm, corrected: true };
   438→  const hm = sites.find((s) => ["https", "http"].some((p) => norm(s) === `${p}://${bare}`));
   439→  if (hm) return { match: hm, corrected: true };
   440→
   441→  // ۵) متغیر www — https://fittup.ir ↔ https://www.fittup.ir
   442→  const stripWww = (u: string) => u.replace(/^https?:\/\//, "").replace(/^www\./, "");
   443→  const wm = sites.find((s) => {
   444→    const ns = norm(s);
   445→    return !ns.startsWith("sc-domain:") && stripWww(ns) === stripWww(nE);
   446→  });
   447→  if (wm) return { match: wm, corrected: true };
   448→
   449→  return { match: null, corrected: false };
   450→}

   545→/** واکشی کامل داده‌های ۲۸ روز: خلاصه + روند روزانه + کوئری‌ها + صفحه‌ها */
   546→async function fetchFreshData(sa: ServiceAccountJson, siteUrl: string): Promise<GscOverview> {
   547→  const token = await getAccessToken(sa);
   548→  const endDate = new Date();
   549→  const startDate = new Date(endDate.getTime() - REPORT_DAYS * 86400_000);
   550→
   551→  const base = { startDate: isoDate(startDate), endDate: isoDate(endDate), dataState: "final" };
   552→
   553→  /**
   554→   * صفحه‌بندی پیشرونده‌ی کوئری/صفحه (درخواست مالک):
   555→   * «باید قابلیت خونده شدن تا ۲۰۰۰ ردیف رو داشته باشه نه اینکه هر سری ۲۰۰۰ ردیفو
   556→   * بخونه — یعنی سقفش تا ۲۰۰۰ ردیف باشه و تا هر تعداد که ردیف داشت خونده بشه»
   557→   *
   558→   * یعنی: صفحات ۵۰۰تایی پشت‌سرهم (startRow: 0، ۵۰۰، ۱۰۰۰، ۱۵۰۰) — فقط تا جایی
   559→   * که داده‌ی واقعی وجود دارد (صفحه‌ی ناقص = اتمام داده) و حداکثر تا سقف ۲۰۰۰.
   560→   * هیچ‌وقت بلوک ثابتی خوانده نمی‌شود؛ صفحات اول هر دو بُعد همزمان با تجمیع و
   561→   * روند روزانه واکشی می‌شوند و فقط اگر صفحه پر بود، صفحات بعدی خوانده می‌شوند.
   562→   */
   563→  const [aggRes, dailyRes, queriesFirstPage, pagesFirstPage] = await Promise.all([
   564→    querySearchAnalytics(token, siteUrl, { ...base, dimensions: [] }),
   565→    querySearchAnalytics(token, siteUrl, { ...base, dimensions: ["date"] }),
   566→    querySearchAnalytics(token, siteUrl, { ...base, dimensions: ["query"], rowLimit: GSC_PAGE_SIZE, startRow: 0 }),
   567→    querySearchAnalytics(token, siteUrl, { ...base, dimensions: ["page"], rowLimit: GSC_PAGE_SIZE, startRow: 0 }),
   568→  ]);
   569→
   570→  const agg = aggRes.rows[0];
   571→  const daily: GscDailyRow[] = dailyRes.rows
   572→    .map((r) => ({ date: r.keys[0], clicks: r.clicks, impressions: r.impressions, ctr: r.ctr, position: r.position }))
   573→    .sort((a, b) => (a.date < b.date ? -1 : 1));
   574→
   575→  // ادامه‌ی صفحه‌بندی فقط وقتی صفحه‌ی اول پر بود (یعنی احتمال داده‌ی بیشتر هست) —
   576→  // startRow دقیقاً از جایی که صفحه‌ی اول تمام شده ادامه می‌یابد (بدون هم‌پوشانی).
   577→  const [queriesRest, pagesRest] = await Promise.all([
   578→    queriesFirstPage.rows.length >= GSC_PAGE_SIZE
   579→      ? fetchRemainingRows(token, siteUrl, base, "query")
   580→      : Promise.resolve([] as GscQueryRow[]),
   581→    pagesFirstPage.rows.length >= GSC_PAGE_SIZE
   582→      ? fetchRemainingRows(token, siteUrl, base, "page")
   583→      : Promise.resolve([] as GscQueryRow[]),
   584→  ]);
   585→
   586→  const queries = [...queriesFirstPage.rows, ...queriesRest];
   587→  const pages = [...pagesFirstPage.rows, ...pagesRest];
   588→
   589→  return {
   590→    clicks: agg?.clicks ?? 0,
   591→    impressions: agg?.impressions ?? 0,
   592→    ctr: agg?.ctr ?? 0,
   593→    position: agg?.position ?? 0,
   594→    daily,
   595→    queries: queries.sort((a, b) => b.clicks - a.clicks),
   596→    pages: pages.sort((a, b) => b.clicks - a.clicks),
   597→  };
   598→}
   599→
   600→/**
   601→ * خواندن صفحات ۲، ۳ و ۴ (startRow: ۵۰۰/۱۰۰۰/۱۵۰۰) برای یک بُعد — صفحه‌ی اول
   602→ * (startRow: 0) قبلاً خوانده شده. توقف: اتمام داده (صفحه‌ی ناقص) یا رسیدن
   603→ * تعداد کل به سقف ۲۰۰۰ ردیف.
   604→ */
   605→async function fetchRemainingRows(
   606→  token: string,
   607→  siteUrl: string,
   608→  base: { startDate: string; endDate: string; dataState: string },
   609→  dimension: "query" | "page"
   610→): Promise<GscQueryRow[]> {
   611→  const all: GscQueryRow[] = [];
   612→  for (let startRow = GSC_PAGE_SIZE; startRow < MAX_GSC_ROWS; startRow += GSC_PAGE_SIZE) {
   613→    const res = await querySearchAnalytics(token, siteUrl, {
   614→      ...base,
   615→      dimensions: [dimension],
   616→      rowLimit: Math.min(GSC_PAGE_SIZE, MAX_GSC_ROWS - startRow),
   617→      startRow,
   618→    });
   619→    const pageRows = res.rows ?? [];
   620→    all.push(...pageRows);
   621→    // exhausted: صفحه‌ی ناقص یعنی داده‌ی بیشتری وجود ندارد
   622→    if (pageRows.length < GSC_PAGE_SIZE) break;
   623→  }
   624→  return all;
   625→}

   638→export async function getSearchConsoleData(force = false): Promise<GscResult> {

   659→
   660→  const statusBase: GscStatus = {
   661→    configured,
   662→    siteUrl: siteUrl || null,

   673→  }

   698→    await setSetting(SETTING_CACHE, JSON.stringify(payload), "GSC cache");
   699→    return {
   700→      ok: true,
   701→      data,
   702→      status: { ...statusBase, cache: { fetchedAt: payload.fetchedAt, stale: false, ttlHours: CACHE_TTL_HOURS } },
   703→    };
   704→  } catch (e) {
   705→    const msg = e instanceof Error ? e.message : "خطای ناشناخته در Search Console";
   706→    // خطا → اگر کشِ کهنه داریم همان را برگردان
   707→    return {
   708→      ok: false,
   709→      error: msg,
   710→      status: { ...statusBase, lastError: msg },
   711→      data: cached?.data,
   712→    };
   713→  }

   743→پرکلیک‌ترین کوئری‌های واقعی کاربران:

   747→${opportunities || "(یافت نشد)"}

   752→  } catch {
