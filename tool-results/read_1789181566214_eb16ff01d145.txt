   279→    /**
   280→     * v1.2.6 — لینک عمیق ورودی (App Link یا اسکیم fitup:// یا اعلان) یا null:
   281→     *  ۱) extra «fitup_link» از NotificationSync — مسیر/کوئری نسبی مثل
   282→     *     "?tab=progress&section=checkup" یا "/panel" → SITE_URL + آن
   283→     *  ۲) dataِ https روی دامنهٔ خودمان (fittup.ir) → همان URL — پارامترهای
   284→     *     ?tab= ?ref= ?renewal= ?screen= ?article= دست‌نخورده می‌مانند
   285→     *  ۳) v1.2.6 — اسکیم اختصاصی fitup://open?url=<encoded> از صفحهٔ /go سایت
   286→     *     (لینک‌های هوشمند پیامک — فقط URL دامنهٔ خودمان پذیرفته می‌شود)
   287→     * (null-امن — هرگز به‌خاطر لینک خراب کرش نمی‌کند)
   288→     */
   289→    private fun deepLinkUrl(from: Intent?): String? {
   290→        if (from == null) return null
   291→        try {
   292→            val extra = from.getStringExtra("fitup_link")
   293→            if (!extra.isNullOrBlank() && (extra.startsWith("/") || extra.startsWith("?"))) {
   294→                return BuildConfig.SITE_URL.trimEnd('/') + extra
   295→            }
   296→            val data = from.data ?: return null
   297→            val scheme = data.scheme?.lowercase()
   298→            // ─── v1.2.6: fitup://open?url=<encoded> — از لینک هوشمند /go سایت ───
   299→            if (scheme == "fitup") {
   300→                val target = data.getQueryParameter("url") ?: return null
   301→                if (target.startsWith("/")) {
   302→                    // مسیر نسبی روی دامنهٔ خودمان
   303→                    return BuildConfig.SITE_URL.trimEnd('/') + target
   304→                }
   305→                val t = android.net.Uri.parse(target)
   306→                val tScheme = t.scheme?.lowercase()
   307→                val tHost = t.host?.lowercase() ?: return null
   308→                val isOurHost = tHost == "fittup.ir" || tHost.endsWith(".fittup.ir")
   309→                if (isOurHost && (tScheme == "https")) return target
   310→                return null
   311→            }
   312→            if (scheme != "https") return null
   313→            val host = data.host?.lowercase() ?: return null
   314→            // فقط دامنهٔ خودمان — همان منطق routeUrl (fittup.ir و زیردامنه‌ها)
   315→            if (host == "fittup.ir" || host.endsWith(".fittup.ir")) return data.toString()
   316→        } catch (_: Exception) {
   317→        }
   318→        return null
   319→    }

   672→    /** شروع دانلود فایل معمولی با DownloadManager → پوشه Download اپ (بدون مجوز نوشتن).

   675→    private fun startNativeDownload(url: String, fileName: String): Boolean {
   676→        val safeName = fileName.replace(Regex("[^A-Za-z0-9._-]"), "_").ifBlank {
   677→            "fitup-${System.currentTimeMillis()}"

   681→                .setTitle(safeName)

   697→    /** جلوی دوبار اجرای پشت‌سرهم (برادکست + onResume هم‌زمان) */
   698→    private var handlingDownloadFinish = false
   699→
   700→    /** پایان یک دانلود: فقط پاک‌سازی وضعیت pending.
   701→     *  فایل‌های معمولی نوتیف خود DownloadManager را دارند؛ APK دیگر اصلاً از
   702→     *  این مسیر نمی‌گذرد (v1.2.8 → مرورگر بیرونی). */
   703→    private fun handleDownloadFinished(id: Long) {
   704→        if (id <= 0 || id != pendingDownloadId()) return
   705→        if (handlingDownloadFinish) return
   706→        handlingDownloadFinish = true
   707→        try {
   708→            clearPendingDownload()
   709→        } finally {
   710→            handlingDownloadFinish = false
   711→        }
   712→    }

   715→     *  بازسازی اکتیویتی/بستن اپ) — فقط پاک‌سازی وضعیت pending (v1.2.8). */
   716→    private fun checkPendingDownload() {
   717→        val id = pendingDownloadId()
   718→        if (id > 0) handleDownloadFinished(id)
   719→    }

   721→    /** گیرندهٔ پایان دانلود (فایل‌های معمولی) → پاک‌سازی وضعیت pending.
   722→     *  ⚠ RECEIVER_EXPORTED (نه NOT_EXPORTED): برادکست ACTION_DOWNLOAD_COMPLETE
   723→     *  «protected» است و فقط دانلودمنیجر/سیستم می‌تواند بفرستد؛ در اندروید ۱۴+
   724→     *  گیرندهٔ NOT_EXPORTED این برادکست را نمی‌گیرد (فرستنده DownloadProvider است). */
   725→    private fun setupDownloadCompleteReceiver() {
   726→        val receiver = object : BroadcastReceiver() {
   727→            override fun onReceive(context: Context?, intent: Intent?) {
   728→                if (intent?.action != DownloadManager.ACTION_DOWNLOAD_COMPLETE) return
   729→                val id = intent.getLongExtra(DownloadManager.EXTRA_DOWNLOAD_ID, -1L)
   730→                handleDownloadFinished(id)
   731→            }
   732→        }
   733→        try {
   734→            if (Build.VERSION.SDK_INT >= 33) {
   735→                registerReceiver(
   736→                    receiver,
   737→                    IntentFilter(DownloadManager.ACTION_DOWNLOAD_COMPLETE),
   738→                    Context.RECEIVER_EXPORTED
   739→                )
   740→            } else {
   741→                registerReceiver(receiver, IntentFilter(DownloadManager.ACTION_DOWNLOAD_COMPLETE))
   742→            }
   743→        } catch (_: Exception) {}
   744→    }

   746→    /** نام فایل از Content-Disposition یا URL */
   747→    private fun guessFileName(contentDisposition: String?, mimeType: String?, url: String): String {
   748→        try {
   749→            if (!contentDisposition.isNullOrBlank()) {
   750→                val m = Regex("filename\\*=UTF-8''([^;]+)|filename=\"?([^\";]+)\"?", RegexOption.IGNORE_CASE)
   751→                    .find(contentDisposition)
   752→                if (m != null) {
   753→                    val name = (m.groupValues[1].ifBlank { m.groupValues[2] })
   754→                        .replace("+", " ")
   755→                    if (name.isNotBlank()) return java.net.URLDecoder.decode(name, "UTF-8")
   756→                }
   757→            }
   758→            if (!mimeType.isNullOrBlank() && mimeType.equals("application/vnd.android.package-archive", true)) {
   759→                return "fitup-${System.currentTimeMillis()}.apk"
   760→            }
   761→            val path = URI(url).path ?: return "fitup-file"
   762→            return path.substringAfterLast('/').ifBlank { "fitup-file" }
   763→        } catch (_: Exception) {
   764→            return "fitup-file"
   765→        }
   766→    }

   768→    /* ───────────── ذخیره data URL (PNG/PDF) ───────────── */
   769→
   770→    @SuppressLint("InlinedApi")
   771→    private fun downloadDataUrl(filename: String, dataUrl: String) {
   772→        try {
   773→            val safeName = filename.ifBlank { "fitup-${System.currentTimeMillis()}" }
   774→                .replace(Regex("[^A-Za-z0-9._-\\u0600-\\u06FF ]"), "_")
   775→            val comma = dataUrl.indexOf(',')
   776→            if (comma < 0 || !dataUrl.startsWith("data:", ignoreCase = true)) {
   777→                runOnUiThread { toast("فایل قابل ذخیره نیست") }
   778→                return
   779→            }
   780→            val meta = dataUrl.substring(5, comma)
   781→            val mime = meta.substringBefore(";").ifBlank { "application/octet-stream" }
   782→            val isBase64 = meta.contains("base64", ignoreCase = true)
   783→            val bytes = if (isBase64) {
   784→                Base64.decode(dataUrl.substring(comma + 1), Base64.DEFAULT)
   785→            } else {
   786→                dataUrl.substring(comma + 1).toByteArray(Charsets.UTF_8)
   787→            }
   788→
   789→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   790→                val values = ContentValues().apply {
   791→                    put(MediaStore.Downloads.DISPLAY_NAME, safeName)
   792→                    put(MediaStore.Downloads.MIME_TYPE, mime)
   793→                    put(MediaStore.Downloads.IS_PENDING, 1)
   794→                }
   795→                val uri = contentResolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, values)
   796→                if (uri == null) {
   797→                    runOnUiThread { toast("خطا در ذخیره فایل") }
   798→                    return
   799→                }
   800→                contentResolver.openOutputStream(uri)?.use { it.write(bytes) }
   801→                values.clear()
   802→                values.put(MediaStore.Downloads.IS_PENDING, 0)
   803→                contentResolver.update(uri, values, null, null)
   804→            } else {
   805→                val dir = getExternalFilesDir(Environment.DIRECTORY_DOCUMENTS) ?: filesDir
   806→                val file = File(dir, safeName)
   807→                FileOutputStream(file).use { it.write(bytes) }
   808→            }
   809→            runOnUiThread { toast("«$safeName» ذخیره شد ✓") }
   810→        } catch (e: Exception) {
   811→            Log.e("FitUpApp", "download failed: ${e.message}")
   812→            runOnUiThread { toast("خطا در ذخیره فایل") }
   813→        }

   932→            val text = cm.primaryClip?.getItemAt(0)?.coerceToText(this)?.toString()?.trim() ?: return
   933→            if (text == lastClipboardDispatched) return

   936→                text.contains("فیتاپ") || text.contains("fittup", ignoreCase = true) ->
   937→                    Regex("(?<!\\d)\\d{4}(?!\\d)").findAll(text).map { it.value }.lastOrNull()
   938→                else -> null
   939→            }
   940→            if (code != null) {
   941→                lastClipboardDispatched = text
   942→                dispatchOtpCode(code)
   943→            }
   944→        } catch (_: Exception) {}

  1018→    private fun checkAppVersion() {
  1019→        thread {
  1020→            try {
  1021→                val url = URL(BuildConfig.SITE_URL.trimEnd('/') + "/api/app/own/latest")
  1022→                val conn = url.openConnection() as HttpURLConnection
  1023→                conn.connectTimeout = 10_000
  1024→                conn.readTimeout = 10_000
  1025→                conn.instanceFollowRedirects = true
  1026→                val body = conn.inputStream.bufferedReader().use { it.readText() }
  1027→                conn.disconnect()
  1028→                val json = JSONObject(body)
  1029→                if (!json.optBoolean("available", false)) return@thread
  1030→                val latest = json.optInt("latestVersionCode", 1)
  1031→                val forced = json.optBoolean("forceUpdate", false)
  1032→                val apkUrl = BuildConfig.SITE_URL.trimEnd('/') + "/api/app/own/download"
  1033→                if (BuildConfig.VERSION_CODE < latest && forced) {
  1034→                    runOnUiThread {
  1035→                        // فقط وقتی سایت بالا نیامده (وگرنه مودال سایت خودش را نشان می‌دهد)
  1036→                        if (errorView.visibility == View.VISIBLE && !forceDialogShown) {
  1037→                            forceDialogShown = true
  1038→                            showForceUpdateDialog(apkUrl)
  1039→                        }
  1040→                    }
  1041→                }
  1042→            } catch (e: Exception) {
  1043→                // آفلاین/خطا — نادیده بگیر؛ در اجرای بعدی دوباره تلاش می‌شود
  1044→            }
