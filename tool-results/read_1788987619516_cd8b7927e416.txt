   186→   186→     * v1.5.5 — لینک عمیق ورودی (App Link یا اسکیم fitup:// یا اعلان) یا null:
   187→   187→     *  ۱) extra «fitup_link» از NotificationSync — مسیر/کوئری نسبی مثل
   188→   188→     *     "?tab=progress&section=checkup" یا "/panel" → SITE_URL + آن
   189→   189→     *  ۲) dataِ https روی دامنهٔ خودمان (fittup.ir) → همان URL — پارامترهای
   190→   190→     *     ?tab= ?ref= ?renewal= ?screen= ?article= دست‌نخورده می‌مانند
   191→   191→     *  ۳) v1.5.5 — اسکیم اختصاصی fitup://open?url=<encoded> از صفحهٔ /go سایت
   192→   192→     *     (لینک‌های هوشمند پیامک — فقط URL دامنهٔ خودمان پذیرفته می‌شود)
   193→   193→     * (null-امن — هرگز به‌خاطر لینک خراب کرش نمی‌کند)
   194→   194→     */
   195→   195→    private fun deepLinkUrl(from: Intent?): String? {
   196→   196→        if (from == null) return null
   197→   197→        try {
   198→   198→            val extra = from.getStringExtra("fitup_link")
   199→   199→            if (!extra.isNullOrBlank() && (extra.startsWith("/") || extra.startsWith("?"))) {
   200→   200→                return BuildConfig.SITE_URL.trimEnd('/') + extra
   201→   201→            }
   202→   202→            val data = from.data ?: return null
   203→   203→            val scheme = data.scheme?.lowercase()
   204→   204→            // ─── v1.5.5: fitup://open?url=<encoded> — از لینک هوشمند /go سایت ───
   205→   205→            if (scheme == "fitup") {
   206→   206→                val target = data.getQueryParameter("url") ?: return null
   207→   207→                if (target.startsWith("/")) {
   208→   208→                    // مسیر نسبی روی دامنهٔ خودمان
   209→   209→                    return BuildConfig.SITE_URL.trimEnd('/') + target
   210→   210→                }
   211→   211→                val t = android.net.Uri.parse(target)
   212→   212→                val tScheme = t.scheme?.lowercase()
   213→   213→                val tHost = t.host?.lowercase() ?: return null
   214→   214→                if (isAllowedHost(tHost) && tScheme == "https") return target
   215→   215→                return null
   216→   216→            }
   217→   217→            if (scheme != "https") return null
   218→   218→            val host = data.host?.lowercase() ?: return null
   219→   219→            // فقط دامنه‌های مجاز — همان منطق isAllowedHost (فیتاپ و زیردامنه‌ها)
   220→   220→            if (isAllowedHost(host)) return data.toString()
   221→   221→        } catch (_: Exception) {
   222→   222→        }
   223→   223→        return null
   224→   224→    }

   625→   625→    /** شروع خرید از بازار — نتیجه به وب برمی‌گردد. fail-closed بدون کلید RSA.
   626→   626→     *  dynamicPriceToken: شناسه قیمت پویا (تخفیف/اعتبار ارتقا) — خالی = قیمت پایه SKU */
   627→   627→    private fun startBazaarPurchase(sku: String, payloadJson: String, callbackId: String, dynamicPriceToken: String) {
   628→   628→        // پاسخ به وب: window.__bazaarPaymentResult(id, resultObject)
   629→   629→        fun respond(json: JSONObject) {
   630→   630→            runOnUiThread {
   631→   631→                webView.evaluateJavascript(
   632→   632→                    "window.__bazaarPaymentResult && window.__bazaarPaymentResult('$callbackId', $json);",
   633→   633→                    null
   634→   634→                )
   635→   635→            }
   636→   636→        }
   637→   637→
   638→   638→        if (!rsaKeyValid) {
   639→   639→            // fail-closed (ممیزی 2-d باگ #1): بدون کلید RSA، خرید هرگز انجام نمی‌شود —
   640→   640→            // نه SecurityCheck.Disable با «خرید بدون وریفای محلی».
   641→   641→            respond(
   642→   642→                JSONObject()
   643→   643→                    .put("ok", false)
   644→   644→                    .put("error", "پرداخت درون‌برنامه‌ای پیکربندی نشده است (کلید RSA). لطفاً از نسخه وب سایت خرید کنید یا اپ را به‌روزرسانی کنید.")
   645→   645→            )
   646→   646→            return
   647→   647→        }
   648→   648→        if (!paymentReady) {
   649→   649→            respond(
   650→   650→                JSONObject()
   651→   651→                    .put("ok", false)
   652→   652→                    .put("error", "اتصال به پرداخت بازار برقرار نیست. برنامه بازار را باز کنید، وارد حساب شوید و دوباره تلاش کنید.")
   653→   653→            )
   654→   654→            return
   655→   655→        }
   656→   656→        if (activePaymentCallbackId != null) {
   657→   657→            respond(JSONObject().put("ok", false).put("error", "یک پرداخت دیگر در حال انجام است."))
   658→   658→            return
   659→   659→        }
   660→   660→        activePaymentCallbackId = callbackId
   661→   661→
   662→   662→        val finish: (JSONObject) -> Unit = { json ->
   663→   663→            activePaymentCallbackId = null
   664→   664→            respond(json)
   665→   665→        }
   666→   666→
   667→   667→        val request = PurchaseRequest(
   668→   668→            productId = sku,
   669→   669→            payload = payloadJson,
   670→   670→            dynamicPriceToken = dynamicPriceToken.ifBlank { null } // قیمت پویا (تخفیف) اگر سایت ثبت کرده باشد
   671→   671→        )
   672→   672→        payment?.purchaseProduct(
   673→   673→            registry = activityResultRegistry,
   674→   674→            request = request,
   675→   675→        ) {
   676→   676→            purchaseSucceed { purchaseInfo ->
   677→   677→                val json = JSONObject()
   678→   678→                    .put("ok", true)
   679→   679→                    .put("purchaseToken", purchaseInfo.purchaseToken ?: "")
   680→   680→                    .put("orderId", purchaseInfo.orderId ?: "")
   681→   681→                    .put("productId", purchaseInfo.productId ?: sku)
   682→   682→                    .put("payload", purchaseInfo.payload ?: payloadJson)
   683→   683→                    .put("purchaseTime", purchaseInfo.purchaseTime ?: 0L)
   684→   684→                    // v1.5.7 — امضای خرید (originalJson + signature) به سایت می‌رود تا
   685→   685→                    // سرور بتواند در نبود API بازار، امضا را با کلید RSA پنل
   686→   686→                    // راستی‌آزمایی کند (رفع ایراد بازرسی بازار درباره پیغام
   687→   687→                    // «اعتبارنامه‌های API بازار تنظیم نیست» بعد از خرید موفق)
   688→   688→                    .put("dataJson", purchaseInfo.originalJson ?: "")
   689→   689→                    .put("signature", purchaseInfo.dataSignature ?: "")
   690→   690→                // سایت خودش purchaseToken را برای فعال‌سازی به /api/payment/bazaar/purchase
   691→   691→                // می‌فرستد و بعد از موفقیت consumePurchase را صدا می‌زند (تمدید ممکن می‌شود)
   692→   692→                finish(json)
   693→   693→            }
   694→   694→            purchaseFailed { throwable ->
   695→   695→                finish(
   696→   696→                    JSONObject()
   697→   697→                        .put("ok", false)
   698→   698→                        .put("error", "پرداخت ناموفق: ${throwable?.message ?: "خطای نامشخص"}")
   699→   699→                )
   700→   700→            }
   701→   701→            purchaseCanceled {
   702→   702→                finish(
   703→   703→                    JSONObject()
   704→   704→                        .put("ok", false)
   705→   705→                        .put("canceled", true)
   706→   706→                        .put("error", "پرداخت توسط شما لغو شد.")
   707→   707→                )

   742→   742→    @SuppressLint("InlinedApi")
   743→   743→    private fun downloadDataUrl(filename: String, dataUrl: String) {
   744→   744→        try {
   745→   745→            val safeName = filename.ifBlank { "fitup-${System.currentTimeMillis()}" }
   746→   746→                .replace(Regex("[^A-Za-z0-9._-\\u0600-\\u06FF ]"), "_")
   747→   747→            val comma = dataUrl.indexOf(',')
   748→   748→            if (comma < 0 || !dataUrl.startsWith("data:", ignoreCase = true)) {
   749→   749→                runOnUiThread { toast("فایل قابل ذخیره نیست") }
   750→   750→                return
   751→   751→            }
   752→   752→            val meta = dataUrl.substring(5, comma)
   753→   753→            val mime = meta.substringBefore(";").ifBlank { "application/octet-stream" }
   754→   754→            val isBase64 = meta.contains("base64", ignoreCase = true)
   755→   755→            val bytes = if (isBase64) {
   756→   756→                Base64.decode(dataUrl.substring(comma + 1), Base64.DEFAULT)
   757→   757→            } else {
   758→   758→                dataUrl.substring(comma + 1).toByteArray(Charsets.UTF_8)
   759→   759→            }
   760→   760→
   761→   761→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   762→   762→                // اندروید ۱۰+ — MediaStore (بدون نیاز به مجوز)
   763→   763→                val values = ContentValues().apply {
   764→   764→                    put(MediaStore.Downloads.DISPLAY_NAME, safeName)
   765→   765→                    put(MediaStore.Downloads.MIME_TYPE, mime)
   766→   766→                    put(MediaStore.Downloads.IS_PENDING, 1)
   767→   767→                }
   768→   768→                val uri = contentResolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, values)
   769→   769→                if (uri == null) {
   770→   770→                    runOnUiThread { toast("خطا در ذخیره فایل") }
   771→   771→                    return
   772→   772→                }
   773→   773→                contentResolver.openOutputStream(uri)?.use { it.write(bytes) }
   774→   774→                values.clear()
   775→   775→                values.put(MediaStore.Downloads.IS_PENDING, 0)
   776→   776→                contentResolver.update(uri, values, null, null)
   777→   777→            } else {
   778→   778→                // اندروید ۹ و پایین‌تر — پوشه اپ (بدون مجوز نوشتن)
   779→   779→                val dir = getExternalFilesDir(Environment.DIRECTORY_DOCUMENTS) ?: filesDir
   780→   780→                val file = File(dir, safeName)
   781→   781→                FileOutputStream(file).use { it.write(bytes) }
   782→   782→            }
   783→   783→            runOnUiThread { toast("«$safeName» ذخیره شد ✓") }
   784→   784→        } catch (e: Exception) {
   785→   785→            Log.e("FitUp", "download failed: ${e.message}")
   786→   786→            runOnUiThread { toast("خطا در ذخیره فایل") }
   787→   787→        }
   788→   788→    }
