   323→    private fun handleExternalScheme(uri: Uri): Boolean {
   324→        try {
   325→            when (uri.scheme?.lowercase()) {
   326→                "tel" -> startActivity(Intent(Intent.ACTION_DIAL, uri))
   327→                "mailto" -> startActivity(Intent(Intent.ACTION_SENDTO, uri))
   328→                "sms" -> startActivity(Intent(Intent.ACTION_SENDTO, uri))
   329→                "intent" -> {
   330→                    val intent = Intent.parseUri(uri.toString(), Intent.URI_INTENT_SCHEME)
   331→                    startActivity(intent)
   332→                }
   333→                else -> startActivity(Intent(Intent.ACTION_VIEW, uri))
   334→            }
   335→        } catch (_: Exception) {
   336→            // اپی برای این لینک نیست — نادیده بگیر (صفحه خطای WebView نشان نده)
   337→        }
   338→        return true

   365→    /** helper سمت سایت — پرامیس تمیز برای خرید بازار + هندلر restore */
   366→    private fun injectBridgeHelper() {
   367→        val js = """
   368→            (function() {
   369→              if (window.__fitupBazaarInjected) return;
   370→              window.__fitupBazaarInjected = true;
   371→              window.__bazaarPending = {};
   372→              // نتیجه پرداخت از نیتیو — id + آبجکت نتیجه
   373→              window.__bazaarPaymentResult = function(id, result) {
   374→                var p = window.__bazaarPending[id];
   375→                if (p) { delete window.__bazaarPending[id]; p(result); }
   376→              };
   377→              // آیا داخل اپ بازار هستیم؟
   378→              window.isFitUpBazaarApp = function() {
   379→                try { return !!(window.FitUpNative && window.FitUpNative.isBazaarApp && window.FitUpNative.isBazaarApp()); }
   380→                catch (e) { return false; }
   381→              };
   382→              // خرید از بازار → پرامیس {ok, purchaseToken, orderId, productId, error}
   383→              // dynamicPriceToken (اختیاری): شناسه قیمت پویا برای پرداخت با مبلغ تخفیف‌دار
   384→              window.fitupBazaarPurchase = function(sku, payload, dynamicPriceToken) {
   385→                return new Promise(function(resolve) {
   386→                  var id = 'cb' + Date.now() + Math.floor(Math.random() * 1000);
   387→                  window.__bazaarPending[id] = resolve;
   388→                  try {
   389→                    var token = (dynamicPriceToken == null) ? '' : String(dynamicPriceToken);
   390→                    window.FitUpNative.purchaseSubscription(String(sku), JSON.stringify(payload || {}), id, token);
   391→                  } catch (e) {
   392→                    delete window.__bazaarPending[id];
   393→                    resolve({ ok: false, error: String(e) });
   394→                  }
   395→                });
   396→              };
   397→            })();
   398→        """.trimIndent()
   399→        runOnUiThread {
   400→            webView.evaluateJavascript(js, null)
   401→        }
   402→    }

   424→    /* ───────────── پرداخت درون‌برنامه‌ای بازار (پولکی) ───────────── */
   425→
   426→    private fun setupPayment() {
   427→        // کلید RSA از پیشخان توسعه‌دهندگان بازار (تب «پرداخت درون‌برنامه‌ای» برنامه)
   428→        val configuredKey = BuildConfig.BAZAAR_RSA_PUBLIC_KEY.trim()
   429→        rsaKeyValid = configuredKey.isNotEmpty() && configuredKey != "PASTE_YOUR_RSA_KEY_HERE"
   430→        if (!rsaKeyValid) {
   431→            Log.w("FitUp", "⚠️ BAZAAR_RSA_PUBLIC_KEY تنظیم نشده — خرید fail-closed می‌شود (امنیت). کلید را از پیشخان بازار بگیرید و در app/build.gradle.kts قرار دهید.")
   432→        }
   433→        val securityCheck = if (rsaKeyValid) {
   434→            SecurityCheck.Enable(rsaPublicKey = configuredKey)
   435→        } else {
   436→            // اتصال برای query/restore باز می‌ماند؛ خود خرید fail-closed است (startBazaarPurchase)
   437→            SecurityCheck.Disable
   438→        }
   439→
   440→        val config = PaymentConfiguration(localSecurityCheck = securityCheck)
   441→        payment = Payment(context = this, config = config)
   442→

   499→    /** شروع خرید از بازار — نتیجه به وب برمی‌گردد. fail-closed بدون کلید RSA.
   500→     *  dynamicPriceToken: شناسه قیمت پویا (تخفیف/اعتبار ارتقا) — خالی = قیمت پایه SKU */
   501→    private fun startBazaarPurchase(sku: String, payloadJson: String, callbackId: String, dynamicPriceToken: String) {
   502→        // پاسخ به وب: window.__bazaarPaymentResult(id, resultObject)
   503→        fun respond(json: JSONObject) {
   504→            runOnUiThread {
   505→                webView.evaluateJavascript(
   506→                    "window.__bazaarPaymentResult && window.__bazaarPaymentResult('$callbackId', $json);",
   507→                    null
   508→                )
   509→            }
   510→        }
   511→
   512→        if (!rsaKeyValid) {
   513→            // fail-closed (ممیزی 2-d باگ #1): بدون کلید RSA، خرید هرگز انجام نمی‌شود —
   514→            // نه SecurityCheck.Disable با «خرید بدون وریفای محلی».
   515→            respond(
   516→                JSONObject()
   517→                    .put("ok", false)
   518→                    .put("error", "پرداخت درون‌برنامه‌ای پیکربندی نشده است (کلید RSA). لطفاً از نسخه وب سایت خرید کنید یا اپ را به‌روزرسانی کنید.")
   519→            )
   520→            return
   521→        }
   522→        if (!paymentReady) {
   523→            respond(
   524→                JSONObject()
   525→                    .put("ok", false)
   526→                    .put("error", "اتصال به پرداخت بازار برقرار نیست. برنامه بازار را باز کنید، وارد حساب شوید و دوباره تلاش کنید.")
   527→            )
   528→            return
   529→        }
   530→        if (activePaymentCallbackId != null) {
   531→            respond(JSONObject().put("ok", false).put("error", "یک پرداخت دیگر در حال انجام است."))
   532→            return
   533→        }
   534→        activePaymentCallbackId = callbackId
   535→
   536→        val finish: (JSONObject) -> Unit = { json ->
   537→            activePaymentCallbackId = null
   538→            respond(json)
   539→        }
   540→
   541→        val request = PurchaseRequest(
   542→            productId = sku,
   543→            payload = payloadJson,
   544→            dynamicPriceToken = dynamicPriceToken.ifBlank { null } // قیمت پویا (تخفیف) اگر سایت ثبت کرده باشد
   545→        )
   546→        payment?.purchaseProduct(
   547→            registry = activityResultRegistry,
   548→            request = request,
   549→        ) {
   550→            purchaseSucceed { purchaseInfo ->
   551→                val json = JSONObject()
   552→                    .put("ok", true)
   553→                    .put("purchaseToken", purchaseInfo.purchaseToken ?: "")
   554→                    .put("orderId", purchaseInfo.orderId ?: "")
   555→                    .put("productId", purchaseInfo.productId ?: sku)
   556→                    .put("payload", purchaseInfo.payload ?: payloadJson)
   557→                    .put("purchaseTime", purchaseInfo.purchaseTime ?: 0L)
   558→                // سایت خودش purchaseToken را برای فعال‌سازی به /api/payment/bazaar/purchase
   559→                // می‌فرستد و بعد از موفقیت consumePurchase را صدا می‌زند (تمدید ممکن می‌شود)
   560→                finish(json)
   561→            }
   562→            purchaseFailed { throwable ->
   563→                finish(
   564→                    JSONObject()
   565→                        .put("ok", false)
   566→                        .put("error", "پرداخت ناموفق: ${throwable?.message ?: "خطای نامشخص"}")
   567→                )
   568→            }
   569→            purchaseCanceled {
   570→                finish(
   571→                    JSONObject()
   572→                        .put("ok", false)
   573→                        .put("canceled", true)
   574→                        .put("error", "پرداخت توسط شما لغو شد.")
   575→                )
   576→            }
   577→            purchaseFlowBegan {
   578→                // جریان خرید شروع شد — نیازی به عمل نیست
   579→            }
   580→            failedToBeginFlow { throwable ->
   581→                // شروع جریان شکست خورد (بازار قدیمی/نصب‌نشده) — گارد single-flight آزاد می‌شود
   582→                finish(
   583→                    JSONObject()
   584→                        .put("ok", false)
   585→                        .put("error", "امکان شروع پرداخت نبود: ${throwable?.message ?: "برنامه بازار به‌روز نیست یا نصب نیست"}")
   586→                )
   587→            }
   588→        }
   589→    }

   609→    /** ذخیره data URL (PNG/PDF) در Downloads اندروید */
   610→    @SuppressLint("InlinedApi")
   611→    private fun downloadDataUrl(filename: String, dataUrl: String) {
   612→        try {
   613→            val safeName = filename.ifBlank { "fitup-${System.currentTimeMillis()}" }
   614→                .replace(Regex("[^A-Za-z0-9._-\\u0600-\\u06FF ]"), "_")
   615→            val comma = dataUrl.indexOf(',')
   616→            if (comma < 0 || !dataUrl.startsWith("data:", ignoreCase = true)) {
   617→                runOnUiThread { toast("فایل قابل ذخیره نیست") }
   618→                return
   619→            }
   620→            val meta = dataUrl.substring(5, comma)
   621→            val mime = meta.substringBefore(";").ifBlank { "application/octet-stream" }
   622→            val isBase64 = meta.contains("base64", ignoreCase = true)
   623→            val bytes = if (isBase64) {
   624→                Base64.decode(dataUrl.substring(comma + 1), Base64.DEFAULT)
   625→            } else {
   626→                dataUrl.substring(comma + 1).toByteArray(Charsets.UTF_8)
   627→            }
   628→
   629→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   630→                // اندروید ۱۰+ — MediaStore (بدون نیاز به مجوز)
   631→                val values = ContentValues().apply {
   632→                    put(MediaStore.Downloads.DISPLAY_NAME, safeName)
   633→                    put(MediaStore.Downloads.MIME_TYPE, mime)
   634→                    put(MediaStore.Downloads.IS_PENDING, 1)
   635→                }
   636→                val uri = contentResolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, values)
   637→                if (uri == null) {
   638→                    runOnUiThread { toast("خطا در ذخیره فایل") }
   639→                    return
   640→                }
   641→                contentResolver.openOutputStream(uri)?.use { it.write(bytes) }
   642→                values.clear()
   643→                values.put(MediaStore.Downloads.IS_PENDING, 0)
   644→                contentResolver.update(uri, values, null, null)
   645→            } else {
   646→                // اندروید ۹ و پایین‌تر — پوشه اپ (بدون مجوز نوشتن)
   647→                val dir = getExternalFilesDir(Environment.DIRECTORY_DOCUMENTS) ?: filesDir
   648→                val file = File(dir, safeName)
   649→                FileOutputStream(file).use { it.write(bytes) }
   650→            }
   651→            runOnUiThread { toast("«$safeName» ذخیره شد ✓") }
   652→        } catch (e: Exception) {
   653→            Log.e("FitUp", "download failed: ${e.message}")
   654→            runOnUiThread { toast("خطا در ذخیره فایل") }
   655→        }
   656→    }

   708→     * مجوز runtime دوربین/میکروفون — دقیقاً وقتی سایت ضبط صدا/ویدیو می‌خواهد
   709→     * (WebChromeClient.onPermissionRequest، بعد از دیالوگ اجازه فارسی).
   710→     * اگر همه مجوزها از قبل داده شده باشند مستقیم grant می‌شود؛ وگرنه
   711→     * دیالوگ سیستمی اندروید.
   712→     */
   713→    private fun requestMediaRuntimePermissions(webRequest: PermissionRequest) {
   714→        // درخواست معلق قبلی را رد کن (درخواست همزمان/تکراری از صفحه)
   715→        pendingWebPermissionRequest?.let { if (it !== webRequest) it.deny() }
   716→        pendingWebPermissionRequest = webRequest
   717→        val perms = mutableListOf<String>()
   718→        if (webRequest.resources.contains(PermissionRequest.RESOURCE_VIDEO_CAPTURE)) {
   719→            perms.add(android.Manifest.permission.CAMERA)
   720→        }
   721→        if (webRequest.resources.contains(PermissionRequest.RESOURCE_AUDIO_CAPTURE)) {
   722→            perms.add(android.Manifest.permission.RECORD_AUDIO)
   723→        }
   724→        val alreadyGranted = perms.all {
   725→            ContextCompat.checkSelfPermission(this, it) == PackageManager.PERMISSION_GRANTED
   726→        }
   727→        if (perms.isEmpty() || alreadyGranted) {
   728→            pendingWebPermissionRequest = null
   729→            runOnUiThread { webRequest.grant(webRequest.resources) }
   730→        } else {
   731→            try {
   732→                mediaPermissionLauncher.launch(perms.toTypedArray())
   733→            } catch (_: Exception) {
   734→                // دیالوگ سیستم باز نشد (مثلاً گوشی خاص) — fail بسته
   735→                pendingWebPermissionRequest = null
   736→                runOnUiThread { webRequest.deny() }
   737→            }
   738→        }
   739→    }

   822→    /** ثبت receiver پیامک — در onResume (فقط با مجوز)؛ در onPause unreg */
   823→    private fun registerSmsReceiver() {
   824→        if (smsReceiver != null) return
   825→        val receiver = object : BroadcastReceiver() {
   826→            override fun onReceive(context: Context?, intent: Intent?) {
   827→                if (intent?.action != "android.provider.Telephony.SMS_RECEIVED") return
   828→                try {
   829→                    val msgs = android.provider.Telephony.Sms.Intents.getMessagesFromIntent(intent) ?: return
   830→                    for (m in msgs) {
   831→                        val body = m?.displayMessageBody?.toString() ?: continue
   832→                        // فقط پیامک‌های حاوی «فیتاپ» (قالب SMS.ir) — کد ۴ رقمی را بگیر
   833→                        if (!body.contains("فیتاپ") && !body.contains("FitUp", true)) continue
   834→                        val match = Regex("(^|\\D)(\\d{4})(\\D|$)").find(body) ?: continue
   835→                        dispatchOtpCode(match.groupValues[2])
   836→                        break
   837→                    }
   838→                } catch (_: Exception) {}
   839→            }
   840→        }
   841→        try {
   842→            registerReceiver(receiver, android.content.IntentFilter("android.provider.Telephony.SMS_RECEIVED"))
   843→            smsReceiver = receiver
   844→        } catch (_: Exception) {}
   845→    }
   846→
   847→    private fun unregisterSmsReceiver() {
   848→        smsReceiver?.let {
   849→            try { unregisterReceiver(it) } catch (_: Exception) {}
   850→        }
   851→        smsReceiver = null
   852→    }

   873→     * در هر اجرا /api/app/version را می‌خواند:
   874→     *  - versionCode < min → دیالوگ آپدیت اجباری (غیرقابل رد)
   875→     *  - versionCode < latest → toast ملایم (یک‌بار در هر session)
   876→     */
   877→    private fun checkAppVersion() {
   878→        thread {
   879→            try {
   880→                val url = URL(BuildConfig.SITE_URL.trimEnd('/') + "/api/app/version")
   881→                val conn = url.openConnection() as HttpURLConnection
   882→                conn.connectTimeout = 10_000
   883→                conn.readTimeout = 10_000
   884→                conn.instanceFollowRedirects = true
   885→                val body = conn.inputStream.bufferedReader().use { it.readText() }
   886→                conn.disconnect()
   887→                val json = JSONObject(body)
   888→                val min = json.optInt("minVersionCode", 1)
   889→                val latest = json.optInt("latestVersionCode", 1)
   890→                val updateUrl = json.optString("updateUrl", "https://cafebazaar.ir/app/ir.fittup.app")
   891→                val title = json.optString("forceUpdateTitle", "به‌روزرسانی لازم است")
   892→                val msg = json.optString("forceUpdateBody", "برای ادامه استفاده از فیتاپ، نسخه جدید را از کافه‌بازار نصب کنید.")
   893→                runOnUiThread {
   894→                    when {
   895→                        BuildConfig.VERSION_CODE < min -> showForceUpdateDialog(updateUrl, title, msg)
   896→                        BuildConfig.VERSION_CODE < latest && !softUpdateToastShown -> {
   897→                            softUpdateToastShown = true
   898→                            toast("نسخه جدید فیتاپ در کافه‌بازار موجود است 🎉")
   899→                        }
   900→                    }
   901→                }
   902→            } catch (e: Exception) {
   903→                // آفلاین/خطا — نادیده بگیر؛ در اجرای بعدی دوباره تلاش می‌شود
   904→            }
   905→        }
   906→    }

   908→    /** دیالوگ آپدیت اجباری — تنها دکمه «به‌روزرسانی» (باز کردن صفحه بازار) */
   909→    private fun showForceUpdateDialog(updateUrl: String, title: String, message: String) {

   917→        dialog.show()
   918→    }
