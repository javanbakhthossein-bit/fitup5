   335→            )}
   336→          </motion.div>
   337→        </AnimatePresence>
   338→      </div>
   339→
   340→      {/* Bottom action — v49: همیشه قابل‌مشاهده (داخل پوستهٔ مقید) + safe-area iOS */}

   344→            onClick={next}
   345→            disabled={!canNext()}
   346→            className="w-full h-12 rounded-2xl text-base font-bold text-white shadow-lg shadow-orange-500/30 transition-all hover:scale-[1.02] hover:shadow-xl hover:shadow-orange-500/40 disabled:opacity-50 disabled:hover:scale-100 disabled:hover:shadow-lg"
   347→            style={GOLD_GRADIENT}
   348→          >

   355→            className="w-full h-12 rounded-2xl text-base font-bold text-white shadow-lg shadow-orange-500/30 transition-all hover:scale-[1.02] hover:shadow-xl hover:shadow-orange-500/40 disabled:opacity-50 disabled:hover:scale-100 disabled:hover:shadow-lg flex items-center justify-center gap-2"
