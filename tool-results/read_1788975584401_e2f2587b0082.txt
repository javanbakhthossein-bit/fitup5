   182→   182→        val base = BuildConfig.SITE_URL.trimEnd('/')

   186→   186→    /**
   187→   187→     * v1.2.6 — لینک عمیق ورودی (App Link یا اسکیم fitup:// یا اعلان) یا null:
   188→   188→     *  ۱) extra «fitup_link» از NotificationSync — مسیر/کوئری نسبی مثل
   189→   189→     *     "?tab=progress&section=checkup" یا "/panel" → SITE_URL + آن
   190→   190→     *  ۲) dataِ https روی دامنهٔ خودمان (fittup.ir) → همان URL — پارامترهای
   191→   191→     *     ?tab= ?ref= ?renewal= ?screen= ?article= دست‌نخورده می‌مانند
   192→   192→     *  ۳) v1.2.6 — اسکیم اختصاصی fitup://open?url=<encoded> از صفحهٔ /go سایت
   193→   193→     *     (لینک‌های هوشمند پیامک — فقط URL دامنهٔ خودمان پذیرفته می‌شود)
   194→   194→     * (null-امن — هرگز به‌خاطر لینک خراب کرش نمی‌کند)
   195→   195→     */
   196→   196→    private fun deepLinkUrl(from: Intent?): String? {
   197→   197→        if (from == null) return null
   198→   198→        try {
   199→   199→            val extra = from.getStringExtra("fitup_link")
   200→   200→            if (!extra.isNullOrBlank() && (extra.startsWith("/") || extra.startsWith("?"))) {
   201→   201→                return BuildConfig.SITE_URL.trimEnd('/') + extra
   202→   202→            }
   203→   203→            val data = from.data ?: return null
   204→   204→            val scheme = data.scheme?.lowercase()
   205→   205→            // ─── v1.2.6: fitup://open?url=<encoded> — از لینک هوشمند /go سایت ───
   206→   206→            if (scheme == "fitup") {
   207→   207→                val target = data.getQueryParameter("url") ?: return null
   208→   208→                if (target.startsWith("/")) {
   209→   209→                    // مسیر نسبی روی دامنهٔ خودمان
   210→   210→                    return BuildConfig.SITE_URL.trimEnd('/') + target
   211→   211→                }
   212→   212→                val t = android.net.Uri.parse(target)
   213→   213→                val tScheme = t.scheme?.lowercase()
   214→   214→                val tHost = t.host?.lowercase() ?: return null
   215→   215→                val isOurHost = tHost == "fittup.ir" || tHost.endsWith(".fittup.ir")
   216→   216→                if (isOurHost && (tScheme == "https")) return target
   217→   217→                return null
   218→   218→            }
   219→   219→            if (scheme != "https") return null
   220→   220→            val host = data.host?.lowercase() ?: return null
   221→   221→            // فقط دامنهٔ خودمان — همان منطق routeUrl (fittup.ir و زیردامنه‌ها)
   222→   222→            if (host == "fittup.ir" || host.endsWith(".fittup.ir")) return data.toString()
   223→   223→        } catch (_: Exception) {
   224→   224→        }
   225→   225→        return null
   226→   226→    }
