   298→    private fun routeUrl(url: Uri): Boolean {
   299→        val host = url.host?.lowercase() ?: return false
   300→        val siteHost = try { URI(BuildConfig.SITE_URL).host?.lowercase() } catch (_: Exception) { null }
   301→
   302→        // سایت خودمان — همیشه داخل WebView
   303→        if (host == "fittup.ir" || host.endsWith(".fittup.ir")) return false
   304→        if (siteHost != null && (host == siteHost || host.endsWith(".$siteHost"))) return false
   305→
   306→        // درگاه پرداخت سایت (زرین‌پال) و بانک‌ها (شاپراک) — داخل WebView تا کاربر
   307→        // بعد از پرداخت به پنل برگردد (payment_verify در همان WebView اجرا می‌شود)
   308→        if (host == "zarinpal.com" || host.endsWith(".zarinpal.com")) return false
   309→        if (host == "zarin.link" || host.endsWith(".zarin.link")) return false
   310→        if (host == "shaparak.ir" || host.endsWith(".shaparak.ir")) return false
   311→
   312→        // شبکه‌های اجتماعی/استورها/سایت‌های دیگر → مرورگر بیرونی
   313→        openExternal(url)
   314→        return true
   315→    }

   317→    /** اسکیم‌های غیر وب — tel/mailto/intent و غیره */
   318→    private fun handleExternalScheme(uri: Uri): Boolean {
   319→        try {
   320→            when (uri.scheme?.lowercase()) {
   321→                "tel" -> startActivity(Intent(Intent.ACTION_DIAL, uri))
   322→                "mailto" -> startActivity(Intent(Intent.ACTION_SENDTO, uri))
   323→                "sms" -> startActivity(Intent(Intent.ACTION_SENDTO, uri))
   324→                "intent" -> {
   325→                    val intent = Intent.parseUri(uri.toString(), Intent.URI_INTENT_SCHEME)
   326→                    startActivity(intent)
   327→                }
   328→                else -> startActivity(Intent(Intent.ACTION_VIEW, uri))
   329→            }
   330→        } catch (_: Exception) {
   331→            // اپی برای این لینک نیست — نادیده بگیر
   332→        }
   333→        return true
   334→    }

   361→    private fun injectBridgeHelper() {
   362→        val js = """
   363→            (function() {
   364→              if (window.__fitupOwnAppInjected) return;
   365→              window.__fitupOwnAppInjected = true;
   366→              // آیا داخل اپ اختصاصی فیتاپ هستیم؟
   367→              window.isFitUpOwnApp = function() {
   368→                try { return !!(window.FitUpNative && window.FitUpNative.isOwnApp && window.FitUpNative.isOwnApp()); }
   369→                catch (e) { return false; }
   370→              };
   371→              // کوکی pwa_standalone — سرور برای URL «/» مستقیم صفحهٔ auth را
   372→              // رندر می‌کند (نه لندینگ) — همان رفتار وب‌اپ. اپ همیشه با
   373→              // ?screen=auth شروع می‌شود؛ این کوکی فقط حالت رفرش/بعد از خروج
   374→              // را هم درست نگه می‌دارد.
   375→              try {
   376→                if (document.cookie.indexOf('pwa_standalone=1') === -1) {
   377→                  document.cookie = 'pwa_standalone=1; path=/; max-age=31536000; samesite=lax';
   378→                }
   379→              } catch (e) {}
   380→            })();
   381→        """.trimIndent()
   382→        runOnUiThread {
   383→            webView.evaluateJavascript(js, null)
   384→        }
   385→    }

   420→    /** شروع دانلود با DownloadManager → پوشه Download اپ (بدون مجوز نوشتن) */
   421→    private fun startNativeDownload(url: String, fileName: String, isApk: Boolean) {
   422→        val safeName = fileName.replace(Regex("[^A-Za-z0-9._-]"), "_").ifBlank {
   423→            "fitup-${System.currentTimeMillis()}" + if (isApk) ".apk" else ""
   424→        }
   425→        val request = DownloadManager.Request(Uri.parse(url))
   426→            .setTitle(if (isApk) "نسخه جدید فیتاپ" else safeName)
   427→            .setDescription(if (isApk) "دانلود مستقیم از فیتاپ" else "دانلود فایل")
   428→            .setNotificationVisibility(DownloadManager.Request.VISIBILITY_VISIBLE_NOTIFY_COMPLETED)
   429→            .setDestinationInExternalFilesDir(this, Environment.DIRECTORY_DOWNLOADS, safeName)
   430→            .setAllowedOverMetered(true)
   431→            .setAllowedOverRoaming(true)
   432→        try {
   433→            val dm = getSystemService(Context.DOWNLOAD_SERVICE) as DownloadManager
   434→            lastDownloadId = dm.enqueue(request)
   435→            lastApkFile = File(
   436→                getExternalFilesDir(Environment.DIRECTORY_DOWNLOADS) ?: filesDir,
   437→                safeName
   438→            )
   439→        } catch (e: Exception) {
   440→            Log.e("FitUpApp", "download enqueue failed: ${e.message}")
   441→            toast("دانلود ممکن نشد")
   442→        }

   465→            if (Build.VERSION.SDK_INT >= 33) {

   467→            } else {

   521→    /** نام فایل از Content-Disposition یا URL */
   522→    private fun guessFileName(contentDisposition: String?, mimeType: String?, url: String): String {
   523→        try {
   524→            if (!contentDisposition.isNullOrBlank()) {
   525→                val m = Regex("filename\\*=UTF-8''([^;]+)|filename=\"?([^\";]+)\"?", RegexOption.IGNORE_CASE)
   526→                    .find(contentDisposition)
   527→                if (m != null) {
   528→                    val name = (m.groupValues[1].ifBlank { m.groupValues[2] })
   529→                        .replace("+", " ")
   530→                    if (name.isNotBlank()) return java.net.URLDecoder.decode(name, "UTF-8")
   531→                }
   532→            }
   533→            if (!mimeType.isNullOrBlank() && mimeType.equals("application/vnd.android.package-archive", true)) {
   534→                return "fitup-${System.currentTimeMillis()}.apk"
   535→            }
   536→            val path = URI(url).path ?: return "fitup-file"
   537→            return path.substringAfterLast('/').ifBlank { "fitup-file" }
   538→        } catch (_: Exception) {
   539→            return "fitup-file"
   540→        }
   541→    }

   543→    /* ───────────── ذخیره data URL (PNG/PDF) ───────────── */
   544→
   545→    @SuppressLint("InlinedApi")
   546→    private fun downloadDataUrl(filename: String, dataUrl: String) {
   547→        try {
   548→            val safeName = filename.ifBlank { "fitup-${System.currentTimeMillis()}" }
   549→                .replace(Regex("[^A-Za-z0-9._-\\u0600-\\u06FF ]"), "_")
   550→            val comma = dataUrl.indexOf(',')
   551→            if (comma < 0 || !dataUrl.startsWith("data:", ignoreCase = true)) {
   552→                runOnUiThread { toast("فایل قابل ذخیره نیست") }
   553→                return
   554→            }
   555→            val meta = dataUrl.substring(5, comma)
   556→            val mime = meta.substringBefore(";").ifBlank { "application/octet-stream" }
   557→            val isBase64 = meta.contains("base64", ignoreCase = true)
   558→            val bytes = if (isBase64) {
   559→                Base64.decode(dataUrl.substring(comma + 1), Base64.DEFAULT)
   560→            } else {
   561→                dataUrl.substring(comma + 1).toByteArray(Charsets.UTF_8)
   562→            }
   563→
   564→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   565→                val values = ContentValues().apply {
   566→                    put(MediaStore.Downloads.DISPLAY_NAME, safeName)
   567→                    put(MediaStore.Downloads.MIME_TYPE, mime)
   568→                    put(MediaStore.Downloads.IS_PENDING, 1)
   569→                }
   570→                val uri = contentResolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, values)
   571→                if (uri == null) {
   572→                    runOnUiThread { toast("خطا در ذخیره فایل") }
   573→                    return
   574→                }
   575→                contentResolver.openOutputStream(uri)?.use { it.write(bytes) }
   576→                values.clear()
   577→                values.put(MediaStore.Downloads.IS_PENDING, 0)
   578→                contentResolver.update(uri, values, null, null)
   579→            } else {
   580→                val dir = getExternalFilesDir(Environment.DIRECTORY_DOCUMENTS) ?: filesDir
   581→                val file = File(dir, safeName)
   582→                FileOutputStream(file).use { it.write(bytes) }
   583→            }
   584→            runOnUiThread { toast("«$safeName» ذخیره شد ✓") }
   585→        } catch (e: Exception) {
   586→            Log.e("FitUpApp", "download failed: ${e.message}")
   587→            runOnUiThread { toast("خطا در ذخیره فایل") }
   588→        }
   589→    }

   730→    private fun registerSmsReceiver() {
   731→        if (smsReceiver != null) return
   732→        val receiver = object : BroadcastReceiver() {
   733→            override fun onReceive(context: Context?, intent: Intent?) {
   734→                if (intent?.action != "android.provider.Telephony.SMS_RECEIVED") return
   735→                try {
   736→                    val msgs = android.provider.Telephony.Sms.Intents.getMessagesFromIntent(intent) ?: return
   737→                    for (m in msgs) {
   738→                        val body = m?.displayMessageBody?.toString() ?: continue
   739→                        if (!body.contains("فیتاپ") && !body.contains("FitUp", true)) continue
   740→                        val match = Regex("(^|\\D)(\\d{4})(\\D|$)").find(body) ?: continue
   741→                        dispatchOtpCode(match.groupValues[2])
   742→                        break
   743→                    }
   744→                } catch (_: Exception) {}
   745→            }
   746→        }
   747→        try {
   748→            registerReceiver(receiver, android.content.IntentFilter("android.provider.Telephony.SMS_RECEIVED"))
   749→            smsReceiver = receiver
   750→        } catch (_: Exception) {}
   751→    }

   774→    /**
   775→     * در هر اجرا /api/app/own/latest را می‌خواند.
   776→     * مودال زیبای آپدیت را خود سایت (AppUpdateModal) نشان می‌دهد؛ این چک نیتیو
   777→     * فقط safety-net است: اگر صفحه سایت بالا نیامده باشد (خطای شبکه) و نسخهٔ
   778→     * جدید «اجباری» بود، دیالوگ نیتیو با دانلود مستقیم نشان می‌دهیم.
   779→     */
   780→    private fun checkAppVersion() {
   781→        thread {
   782→            try {
   783→                val url = URL(BuildConfig.SITE_URL.trimEnd('/') + "/api/app/own/latest")
   784→                val conn = url.openConnection() as HttpURLConnection
   785→                conn.connectTimeout = 10_000
   786→                conn.readTimeout = 10_000
   787→                conn.instanceFollowRedirects = true
   788→                val body = conn.inputStream.bufferedReader().use { it.readText() }
   789→                conn.disconnect()
   790→                val json = JSONObject(body)
   791→                if (!json.optBoolean("available", false)) return@thread
   792→                val latest = json.optInt("latestVersionCode", 1)
   793→                val forced = json.optBoolean("forceUpdate", false)
   794→                val apkUrl = BuildConfig.SITE_URL.trimEnd('/') + "/api/app/own/download"
   795→                if (BuildConfig.VERSION_CODE < latest && forced) {
   796→                    runOnUiThread {
   797→                        // فقط وقتی سایت بالا نیامده (وگرنه مودال سایت خودش را نشان می‌دهد)
   798→                        if (errorView.visibility == View.VISIBLE && !forceDialogShown) {
   799→                            forceDialogShown = true
   800→                            showForceUpdateDialog(apkUrl)
   801→                        }
   802→                    }
   803→                }
   804→            } catch (e: Exception) {
   805→                // آفلاین/خطا — نادیده بگیر؛ در اجرای بعدی دوباره تلاش می‌شود
   806→            }
   807→        }
   808→    }
