    33→// ۰۹:۵۹ → "۹:۵۹" فارسی — شمارش معکوس اعتبار کد
    34→function formatCountdown(totalSec: number): string {
    35→  const m = Math.floor(totalSec / 60);
    36→  const s = totalSec % 60;
    37→  return `${m}:${String(s).padStart(2, "0")}`;
    38→}

    40→// FIX: تبدیل ارقام فارسی (۰-۹) و عربی (٠-٩) به لاتین قبل از فیلتر.
    41→// regex \d فقط ارقام ASCII را می‌گیرد و قبلاً ارقام فارسی (کیبورد فارسی)
    42→// بی‌صدا حذف می‌شدند — کاربر نمی‌توانست شماره/کد را تایپ کند.
    43→function normalizeDigits(input: string): string {
    44→  return input
    45→    .replace(/[\u06F0-\u06F9]/g, (d) => String(d.charCodeAt(0) - 0x06f0)) // ۰-۹
    46→    .replace(/[\u0660-\u0669]/g, (d) => String(d.charCodeAt(0) - 0x0660)); // ٠-٩
    47→}

    49→// استخراج کد OTP از هر ورودی خام (تایپ، پیست، WebOTP، autofill):
    50→// ارقام فارسی/عربی → لاتین، حذف غیررقمی؛ اگر بیش از ۴ رقم بود «آخرین ۴ رقم» —
    51→// چون در متن پیامک (فرمت @دامنه #کد) کد در انتهاست.
    52→function extractOtpDigits(raw: string): string {
    53→  const digits = normalizeDigits(String(raw ?? "")).replace(/\D/g, "");
    54→  return digits.length > 4 ? digits.slice(-4) : digits;
    55→}

   266→  // ---- Send OTP ----
   267→  async function handleSendOtp(e?: React.FormEvent) {
   268→    e?.preventDefault();
   269→    // گارد تک‌ارسال: Enter در فرم یا دابل‌کلیک نباید دو درخواست بفرستد

   275→      toast.error("شماره موبایل نامعتبر است.");
   276→      return;
   277→    }
   278→
   279→    setSending(true);
   280→    // 🩹 v45: timeout ۱۲ثانیه‌ای — سمت سرور حالا ≤ ~۹ثانیه جواب می‌دهد؛
   281→    // اگر شبکه قطع شد به‌جای «failed to fetch» خام، پیام فارسی واضح
   282→    const controller = new AbortController();
   283→    const timer = setTimeout(() => controller.abort(), 12_000);
   284→    try {
   285→      const res = await fetch("/api/auth/send-otp", {
   286→        method: "POST",
   287→        headers: { "Content-Type": "application/json" },
   288→        body: JSON.stringify({ mobile }),
   289→        signal: controller.signal,
   290→      });
   291→      const data = await res.json().catch(() => ({}));
   292→      if (!res.ok) {
   293→        const msg = data?.error || "ارسال کد با خطا مواجه شد.";
   294→        setError(msg);
   295→        toast.error(msg);
   296→        // If resend-too-soon, jump to OTP step anyway and apply the wait time
   297→        if (data?.code === "RESEND_TOO_SOON" && typeof data.waitSeconds === "number") {
   298→          setStep("otp");
   299→          setResendIn(Math.min(RESEND_COOLDOWN_SEC, Math.max(1, data.waitSeconds)));
   300→          // کد قبلی هنوز در راه است — تایمر اعتبارش را هم شروع کن
   301→          setValidityIn(OTP_VALIDITY_FALLBACK_SEC);
   302→        }
   303→        return;
   304→      }
   305→      toast.success("کد ۴ رقمی به شماره شما ارسال شد.");
   306→      setStep("otp");
   307→      setCode("");
   308→      setResendIn(RESEND_COOLDOWN_SEC);
   309→      setValidityIn(
   310→        typeof data?.expiresIn === "number" ? data.expiresIn : OTP_VALIDITY_FALLBACK_SEC
   311→      );
   312→      // دور تازهٔ پیامک → listener وب‌OTP برای کد جدید دوباره arm می‌شود
   313→      setOtpNonce((n) => n + 1);
   314→    } catch (err) {
   315→      const msg =
   316→        err instanceof DOMException && err.name === "AbortError"
   317→          ? "ارسال کد بیش از حد طول کشید. لطفاً اتصال اینترنت را بررسی و دوباره تلاش کنید."
   318→          : err instanceof TypeError
   319→            ? "ارتباط با سرور برقرار نشد. لطفاً اتصال اینترنت را بررسی و دوباره تلاش کنید."
   320→            : err instanceof Error
   321→              ? err.message
   322→              : "خطای ناشناخته";
   323→      setError(msg);
   324→      toast.error(msg);
   325→    } finally {
   326→      clearTimeout(timer);
   327→      setSending(false);
   328→    }
   329→  }
