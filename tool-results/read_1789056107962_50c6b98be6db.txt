   699→   699→        status: "failed",
   700→   700→        error,
   701→   701→      },
   702→   702→      update: {
   703→   703→        status: "failed",
   704→   704→        error,
   705→   705→      },
   706→   706→    });
   707→   707→  } catch (e) {
   708→   708→    console.warn("[sendAbandonedCartSms] failed-record upsert error:", e);
   709→   709→  }
   710→   710→}
   711→   711→
   712→   712→/**
   713→   713→ * ارسال پیامک ترک درگاه — فقط با قالب sms.ir (نردبان خودترمیم پارامترها).
   714→   714→ * v61: مسیر bulk/متن آزاد حذف شد (دیریکتیو مالک + خط خدماتی که bulk را
   715→   715→ * پشتیبانی نمی‌کند). شکست قالب = رکورد failed و تلاشِ sweep بعدی.
   716→   716→ */
   717→   717→export async function sendAbandonedCartSms(
   718→   718→  opts: AbandonedCartSmsOptions
   719→   719→): Promise<AbandonedCartSmsResult> {
   720→   720→  const { user, dedupeKey, link, templateId } = opts;
   721→   721→
   722→   722→  const normalized = normalizeMobileForSmsIr(user.mobile);
   723→   723→  if (!/^9\d{9}$/.test(normalized)) {
   724→   724→    return { sent: false, skipped: "invalid_mobile", error: `شماره نامعتبر: ${normalized}` };
   725→   725→  }
   726→   726→
   727→   727→  const name = smsFirstName(user.name);
   728→   728→
   729→   729→  // ─── مسیر واحد — قالب sms.ir (v61: قالبی الزامی؛ bulk حذف شد) ───
   730→   730→  const templateIdSafe = templateId?.trim() || "";
   731→   731→  if (!templateIdSafe) {
   732→   732→    const noTpl =
   733→   733→      "شناسه قالب ترک خرید تنظیم نیست (SMSIR_ABANDONED_CART_TEMPLATE_ID یا SiteSetting: abandoned_cart_sms_template_id) — پیامک قالبی ارسال نشد.";
   734→   734→    console.warn(`[sendAbandonedCartSms] (key=${dedupeKey}) ${noTpl}`);
   735→   735→    await recordAbandonedCartFailure(normalized, dedupeKey, "", user.id, noTpl);
   736→   736→    return { sent: false, error: noTpl };
   737→   737→  }
   738→   738→
   739→   739→  // نردبان خودترمیم پارامترهای قالب: [NAME+LINK] → [NAME] → []
   740→   740→  const attempts: Array<Array<{ name: string; value: string }>> = [
   741→   741→    [
   742→   742→      { name: "NAME", value: name },
   743→   743→      { name: "LINK", value: link },
   744→   744→    ],
   745→   745→    [{ name: "NAME", value: name }],
   746→   746→    [],
   747→   747→  ];
   748→   748→  let last: SendTemplateSmsResult = { sent: false, error: "no attempt" };
   749→   749→  for (const params of attempts) {
   750→   750→    last = await sendTemplateSms({
   751→   751→      key: dedupeKey,
   752→   752→      templateEnvKey: undefined,
   753→   753→      templateIdOverride: templateIdSafe,
   754→   754→      mobile: user.mobile,
   755→   755→      params,
   756→   756→      userId: user.id,
   757→   757→    });
   758→   758→    if (last.sent) return { sent: true, via: "template" };
   759→   759→    if (last.skipped === "already_sent" || last.skipped === "invalid_mobile") {
   760→   760→      return {
   761→   761→        sent: false,
   762→   762→        skipped: last.skipped === "invalid_mobile" ? "invalid_mobile" : "already_sent",
   763→   763→        error: last.error,
   764→   764→      };
   765→   765→    }
   766→   766→    // no_template / no_name / خطای درگاه → رانگ بعدی نردبان
   767→   767→  }
   768→   768→
   769→   769→  // ─── شکست نهایی قالب — رکورد failed برای دیده‌شدن؛ sweep بعدی دوباره تلاش می‌کند ───
   770→   770→  // (sendTemplateSms خودش در هر تلاش رکورد ثبت می‌کند؛ این upsert فقط گارانتی
   771→   771→  //  می‌کند که حتماً یک رکورد failed با آخرین خطا موجود باشد.)
   772→   772→  await recordAbandonedCartFailure(
   773→   773→    normalized,
   774→   774→    dedupeKey,
   775→   775→    templateIdSafe,
   776→   776→    user.id,
   777→   777→    last.error ?? "abandoned_cart sms failed"
   778→   778→  );
   779→   779→
   780→   780→  return { sent: false, error: last.error ?? "abandoned_cart sms failed" };
   781→   781→}
