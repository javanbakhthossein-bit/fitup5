    33→// ۰۹:۵۹ → "۹:۵۹" فارسی — شمارش معکوس اعتبار کد
    34→function formatCountdown(totalSec: number): string {
    35→  const m = Math.floor(totalSec / 60);
    36→  const s = totalSec % 60;
    37→  return `${m}:${String(s).padStart(2, "0")}`;
    38→}

    40→// FIX: تبدیل ارقام فارسی (۰-۹) و عربی (٠-٩) به لاتین قبل از فیلتر.

    42→// بی‌صدا حذف می‌شدند — کاربر نمی‌توانست شماره/کد را تایپ کند.
    43→function normalizeDigits(input: string): string {
    44→  return input
    45→    .replace(/[\u06F0-\u06F9]/g, (d) => String(d.charCodeAt(0) - 0x06f0)) // ۰-۹
    46→    .replace(/[\u0660-\u0669]/g, (d) => String(d.charCodeAt(0) - 0x0660)); // ٠-٩
    47→}

    49→// استخراج کد OTP از هر ورودی خام (تایپ، پیست، WebOTP، autofill):
    50→// ارقام فارسی/عربی → لاتین، حذف غیررقمی؛ اگر بیش از ۴ رقم بود «آخرین ۴ رقم» —
    51→// چون در متن پیامک (فرمت @دامنه #کد) کد در انتهاست.
    52→function extractOtpDigits(raw: string): string {
    53→  const digits = normalizeDigits(String(raw ?? "")).replace(/\D/g, "");
    54→  return digits.length > 4 ? digits.slice(-4) : digits;
    55→}

   266→  // ---- Send OTP ----
   267→  async function handleSendOtp(e?: React.FormEvent) {
   268→    e?.preventDefault();

   276→      return;
   277→    }
   278→
   279→    setSending(true);
   280→    // 🩹 v45: timeout ۱۲ثانیه‌ای — سمت سرور حالا ≤ ~۹ثانیه جواب می‌دهد؛
   281→    // اگر شبکه قطع شد به‌جای «failed to fetch» خام، پیام فارسی واضح
   282→    const controller = new AbortController();
   283→    const timer = setTimeout(() => controller.abort(), 12_000);
   284→    try {
   285→      // v57 — کلید bridge اپ (فقط داخل اپ نیتیو مقدار دارد): سرور با تطبیقش
   286→      // کد را در همان پاسخ برمی‌گرداند (bridgeCode) → ورود خودکار از سرور.
   287→      const bridgeKey = getNativeOtpBridgeKey();
   288→      const res = await fetch("/api/auth/send-otp", {
   289→        method: "POST",
   290→        headers: {
   291→          "Content-Type": "application/json",
   292→          ...(bridgeKey ? { "x-fitup-otp-bridge": bridgeKey } : {}),
   293→        },
   294→        body: JSON.stringify({ mobile }),
   295→        signal: controller.signal,
   296→      });
   297→      const data = await res.json().catch(() => ({}));
   298→      if (!res.ok) {
   299→        const msg = data?.error || "ارسال کد با خطا مواجه شد.";
   300→        setError(msg);
   301→        toast.error(msg);
   302→        // If resend-too-soon, jump to OTP step anyway and apply the wait time
   303→        if (data?.code === "RESEND_TOO_SOON" && typeof data.waitSeconds === "number") {
   304→          setStep("otp");
   305→          setResendIn(Math.min(RESEND_COOLDOWN_SEC, Math.max(1, data.waitSeconds)));
   306→          // کد قبلی هنوز در راه است — تایمر اعتبارش را هم شروع کن
   307→          setValidityIn(OTP_VALIDITY_FALLBACK_SEC);
   308→        }
   309→        return;
   310→      }
   311→      toast.success("کد ۴ رقمی به شماره شما ارسال شد.");
   312→      setStep("otp");
   313→      setResendIn(RESEND_COOLDOWN_SEC);
   314→      setValidityIn(
   315→        typeof data?.expiresIn === "number" ? data.expiresIn : OTP_VALIDITY_FALLBACK_SEC
   316→      );
   317→      // دور تازهٔ پیامک → listener وب‌OTP برای کد جدید دوباره arm می‌شود
   318→      setOtpNonce((n) => n + 1);
   319→      // v57 — ورود خودکار از سرور (فقط داخل اپ): کد از پاسخ می‌آید، جا می‌افتد
   320→      // و اثر اتو-وریفای بالایی خودش verify می‌کند → کاربر مستقیم وارد پنل می‌شود.
   321→      // setCode بعد از setStep تا اثر auto-verify با step="otp" فعال شود.
   322→      if (typeof data?.bridgeCode === "string" && /^\d{4}$/.test(data.bridgeCode)) {
   323→        setCode(data.bridgeCode);
   324→      } else {
   325→        setCode("");
   326→      }
   327→    } catch (err) {
   328→      const msg =
   329→        err instanceof DOMException && err.name === "AbortError"
   330→          ? "ارسال کد بیش از حد طول کشید. لطفاً اتصال اینترنت را بررسی و دوباره تلاش کنید."
   331→          : err instanceof TypeError
   332→            ? "ارتباط با سرور برقرار نشد. لطفاً اتصال اینترنت را بررسی و دوباره تلاش کنید."
   333→            : err instanceof Error
   334→              ? err.message
   335→              : "خطای ناشناخته";
   336→      setError(msg);
   337→      toast.error(msg);
   338→    } finally {
   339→      clearTimeout(timer);
   340→      setSending(false);
   341→    }
   342→  }

   345→  // الگوی ناوبری SPA: setScreen از store + به‌روزرسانی URL با پارامتر doc

   348→  function openLegalDoc(doc: "terms" | "privacy") {

   365→  async function handleResend() {
   366→    if (resendIn > 0) return;
   367→    setError("");
   368→    setSending(true);
   369→    const controller = new AbortController();
   370→    const timer = setTimeout(() => controller.abort(), 12_000);
   371→    try {
   372→      // v57 — کلید bridge اپ برای ورود خودکار (فقط داخل اپ مقدار دارد)
   373→      const bridgeKey = getNativeOtpBridgeKey();
   374→      const res = await fetch("/api/auth/send-otp", {
   375→        method: "POST",
   376→        headers: {
   377→          "Content-Type": "application/json",
   378→          ...(bridgeKey ? { "x-fitup-otp-bridge": bridgeKey } : {}),
   379→        },
   380→        body: JSON.stringify({ mobile }),
   381→        signal: controller.signal,
   382→      });
   383→      const data = await res.json().catch(() => ({}));
   384→      if (!res.ok) {
   385→        const msg = data?.error || "ارسال مجدد با خطا مواجه شد.";
   386→        if (data?.code === "RESEND_TOO_SOON" && typeof data.waitSeconds === "number") {
   387→          setResendIn(Math.min(RESEND_COOLDOWN_SEC, Math.max(1, data.waitSeconds)));
   388→          toast.error(msg);
   389→          return;
   390→        }
   391→        setError(msg);
   392→        toast.error(msg);
   393→        return;
   394→      }
   395→      toast.success("کد جدید ارسال شد.");
   396→      setResendIn(RESEND_COOLDOWN_SEC);
   397→      setValidityIn(
   398→        typeof data?.expiresIn === "number" ? data.expiresIn : OTP_VALIDITY_FALLBACK_SEC
   399→      );
   400→      // دور تازهٔ پیامک → listener وب‌OTP برای کد جدید دوباره arm می‌شود
   401→      setOtpNonce((n) => n + 1);
   402→      // v57 — ورود خودکار از سرور در ریسند هم (فقط داخل اپ)
   403→      if (typeof data?.bridgeCode === "string" && /^\d{4}$/.test(data.bridgeCode)) {
   404→        setCode(data.bridgeCode);
   405→      } else {
   406→        setCode("");
   407→      }
   408→    } catch (err) {
   409→      const msg =
   410→        err instanceof DOMException && err.name === "AbortError"
   411→          ? "ارسال کد بیش از حد طول کشید. لطفاً دوباره تلاش کنید."
   412→          : err instanceof TypeError
   413→            ? "ارتباط با سرور برقرار نشد. لطفاً دوباره تلاش کنید."
   414→            : err instanceof Error
   415→              ? err.message
   416→              : "خطای ناشناخته";
   417→      setError(msg);
   418→      toast.error(msg);
   419→    } finally {
   420→      clearTimeout(timer);
   421→      setSending(false);
   422→    }
   423→  }
