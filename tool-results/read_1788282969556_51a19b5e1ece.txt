    63→/** ساعت فعلی تهران (۰-۲۳) — با Intl و timeZone Asia/Tehran، نه timezone مرورگر */
    64→function getTehranHour(): number {
    65→  try {
    66→    const h = new Intl.DateTimeFormat("en-US", {
    67→      timeZone: "Asia/Tehran",
    68→      hour: "numeric",
    69→      hour12: false,
    70→    }).format(new Date());
    71→    const parsed = parseInt(h, 10);
    72→    if (Number.isFinite(parsed)) return parsed % 24;
    73→  } catch {
    74→    // Intl در دسترس نیست — fallback به ساعت محلی
    75→  }
    76→  return new Date().getHours();
    77→}

    79→/** Format a JS Date as a Persian (Jalali) string with day-of-week + date + Tehran time */
    80→function formatPersianDateTime(d: Date): string {
    81→  try {
    82→    const weekdayNames = ["یکشنبه", "دوشنبه", "سه‌شنبه", "چهارشنبه", "پنجشنبه", "جمعه", "شنبه"];
    83→    // Intl supports islamic-civil but for Persian (Jalali) use fa-IR with persian calendar
    84→    const parts = new Intl.DateTimeFormat("fa-IR-u-ca-persian", {
    85→      weekday: "long",
    86→      day: "numeric",
    87→      month: "long",
    88→      year: "numeric",
    89→      timeZone: "Asia/Tehran",
    90→    }).formatToParts(d);
    91→    // Extract parts
    92→    const get = (t: string) => parts.find((p) => p.type === t)?.value ?? "";
    93→    const day = get("weekday");
    94→    const date = `${get("day")} ${get("month")} ${get("year")}`;
    95→    // Time — بر اساس ساعت تهران (نه timezone مرورگر)
    96→    const timeParts = new Intl.DateTimeFormat("en-US", {
    97→      timeZone: "Asia/Tehran",
    98→      hour: "2-digit",
    99→      minute: "2-digit",
   100→      hour12: false,
   101→    }).formatToParts(d);
   102→    const hh = timeParts.find((p) => p.type === "hour")?.value ?? "00";
   103→    const mm = timeParts.find((p) => p.type === "minute")?.value ?? "00";
   104→    return `${day}، ${date} - ${toPersianDigits(hh)}:${toPersianDigits(mm)}`;
   105→  } catch {
   106→    // Fallback if Intl calendar is unavailable
   107→    const dayIdx = (d.getDay() + 1) % 7;
   108→    const wd = PERSIAN_WEEKDAYS[dayIdx];
   109→    const hh = String(d.getHours()).padStart(2, "0");
   110→    const mm = String(d.getMinutes()).padStart(2, "0");
   111→    return `${wd} - ${toPersianDigits(hh)}:${toPersianDigits(mm)}`;
   112→  }
   113→}

   115→/**
   116→ * ساعت زنده تهران — کامپوننت ایزوله (FE-M2):
   117→ * interval یک‌ثانیه‌ای فقط این کامپوننت کوچک را re-render می‌کند، نه کل
   118→ * داشبورد (که شامل LineChart ری‌چارت و کارت‌های متعدد است).
   119→ */
   120→const TehranClock = memo(function TehranClock() {
   121→  const [now, setNow] = useState<string>("");
   122→  useEffect(() => {
   123→    const tick = () => setNow(formatPersianDateTime(new Date()));
   124→    tick();
   125→    const id = setInterval(tick, 1000);
   126→    return () => clearInterval(id);
   127→  }, []);
   128→  return <span className="font-stat truncate">{now || "در حال بارگذاری..."}</span>;
   129→});

   131→/** Estimate calories burned during a workout using MET × weight × hours */
   132→function estimateCaloriesBurned(
   133→  minutes: number,
   134→  userWeightKg: number,
   135→  metValue = 6.5
   136→): number {
   137→  const hours = minutes / 60;
   138→  return Math.round(metValue * userWeightKg * hours);
   139→}

   771→  workout: number;
   772→  calories: number;
   773→  water: number;

   789→                cy="60"
   790→                r={ring.radius}

   801→                stroke={ring.color}
   802→                strokeWidth="9"
   803→                strokeLinecap="round"
   804→                strokeDasharray={c}
   805→                initial={{ strokeDashoffset: c }}
   806→                animate={{ strokeDashoffset: c * (1 - ring.progress) }}
   807→                transition={{ duration: 1.2, delay: i * 0.15, ease: "easeOut" }}
   808→                style={{ filter: `drop-shadow(0 0 4px ${ring.color}80)` }}

  1451→/** Circular progress indicator for body fat percentage */
  1452→function BodyFatRing({ percent }: { percent: number }) {
  1453→  const size = 56;
  1454→  const stroke = 5;
  1455→  const radius = (size - stroke) / 2;
  1456→  const circumference = 2 * Math.PI * radius;
  1457→  // درصد چربی را به فرکانس بصری (5-35) تبدیل می‌کنیم تا نمودار معنادار باشد
  1458→  const visualFraction = Math.min(1, Math.max(0.05, (percent - 3) / 35));
  1459→  const offset = circumference * (1 - visualFraction);
  1460→  const color = percent < 14 ? "#06b6d4" : percent < 21 ? "#10b981" : percent < 26 ? "#f59e0b" : "#ef4444";
  1461→
  1462→  return (
  1463→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
  1464→      <svg width={size} height={size} className="-rotate-90">
  1465→        <circle
  1466→          cx={size / 2}
  1467→          cy={size / 2}
  1468→          r={radius}
  1469→          fill="none"
  1470→          stroke="#f1f5f9"
  1471→          strokeWidth={stroke}
  1472→        />
  1473→        <motion.circle
  1474→          cx={size / 2}
  1475→          cy={size / 2}
  1476→          r={radius}
  1477→          fill="none"
  1478→          stroke={color}
  1479→          strokeWidth={stroke}
  1480→          strokeLinecap="round"
  1481→          strokeDasharray={circumference}
  1482→          initial={{ strokeDashoffset: circumference }}
  1483→          animate={{ strokeDashoffset: offset }}
  1484→          transition={{ duration: 1, ease: "easeOut" }}
  1485→        />
  1486→      </svg>
  1487→      <div className="absolute inset-0 flex items-center justify-center">
  1488→        <span className="text-base font-black font-stat text-slate-900 leading-none">
  1489→          {toPersianDigits(percent.toFixed(1))}
  1490→        </span>
  1491→      </div>
  1492→    </div>
  1493→  );
  1494→}
