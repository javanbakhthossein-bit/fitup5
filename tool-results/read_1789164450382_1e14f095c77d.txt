   124→export function SportsProfileModal({ open, onClose }: { open: boolean; onClose: () => void }) {
   125→  useScrollLock(open);
   126→  const { user } = useAppStore();
   127→  const [loading, setLoading] = useState(true);
   128→  const [profile, setProfile] = useState<OnboardingProfileDto | null>(null);
   129→  const [baseline, setBaseline] = useState<BaselineDto | null>(null);
   130→  const [analysis, setAnalysis] = useState<string>("");
   131→  const [progressPhotos, setProgressPhotos] = useState<any[]>([]);
   132→  const [userMedia, setUserMedia] = useState<any>({
   133→    bodyPhotos: [], bloodTests: [], videoAnalysis: [], bodyAnalysis: [],
   134→  });
   135→  const [editingSection, setEditingSection] = useState<SectionId | null>(null);
   136→  const [editData, setEditData] = useState<Record<string, string>>({});
   137→  const [saving, setSaving] = useState(false);
   138→  // v15: تاریخ عضویت در فیتاپ — از پاسخ /api/onboarding/analysis (memberSince)
   139→  const [memberSince, setMemberSince] = useState<string | null>(null);
   140→  const memberSinceText = (() => {
   141→    if (!memberSince) return null;
   142→    try {
   143→      return new Date(memberSince).toLocaleDateString("fa-IR", { year: "numeric", month: "long", day: "numeric" });
   144→    } catch {
   145→      return null;
   146→    }
   147→  })();
   148→
   149→  const loadData = useCallback(async () => {
   150→    try {
   151→      const [onboardingRes, progressRes, mediaRes] = await Promise.all([
   152→        fetch("/api/onboarding/analysis"),
   153→        fetch("/api/progress"),
   154→        fetch("/api/user-media"),
   155→      ]);
   156→      const onboardingData = await onboardingRes.json().catch(() => null);
   157→      const progressData = await progressRes.json().catch(() => null);
   158→      const mediaData = await mediaRes.json().catch(() => null);
   159→      if (onboardingData?.profile) setProfile(onboardingData.profile);
   160→      if (onboardingData?.baseline) setBaseline(onboardingData.baseline);
   161→      if (onboardingData?.analysis) setAnalysis(onboardingData.analysis);
   162→      // v15: تاریخ عضویت در فیتاپ
   163→      setMemberSince(onboardingData?.memberSince ?? null);
   164→      // C1-fix: کلید درست پاسخ GET /api/progress همان «photos» است (نه progressPhotos) —
   165→      // قبلاً کارت «عکس‌های پیشرفت بدن» همیشه خالی می‌ماند. اگر photos نبود،
   166→      // fallback به bodyPhotos از /api/user-media (همان شکل داده: id/imageUrl/type/takenAt).
   167→      if (Array.isArray(progressData?.photos)) {
   168→        setProgressPhotos(progressData.photos);
   169→      } else if (Array.isArray(mediaData?.bodyPhotos)) {
   170→        setProgressPhotos(mediaData.bodyPhotos);
   171→      }
   172→      if (mediaData) setUserMedia(mediaData);
   173→    } catch {
   174→    } finally {
   175→      setLoading(false);
   176→    }
   177→  }, []);
   178→
   179→  useEffect(() => {
   180→    if (open) {
   181→      setLoading(true);
   182→      setEditingSection(null);
   183→      loadData();
   184→    }
   185→  }, [open, loadData]);
   186→
   187→  function startEditing(section: SectionId) {
   188→    const p = profile;
   189→    if (!p) return;
   190→    const d: Record<string, string> = {};
   191→    switch (section) {
   192→      case "basic":
   193→        d.age = String(p.age ?? "");
   194→        d.height = String(p.height ?? "");
   195→        d.weight = String(p.weight ?? "");
   196→        d.targetWeight = String(p.targetWeight ?? "");
   197→        d.gender = p.gender || "male";
   198→        break;
   199→      case "goal":
   200→        d.goal = p.goal || "fitness";
   201→        d.activityLevel = p.activityLevel || "moderate";
   202→        d.workoutDays = String(p.workoutDays ?? "");
   203→        d.workoutPlace = p.workoutPlace || "gym";
   204→        d.workoutTime = p.workoutTime || "";
   205→        d.workoutDaysList = (p.workoutDaysList || []).join(",");
   206→        break;
   207→      case "equipment":
   208→        d.equipment = (p.equipment || []).join(", ");
   209→        break;
   210→      case "nutrition":
   211→        d.dietType = p.dietType || "standard";
   212→        d.preferredCuisine = p.preferredCuisine || "";
   213→        d.dislikedFoods = p.dislikedFoods || "";
   214→        d.allergies = p.allergies || "";
   215→        break;
   216→      case "health":
   217→        d.injuries = p.injuries || "";
   218→        d.diseases = p.diseases || "";
   219→        d.drugAllergies = p.drugAllergies || "";
   220→        d.currentMedications = p.currentMedications || "";
   221→        d.medicalConditions = (p.medicalConditions || []).join(",");
   222→        break;
   223→      case "recovery":
   224→        d.sleepHours = String(p.sleepHours ?? "");
   225→        d.stressLevel = String(p.stressLevel ?? "");
   226→        d.waterHabit = String(p.waterHabit ?? "");
   227→        d.bodyFrame = p.bodyFrame || "";
   228→        break;
   229→      case "experience":
   230→        d.trainingExperience = p.trainingExperience || "beginner";
   231→        d.previousTrainingType = p.previousTrainingType || "";
   232→        d.maxLifts = p.maxLifts || "";
   233→        break;
   234→      case "target":
   235→        d.targetDate = p.targetDate || "";
   236→        break;
   237→      case "supplements":
   238→        d.currentSupplements = p.currentSupplements || "";
   239→        break;
   240→    }
   241→    setEditData(d);
   242→    setEditingSection(section);
   243→  }
   244→
   245→  function cancelEditing() {
   246→    setEditingSection(null);
   247→    setEditData({});
   248→  }
   249→
   250→  async function saveSection() {
   251→    if (!editingSection) return;
   252→    setSaving(true);
   253→    try {
   254→      const payload: Record<string, any> = {};
   255→      const numFields: Record<SectionId, string[]> = {
   256→        basic: ["age", "height", "weight", "targetWeight"],
   257→        goal: ["workoutDays"],
   258→        equipment: [],
   259→        nutrition: [],
   260→        health: [],
   261→        recovery: ["sleepHours", "stressLevel", "waterHabit"],
   262→        experience: [],
   263→        target: [],
   264→        supplements: [],
   265→      };
   266→      for (const [k, v] of Object.entries(editData)) {
   267→        if (numFields[editingSection].includes(k)) {
   268→          payload[k] = v === "" ? null : Number(v);
   269→        } else {
   270→          payload[k] = v;
   271→        }
   272→      }
   273→      const res = await fetch("/api/onboarding/profile", {
   274→        method: "PUT",
   275→        headers: { "Content-Type": "application/json" },
   276→        body: JSON.stringify(payload),
   277→      });
   278→      if (!res.ok) throw new Error();
   279→      const data = await res.json();
   280→      if (data?.ok) {
   281→        toast.success("پروفایل به‌روزرسانی شد و به مربی هوشمند تزریق شد ✅");
   282→      } else {
   283→        toast.success("پرونده ورزشی به‌روزرسانی شد ✅");
   284→      }
   285→      setEditingSection(null);
   286→      setEditData({});
   287→      await loadData();
   288→    } catch {
   289→      toast.error("خطا در ذخیره پروفایل");
   290→    } finally {
   291→      setSaving(false);
   292→    }
   293→  }
   294→
   295→  const p = profile;
   296→
   297→  if (!open) return null;
   298→
   299→  return (
   300→    <motion.div
   301→      initial={{ opacity: 0 }}
   302→      animate={{ opacity: 1 }}
   303→      exit={{ opacity: 0 }}
   304→      className="fixed inset-0 z-[70] bg-black/40 backdrop-blur-sm flex items-end sm:items-center justify-center"
   305→      onClick={onClose}
   306→      role="dialog"
   307→      aria-modal="true"
   308→      aria-label="پرونده ورزشی"
   309→    >
   310→      <motion.div
   311→        initial={{ y: 60, opacity: 0, scale: 0.98 }}
   312→        animate={{ y: 0, opacity: 1, scale: 1 }}
   313→        exit={{ y: 40, opacity: 0 }}
   314→        transition={{ type: "spring", damping: 26, stiffness: 300 }}
   315→        onClick={(e) => e.stopPropagation()}
   316→        className="w-full sm:max-w-lg h-[94vh] sm:h-[90vh] bg-white sm:rounded-3xl rounded-t-3xl flex flex-col overflow-hidden shadow-2xl"
   317→        dir="rtl"
   318→      >
   319→        {/* Header */}
   320→        <div
   321→          className="relative flex items-center justify-between p-4 text-white shrink-0 overflow-hidden"
   322→          style={{ background: "linear-gradient(135deg, #f59e0b, #f97316)" }}
   323→        >
   324→          <div className="absolute -left-8 -top-8 w-32 h-32 rounded-full bg-white/10" />
   325→          <div className="relative flex items-center gap-2">
   326→            <div className="w-9 h-9 rounded-xl bg-white/20 flex items-center justify-center">
   327→              <Dumbbell className="w-5 h-5 text-white" />
   328→            </div>
   329→            <div>
   330→              <h2 className="font-black text-base leading-tight">پرونده ورزشی</h2>
   331→              <p className="text-[10px] opacity-80">همه‌چیز درباره بدن و اهدافت — با ویرایش سریع</p>
   332→            </div>
   333→          </div>
   334→          <Button
   335→            variant="ghost"
   336→            size="icon"
   337→            onClick={onClose}
   338→            className="relative rounded-full text-white hover:bg-white/20 hover:text-white"
   339→            aria-label="بستن پرونده ورزشی"
   340→          >
   341→            <X className="w-5 h-5" />
   342→          </Button>
   343→        </div>
   344→
   345→        {/* Content */}
   346→        <div className="flex-1 overflow-y-auto custom-scrollbar p-4 space-y-3 bg-slate-50/50">
   347→          {loading ? (
   348→            <div className="space-y-3">
   349→              <Skeleton className="h-28 rounded-2xl" />
   350→              <Skeleton className="h-40 rounded-2xl" />
   351→              <Skeleton className="h-40 rounded-2xl" />
   352→              <Skeleton className="h-40 rounded-2xl" />
   353→            </div>
   354→          ) : !p ? (
   355→            <div className="flex flex-col items-center justify-center py-20 text-center gap-3">
   356→              <div className="w-14 h-14 rounded-2xl bg-orange-100 flex items-center justify-center">
   357→                <Dumbbell className="w-7 h-7 text-orange-500" />
   358→              </div>
   359→              <p className="text-sm text-slate-500">پرونده ورزشی یافت نشد — اول آنبوردینگ را کامل کنید.</p>
   360→            </div>
   361→          ) : (
   362→            <>
   363→              {/* خلاصه بالا */}
   364→              <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }}>
   365→                <Card className="p-4 bg-white">
   366→                  {/* v15: تاریخ عضویت در فیتاپ (درخواست مالک — آنبوردینگ/پرونده کامل) */}
   367→                  {memberSinceText && (
   368→                    <div className="flex items-center justify-center gap-1.5 mb-3 pb-3 border-b border-orange-50">
   369→                      <CalendarDays className="w-3.5 h-3.5 text-orange-500" />
   370→                      <p className="text-[11px] text-orange-700 font-bold">عضو فیتاپ از {memberSinceText}</p>
   371→                    </div>
   372→                  )}
   373→                  <div className="grid grid-cols-4 gap-2 text-center">
   374→                    <SummaryChip label="سن" value={p.age ? `${toPersianDigits(p.age)}` : "—"} />
   375→                    <SummaryChip label="قد" value={p.height ? `${toPersianDigits(p.height)}` : "—"} unit="cm" />
   376→                    <SummaryChip label="وزن" value={p.weight ? `${toPersianDigits(p.weight)}` : "—"} unit="kg" />
   377→                    <SummaryChip label="هدف" value={p.goalLabel || "—"} />
   378→                  </div>
   379→                </Card>
   380→              </motion.div>
   381→
   382→              {/* ۹ سکشن پرونده */}
   383→              <Card className="p-4 space-y-4 bg-white">
   384→                {/* ۱) اطلاعات پایه */}
   385→                <Section
   386→                  title="اطلاعات پایه"
   387→                  icon={<Activity className="w-4 h-4 text-primary" />}
   388→                  editing={editingSection === "basic"}
   389→                  onEdit={() => startEditing("basic")}
   390→                  onCancel={cancelEditing}
   391→                  onSave={saveSection}
   392→                  saving={saving}
   393→                >
   394→                  {editingSection === "basic" ? (
   395→                    <div className="grid grid-cols-2 gap-2">
   396→                      <EditSelect
   397→                        label="جنسیت"
   398→                        value={editData.gender || "male"}
   399→                        onChange={(v) => setEditData({ ...editData, gender: v })}
   400→                        options={[
   401→                          { value: "male", label: "آقا" },
   402→                          { value: "female", label: "خانم" },
   403→                        ]}
   404→                      />
   405→                      <EditField label="سن (سال)" value={editData.age || ""} onChange={(v) => setEditData({ ...editData, age: v })} type="number" />
   406→                      <EditField label="قد (cm)" value={editData.height || ""} onChange={(v) => setEditData({ ...editData, height: v })} type="number" />
   407→                      <EditField label="وزن (kg)" value={editData.weight || ""} onChange={(v) => setEditData({ ...editData, weight: v })} type="number" />
   408→                      <EditField label="وزن هدف (kg)" value={editData.targetWeight || ""} onChange={(v) => setEditData({ ...editData, targetWeight: v })} type="number" />
   409→                    </div>
   410→                  ) : (
   411→                    <div className="grid grid-cols-2 gap-2">
   412→                      <InfoRow label="جنسیت" value={p.genderLabel || "—"} />
   413→                      <InfoRow label="سن" value={`${toPersianDigits(p.age)} سال`} />
   414→                      <InfoRow label="قد" value={`${toPersianDigits(p.height)} cm`} />
   415→                      <InfoRow label="وزن فعلی" value={`${toPersianDigits(p.weight)} kg`} />
   416→                      {p.targetWeight != null && <InfoRow label="وزن هدف" value={`${toPersianDigits(p.targetWeight)} kg`} />}
   417→                    </div>
   418→                  )}
   419→                </Section>
   420→
   421→                {/* ۲) هدف و فعالیت */}
   422→                <Section
   423→                  title="هدف و فعالیت"
   424→                  icon={<Target className="w-4 h-4 text-amber-500" />}
   425→                  editing={editingSection === "goal"}
   426→                  onEdit={() => startEditing("goal")}
   427→                  onCancel={cancelEditing}
   428→                  onSave={saveSection}
   429→                  saving={saving}
   430→                >
   431→                  {editingSection === "goal" ? (
   432→                    <div className="grid grid-cols-2 gap-2">
   433→                      <EditSelect
   434→                        label="هدف اصلی"
   435→                        value={editData.goal || "fitness"}
   436→                        onChange={(v) => setEditData({ ...editData, goal: v })}
   437→                        options={(Object.keys(GOAL_LABELS) as Goal[]).map((g) => ({ value: g, label: GOAL_LABELS[g] }))}
   438→                      />
   439→                      <EditSelect
   440→                        label="سطح فعالیت"
   441→                        value={editData.activityLevel || "moderate"}
   442→                        onChange={(v) => setEditData({ ...editData, activityLevel: v })}
   443→                        options={(Object.keys(ACTIVITY_LABELS) as ActivityLevel[]).map((g) => ({ value: g, label: ACTIVITY_LABELS[g] }))}
   444→                      />
   445→                      <EditField label="روزهای تمرین در هفته" value={editData.workoutDays || ""} onChange={(v) => setEditData({ ...editData, workoutDays: v })} type="number" />
   446→                      <EditSelect
   447→                        label="محیط تمرین"
   448→                        value={editData.workoutPlace || "gym"}
   449→                        onChange={(v) => setEditData({ ...editData, workoutPlace: v })}
   450→                        options={(Object.keys(WORKOUT_PLACE_LABELS) as WorkoutPlace[]).map((g) => ({ value: g, label: WORKOUT_PLACE_LABELS[g] }))}
   451→                      />
   452→                      <EditSelect
   453→                        label="زمان ترجیحی تمرین"
   454→                        value={editData.workoutTime || ""}
   455→                        onChange={(v) => setEditData({ ...editData, workoutTime: v })}
   456→                        options={[
   457→                          { value: "", label: "—" },
   458→                          ...(Object.keys(WORKOUT_TIME_LABELS) as WorkoutTime[]).map((g) => ({ value: g, label: WORKOUT_TIME_LABELS[g] })),
   459→                        ]}
   460→                      />
   461→                      <div className="col-span-2">
   462→                        <Label className="mb-1 block text-[11px] text-muted-foreground">روزهای انتخابی هفته</Label>
   463→                        <div className="flex flex-wrap gap-1.5">
   464→                          {PERSIAN_WEEKDAYS.map((day) => {
   465→                            const selected = (editData.workoutDaysList || "").split(",").map((s) => s.trim()).includes(day);
   466→                            return (
   467→                              <button
   468→                                key={day}
   469→                                type="button"
   470→                                onClick={() => {
   471→                                  const current = (editData.workoutDaysList || "").split(",").map((s) => s.trim()).filter(Boolean);

  1033→        onValueChange={(v) => onChange(fromSentinel(v))}
  1034→      >
  1035→        <SelectTrigger className="h-9 rounded-lg text-sm">
  1036→          <SelectValue placeholder="—" />
  1037→        </SelectTrigger>
  1038→        <SelectContent dir="rtl">

  1040→            <SelectItem
