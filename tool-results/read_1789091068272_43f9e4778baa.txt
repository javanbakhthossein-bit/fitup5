   106→   106→function isVisionRequest(params: Record<string, unknown>): boolean {
   107→   107→  const messages = params.messages;
   108→   108→  if (!Array.isArray(messages)) return false;
   109→   109→  return messages.some((m) => {
   110→   110→    if (!m || typeof m !== "object") return false;
   111→   111→    const content = (m as { content?: unknown }).content;
   112→   112→    if (!Array.isArray(content)) return false;
   113→   113→    return content.some(
   114→   114→      (p) =>
   115→   115→        p &&
   116→   116→        typeof p === "object" &&
   117→   117→        (p as { type?: unknown }).type === "image_url"
   118→   118→    );
   119→   119→  });
   120→   120→}

   233→   233→  const directives = getSystemDirectives();
   234→   234→  if (!systemPrompt) return directives;
   235→   235→  // چک کردن idempotency: اگر prompt با "قوانین مهم سیستم:" شروع می‌شود، یعنی قبلاً تزریق شده.

   237→   237→  // backward compatibility: اگر با دایرکتیو قدیمی برند شروع شده، کل دایرکتیو جدید را جایگزین می‌کنیم
   238→   238→  if (systemPrompt.startsWith("قانون مهم برند:")) {

   240→   240→    const withoutOldDirective = systemPrompt.replace(/^قانون مهم برند:[^\n]*\n\n/, "");
   241→   241→    return directives + withoutOldDirective;
   242→   242→  }
   243→   243→  return directives + systemPrompt;
   244→   244→}

   266→   266→/**
   267→   267→ * Proxy هوشمند برای avalaiClient.
   268→   268→ *
   269→   269→ * این Proxy تمام کال‌های `chat.completions.create` را intercept می‌کند و به‌صورت
   270→   270→ * خودکار پارامترهای مخصوص هر مدل را اضافه می‌کند:
   271→   271→ *
   272→   272→ *  - مدل‌های gemini-3.x (gemini-3.8-flash، gemini-3.5-flash و ...):
   273→   273→ *    max_tokens و temperature را به generationConfig منتقل می‌کند و
   274→   274→ *    thinkingConfig.thinkingLevel = "low" اضافه می‌کند (دیریکتیو: تفکر LOW — سرعت).
   275→   275→ *
   276→   276→ *  - مدل‌های deepseek-v4 (deepseek-v4-flash / DeepSeek-V4-Flash-0731):
   277→   277→ *    reasoning_effort = "low" اضافه می‌کند (دیریکتیو: تفکر LOW — سرعت).
   278→   278→ *    این مدل از reasoning_effort با مقادیر low | high | max پشتیبانی می‌کند، اما
   279→   279→ *    "max" باعث تفکر بیش از ~۱۰۰ ثانیه و 504 کلادفلر/تایم‌اوت می‌شد.
   280→   280→ *
   281→   281→ *  - برای مدل‌های دیگر: رفتار استاندارد OpenAI حفظ می‌شود.
   282→   282→ *
   283→   283→ * مهم: تابع `create` در OpenAI SDK به `this._client` وابسته است. اگر آن را bind
   284→   284→ * نکنیم، خطای "Cannot read properties of undefined (reading '_client')" رخ می‌دهد.
   285→   285→ * بنابراین `originalCreate` را قبل از wrap کردن به `completionsObj` bind می‌کنیم.
   286→   286→ */
   287→   287→function wrapCreateWithGemini3Support<T extends (...args: any[]) => any>(
   288→   288→  originalCreate: T,
   289→   289→  thisArg: any
   290→   290→): T {
   291→   291→  // bind کردن به thisArg (completions object) تا this._client در دسترس باشد
   292→   292→  const boundCreate = originalCreate.bind(thisArg) as T;
   293→   293→  return ((async (params: any, options?: any) => {
   294→   294→    const model: string = params?.model || TEXT_MODEL;
   295→   295→    const isGemini3 = isGemini3Model(model);
   296→   296→    const isDeepseekV4 = isDeepseekV4Model(model);
   297→   297→
   298→   298→    if (isGemini3) {
   299→   299→      // برای gemini-3.x: پارامترها به generationConfig منتقل شوند
   300→   300→      // thinkingLevel پیش‌فرض "low" است (دیریکتیو کاربر: تفکر LOW — سرعت)،
   301→   301→      // اما اگر caller مقدار دیگری (مثلاً "high" برای تولید برنامه) داده باشد،
   302→   302→      // آن مقدار محترم شمرده می‌شود و override نمی‌شود.
   303→   303→      const callerThinkingLevel = (params as any)?.extra_body?.generationConfig?.thinkingConfig?.thinkingLevel;
   304→   304→      const thinkingLevel = typeof callerThinkingLevel === "string" && callerThinkingLevel
   305→   305→        ? callerThinkingLevel
   306→   306→        : "low";
   307→   307→      const generationConfig: Record<string, unknown> = {
   308→   308→        thinkingConfig: { thinkingLevel },
   309→   309→      };
   310→   310→      if (params?.max_tokens != null) {
   311→   311→        generationConfig.maxOutputTokens = params.max_tokens;
   312→   312→      }
   313→   313→      if (params?.temperature != null) {
   314→   314→        generationConfig.temperature = params.temperature;
   315→   315→      }
   316→   316→      if (params?.top_p != null) {
   317→   317→        generationConfig.topP = params.top_p;
   318→   318→      }
   319→   319→
   320→   320→      // حذف پارامترهای top-level که gemini-3.x نمی‌پذیرد
   321→   321→      const { max_tokens, temperature, top_p, ...rest } = params;
   322→   322→
   323→   323→      // ادغام با extra_body موجود (اگر کاربر قبلاً چیزی فرستاده)
   324→   324→      const existingExtraBody = (params as any).extra_body || {};
   325→   325→      const mergedParams = {
   326→   326→        ...rest,
   327→   327→        extra_body: {
   328→   328→          ...existingExtraBody,
   329→   329→          generationConfig: {
   330→   330→            ...(existingExtraBody.generationConfig || {}),
   331→   331→            ...generationConfig,
   332→   332→          },
   333→   333→        },
   334→   334→      };
   335→   335→
   336→   336→      return boundCreate(mergedParams, options);
   337→   337→    }
   338→   338→
   339→   339→    if (isDeepseekV4) {
   340→   340→      // برای deepseek-v4-flash: reasoning_effort = "low" اضافه کن (تفکر سبک — دیریکتیو کاربر: LOW).
   341→   341→      // ⚠️ مقدار قبلی "max" باعث تفکر بیش از ~۱۰۰ ثانیه و 504 کلادفلر/تایم‌اوت ۱۰ دقیقه‌ای می‌شد.
   342→   342→      // اگر کاربر قبلاً reasoning_effort تنظیم کرده، آن را override نمی‌کنیم.
   343→   343→      if (params?.reasoning_effort == null) {
   344→   344→        return boundCreate({ ...params, reasoning_effort: "low" }, options);
   345→   345→      }
   346→   346→      return boundCreate(params, options);
   347→   347→    }
   348→   348→
   349→   349→    // برای مدل‌های دیگر: بدون تغییر
   350→   350→    return boundCreate(params, options);
   351→   351→  }) as unknown as T);
   352→   352→}

   357→   357→ * چرا: گیت‌وی api.avalai.ir (کلادفلر) برای پاسخ‌های طولانیِ تولید برنامه

   359→   359→ * (تست واقعی: پاسخ‌های ۹۷-۱۱۰ ثانیه‌ای گاهی رد می‌شوند، گاهی در ۶۳s قطع).

   372→   372→ * v39: زنجیرهٔ تفکر فقط روی «پاسخِ واقعاً قابل‌استفاده» توقف می‌کند، نه هر پاسخ 200.
   373→   373→ */
   374→   374→export type PlanContentValidator = (text: string) => string | null;

   376→   376→export async function createPlanCompletionWithRetry(
   377→   377→  params: Record<string, unknown>,
   378→   378→  logTag: string,
   379→   379→  maxAttempts = 3,
   380→   380→  opts?: {
   381→   381→    timeoutMs?: number;
   382→   382→    fallbackParams?: Record<string, unknown>;
   383→   383→    validateContent?: PlanContentValidator;
   384→   384→    userId?: string | null; // v70 حسابداری
   385→   385→  }
   386→   386→): Promise<string> {
   387→   387→  let lastErr: unknown = null;
   388→   388→  // v58 — پارامتر داخلی fallback_model قبل از ارسال حذف می‌شود (هرگز به API نمی‌رود)
   389→   389→  const apiParams: Record<string, unknown> = { ...params };
   390→   390→  delete apiParams.fallback_model;
   391→   391→  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
   392→   392→    const attemptStartedAt = Date.now(); // v70 حسابداری — اندازه‌گیری تأخیر
   393→   393→    try {
   394→   394→      const completion = await avalaiClient.chat.completions.create(apiParams as any, {
   395→   395→        maxRetries: 0, // retry دستی اینجا — دوباره‌کاری نکن
   396→   396→        ...(opts?.timeoutMs ? { timeout: opts.timeoutMs } : {}),
   397→   397→      });
   398→   398→      // ─── v39-fix: پاسخ «خالی» دیگر موفقیت حساب نمی‌شود ───
   399→   399→      // باگ ریشه‌ای تولید برنامه: AvalAI با 200 و content:null برمی‌گشت
   400→   400→      // (مدل تفکری gemini-3.8، کل بودجهٔ خروجی را با reasoning می‌خورد و
   401→   401→      // صفر توکن متن تولید می‌کرد → finish_reason=length) و کد قدیمی
   402→   402→      // `content || ""` آن را «موفق» می‌گرفت → زنجیره همان لحظه متوقف،
   403→   403→      // بدون سقوط به سطح تفکر بعدی و بدون فال‌بک مدل → «برنامه خالی».
   404→   404→      const choice: any = (completion as any)?.choices?.[0];
   405→   405→      const rawContent = choice?.message?.content;
   406→   406→      const text = typeof rawContent === "string" ? rawContent : "";
   407→   407→      if (!text.trim()) {
   408→   408→        const usage: any = (completion as any)?.usage || {};
   409→   409→        const cd = usage.completion_tokens_details || {};
   410→   410→        const diag = `finish_reason=${choice?.finish_reason || "?"} completion_tokens=${usage.completion_tokens ?? "?"} reasoning=${cd.reasoning_tokens ?? "?"} text=${cd.text_tokens ?? "?"}`;
   411→   411→        console.error(`[${logTag}] attempt ${attempt}/${maxAttempts}: EMPTY content (${diag})`);
   412→   412→        // قطعی است؛ تلاش دوباره با همین پارامترها بی‌فایده → سطح بعدی زنجیره
   413→   413→        lastErr = new Error(`پاسخ خالی از مدل هوش مصنوعی (${diag})`);
   414→   414→        break;
   415→   415→      }
   416→   416→      if (choice?.finish_reason === "length") {
   417→   417→        // متن هست ولی سقف خروجی خورده — احتمالاً JSON بریده؛ هنوز برمی‌گردانیم
   418→   418→        // چون ممکن است JSON کامل باشد و فقط توضیح انتهایی بریده شده باشد.
   419→   419→        console.warn(`[${logTag}] attempt ${attempt}: finish_reason=length (len=${text.length}) — may be truncated`);
   420→   420→      }
   421→   421→      if (opts?.validateContent) {
   422→   422→        const vErr = opts.validateContent(text);
   423→   423→        if (vErr) {
   424→   424→          console.error(
   425→   425→            `[${logTag}] attempt ${attempt}/${maxAttempts}: validation failed — ${vErr} (len=${text.length}, finish_reason=${choice?.finish_reason || "?"}, head=${text.slice(0, 160).replace(/\s+/g, " ")})`
   426→   426→          );
   427→   427→          lastErr = new Error(vErr);
   428→   428→          if (attempt === maxAttempts) break;
   429→   429→          await new Promise((r) => setTimeout(r, 2000));
   430→   430→          continue; // پاسخ نامعتبر می‌تواند تصادفی باشد → retry همان سطح
   431→   431→        }
   432→   432→      }
   433→   433→      // ─── v70 حسابداری — لاگ مصرف AI (best-effort؛ زنجیرهٔ برنامه گران‌ترین مسیر است
   434→   434→      // و تا امروز هیچ ردیفی در داشبورد حسابداری نمی‌گذاشت) ───
   435→   435→      try {
   436→   436→        const usage: any = (completion as any)?.usage;
   437→   437→        if (usage && typeof usage === "object") {
   438→   438→          const promptTokens = Number(usage.prompt_tokens ?? 0) || 0;
   439→   439→          const completionTokens = Number(usage.completion_tokens ?? 0) || 0;
   440→   440→          await logAiUsage({
   441→   441→            route: logTag.split("(fallback")[0],
   442→   442→            model: String(apiParams.model ?? TEXT_MODEL),
   443→   443→            promptTokens,
   444→   444→            completionTokens,
   445→   445→            totalTokens: Number(usage.total_tokens ?? promptTokens + completionTokens) || promptTokens + completionTokens,
   446→   446→            latencyMs: Date.now() - attemptStartedAt,
   447→   447→            userId: opts?.userId ?? null,
   448→   448→          });
   449→   449→        }
   450→   450→      } catch {
   451→   451→        // هرگز جریان اصلی را نمی‌شکند
   452→   452→      }
   453→   453→      return text;
   454→   454→    } catch (err) {
   455→   455→      lastErr = err;
   456→   456→      const msg = String((err as Error)?.message || err);
   457→   457→      const retriable = /50[234]|429|timed out|timeout|ECONNRESET|ECONNREFUSED|fetch failed|socket hang up/i.test(msg);
   458→   458→      console.error(`[${logTag}] AvalAI error (attempt ${attempt}/${maxAttempts}):`, msg.slice(0, 200));
   459→   459→      if (!retriable || attempt === maxAttempts) break;
   460→   460→      await new Promise((r) => setTimeout(r, 3000)); // backoff کوتاه
   461→   461→    }
   462→   462→  }
   463→   463→
   464→   464→  // تلاش نهایی با پارامتر جایگزین (مثلاً تفکر low) — بهتر از هیچ برنامه‌ای نیست
   465→   465→  if (opts?.fallbackParams) {
   466→   466→    console.warn(`[${logTag}] falling back to alternate params (e.g. low thinking)...`);
   467→   467→    const fbStartedAt = Date.now();
   468→   468→    try {
   469→   469→      const completion = await avalaiClient.chat.completions.create(opts.fallbackParams as any, {
   470→   470→        maxRetries: 0,
   471→   471→        ...(opts?.timeoutMs ? { timeout: opts.timeoutMs } : {}),
   472→   472→      });
   473→   473→      const choice: any = (completion as any)?.choices?.[0];
   474→   474→      const rawContent = choice?.message?.content;
   475→   475→      const text = typeof rawContent === "string" ? rawContent : "";
   476→   476→      if (text.trim() && (!opts.validateContent || !opts.validateContent(text))) {
   477→   477→        // v70 حسابداری — لاگ مصرف تلاش جایگزین
   478→   478→        try {
   479→   479→          const usage: any = (completion as any)?.usage;
   480→   480→          if (usage && typeof usage === "object") {
   481→   481→            const promptTokens = Number(usage.prompt_tokens ?? 0) || 0;
   482→   482→            const completionTokens = Number(usage.completion_tokens ?? 0) || 0;
   483→   483→            await logAiUsage({
   484→   484→              route: logTag.split("(fallback")[0],
   485→   485→              model: String((opts.fallbackParams as Record<string, unknown>).model ?? TEXT_MODEL),
   486→   486→              promptTokens,
   487→   487→              completionTokens,
   488→   488→              totalTokens: Number(usage.total_tokens ?? promptTokens + completionTokens) || promptTokens + completionTokens,
   489→   489→              latencyMs: Date.now() - fbStartedAt,
   490→   490→              userId: opts?.userId ?? null,
   491→   491→            });
   492→   492→          }
   493→   493→        } catch {}
   494→   494→        return text;
   495→   495→      }
   496→   496→      console.error(`[${logTag}] fallback attempt returned empty/invalid content (finish_reason=${choice?.finish_reason || "?"})`);
   497→   497→      lastErr = new Error("پاسخ خالی/نامعتبر از مدل (fallbackParams)");
   498→   498→    } catch (err) {
   499→   499→      lastErr = err;
   500→   500→      console.error(`[${logTag}] fallback attempt failed:`, String((err as Error)?.message || err).slice(0, 200));
   501→   501→    }
   502→   502→  }
   503→   503→
   504→   504→  console.error(`[${logTag}] all attempts failed:`, lastErr);
   505→   505→  throw lastErr instanceof Error
   506→   506→    ? lastErr
   507→   507→    : new Error("خطا در ارتباط با سرویس هوش مصنوعی. لطفاً کمی بعد دوباره تلاش کنید.");
   508→   508→}

   528→   528→const PLAN_ATTEMPT_TIMEOUT_MS = 270_000; // ۴.۵ دقیقه برای هر تلاش
   529→   529→
   530→   530→// v39-fix: بودجهٔ صریح خروجی برای تولید برنامه.
   531→   531→// بدون این، AvalAI برای gemini-3.8 یک بودجهٔ پیش‌فرض کوچک اعمال می‌کند که
   532→   532→// reasoning (تفکر max/high) به‌تنهایی آن را می‌بلعد → content:null → «برنامه خالی».
   533→   533→// تست واقعی (سپتامبر ۲۰۲۶): maxOutputTokens=65536 پذیرفته می‌شود و reasoning+JSON
   534→   534→// بزرگ (~۲۵هزار توکن) را با هم جا می‌دهد. برای gemini-3.x پروکسیِ پایین‌تر این
   535→   535→// مقدار را به generationConfig.maxOutputTokens منتقل می‌کند.
   536→   536→const PLAN_MAX_OUTPUT_TOKENS = 65536;
   537→   537→
   538→   538→/**
   539→   539→ * ساخت پارامترهای chat.completions برای تولید برنامه با سطح تفکر مشخص.
   540→   540→ * v40: پارامتر model اضافه شد — فال‌بک deepseek-v4-flash هم با همین سازنده
   541→   541→ * ساخته می‌شود تا reasoning_effort=high بگیرد (پروکسیِ پایین‌تر مقدار non-null
   542→   542→ * را override نمی‌کند) و extra_bodyِ تفکرِ gemini فقط برای gemini-3.x ست شود.
   543→   543→ */
   544→   544→function buildPlanParams(
   545→   545→  systemPrompt: string,
   546→   546→  userPrompt: string,
   547→   547→  thinkingLevel: string,
   548→   548→  model: string = TEXT_MODEL
   549→   549→): Record<string, unknown> {
   550→   550→  return {
