   298→// ═══ ردیابی نوشته‌های در-پرواز /api/nutrition/log (مهار رقابت‌های خوش‌بینانه) ═══
   299→//  ۱) GET زمان mount با POST در-پرواز: جایگزینی کل لیست، ورودی تازه‌اضافه‌شده
   300→//     را پاک می‌کرد (غذا بدون reload ناپدید می‌شد) → تا settle شدن نوشته‌ها
   301→//     جایگزینی نمی‌کنیم و بعد از آن دوباره fetch می‌کنیم (loadTodayFoodLogs).
   302→//  ۲) حذفِ آیتمی که POST آن هنوز برنگشته: DELETE سمت سرور skip می‌شد و ردیف
   303→//     به‌صورت «شبح» در بارگذاری بعدی برمی‌گشت → DELETE را بعد از resolve شدن
   304→//     POST با id واقعی سرور اجرا می‌کنیم (addLoggedFood).
   305→const pendingFoodWrites = new Set<Promise<void>>();
   306→// tempIdهایی (food_*) که قبل از resolve شدن POST حذف شده‌اند (DELETE معلق)
   307→const deletedFoodTemps = new Set<string>();
   308→
   309→export const useAppStore = create<AppState>((set, get) => ({
   310→  // همیشه "loading" شروع می‌شود تا server و client هماهنگ باشند (جلوگیری از hydration mismatch).
   311→  // URL parsing در useEffect مربوط به page-client انجام می‌شود (applyUrlToScreen).
   312→  // این الگو مطابق مستندات Next.js برای hydration-safe state است.
   313→  screen: "loading",
   314→  // 🩹 v45: شمارندهٔ session برای باطل‌کردن پاسخ‌های async بعد از خروج
   315→  sessionEpoch: 0,
   316→  user: null,
   317→  setUser: (u) => set({ user: u }),
   318→
   319→  setScreen: (s) => set({ screen: s }),
   320→
   321→  // New-terms modal flag
   322→  termsUpdateRequired: false,
   323→  setTermsUpdateRequired: (v) => set({ termsUpdateRequired: v }),
   324→  articleSlug: null,
   325→  setArticleSlug: (s) => set({ articleSlug: s }),
   326→  exerciseId: null,
   327→  setExerciseId: (s) => set({ exerciseId: s }),
   328→  foodId: null,
   329→  setFoodId: (s) => set({ foodId: s }),
   330→
   331→  mainTab: "dashboard",
   332→  setMainTab: (t) => set({ mainTab: t }),
   333→
   334→  chatMode: "coach",
   335→  setChatMode: (m) => set({ chatMode: m }),
   336→  nikaMessages: [],
   337→  setNikaMessages: (m) => set({ nikaMessages: m }),
   338→  addNikaMessage: (m) => set((s) => ({ nikaMessages: [...s.nikaMessages, m] })),
   339→
   340→  // Daily tracking defaults
   341→  waterMl: 0,
   342→  addWater: (ml) => set((s) => ({ waterMl: Math.max(0, Math.min(5000, s.waterMl + ml)) })),
   343→  caloriesConsumed: 0,
   344→  caloriesBurned: 0,
   345→  setCaloriesConsumed: (n) => set({ caloriesConsumed: Math.max(0, n) }),
   346→  setCaloriesBurned: (n) => set({ caloriesBurned: Math.max(0, n) }),
   347→  loggedFoods: [],
   348→  addLoggedFood: (f) => {
   349→    const tempId = f.id;
   350→    // به‌روزرسانی خوش‌بینانه‌ی محلی (UI فوراً واکنش نشان می‌دهد)
   351→    set((s) => ({
   352→      loggedFoods: [...s.loggedFoods, f],
   353→      caloriesConsumed: s.caloriesConsumed + f.calories,
   354→    }));
   355→    // ذخیره در سرور (background, non-blocking) با keepalive برای ادامه در پس‌زمینه
   356→    const write = (async () => {
   357→      try {
   358→        const r = await fetch("/api/nutrition/log", {
   359→          method: "POST",
   360→          headers: { "Content-Type": "application/json" },
   361→          body: JSON.stringify({
   362→            name: f.name,
   363→            meal: f.meal,
   364→            calories: f.calories,
   365→            protein: f.protein,
   366→            carbs: f.carbs,
   367→            fat: f.fat,
   368→            servingSize: f.servingSize ?? "۱ وعده",
   369→            source: f.source ?? "manual",
   370→            foodLibraryId: f.foodLibraryId ?? null,
   371→            imageUrl: f.imageUrl ?? null,
   372→          }),
   373→          keepalive: true,
   374→        });
   375→        if (!r.ok) {
   376→          // ردیفی در سرور ساخته نشده — DELETE معلق (اگر بود) لازم نیست
   377→          deletedFoodTemps.delete(tempId);
   378→          return;
   379→        }
   380→        const data = await r.json().catch(() => null);
   381→        // جایگزینی id موقت با id سرور (تا حذف بعدی درست کار کند)
   382→        if (data?.food?.id && data.food.id !== tempId) {
   383→          const serverId = data.food.id as string;
   384→          set((s) => ({
   385→            loggedFoods: s.loggedFoods.map((x) =>
   386→              x.id === tempId ? { ...x, id: serverId } : x
   387→            ),
   388→          }));
   389→          // کاربر همین‌حین (قبل از resolve شدن POST) حذف کرده بود → DELETE
   390→          // معلق را با id واقعی سرور اجرا کن تا ردیف شبح برنگردد
   391→          if (deletedFoodTemps.has(tempId)) {
   392→            deletedFoodTemps.delete(tempId);
   393→            const del = (async () => {
   394→              try {
   395→                await fetch(`/api/nutrition/log/${serverId}`, { method: "DELETE", keepalive: true });
   396→              } catch {}
   397→            })();
   398→            pendingFoodWrites.add(del);
   399→            void del.finally(() => pendingFoodWrites.delete(del));
   400→          }
   401→        }
   402→      } catch {
   403→        // سکوت — state محلی حفظ می‌شود تا کاربر داده را از دست ندهد
   404→        // در بارگذاری بعدی صفحه، با سرور reconcile می‌شود
   405→        deletedFoodTemps.delete(tempId);
   406→      }
   407→    })();
   408→    pendingFoodWrites.add(write);
   409→    void write.finally(() => pendingFoodWrites.delete(write));
   410→  },
   411→  removeLoggedFood: (id) => {
   412→    set((s) => {
   413→      const food = s.loggedFoods.find((x) => x.id === id);
   414→      return {
   415→        loggedFoods: s.loggedFoods.filter((x) => x.id !== id),
   416→        caloriesConsumed: Math.max(0, s.caloriesConsumed - (food?.calories ?? 0)),
   417→      };
   418→    });
   419→    // حذف از سرور فقط اگر id موقت نباشد (id موقت با "food_" شروع می‌شود)
   420→    if (id.startsWith("food_")) {
   421→      // POST این آیتم هنوز resolve نشده — DELETE فعلاً ممکن نیست؛ علامت می‌زنیم
   422→      // تا بعد از resolve شدن POST با id واقعی سرور حذف شود (addLoggedFood)
   423→      deletedFoodTemps.add(id);
   424→      return;
   425→    }
   426→    const del = (async () => {
   427→      try {
   428→        await fetch(`/api/nutrition/log/${id}`, { method: "DELETE", keepalive: true });
   429→      } catch {}
   430→    })();
   431→    pendingFoodWrites.add(del);
   432→    void del.finally(() => pendingFoodWrites.delete(del));
   433→  },
   434→  loadTodayFoodLogs: async () => {
   435→    try {
   436→      // اگر نوشته‌ای (POST ثبت یا DELETE) هنوز در پرواز است، جایگزینی کل لیست
   437→      // می‌تواند ورودی خوش‌بینانه را پاک کند یا ردیف حذف‌شده را برگرداند —
   438→      // این بار جایگزینی نمی‌کنیم؛ بعد از settle شدن نوشته‌ها دوباره fetch می‌کنیم.

   624→//  Persist جلسه تمرین فعال در localStorage (FE-H7)
   625→//  ست‌های ثبت‌شده قبلاً با refresh/endSession از بین می‌رفتند؛
   626→//  حالا با هر تغییر activeSession (start/logSet/endSession/تغییر حرکت)
   627→//  در localStorage ذخیره می‌شود و بعد از refresh (تا ۲۴ ساعت) بازیابی می‌گردد.
   628→// ═══════════════════════════════════════════════════════════════
   629→const ACTIVE_SESSION_KEY = "fitup_active_session";
   630→const ACTIVE_SESSION_MAX_AGE_MS = 24 * 60 * 60 * 1000; // ۲۴ ساعت
   631→
   632→function persistActiveSession(session: ActiveSession | null): void {
   633→  if (typeof window === "undefined") return;
   634→  try {
   635→    if (!session) {
   636→      window.localStorage.removeItem(ACTIVE_SESSION_KEY);
   637→    } else {
   638→      window.localStorage.setItem(
   639→        ACTIVE_SESSION_KEY,
   640→        JSON.stringify({ activeSession: session, savedAt: Date.now() })
   641→      );
   642→    }
   643→  } catch {
   644→    // localStorage پر است یا غیرفعال — بی‌صدا رد شو
   645→  }
   646→}
   647→
   648→// subscribe همه مسیرهای تغییر activeSession را پوشش می‌دهد
   649→// (startSession / logSet / endSession / setState مستقیم در active-workout-session)
   650→// t8c-5: نوشتن با تأخیر trailing ~۵۰۰ms — هر تیک ناوبری/تایپ ست‌ها دیگر
   651→// localStorage synchronous write نمی‌زند (عامل لگ ناوبری در پلیر).Flush فوری
   652→// روی pagehide/beforeunload تا بستن تب/اپ وسط تایمر داده گم نکند.
   653→if (typeof window !== "undefined") {
   654→  let persistTimer: ReturnType<typeof setTimeout> | null = null;
   655→
   656→  const schedulePersistActiveSession = () => {
   657→    if (persistTimer) clearTimeout(persistTimer);
   658→    persistTimer = setTimeout(() => {
   659→      persistTimer = null;
   660→      persistActiveSession(useAppStore.getState().activeSession);
   661→    }, 500);
   662→  };
   663→
   664→  const flushActiveSession = () => {
   665→    if (persistTimer) {
   666→      clearTimeout(persistTimer);
   667→      persistTimer = null;
   668→    }
   669→    persistActiveSession(useAppStore.getState().activeSession);
   670→  };
   671→
   672→  useAppStore.subscribe((state, prev) => {
   673→    if (state.activeSession !== prev.activeSession) {
   674→      schedulePersistActiveSession();
   675→    }
   676→  });
   677→
   678→  window.addEventListener("pagehide", flushActiveSession);
   679→  window.addEventListener("beforeunload", flushActiveSession);
   680→}

   682→/**
   683→ * بازیابی جلسه تمرین فعال از localStorage — فقط سمت client.
   684→ * در mount پنل (MainApp) صدا زده می‌شود؛ اگر جلسه‌ای <۲۴ ساعت وجود داشت
   685→ * و store خالی بود، بازیابی می‌کند تا پیشرفت کاربر با refresh از دست نرود.
   686→ */
   687→export function restoreActiveSession(): void {
   688→  if (typeof window === "undefined") return;
   689→  try {
   690→    const raw = window.localStorage.getItem(ACTIVE_SESSION_KEY);
   691→    if (!raw) return;
   692→    const parsed = JSON.parse(raw) as {
   693→      activeSession?: ActiveSession | null;
   694→      savedAt?: number;
   695→    };
   696→    const savedAt = typeof parsed.savedAt === "number" ? parsed.savedAt : 0;
   697→    const fresh = Date.now() - savedAt < ACTIVE_SESSION_MAX_AGE_MS;
   698→    if (!parsed.activeSession || !fresh) {
   699→      window.localStorage.removeItem(ACTIVE_SESSION_KEY);
   700→      return;
   701→    }
   702→    // فقط وقتی store فعلی جلسه‌ای ندارد (که جلسه جدید شروع‌شده را خراب نکند)
   703→    if (!useAppStore.getState().activeSession) {
   704→      // ─── اعتبارسنجی جلسه در برابر برنامه فعلی (اگر لود شده باشد) ───
   705→      // اگر برنامه regenerate شده باشد (روز حذف/حرکات کمتر)، ایندکس یا
   706→      // روز restore‌شده می‌تواند خارج از محدوده باشد → crash loop رندر.
   707→      // روز نامعتبر → جلسه پاک؛ ایندکس بزرگ → clamp به آخرین حرکت.
   708→      // (اگر برنامه هنوز لود نشده، ویو خودش ایندکس را clamp می‌کند)
   709→      const plan = useAppStore.getState().workoutPlan;
   710→      let session = parsed.activeSession;
   711→      if (plan && Array.isArray(plan.days) && plan.days.length > 0) {
   712→        const day = plan.days.find((d) => d.day === session.dayId);
   713→        const exCount = day?.exercises?.length ?? 0;
   714→        if (!day || exCount === 0) {
   715→          // روز جلسه دیگر در برنامه وجود ندارد — جلسه بی‌اعتبار است
   716→          window.localStorage.removeItem(ACTIVE_SESSION_KEY);
   717→          return;
   718→        }
   719→        session = {
   720→          ...session,
   721→          currentExerciseIdx: Math.min(Math.max(0, session.currentExerciseIdx ?? 0), exCount - 1),
   722→        };
   723→      }
   724→      useAppStore.setState({ activeSession: session });
   725→    }
   726→  } catch {
   727→    // داده خراب — کلید را پاک کن
   728→    try {
   729→      window.localStorage.removeItem(ACTIVE_SESSION_KEY);
   730→    } catch {}
   731→  }
   732→}

   758→export function savePlanCache(): void {
   759→  if (typeof window === "undefined") return;

   761→    const { user, workoutPlan, mealPlan } = useAppStore.getState();
   762→    if (!user?.id) return; // کاربر لاگین نیست — چیزی برای ذخیره نیست

   780→      workoutPlan,
   781→      mealPlan,

   787→  }

   861→//  v48 — به‌روزرسانی زندهٔ اطلاعات کاربر (رفع باگ بحرانی مالک)
   862→//  «اپ/سایت را باز می‌کنم، صفحهٔ چت می‌آید و می‌نویسد چت با فیتاپ قفل است
   863→//   با اینکه پلن حرفه‌ای دارم — باید یک بار رفرش کنم تا اطلاعات برگرده.»
   864→//
   865→//  ریشه: در اپ‌های نیتیو (WebView زنده) و PWA، اپ ممکن است ساعت‌ها باز بماند؛
   866→//  اطلاعات کاربر در store فقط هنگام لود کامل صفحه از /api/auth/me خوانده
   867→//  می‌شود. اگر session قدیمی/ناقص باشد (شبکهٔ ضعیف موقع باز شدن، session
   868→//  restore اپ، تغییر پلن در دستگاه دیگر و…)، همهٔ گیت‌های پلن (چت/باشگاه/…)
   869→//  بر اساس دادهٔ کهنه «قفل» نشان می‌دهند تا کاربر دستی رفرش کند.
   870→//
   871→//  راه‌حل: refreshUserIfNeeded — هر بار اپ به foreground برمی‌گردد (و در
   872→//  mount)، اطلاعات کاربر بی‌صدا از سرور تازه می‌شود؛ همهٔ گیت‌ها همان لحظه
   873→//  زنده می‌شوند. throttle ۶۰ ثانیه‌ای + گارد sessionEpoch (ضد پاسخِ کهنه
   874→//  بعد از logout — الگوی v45).
   875→// ═══════════════════════════════════════════════════════════════
   876→let _lastUserRefreshAt = 0;
   877→let _refreshInFlight = false;
   878→
   879→export async function refreshUserIfNeeded(force = false): Promise<void> {
   880→  if (typeof window === "undefined") return;
   881→  if (_refreshInFlight) return;
   882→  const st = useAppStore.getState();
   883→  // کاربر لاگین نیست → کاری نیست (auth screen خودش flow ورود را دارد)
   884→  if (!st.user) {
   885→    if (!force) return;
   886→  }
   887→  const nowMs = Date.now();
   888→  if (!force && nowMs - _lastUserRefreshAt < 60_000) return; // throttle ۶۰s
   889→  _lastUserRefreshAt = nowMs;
   890→  _refreshInFlight = true;
   891→  const epochAtStart = useAppStore.getState().sessionEpoch;
   892→  try {
   893→    const res = await fetch("/api/auth/me", { cache: "no-store" });
   894→    if (!res.ok) return;
   895→    if (useAppStore.getState().sessionEpoch !== epochAtStart) return; // وسطش logout شد
   896→    const data = await res.json();
   897→    if (useAppStore.getState().sessionEpoch !== epochAtStart) return;
   898→    // v51 (مسدودسازی): اگر کاربر وسط سشن توسط ادمین مسدود شود، همان لحظه که
   899→    // اپ به foreground برمی‌گردد صفحهٔ زیبای «مسدود» جای کل پنل را می‌گیرد.
   900→    if (data?.blocked && !data?.user) {
   901→      useAppStore.getState().setUser(null);
   902→      useAppStore.getState().setScreen("blocked");
   903→      return;
   904→    }
   905→    if (data?.user) {
   906→      const cur = useAppStore.getState().user;
   907→      // فقط اگر چیزی عوض شده setUser کن — از re-render بی‌مورد سراسری جلوگیری می‌کند
   908→      const changed =
   909→        !cur ||
   910→        cur.planName !== data.user.planName ||
   911→        cur.subscriptionEnd !== data.user.subscriptionEnd ||
   912→        cur.hasActiveSubscription !== data.user.hasActiveSubscription ||
   913→        cur.hasPendingSubscription !== data.user.hasPendingSubscription ||
   914→        cur.avatarUrl !== data.user.avatarUrl ||
   915→        cur.name !== data.user.name ||
   916→        cur.walletBalance !== data.user.walletBalance ||
   917→        cur.videoAnalysisUsed !== data.user.videoAnalysisUsed ||
   918→        cur.bloodTestUsed !== data.user.bloodTestUsed ||
   919→        cur.lastPlanExpiresAt !== data.user.lastPlanExpiresAt;
   920→      if (changed) useAppStore.getState().setUser(data.user);
   921→    }
   922→  } catch {
   923→    // شبکه در دسترس نیست — دفعهٔ بعد foreground دوباره تلاش می‌شود
   924→  } finally {
   925→    _refreshInFlight = false;
   926→  }
   927→}
