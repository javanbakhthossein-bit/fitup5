    79→// Macro ring — circular progress for protein/carbs/fat (pure SVG, no deps)
    80→// ─────────────────────────────────────────────────────────────────────────────
    81→function MacroRing({
    82→  label,
    83→  grams,
    84→  calories,
    85→  color,
    86→  icon,
    87→}: {
    88→  label: string;
    89→  grams: number;
    90→  calories: number;
    91→  color: string;
    92→  icon: React.ReactNode;
    93→}) {
    94→  const size = 84;
    95→  const stroke = 7;
    96→  const radius = (size - stroke) / 2;
    97→  const circumference = 2 * Math.PI * radius;
    98→  // Each macro is shown as a fraction of its calorie contribution vs the
    99→  // other two. This gives a sense of proportion rather than a fixed max.
   100→  // We just render a fully-stroked ring with the grams label inside.
   101→  const fillFraction = 0.78; // visual fill — purely aesthetic
   102→
   103→  return (
   104→    <div className="flex flex-col items-center gap-1.5">
   105→      <div className="relative" style={{ width: size, height: size }}>
   106→        <svg width={size} height={size} className="-rotate-90">
   107→          <circle
   108→            cx={size / 2}
   109→            cy={size / 2}
   110→            r={radius}
   111→            fill="none"
   112→            stroke="#f1f5f9"
   113→            strokeWidth={stroke}
   114→          />
   115→          <motion.circle
   116→            cx={size / 2}
   117→            cy={size / 2}
   118→            r={radius}
   119→            fill="none"
   120→            stroke={color}
   121→            strokeWidth={stroke}
   122→            strokeLinecap="round"
   123→            strokeDasharray={circumference}
   124→            initial={{ strokeDashoffset: circumference }}
   125→            animate={{ strokeDashoffset: circumference * (1 - fillFraction) }}
   126→            transition={{ duration: 1, ease: "easeOut" }}
   127→          />
   128→        </svg>
   129→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   130→          <span className="text-slate-400">{icon}</span>
   131→          <span className="text-base font-black font-stat text-slate-900 leading-none">
   132→            {toPersianDigits(grams)}
   133→          </span>
   134→          <span className="text-[9px] text-slate-400">گرم</span>
   135→        </div>
   136→      </div>
   137→      <p className="text-xs font-bold text-slate-700">{label}</p>
   138→      <p className="text-[10px] text-slate-400">{toPersianDigits(calories)} کیلوکالری</p>
   139→    </div>
   140→  );
   141→}

   323→  // ریشه: تولید تحلیل همگام داخل GET است (چند ثانیه تا چند دقیقه) و WebView
   324→  // اینستا fetchهای طولانی را قطع می‌کند؛ قبلاً فقط «یک» fetch بی‌retry بود و

   328→  // دقیقه صبر) — سمت سرور هم همهٔ درخواست‌های هم‌زمان به «یک» تولید در-جریان
   329→  // می‌پیوندند (قفل مشترک) پس هر retry هزینهٔ AI اضافه ندارد و به‌محض

   331→  const ANALYSIS_MAX_ATTEMPTS = 10;

   334→  async function fetchAnalysisOnce(url: string): Promise<{ ok: boolean; permanentFail: boolean; json: any }> {
   335→    const controller = new AbortController();
   336→    const timer = setTimeout(() => controller.abort(), ANALYSIS_ATTEMPT_TIMEOUT_MS);
   337→    try {
   338→      const res = await fetch(url, { signal: controller.signal });
   339→      const json = await res.json().catch(() => null);
   340→      if (res.ok && json?.analysis) return { ok: true, permanentFail: false, json };
   341→      // خطاهای دائمی — retry بی‌فایده است (پروفایل نبود / سشن نیست)
   342→      if (res.status === 400 || res.status === 401) {
   343→        return { ok: false, permanentFail: true, json };
   344→      }
   345→      return { ok: false, permanentFail: false, json };
   346→    } catch {
   347→      // abort / network — قابل‌retry
   348→      return { ok: false, permanentFail: false, json: null };
   349→    } finally {
   350→      clearTimeout(timer);
   351→    }
   352→  }

   354→  async function loadAnalysis(force = false) {
   355→    if (force) setRefreshing(true);
   356→    try {
   357→      const url = force ? "/api/onboarding/analysis?force=1" : "/api/onboarding/analysis";
   358→      for (let attempt = 1; attempt <= ANALYSIS_MAX_ATTEMPTS; attempt++) {
   359→        const { ok, permanentFail, json } = await fetchAnalysisOnce(url);
   360→        if (ok) {
   361→          setData(json as AnalysisResponse);
   362→          return;
   363→        }
   364→        if (permanentFail) break;
   365→        // مکث بین تلاش‌ها (backoff ملایم) — در طول retryها صفحهٔ «در حال تحلیل»
   366→        // جذاب نشان داده می‌شود، نه کارت خطا
   367→        if (attempt < ANALYSIS_MAX_ATTEMPTS) {
   368→          await new Promise((r) => setTimeout(r, attempt <= 4 ? 2_500 : 6_000));
   369→        }
   370→      }
   371→      if (force) toast.error("خطا در بازخوانی تحلیل");
   372→    } finally {
   373→      setLoading(false);
   374→      setRefreshing(false);
   375→    }
   376→  }
