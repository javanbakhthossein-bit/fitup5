   105→/** تشخیص درخواست ویژن — آیا بعضی پیام‌ها شامل تصویر (image_url) هستند؟ */
   106→function isVisionRequest(params: Record<string, unknown>): boolean {
   107→  const messages = params.messages;
   108→  if (!Array.isArray(messages)) return false;
   109→  return messages.some((m) => {
   110→    if (!m || typeof m !== "object") return false;
   111→    const content = (m as { content?: unknown }).content;
   112→    if (!Array.isArray(content)) return false;
   113→    return content.some(
   114→      (p) =>
   115→        p &&
   116→        typeof p === "object" &&
   117→        (p as { type?: unknown }).type === "image_url"
   118→    );
   119→  });
   120→}

   167→function gregorianToJalaliYear(gYear: number): number {

   224→/**
   225→ * تزریق دایرکتیو سیستم (برند + سال جاری) به ابتدای یک system prompt.
   226→ * اگر prompt قبلاً تزریق شده باشد، دوباره تزریق نمی‌کند.
   227→ *
   228→ * این تابع تمام دایرکتیوهای سیستمی را شامل می‌شود:
   229→ *  - قانون برند (فیتاپ / FitUp)
   230→ *  - قانون سال جاری (میلادی و شمسی)
   231→ */
   232→export function withSystemDirectives(systemPrompt: string): string {
   233→  const directives = getSystemDirectives();
   234→  if (!systemPrompt) return directives;
   235→  // چک کردن idempotency: اگر prompt با "قوانین مهم سیستم:" شروع می‌شود، یعنی قبلاً تزریق شده.
   236→  if (systemPrompt.startsWith("قوانین مهم سیستم:")) return systemPrompt;
   237→  // backward compatibility: اگر با دایرکتیو قدیمی برند شروع شده، کل دایرکتیو جدید را جایگزین می‌کنیم
   238→  if (systemPrompt.startsWith("قانون مهم برند:")) {
   239→    // حذف دایرکتیو قدیمی و تزریق جدید
   240→    const withoutOldDirective = systemPrompt.replace(/^قانون مهم برند:[^\n]*\n\n/, "");
   241→    return directives + withoutOldDirective;
   242→  }
   243→  return directives + systemPrompt;
   244→}

   266→/**
   267→ * Proxy هوشمند برای avalaiClient.
   268→ *
   269→ * این Proxy تمام کال‌های `chat.completions.create` را intercept می‌کند و به‌صورت
   270→ * خودکار پارامترهای مخصوص هر مدل را اضافه می‌کند:
   271→ *
   272→ *  - مدل‌های gemini-3.x (gemini-3.8-flash، gemini-3.5-flash و ...):
   273→ *    max_tokens و temperature را به generationConfig منتقل می‌کند و
   274→ *    thinkingConfig.thinkingLevel = "low" اضافه می‌کند (دیریکتیو: تفکر LOW — سرعت).
   275→ *
   276→ *  - مدل‌های deepseek-v4 (deepseek-v4-flash / DeepSeek-V4-Flash-0731):
   277→ *    reasoning_effort = "low" اضافه می‌کند (دیریکتیو: تفکر LOW — سرعت).
   278→ *    این مدل از reasoning_effort با مقادیر low | high | max پشتیبانی می‌کند، اما
   279→ *    "max" باعث تفکر بیش از ~۱۰۰ ثانیه و 504 کلادفلر/تایم‌اوت می‌شد.
   280→ *
   281→ *  - برای مدل‌های دیگر: رفتار استاندارد OpenAI حفظ می‌شود.
   282→ *
   283→ * مهم: تابع `create` در OpenAI SDK به `this._client` وابسته است. اگر آن را bind
   284→ * نکنیم، خطای "Cannot read properties of undefined (reading '_client')" رخ می‌دهد.
   285→ * بنابراین `originalCreate` را قبل از wrap کردن به `completionsObj` bind می‌کنیم.
   286→ */
   287→function wrapCreateWithGemini3Support<T extends (...args: any[]) => any>(
   288→  originalCreate: T,
   289→  thisArg: any
   290→): T {
   291→  // bind کردن به thisArg (completions object) تا this._client در دسترس باشد
   292→  const boundCreate = originalCreate.bind(thisArg) as T;
   293→  return ((async (params: any, options?: any) => {
   294→    const model: string = params?.model || TEXT_MODEL;
   295→    const isGemini3 = isGemini3Model(model);
   296→    const isDeepseekV4 = isDeepseekV4Model(model);
   297→
   298→    if (isGemini3) {
   299→      // برای gemini-3.x: پارامترها به generationConfig منتقل شوند
   300→      // thinkingLevel پیش‌فرض "low" است (دیریکتیو کاربر: تفکر LOW — سرعت)،
   301→      // اما اگر caller مقدار دیگری (مثلاً "high" برای تولید برنامه) داده باشد،
   302→      // آن مقدار محترم شمرده می‌شود و override نمی‌شود.
   303→      const callerThinkingLevel = (params as any)?.extra_body?.generationConfig?.thinkingConfig?.thinkingLevel;
   304→      const thinkingLevel = typeof callerThinkingLevel === "string" && callerThinkingLevel
   305→        ? callerThinkingLevel
   306→        : "low";
   307→      const generationConfig: Record<string, unknown> = {
   308→        thinkingConfig: { thinkingLevel },
   309→      };
   310→      if (params?.max_tokens != null) {
   311→        generationConfig.maxOutputTokens = params.max_tokens;
   312→      }
   313→      if (params?.temperature != null) {
   314→        generationConfig.temperature = params.temperature;
   315→      }
   316→      if (params?.top_p != null) {
   317→        generationConfig.topP = params.top_p;
   318→      }
   319→
   320→      // حذف پارامترهای top-level که gemini-3.x نمی‌پذیرد
   321→      const { max_tokens, temperature, top_p, ...rest } = params;
   322→
   323→      // ادغام با extra_body موجود (اگر کاربر قبلاً چیزی فرستاده)
   324→      const existingExtraBody = (params as any).extra_body || {};
   325→      const mergedParams = {
   326→        ...rest,
   327→        extra_body: {
   328→          ...existingExtraBody,
   329→          generationConfig: {
   330→            ...(existingExtraBody.generationConfig || {}),
   331→            ...generationConfig,
   332→          },
   333→        },
   334→      };
   335→
   336→      return boundCreate(mergedParams, options);
   337→    }
   338→
   339→    if (isDeepseekV4) {
   340→      // برای deepseek-v4-flash: reasoning_effort = "low" اضافه کن (تفکر سبک — دیریکتیو کاربر: LOW).
   341→      // ⚠️ مقدار قبلی "max" باعث تفکر بیش از ~۱۰۰ ثانیه و 504 کلادفلر/تایم‌اوت ۱۰ دقیقه‌ای می‌شد.
   342→      // اگر کاربر قبلاً reasoning_effort تنظیم کرده، آن را override نمی‌کنیم.
   343→      if (params?.reasoning_effort == null) {
   344→        return boundCreate({ ...params, reasoning_effort: "low" }, options);
   345→      }
   346→      return boundCreate(params, options);
   347→    }
   348→
   349→    // برای مدل‌های دیگر: بدون تغییر
   350→    return boundCreate(params, options);
   351→  }) as unknown as T);
   352→}

   354→/**
   355→ * فراخوانی LLM با retry دستی برای تولید برنامه‌های تمرینی/غذایی.
   356→ *
   357→ * چرا: گیت‌وی api.avalai.ir (کلادفلر) برای پاسخ‌های طولانیِ تولید برنامه
   358→ * (خروجی JSON بزرگ ≈ ۳۰KB) به‌صورت ناپایدار 504 Gateway Timeout می‌دهد
   359→ * (تست واقعی: پاسخ‌های ۹۷-۱۱۰ ثانیه‌ای گاهی رد می‌شوند، گاهی در ۶۳s قطع).
   360→ * SDK خودش maxRetries دارد ولی خطاهای HTML-دار 504 را همیشه retry نمی‌کند؛
   361→ * اینجا تا ۳ تلاش با backoff کوتاه انجام می‌شود (فقط برای خطاهای گذرا).
   362→ *
   363→ * planTimeoutMs: تولید برنامه در پس‌زمینه انجام می‌شود (نه داخل request کاربر)
   364→ * پس می‌توانیم timeout بلندتری گذاشت — تفکر high ممکن است ۲-۴ دقیقه طول بکشد.
   365→ *
   366→ * fallbackParams: اگر همه تلاش‌ها با پارامتر اصلی (مثلاً تفکر high) شکست خوردند،
   367→ * یک تلاش نهایی با پارامتر جایگزین (تفکر low) انجام می‌شود — کیفیت در اولویت
   368→ * است ولی کاربر هرگز بدون برنامه نمی‌ماند.
   369→ */
   370→/**
   371→ * اعتبارسنج محتوای پاسخ — null/رشتهٔ خالی یعنی OK، رشتهٔ فارسی یعنی توضیح خطا.
   372→ * v39: زنجیرهٔ تفکر فقط روی «پاسخِ واقعاً قابل‌استفاده» توقف می‌کند، نه هر پاسخ 200.
   373→ */
   374→export type PlanContentValidator = (text: string) => string | null;
   375→
   376→export async function createPlanCompletionWithRetry(
   377→  params: Record<string, unknown>,
   378→  logTag: string,
   379→  maxAttempts = 3,
   380→  opts?: {
   381→    timeoutMs?: number;
   382→    fallbackParams?: Record<string, unknown>;
   383→    validateContent?: PlanContentValidator;
   384→  }
   385→): Promise<string> {
   386→  let lastErr: unknown = null;
   387→  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
   388→    try {
   389→      const completion = await avalaiClient.chat.completions.create(params as any, {
   390→        maxRetries: 0, // retry دستی اینجا — دوباره‌کاری نکن
   391→        ...(opts?.timeoutMs ? { timeout: opts.timeoutMs } : {}),
   392→      });
   393→      // ─── v39-fix: پاسخ «خالی» دیگر موفقیت حساب نمی‌شود ───
   394→      // باگ ریشه‌ای تولید برنامه: AvalAI با 200 و content:null برمی‌گشت
   395→      // (مدل تفکری gemini-3.8، کل بودجهٔ خروجی را با reasoning می‌خورد و
   396→      // صفر توکن متن تولید می‌کرد → finish_reason=length) و کد قدیمی
   397→      // `content || ""` آن را «موفق» می‌گرفت → زنجیره همان لحظه متوقف،
   398→      // بدون سقوط به سطح تفکر بعدی و بدون فال‌بک مدل → «برنامه خالی».
   399→      const choice: any = (completion as any)?.choices?.[0];
   400→      const rawContent = choice?.message?.content;
   401→      const text = typeof rawContent === "string" ? rawContent : "";
   402→      if (!text.trim()) {
   403→        const usage: any = (completion as any)?.usage || {};
   404→        const cd = usage.completion_tokens_details || {};
   405→        const diag = `finish_reason=${choice?.finish_reason || "?"} completion_tokens=${usage.completion_tokens ?? "?"} reasoning=${cd.reasoning_tokens ?? "?"} text=${cd.text_tokens ?? "?"}`;
   406→        console.error(`[${logTag}] attempt ${attempt}/${maxAttempts}: EMPTY content (${diag})`);
   407→        // قطعی است؛ تلاش دوباره با همین پارامترها بی‌فایده → سطح بعدی زنجیره
   408→        lastErr = new Error(`پاسخ خالی از مدل هوش مصنوعی (${diag})`);
   409→        break;
   410→      }
   411→      if (choice?.finish_reason === "length") {
   412→        // متن هست ولی سقف خروجی خورده — احتمالاً JSON بریده؛ هنوز برمی‌گردانیم
   413→        // چون ممکن است JSON کامل باشد و فقط توضیح انتهایی بریده شده باشد.
   414→        console.warn(`[${logTag}] attempt ${attempt}: finish_reason=length (len=${text.length}) — may be truncated`);
   415→      }
   416→      if (opts?.validateContent) {
   417→        const vErr = opts.validateContent(text);
   418→        if (vErr) {
   419→          console.error(
   420→            `[${logTag}] attempt ${attempt}/${maxAttempts}: validation failed — ${vErr} (len=${text.length}, finish_reason=${choice?.finish_reason || "?"}, head=${text.slice(0, 160).replace(/\s+/g, " ")})`
   421→          );
   422→          lastErr = new Error(vErr);
   423→          if (attempt === maxAttempts) break;
   424→          await new Promise((r) => setTimeout(r, 2000));
   425→          continue; // پاسخ نامعتبر می‌تواند تصادفی باشد → retry همان سطح
   426→        }
   427→      }
   428→      return text;
   429→    } catch (err) {
   430→      lastErr = err;
   431→      const msg = String((err as Error)?.message || err);
   432→      const retriable = /50[234]|429|timed out|timeout|ECONNRESET|ECONNREFUSED|fetch failed|socket hang up/i.test(msg);
   433→      console.error(`[${logTag}] AvalAI error (attempt ${attempt}/${maxAttempts}):`, msg.slice(0, 200));
   434→      if (!retriable || attempt === maxAttempts) break;
   435→      await new Promise((r) => setTimeout(r, 3000)); // backoff کوتاه
   436→    }
   437→  }
   438→
   439→  // تلاش نهایی با پارامتر جایگزین (مثلاً تفکر low) — بهتر از هیچ برنامه‌ای نیست
   440→  if (opts?.fallbackParams) {
   441→    console.warn(`[${logTag}] falling back to alternate params (e.g. low thinking)...`);
   442→    try {
   443→      const completion = await avalaiClient.chat.completions.create(opts.fallbackParams as any, {
   444→        maxRetries: 0,
   445→        ...(opts?.timeoutMs ? { timeout: opts.timeoutMs } : {}),
   446→      });
   447→      const choice: any = (completion as any)?.choices?.[0];
   448→      const rawContent = choice?.message?.content;
   449→      const text = typeof rawContent === "string" ? rawContent : "";
   450→      if (text.trim() && (!opts.validateContent || !opts.validateContent(text))) {
   451→        return text;
   452→      }
   453→      console.error(`[${logTag}] fallback attempt returned empty/invalid content (finish_reason=${choice?.finish_reason || "?"})`);
   454→      lastErr = new Error("پاسخ خالی/نامعتبر از مدل (fallbackParams)");
   455→    } catch (err) {
   456→      lastErr = err;
   457→      console.error(`[${logTag}] fallback attempt failed:`, String((err as Error)?.message || err).slice(0, 200));
   458→    }
   459→  }
   460→
   461→  console.error(`[${logTag}] all attempts failed:`, lastErr);
   462→  throw lastErr instanceof Error
   463→    ? lastErr
   464→    : new Error("خطا در ارتباط با سرویس هوش مصنوعی. لطفاً کمی بعد دوباره تلاش کنید.");
   465→}

   468→// زنجیرهٔ تولید برنامه‌های تمرینی/تغذیه/مکمل — دیریکتیو مالک (v40):
   469→// «با gemini-3.8-flash ساخته بشه؛ در بدترین حالت با میزان تفکر high.
   470→//  اولویت رو بذار رو اون و اگر یک درصد نشد از DeepSeek 4 Flash استفاده کن»
   471→// مدل gemini-3.8-flash از تلاش استدلالی قابل‌تنظیم پشتیبانی می‌کند. زنجیره:
   472→//   گام ۱ — gemini-3.8-flash با thinkingLevel=high (۲ تلاش برای خطاهای گذرای گیت‌وی)
   473→//   گام ۲ — deepseek-v4-flash با reasoning_effort=high (۲ تلاش) — فقط اگر
   474→//           گام ۱ به‌هرشکلی شکست خورد (پاسخ خالی/نامعتبر، 400 دائمی، 504/تایم‌اوت پایدار)
   475→//
   476→// بودجه‌بندی زمان: watchdog تولید برنامه STUCK_GENERATION_WINDOW_MS=۵۰ دقیقه است؛
   477→// سقف هر تلاش ۲۷۰ ثانیه → بدترین حالت (۲×gemini + ۲×deepseek) ≈ ۱۸ دقیقه < ۵۰ ✓
   478→// (تمرینی و غذایی موازی اجرا می‌شوند و هر دو از همین بودجه استفاده می‌کنند.)
   479→//
   480→// چرا max حذف شد؟ تست واقعی v39: تفکر max بودجهٔ خروجی را با reasoning می‌بلعید
   481→// (finish_reason=length، متن خالی/بریده) — همان ریشهٔ «برنامه خالی»؛ تفکر high
   482→// کیفیت نزدیک max می‌دهد بدون آن مشکل (curl واقعی: finish=stop + JSON کامل).
   483→// چرا low حذف شد؟ دیریکتیو مالک: بدترین حالتِ تفکر باید high بماند — به‌جای
   484→// پایین‌آوردن سطح، مدل عوض می‌شود (deepseek-v4-flash با high تست‌شده ✓).
   485→const PLAN_ATTEMPT_TIMEOUT_MS = 270_000; // ۴.۵ دقیقه برای هر تلاش
   486→
   487→// v39-fix: بودجهٔ صریح خروجی برای تولید برنامه.
   488→// بدون این، AvalAI برای gemini-3.8 یک بودجهٔ پیش‌فرض کوچک اعمال می‌کند که
   489→// reasoning (تفکر max/high) به‌تنهایی آن را می‌بلعد → content:null → «برنامه خالی».
   490→// تست واقعی (سپتامبر ۲۰۲۶): maxOutputTokens=65536 پذیرفته می‌شود و reasoning+JSON
   491→// بزرگ (~۲۵هزار توکن) را با هم جا می‌دهد. برای gemini-3.x پروکسیِ پایین‌تر این
   492→// مقدار را به generationConfig.maxOutputTokens منتقل می‌کند.
   493→const PLAN_MAX_OUTPUT_TOKENS = 65536;
   494→
   495→/**
   496→ * ساخت پارامترهای chat.completions برای تولید برنامه با سطح تفکر مشخص.
   497→ * v40: پارامتر model اضافه شد — فال‌بک deepseek-v4-flash هم با همین سازنده
   498→ * ساخته می‌شود تا reasoning_effort=high بگیرد (پروکسیِ پایین‌تر مقدار non-null
   499→ * را override نمی‌کند) و extra_bodyِ تفکرِ gemini فقط برای gemini-3.x ست شود.
   500→ */
   501→function buildPlanParams(
   502→  systemPrompt: string,
   503→  userPrompt: string,
   504→  thinkingLevel: string,
   505→  model: string = TEXT_MODEL
   506→): Record<string, unknown> {
   507→  return {
   508→    model,
   509→    messages: [
   510→      { role: "system", content: systemPrompt },
   511→      { role: "user", content: userPrompt },
   512→    ],
   513→    // v39-fix: بودجهٔ صریح خروجی (جلوگیری از بلعیده‌شدن بودجه توسط reasoning)
   514→    max_tokens: PLAN_MAX_OUTPUT_TOKENS,
   515→    // برای مدل‌های deepseek-v4 (reasoning_effort پشتیبانی می‌کنند)
   516→    reasoning_effort: thinkingLevel,
   517→    // برای مدل‌های gemini-3.x (thinkingLevel از طریق generationConfig)
   518→    ...(isGemini3Model(model)
   519→      ? { extra_body: { generationConfig: { thinkingConfig: { thinkingLevel } } } }
   520→      : {}),
   521→  };
   522→}
   523→
   524→/**
   525→ * فراخوانی LLM برای تولید برنامه — دیریکتیو مالک (v40):
   526→ * «برنامه با gemini-3.8-flash ساخته بشه؛ در بدترین حالت با میزان تفکر high؛
   527→ *  اگر یک درصد نشد، از DeepSeek 4 Flash استفاده کن.»
   528→ *
   529→ * گام ۱: gemini-3.8-flash با thinkingLevel=high — ۲ تلاش برای خطاهای گذرای
   530→ *        گیت‌وی (504/429/timeout)؛ خطای دائمی (400) و پاسخ خالی/نامعتبر
   531→ *        بلافاصله به گام ۲ می‌روند.
   532→ * گام ۲: deepseek-v4-flash با reasoning_effort=high — ۲ تلاش (تست واقعی curl:
   533→ *        finish=stop + JSON کامل در ~۶.۵ ثانیه). AVALAI_FALLBACK_TEXT_MODEL
   534→ *        اگر ست شده باشد جایگزین deepseek پیش‌فرض می‌شود.
   535→ *
   536→ * v39→v40: زنجیرهٔ قدیمی max→high→low حذف شد — max بودجهٔ خروجی را می‌بلعید
   537→ * (ریشهٔ «برنامه خالی») و low کیفیت را پایین می‌آورد؛ حالا سطح تفکر هرگز از
   538→ * high پایین نمی‌آید و به‌جای آن مدل تعویض می‌شود.
   539→ */
   540→export async function generatePlanContent(
   541→  systemPrompt: string,
   542→  userPrompt: string,
   543→  logTag: string,
   544→  validateContent?: PlanContentValidator
   545→): Promise<string> {
   546→  const chain: Array<{ level: string; attempts: number }> = [
   547→    { level: "high", attempts: 2 }, // دیریکتیو مالک: تفکر high — پایین‌تر نمی‌آییم
   548→  ];
   549→  let lastErr: unknown = null;
   550→  for (const step of chain) {
   551→    try {
   552→      return await createPlanCompletionWithRetry(
   553→        buildPlanParams(systemPrompt, userPrompt, step.level),
   554→        logTag,
   555→        step.attempts,
   556→        // پاسخ فقط وقتی «موفق» است که واقعاً قابل‌استفاده باشد —
   557→        // وگرنه زنجیره به مدل فال‌بک می‌رود
   558→        { timeoutMs: PLAN_ATTEMPT_TIMEOUT_MS, validateContent }
   559→      );
   560→    } catch (err) {
   561→      lastErr = err;
   562→      console.warn(
   563→        `[${logTag}] gemini-3.8-flash (thinking=${step.level}) failed — switching to fallback model:`,
   564→        String((err as Error)?.message || err).slice(0, 180)
   565→      );
   566→    }
   567→  }
   568→
   569→  // ─── گام ۲: فال‌بک مدل — DeepSeek 4 Flash (دیریکتیو مالک v40) ───
   570→  // با همان سطح تفکر high (reasoning_effort=high) — آخرین خط دفاعی قبل از
   571→  // «ناموفق»؛ پرامپت‌ها و اعتبارسنج دست‌نخورده می‌مانند، فقط موتور عوض می‌شود.
   572→  const fallbackModel = process.env.AVALAI_FALLBACK_TEXT_MODEL || "deepseek-v4-flash";
   573→  if (fallbackModel && fallbackModel !== TEXT_MODEL) {
   574→    console.warn(
   575→      `[${logTag}] primary model exhausted — trying fallback model "${fallbackModel}" (thinking=high)...`
   576→    );
   577→    try {
   578→      return await createPlanCompletionWithRetry(
   579→        buildPlanParams(systemPrompt, userPrompt, "high", fallbackModel),
   580→        `${logTag}(fallback:${fallbackModel})`,
   581→        2,
   582→        { timeoutMs: PLAN_ATTEMPT_TIMEOUT_MS, validateContent }
   583→      );
   584→    } catch (err) {
   585→      lastErr = err;
   586→      console.error(`[${logTag}] fallback model also failed:`, String((err as Error)?.message || err).slice(0, 200));
   587→    }
   588→  }
   589→
   590→  throw lastErr instanceof Error
   591→    ? lastErr
   592→    : new Error("خطا در ارتباط با سرویس هوش مصنوعی. لطفاً کمی بعد دوباره تلاش کنید.");
   593→}
   594→
   595→/**
   596→ * فراخوانی چت با retry سبک برای خطاهای گذرا — برای چت‌ها و تحلیل‌های سریع.
   597→ *
   598→ * چرا: گیت‌وی api.avalai.ir (ArvanCloud) به‌صورت ناپایدار درخواست‌ها را بعد از
   599→ * ~۳۰ ثانیه با 504 (صفحه HTML کلادفلر) می‌کُشد. تست واقعی: چت نیکا/مربی گاهی
   600→ * در اولین تلاش 504 می‌خورد و در تلاش دوم همان لحظه جواب می‌دهد. بدون retry،
   601→ * کاربر خطای «ارتباط برقرار نشد» می‌بیند در حالی که یک تلاش دیگر کافی بود.
   602→ *
   603→ * فقط خطاهای گذرا (502/503/504/429/timeout/شبکه) retry می‌شوند؛
   604→ * خطاهای دائمی (401 کلید غلط، 400 پارامتر غلط) بلافاصله fail می‌شوند.
   605→ *
   606→ * v42: بعد از اتمام همهٔ تلاش‌های مدل اول، درخواست با مدل فال‌بک تکرار می‌شود
   607→ * (ویژن → gemini-3.5-flash، متن → deepseek-v4-flash — دیریکتیو مالک).
   608→ * چت نیکا/مربی، تحلیل عکس غذا/بدن و تحلیل ویدیو از این تابع استفاده می‌کنند.
   609→ */
   610→export async function createChatCompletionWithRetry(
   611→  params: Record<string, unknown>,
   612→  logTag: string,
   613→  maxAttempts = 3
   614→): Promise<string> {
   615→  try {
   616→    return await chatCompletionWithRetryAttempt(params, logTag, maxAttempts);
   617→  } catch (primaryErr) {
   618→    // ─── v42 دیریکتیو مالک: هر شکستِ نهایی مدل اول → مدل فال‌بک ───
   619→    // ویژن → gemini-3.5-flash | متن → deepseek-v4-flash (خودکار)
   620→    const fb = fallbackModelFor(params);
   621→    if (!fb) throw primaryErr;
   622→    console.warn(
   623→      `[${logTag}] primary model exhausted — falling back to "${fb}" (v42 directive)...`
   624→    );
   625→    return await chatCompletionWithRetryAttempt(
   626→      withFallbackModel(params, fb),
   627→      `${logTag}(fallback:${fb})`,
   628→      maxAttempts
   629→    );
   630→  }
   631→}

   633→async function chatCompletionWithRetryAttempt(
   634→  params: Record<string, unknown>,
   635→  logTag: string,
   636→  maxAttempts: number
   637→): Promise<string> {
   638→  let lastErr: unknown = null;
   639→  // v58 — پارامتر داخلی fallback_model قبل از ارسال حذف می‌شود (هرگز به API نمی‌رود)
   640→  const apiParams: Record<string, unknown> = { ...params };
   641→  delete apiParams.fallback_model;
   642→  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
   643→    try {
   644→      const completion = await avalaiClient.chat.completions.create(apiParams as any, {
   645→        maxRetries: 0, // retry دستی اینجا
   646→      });
   647→      // v39: پاسخ خالی (content:null — مدل تفکری بودجه را با reasoning می‌خورد)
   648→      // دیگر موفقیت نیست؛ مثل خطای گذرا retry می‌شود (چت سریع است)
   649→      const choice: any = (completion as any)?.choices?.[0];
   650→      const rawContent = choice?.message?.content;
   651→      const text = typeof rawContent === "string" ? rawContent : "";
   652→      if (!text.trim()) {
   653→        const usage: any = (completion as any)?.usage || {};
   654→        const cd = usage.completion_tokens_details || {};
   655→        console.error(
   656→          `[${logTag}] attempt ${attempt}/${maxAttempts}: EMPTY content (finish_reason=${choice?.finish_reason || "?"}, reasoning=${cd.reasoning_tokens ?? "?"}, text=${cd.text_tokens ?? "?"})`
   657→        );
   658→        lastErr = new Error(`پاسخ خالی از مدل هوش مصنوعی (finish_reason=${choice?.finish_reason || "?"})`);
   659→        if (attempt === maxAttempts) break;
   660→        await new Promise((r) => setTimeout(r, 2500));
   661→        continue;
   662→      }
   663→      return text;
   664→    } catch (err) {
   665→      lastErr = err;
   666→      const msg = String((err as Error)?.message || err);
   667→      const retriable = /50[234]|429|timed out|timeout|ECONNRESET|ECONNREFUSED|fetch failed|socket hang up/i.test(msg);
   668→      console.error(`[${logTag}] AvalAI error (attempt ${attempt}/${maxAttempts}):`, msg.slice(0, 200));
   669→      if (!retriable || attempt === maxAttempts) break;
   670→      await new Promise((r) => setTimeout(r, 2500)); // backoff کوتاه
   671→    }
   672→  }
   673→  throw lastErr instanceof Error ? lastErr : new Error(String(lastErr));
   674→}

   715→/**
   716→ * فراخوانی محافظت‌شدهٔ chat.completions — همهٔ نقاط AI باید از این استفاده کنند
   717→ * نه از avalaiClient مستقیم. رفتار: retry خطاهای گذرا (502/503/504/429/timeout)،
   718→ * retry پاسخ خالی (content:null)، اعتبارسنجی اختیاری محتوا، لاگ تشخیصی کامل.
   719→ * v42: بعد از اتمام تلاش‌های مدل اول، مدل فال‌بک (ویژن → gemini-3.5-flash،
   720→ * متن → deepseek-v4-flash) امتحان می‌شود — دیریکتیو مالک «اگر یک درصد نشد…».

   722→export async function createResilientCompletion(
   723→  params: Record<string, unknown>,
   724→  opts: ResilientCompletionOptions
   725→): Promise<string> {
   726→  try {
   727→    return await resilientCompletionAttempt(params, opts);
   728→  } catch (primaryErr) {
   729→    // ─── v42 دیریکتیو مالک: هر شکستِ نهایی مدل اول → مدل فال‌بک ───
   730→    // ویژن → gemini-3.5-flash | متن → deepseek-v4-flash (تشخیص خودکار با image_url)
   731→    const fb = fallbackModelFor(params);
   732→    if (!fb) throw primaryErr;
   733→    console.warn(
   734→      `[${opts.logTag}] primary model exhausted — falling back to "${fb}" (v42 directive)...`
   735→    );
   736→    return await resilientCompletionAttempt(withFallbackModel(params, fb), {
   737→      ...opts,
   738→      logTag: `${opts.logTag}(fallback:${fb})`,
   739→    });
   740→  }

   743→async function resilientCompletionAttempt(
   744→  params: Record<string, unknown>,
   745→  opts: ResilientCompletionOptions
   746→): Promise<string> {
   747→  const maxAttempts = opts.maxAttempts ?? 3;
   748→  const body: Record<string, unknown> = {
   749→    ...(opts.model ? { model: opts.model } : {}),
   750→    ...params,
   751→  };
   752→  if (body.model == null) body.model = TEXT_MODEL;
   753→  // v58 — پارامتر داخلی مسیر فال‌بک؛ هرگز نباید به API برسد
   754→  delete body.fallback_model;
   755→  if (opts.maxTokens != null && body.max_tokens == null) body.max_tokens = opts.maxTokens;
   756→  if (opts.temperature != null && body.temperature == null) body.temperature = opts.temperature;
   757→  if (opts.reasoningEffort != null && body.reasoning_effort == null) body.reasoning_effort = opts.reasoningEffort;
   758→
   759→  let lastErr: unknown = null;
   760→  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
   761→    const attemptStartedAt = Date.now(); // v52 حسابداری — اندازه‌گیری تأخیر برای لاگ مصرف
   762→    try {
   763→      const completion = await avalaiClient.chat.completions.create(body as any, {
   764→        maxRetries: 0, // retry دستی اینجا — دوباره‌کاری نکن
   765→        ...(opts.timeoutMs ? { timeout: opts.timeoutMs } : {}),
   766→      });
   767→      const choice: any = (completion as any)?.choices?.[0];
   768→      const rawContent = choice?.message?.content;
   769→      const text = typeof rawContent === "string" ? rawContent : "";
   770→      if (!text.trim()) {
   771→        // v39-lesson: 200 با content:null = شکست، نه موفقیت
   772→        const usage: any = (completion as any)?.usage || {};
   773→        const cd = usage.completion_tokens_details || {};
   774→        const diag = `finish_reason=${choice?.finish_reason || "?"} completion_tokens=${usage.completion_tokens ?? "?"} reasoning=${cd.reasoning_tokens ?? "?"} text=${cd.text_tokens ?? "?"}`;
   775→        console.error(`[${opts.logTag}] attempt ${attempt}/${maxAttempts}: EMPTY content (${diag})`);
   776→        lastErr = new Error(`پاسخ خالی از مدل هوش مصنوعی (${diag})`);
   777→        if (attempt === maxAttempts) break;
   778→        await new Promise((r) => setTimeout(r, 2000));
   779→        continue;
   780→      }
   781→      if (choice?.finish_reason === "length") {
   782→        console.warn(`[${opts.logTag}] attempt ${attempt}: finish_reason=length (len=${text.length}) — خروجی ممکن است بریده باشد`);
   783→      }
   784→      if (opts.validateContent) {
   785→        const vErr = opts.validateContent(text);
   786→        if (vErr) {
   787→          console.error(
   788→            `[${opts.logTag}] attempt ${attempt}/${maxAttempts}: validation failed — ${vErr} (len=${text.length}, finish_reason=${choice?.finish_reason || "?"}, head=${text.slice(0, 160).replace(/\s+/g, " ")})`
   789→          );
   790→          lastErr = new Error(vErr);
   791→          if (attempt === maxAttempts) break;
   792→          await new Promise((r) => setTimeout(r, 2000));
   793→          continue;
   794→        }
   795→      }
   796→      // ─── v52 حسابداری — لاگ مصرف AI (best-effort؛ بعد از اعتبارسنجیِ موفق) ───
   797→      // هیچ تغییری در منطق retry/فال‌بک/اعتبارسنجی ایجاد نمی‌کند — فقط ثبت هزینه.
   798→      try {
   799→        const usage: any = (completion as any)?.usage;
   800→        if (usage && typeof usage === "object") {
   801→          const promptTokens = Number(usage.prompt_tokens ?? 0) || 0;
   802→          const completionTokens = Number(usage.completion_tokens ?? 0) || 0;
   803→          const totalTokens = Number(usage.total_tokens ?? promptTokens + completionTokens) || promptTokens + completionTokens;
   804→          await logAiUsage({
   805→            route: opts.routeTag ?? opts.logTag ?? "unknown",
   806→            model: String(body.model ?? TEXT_MODEL), // مدل واقعی همین تلاشِ موفق (بعد از fallback هم درست است)
   807→            promptTokens,
   808→            completionTokens,
   809→            totalTokens,
   810→            latencyMs: Date.now() - attemptStartedAt,
   811→            userId: opts.userId ?? null,
   812→          });
   813→        }
   814→      } catch {
   815→        // هرگز جریان اصلی را نمی‌شکند
   816→      }
   817→      return text;
   818→    } catch (err) {
   819→      lastErr = err;
   820→      const msg = String((err as Error)?.message || err);
   821→      const retriable = /50[234]|429|timed out|timeout|ECONNRESET|ECONNREFUSED|fetch failed|socket hang up|Request was aborted/i.test(msg);
   822→      console.error(`[${opts.logTag}] AvalAI error (attempt ${attempt}/${maxAttempts}):`, msg.slice(0, 200));
   823→      if (!retriable || attempt === maxAttempts) break;
   824→      await new Promise((r) => setTimeout(r, 2500));
   825→    }
   826→  }
   827→  throw lastErr instanceof Error
   828→    ? lastErr
   829→    : new Error("خطا در ارتباط با سرویس هوش مصنوعی. لطفاً کمی بعد دوباره تلاش کنید.");
   830→}

   832→// ساخت Proxy عمیق که chat.completions.create را intercept می‌کند
   833→export const avalaiClient = new Proxy({} as OpenAI, {
   834→  get(_, prop) {
   835→    const client = getAvalaiClient();
   836→    const value = (client as any)[prop];
   837→
   838→    if (prop === "chat") {
   839→      // Proxy برای chat
   840→      return new Proxy(value, {
   841→        get(__, chatProp) {
   842→          const chatValue = (value as any)[chatProp];
   843→          if (chatProp === "completions") {

   846→              get(___, compProp) {
   847→                const compValue = (chatValue as any)[compProp];
   848→                if (compProp === "create" && typeof compValue === "function") {
   849→                  // wrap create با مدیریت gemini-3.x — bind به chatValue (completions object)
   850→                  return wrapCreateWithGemini3Support(compValue, chatValue);

   852→                return typeof compValue === "function" ? compValue.bind(chatValue) : compValue;
   853→              },
   854→            });
   855→          }
   856→          return typeof chatValue === "function" ? chatValue.bind(value) : chatValue;
   857→        },
   858→      });
   859→    }
   860→
   861→    return typeof value === "function" ? value.bind(client) : value;
   862→  },
   863→});

  1057→// ─── محاسبه دقیق TDEE و کالری هدف بر اساس هدف و فعالیت ───
  1058→function computeTDEEAndTarget(data: OnboardingData): {
  1059→  bmr: number;
  1060→  tdee: number;
  1061→  targetCalories: number;
  1062→  calorieAdjustment: number;
  1063→  proteinG: number;
  1064→  carbsG: number;
  1065→  fatG: number;
  1066→  proteinPerKg: number;
  1067→  carbsPerKg: number;
  1068→  fatPerKg: number;
  1069→} {
  1070→  // فرمول Mifflin-St Jeor (دقیق‌ترین فرمول BMR)
  1071→  const bmr =
  1072→    data.gender === "male"
  1073→      ? 10 * data.weight + 6.25 * data.height - 5 * data.age + 5
  1074→      : 10 * data.weight + 6.25 * data.height - 5 * data.age - 161;
  1075→  const activityMultiplier = getActivityMultiplier(data.activityLevel);
  1076→  const tdee = bmr * activityMultiplier;
  1077→
  1078→  // تنظیم کالری بر اساس هدف — مازاد/نقصان هوشمند
  1079→  let calorieAdjustment = 0;
  1080→  switch (data.goal) {
  1081→    case "fat_loss":
  1082→      // نقصان ۴۰۰-۵۰۰ کالری برای چربی‌سوزی ایمن (۰.۵ کیلو در هفته)
  1083→      calorieAdjustment = -Math.min(500, Math.round(tdee * 0.2));
  1084→      break;
  1085→    case "cut":
  1086→      // v60 — کات: نقصان ملایم‌تر (حدود ۱۵٪) برای حفظ حداکثری عضله
  1087→      calorieAdjustment = -Math.min(400, Math.round(tdee * 0.15));
  1088→      break;
  1089→    case "muscle_gain":
  1090→      // مازاد ۲۵۰-۴۰۰ کالری برای حجم‌گیری تمیز (به حداقل رساندن چربی‌سازی)
  1091→      calorieAdjustment = 300;
  1092→      break;
  1093→    case "bulk":
  1094→      // v60 — افزایش حجم: مازاد بزرگ‌تر (حدود ۴۰۰) برای رشد سریع‌تر با کنترل چربی
  1095→      calorieAdjustment = 400;
  1096→      break;
  1097→    case "strength":
  1098→      // مازاد ملایم ۲۰۰ کالری برای قدرت
  1099→      calorieAdjustment = 200;
  1100→      break;
  1101→    case "endurance":
  1102→      // تعادل با کمی مازاد برای ریکاوری
  1103→      calorieAdjustment = 100;
  1104→      break;
  1105→    case "fitness":
  1106→    default:
  1107→      calorieAdjustment = 0;
  1108→      break;
  1109→  }
  1110→  const targetCalories = Math.round(tdee + calorieAdjustment);
  1111→
  1112→  // محاسبه درشت‌مغذی‌ها — پروتئین بر اساس هدف، چربی ۲۵٪، کربوهیدرات باقی‌مانده
  1113→  // v60 — کات: بالاترین پروتئین (۲.۴g/kg) برای حفظ عضله در نقصان؛ bulk مثل muscle_gain
  1114→  const proteinPerKg =
  1115→    data.goal === "cut"
  1116→      ? 2.4
  1117→      : data.goal === "fat_loss"
  1118→        ? 2.2
  1119→        : data.goal === "muscle_gain" || data.goal === "bulk"
  1120→          ? 2.0
  1121→          : 1.8;
  1122→  const proteinG = Math.round(data.weight * proteinPerKg);
  1123→  const fatG = Math.round((targetCalories * 0.25) / 9); // ۲۵٪ کالری از چربی
  1124→  const carbsG = Math.max(0, Math.round((targetCalories - proteinG * 4 - fatG * 9) / 4));
  1125→  const carbsPerKg = Math.round((carbsG / data.weight) * 10) / 10;
  1126→  const fatPerKg = Math.round((fatG / data.weight) * 10) / 10;
  1127→
  1128→  return {
  1129→    bmr: Math.round(bmr),
  1130→    tdee: Math.round(tdee),
  1131→    targetCalories,
  1132→    calorieAdjustment,
  1133→    proteinG,
  1134→    carbsG,
  1135→    fatG,
  1136→    proteinPerKg,
  1137→    carbsPerKg,
  1138→    fatPerKg,
  1139→  };
  1140→}

  1149→export async function getAiConfig(key: string, fallback: string): Promise<string> {
  1150→  const now = Date.now();
  1151→  const hit = _aiConfigCache.get(key);
  1152→  if (hit && now - hit.at < AI_CONFIG_CACHE_TTL_MS) {
  1153→    return hit.value || fallback;
  1154→  }
  1155→  const cfg = await db.aiConfig.findUnique({ where: { key } });
  1156→  const value = cfg?.value || fallback;
  1157→  _aiConfigCache.set(key, { value, at: now });
  1158→  return value;
  1159→}
