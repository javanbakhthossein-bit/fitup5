    31→async function saveFramesToDisk(dataUrls: string[]): Promise<string[]> {
    32→  const urls: string[] = [];
    33→  const stamp = Date.now();
    34→  for (let i = 0; i < dataUrls.length; i++) {
    35→    try {
    36→      const parsed = parseDataUrl(dataUrls[i], "image/jpeg");
    37→      const ext = parsed.mime === "image/png" ? "png" : parsed.mime === "image/webp" ? "webp" : "jpg";
    38→      const buffer = Buffer.from(parsed.base64, "base64");
    39→      if (buffer.length === 0) continue;
    40→      const fileName = `chat-vframe-${stamp}-${Math.random().toString(36).slice(2, 8)}-${i}.${ext}`;
    41→      const { url } = await savePrivateMediaFile("chat", fileName, buffer);
    42→      urls.push(url);
    43→    } catch (frameErr) {
    44→      console.error("[chat] frame save failed (skipping):", frameErr);
    45→    }
    46→  }
    47→  return urls;
    48→}

    56→async function buildMediaSyncAttachment(
    57→  history: { mediaUrl: string | null; mediaType: string | null; mediaFrames: string | null }[]
    58→): Promise<{ dataUrls: string[]; kind: "image" | "video-frames"; note: string } | undefined> {
    59→  try {
    60→    const recent = history.slice(-MEDIA_SYNC_WINDOW);
    61→    const lastMedia = [...recent].reverse().find((m) => m.mediaUrl && m.mediaType);
    62→    if (!lastMedia?.mediaUrl) return undefined;
    63→
    64→    if (lastMedia.mediaType === "image") {
    65→      const buf = await readFile(absolutePathForUploadUrl(lastMedia.mediaUrl));
    66→      if (buf.length === 0) return undefined;
    67→      const mime = lastMedia.mediaUrl.endsWith(".png") ? "image/png" : "image/webp";
    68→      return {
    69→        dataUrls: [`data:${mime};base64,${buf.toString("base64")}`],
    70→        kind: "image",
    71→        note: "\n\n[عکسی که کاربر چند پیام قبل در همین گفتگو فرستاده بود، برای مرجع دوباره پیوست شده است. اگر سوال فعلی دربارهٔ همان عکس/غذا/فرم بدن است، با تکیه بر آن پاسخ بده.]",
    72→      };
    73→    }
    74→
    75→    // ویدیو — اولویت: فریم‌های ذخیره‌شده (v70) → استخراج دوباره از فایل (پیام‌های قدیمی‌تر)
    76→    let frameUrls: string[] = [];
    77→    if (lastMedia.mediaFrames) {
    78→      try {
    79→        const parsedFrames = JSON.parse(lastMedia.mediaFrames);
    80→        if (Array.isArray(parsedFrames)) frameUrls = parsedFrames.filter((u: unknown): u is string => typeof u === "string");
    81→      } catch {
    82→        // JSON خراب → فال‌بک به استخراج دوباره
    83→      }
    84→    }
    85→    if (frameUrls.length === 0) {
    86→      try {
    87→        frameUrls = await extractVideoFramesAsDataUrls(absolutePathForUploadUrl(lastMedia.mediaUrl), 6);
    88→      } catch (frameErr) {
    89→        console.error("[chat] media-sync frame re-extraction failed:", frameErr);
    90→        return undefined;
    91→      }
    92→    }
    93→    const dataUrls: string[] = [];
    94→    for (const u of frameUrls.slice(0, MAX_CLIENT_FRAMES)) {
    95→      try {
    96→        const buf = await readFile(absolutePathForUploadUrl(u));
    97→        if (buf.length === 0) continue;
    98→        const mime = u.endsWith(".png") ? "image/png" : u.endsWith(".webp") ? "image/webp" : "image/jpeg";
    99→        dataUrls.push(`data:${mime};base64,${buf.toString("base64")}`);
   100→      } catch {
   101→        // فریم حذف‌شده از دیسک — رد شو
   102→      }
   103→    }
   104→    if (dataUrls.length === 0) return undefined;
   105→    return {
   106→      dataUrls,
   107→      kind: "video-frames",
   108→      note: "\n\n[ویدیویی که کاربر چند پیام قبل در همین گفتگو فرستاده بود، برای مرجع دوباره پیوست شده است (فریم‌ها به ترتیب زمانی). اگر سوال فعلی دربارهٔ همان ویدیو/حرکت است، تحلیل قبلی خود را ادامه بده و دقیق‌تر راهنمایی کن.]",
   109→    };
   110→  } catch (err) {
   111→    console.error("[chat] buildMediaSyncAttachment failed (continuing without sync):", err);
   112→    return undefined;
   113→  }
   114→}

   120→async function validateChatVideoUrl(userId: string, url: string): Promise<string | null> {
   121→  if (!url.startsWith("/uploads/chat/")) return null;
   122→  if (url.includes("..")) return null;
   123→  // مالکیت: نام فایل ویدیوی مسیر multipart الگوی chat-video-{uid}-… دارد
   124→  const fileName = url.split("/").pop() || "";
   125→  if (!fileName.startsWith(`chat-video-${userId}-`)) return null;
   126→  const filePath = absolutePathForUploadUrl(url);
   127→  try {
   128→    const st = await stat(filePath);
   129→    if (!st.isFile() || st.size === 0) return null;
   130→  } catch {
   131→    return null;
   132→  }
   133→  return url;
   134→}

   149→/**
   150→ * پارس data URL به mime و base64.
   151→ * پشتیبانی از هر دو فرمت:
   152→ *  - "data:image/jpeg;base64,...."
   153→ *  - raw base64 با mime جداگانه
   154→ */
   155→function parseDataUrl(value: string, fallbackMime = "image/jpeg"): { mime: string; base64: string } {
   156→  const match = value.match(/^data:([^;]+);base64,(.+)$/);
   157→  if (match) {
   158→    return { mime: match[1], base64: match[2] };
   159→  }
   160→  return { mime: fallbackMime, base64: value };
   161→}

   164→async function saveBase64File(
   165→  base64: string,
   166→  mime: string,
   167→  kind: "image" | "video"
   168→): Promise<{ url: string; mime: string; size: number }> {
   169→  const buffer = Buffer.from(base64, "base64");

   180→    "video/x-matroska": "mkv",

   188→/**
   189→ * ذخیره عکس با کاهش حجم خودکار:
   190→ *  - resize به حداکثر 1600px
   191→ *  - تبدیل به WebP (کیفیت ۷۵)

   196→ * decode کامل رد می‌شوند تا حافظه/پردازنده ایمن بماند؛ خطا → null (نه throw).

   198→const MAX_IMAGE_INPUT_PIXELS = 60_000_000;

   200→async function saveOptimizedImage(
   201→  base64: string,
   202→  mime: string
   203→): Promise<{ url: string; mime: string; size: number; optimizedBase64: string } | null> {
   204→  const sharp = (await import("sharp")).default;
   205→  const buffer = Buffer.from(base64, "base64");
   206→
   207→  // کاهش حجم + resize + WebP (با محدودیت پیکسل — fail-safe)
   208→  let processed: Buffer;
   209→  try {
   210→    processed = await sharp(buffer, { limitInputPixels: MAX_IMAGE_INPUT_PIXELS })
   211→      .resize(1600, 1600, {
   212→        fit: "inside",
   213→        withoutEnlargement: true,
   214→      })
   215→      .webp({ quality: 75 })
   216→      .toBuffer();
   217→  } catch (err) {
   218→    console.error("[chat] image optimization failed (too large/corrupt):", err);
   219→    return null;
   220→  }
   221→
   222→  const fileName = `chat-image-${Date.now()}-${Math.random().toString(36).slice(2, 8)}.webp`;
   223→  const { url } = await savePrivateMediaFile("chat", fileName, processed);
   224→
   225→  const optimizedBase64 = processed.toString("base64");
   226→  return {
   227→    url,
   228→    mime: "image/webp",
   229→    size: processed.length,
   230→    optimizedBase64,
   231→  };
   232→}
