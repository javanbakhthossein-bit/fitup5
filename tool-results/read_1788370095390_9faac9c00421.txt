    28→/**
    29→ * تبدیل ArrayBuffer به base64 بدون سرریز پشته.
    30→ *
    31→ * رفع باگ: قبلاً `String.fromCharCode(...new Uint8Array(buffer))` استفاده می‌شد
    32→ * که برای عکس‌های موبایل (چند مگابایت = میلیون‌ها آرگومان) با خطای
    33→ * `RangeError: Maximum call stack size exceeded` کرش می‌کرد.
    34→ * راه‌حل استاندارد: تبدیل تکه‌تکه (۳۲KB در هر گام).
    35→ */
    36→function arrayBufferToBase64(buffer: ArrayBuffer): string {
    37→  const bytes = new Uint8Array(buffer);
    38→  const CHUNK_SIZE = 0x8000; // ۳۲ کیلوبایت — امن برای پشته فراخوانی
    39→  let binary = "";
    40→  for (let i = 0; i < bytes.length; i += CHUNK_SIZE) {
    41→    binary += String.fromCharCode(...bytes.subarray(i, i + CHUNK_SIZE));
    42→  }
    43→  return btoa(binary);
    44→}
