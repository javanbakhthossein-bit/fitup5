   102→  (window as any).__fitupOverlayBackGuard = true;
   103→  window.addEventListener("popstate", (e: PopStateEvent) => {
   104→    const st = useAppStore.getState();

   106→      st.setOverlay(null);
   107→      // یک entry دوباره push می‌کنیم تا back بعدی مستقیماً از پنل خارج نکند
   108→      try {
   109→        window.history.pushState({ fitupOverlay: true }, "", window.location.href);
   110→      } catch {}
   111→      e.stopImmediatePropagation();
   112→      e.preventDefault();
   113→    }
   114→  });

   130→  const setScreen = useAppStore((s) => s.setScreen);
   131→  const setMainTab = useAppStore((s) => s.setMainTab);
   132→
   133→  // ─── 🩹 v45: گارد کاربر شبح (باگ «بعد از خروج دکمهٔ شروع تا رفرش کار نمی‌کند») ───
   134→  // اگر به هر دلیلی (race پاسخ async، خطای مسیر خروج قدیمی) MainApp بدون
   135→  // user مونت شود، پنل خالی و مرده رندر می‌شد. حالا: رندر نمی‌شود و به
   136→  // صفحهٔ ورود (اپ) یا لندینگ (مرورگر) برمی‌گردیم.
   137→  useEffect(() => {
   138→    if (!user) {

   140→    }
   141→  }, [user, setScreen]);
   142→
   143→  // ─── v48: تازه‌سازی زندهٔ اطلاعات کاربر — ریشه‌کنی «چت قفل است تا رفرش کنم» ───
   144→  // در WebView اپ‌ها و PWA اپ ممکن است ساعت‌ها باز بماند؛ اگر دادهٔ کاربر در
   145→  // store کهنه/ناقص باشد، همهٔ گیت‌های پلن غلط «قفل» نشان می‌دهند. حالا هر بار
   146→  // اپ mount می‌شود یا به foreground برمی‌گردد (visibilitychange/focus)،
   147→  // اطلاعات کاربر بی‌صدا از /api/auth/me تازه می‌شود (throttle ۶۰s داخل lib).
   148→  useEffect(() => {
   149→    void refreshUserIfNeeded(!useAppStore.getState().user);
   150→    const onVisible = () => {
   151→      if (document.visibilityState === "visible") {
   152→        void refreshUserIfNeeded(!useAppStore.getState().user);
   153→      }
   154→    };
   155→    const onFocus = () => void refreshUserIfNeeded(false);
   156→    document.addEventListener("visibilitychange", onVisible);
   157→    window.addEventListener("focus", onFocus);
   158→    return () => {
   159→      document.removeEventListener("visibilitychange", onVisible);
   160→      window.removeEventListener("focus", onFocus);
   161→    };
   162→  }, []);

   164→  // ─── v15: قفل تب‌های نیازمند پلن برای کاربر بدون پلن فعال (درخواست مالک) ───
   165→  // برنامه‌ها / تمرین‌ها / دستیار تغذیه / پیشرفت / چت با فیتاپ — برای کاربری که
   166→  // «هیچ پلن فعالی ندارد» قفل می‌شوند. استثناها: ادمین + کاربر pending
   167→  // (خرید کرده و در جریان تکمیل پیش‌نیازهاست — پول داده و نباید از همه‌چیز
   168→  // محروم شود؛ جریان پیش‌نیازها در داشبورد انجام می‌شود).
   169→  const PLAN_LOCKED_TABS = new Set<MainTab>(["programs", "workouts", "nutrition", "progress", "chat"]);
   170→  const planLocked =
   171→    !!user &&
   172→    user.role !== "ADMIN" &&
   173→    !user.hasActiveSubscription &&
   174→    !user.hasPendingSubscription &&
   175→    PLAN_LOCKED_TABS.has(mainTab);
   176→
   177→  // ─── v51 (قانون کافه‌بازار + درخواست مالک): تب «اپ موبایل» فقط در اپ بازار پنهان است ───
   178→  // بازار آپدیت را خودش مدیریت می‌کند و تبلیغ/دانلود APK داخل اپ بازار تخلف است
   179→  // (قبلاً برخورد گرفته بود). در اپ اختصاصی خودمان این تب «باید» بماند —
   180→  // مالک: «مدال دانلود نسخه جدید اندروید و همچنین اپ موبایل در برنامه خودمون…
   181→  // باید به آخرین نسخه و نسخه درست اشاره کنند». در مرورگر/PWA هم مثل قبل هست.
   182→  const [navItems, setNavItems] = useState(NAV_ITEMS_ALL);
   183→  useEffect(() => {
   184→    try {
   185→      if (isFitUpBazaarApp()) {
   186→        setNavItems(NAV_ITEMS_ALL.filter((n) => n.id !== "mobileapp"));
   187→      }
   188→    } catch {}
   189→  }, []);
   190→
   191→  // گارد رندر: اگر به هر روشی (URL عمیق/نوتیف قدیمی) تب mobileapp داخل اپ بازار
   192→  // فعال شود، به داشبورد برمی‌گردیم — بخش اپ موبایل در نسخه بازار اصلاً وجود ندارد.
   193→  useEffect(() => {
   194→    try {
   195→      if (mainTab === "mobileapp" && isFitUpBazaarApp()) {
   196→        setMainTab("dashboard");
   197→      }
   198→    } catch {}
   199→  }, [mainTab, setMainTab]);
   200→
   201→  // ─── Polling نوتیف‌ها ───
   202→  // وقتی overlay اعلانات باز است، polling سریع‌تر (هر ۱۰ ثانیه) می‌شود تا
   203→  // حس real-time به کاربر بدهد. در حالت عادی هر ۳۰ ثانیه poll می‌کنیم تا
   204→  // بار روی سرور کم باشد. وقتی نوتیف جدید می‌آید، setNotifications صدا
   205→  // زده می‌شود که unreadCount را هم در-store به‌روز می‌کند و badge بلافاصله
   206→  // آپدیت می‌شود.
   207→  useEffect(() => {
   208→    let cancelled = false;
   209→    // ─── نوتیف native در اپ کافه‌بازار ───
   210→    // web-push داخل WebView کار نمی‌کند؛ به‌جای آن پل JS اپ اندروید
   211→    // (window.FitUpNative.showNotification) نوتیف سیستم اندروید را نشان می‌دهد.
   212→    // فقط وقتی تعداد ناخوانده «افزایش» یافته و baseline قبلی موجود است
   213→    // (اولین poll بعد از باز شدن اپ اسپم نمی‌کند).
   214→    //
   215→    // ─── چت/نیکا در اپ‌های نیتیو اعلان سیستم ندارد ───
   216→    // درخواست صریح مالک: نوتیف‌های «چت» (نوع coach = پیام‌های چت ربات/مربی و
   217→    // نیکا) در اپ‌های نیتیو (بازار + اختصاصی) نباید به‌صورت اعلان اندروید
   218→    // نمایش داده شوند — تجربه چت داخل خود اپ کافی است.
   219→    const CHAT_NOTIFICATION_TYPES = new Set(["coach"]);
   220→    const nativeNotify = (items: any[]) => {
   221→      try {
   222→        const native = (window as any).FitUpNative;
   223→        if (!native?.showNotification) return;
   224→        // ─── اپ‌های نیتیو (بازار + اختصاصی): اعلان‌های چت فیلتر می‌شوند ───
   225→        // تجربهٔ چت داخل خود اپ کافی است؛ اعلان سیستم برای پیام‌های چت/نیکا
   226→        // آزاردهنده است (درخواست صریح مالک برای بازار — برای اپ اختصاصی هم
   227→        // همان سیاست). PWA/مرورگر دست‌نخورده.
   228→        if (isFitUpBazaarApp() || isFitUpOwnApp()) {
   229→          const newestNonChat = items.find(
   230→            (n: any) => !n?.read && !CHAT_NOTIFICATION_TYPES.has(n?.type)
   231→          );
   232→          if (newestNonChat?.title) {
   233→            native.showNotification(
   234→              String(newestNonChat.title),
   235→              String(newestNonChat.body || "")
   236→            );
   237→          }
   238→          return;
   239→        }
   240→        const newest = items.find((n: any) => !n?.read);
   241→        if (newest?.title) native.showNotification(String(newest.title), String(newest.body || ""));
   242→      } catch {}
   243→    };
   244→
   245→    const loadNotifications = async () => {
   246→      // همیشه poll می‌کنیم — حتی وقتی overlay باز است.
   247→      // store یکپارچه است و overlay از همان state می‌خواند، بنابراین
   248→      // به‌روزرسانی‌های real-time مستقیماً در overlay و badge نمایش داده می‌شود.
   249→      try {
   250→        const res = await fetch("/api/notifications", { cache: "no-store" });
   251→        if (!res.ok) return;
   252→        const data = await res.json();
   253→        if (!cancelled) {
   254→          // فقط اگر داده واقعاً تغییر کرده باشد، state را آپدیت کن (جلوگیری از flicker)
   255→          // مهم: مقایسه با آخرین state از store (getState) انجام می‌شود نه closure
   256→          // effect — قبلاً آرایه notifications از closure کهنه خوانده می‌شد و
   257→          // مقایسه همیشه با snapshot قدیمی بود (stale closure / flicker).
   258→          const newJson = JSON.stringify(data.notifications || []);
   259→          const oldJson = JSON.stringify(useAppStore.getState().notifications);
   260→          if (newJson !== oldJson) {
   261→            const prevItems = useAppStore.getState().notifications;
   262→            const prevUnread = prevItems.filter((n: any) => !n?.read).length;
   263→            const nextItems: any[] = data.notifications || [];
   264→            const nextUnread = nextItems.filter((n: any) => !n?.read).length;
   265→            useAppStore.getState().setNotifications(nextItems);
   266→            // ─── نوتیف سیستم اندروید (فقط اپ کافه‌بازار) ───
   267→            // فقط وقتی ناخوانده‌ها «افزایش» یافته‌اند و baseline قبلی موجود است
   268→            // (آرایه قبلی غیرخالی) — تا اولین poll بعد از باز شدن اپ اسپم نکند.
   269→            if (nextUnread > prevUnread && prevItems.length > 0) {
   270→              nativeNotify(nextItems);
   271→            }
   272→          }
   273→        }
   274→      } catch {}
   275→    };
   276→    loadNotifications();
   277→    // Polling تطبیقی: وقتی overlay اعلانات باز است، هر ۱۰ ثانیه؛ در غیر این
   278→    // صورت هر ۳۰ ثانیه. فقط یک interval فعال است (نه دو interval همزمان).
   279→    const pollInterval = overlay === "notifications" ? 10000 : 30000;
   280→    const interval = setInterval(loadNotifications, pollInterval);
   281→    // وقتی کاربر به تب برمی‌گردد (بعد از tab switch یا minimize)، فوراً یک
   282→    // poll اجرا می‌کنیم تا badge همیشه به‌روز باشد.
   283→    const onVisibilityChange = () => {
   284→      if (document.visibilityState === "visible" && !cancelled) {
   285→        loadNotifications();
   286→      }
   287→    };
   288→    document.addEventListener("visibilitychange", onVisibilityChange);
   289→    // ─── Real-time push listener ───
   290→    // وقتی service worker یک push notification دریافت می‌کند (حتی اگر اپ باز
   291→    // باشد)، یک پیام 'PUSH_RECEIVED' به صفحه ارسال می‌کند. این listener آن
   292→    // پیام را می‌گیرد و فوراً نوتیف‌ها را از سرور refresh می‌کند — بدون
   293→    // منتظر ماندن برای polling بعدی.
   294→    const onControllerMessage = (event: MessageEvent) => {
   295→      if (event.data?.type === "PUSH_RECEIVED" && !cancelled) {
   296→        loadNotifications();
   297→      }
   298→    };
   299→    if (navigator.serviceWorker) {
   300→      navigator.serviceWorker.addEventListener("message", onControllerMessage);
   301→    }
   302→    return () => {
   303→      cancelled = true;
   304→      clearInterval(interval);
   305→      document.removeEventListener("visibilitychange", onVisibilityChange);
   306→      if (navigator.serviceWorker) {
   307→        navigator.serviceWorker.removeEventListener("message", onControllerMessage);
   308→      }
