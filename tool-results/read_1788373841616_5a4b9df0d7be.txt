    63→/** Format a JS Date as a full Persian (Jalali) string: weekday + day + month + year — بدون ساعت */
    64→function formatPersianDateTime(d: Date): string {
    65→  try {
    66→    const parts = new Intl.DateTimeFormat("fa-IR-u-ca-persian", {
    67→      weekday: "long",
    68→      day: "numeric",
    69→      month: "long",
    70→      year: "numeric",
    71→      timeZone: "Asia/Tehran",
    72→    }).formatToParts(d);
    73→    const get = (t: string) => parts.find((p) => p.type === t)?.value ?? "";
    74→    // درخواست مالک: تاریخ و روزِ کامل — بدون ساعت و دقیقه
    75→    return `${get("weekday")} ${get("day")} ${get("month")} ${get("year")}`;
    76→  } catch {
    77→    // Fallback if Intl calendar is unavailable
    78→    const dayIdx = (d.getDay() + 1) % 7;
    79→    const wd = PERSIAN_WEEKDAYS[dayIdx];
    80→    const jalaliYear = 1400 + Math.floor((d.getTime() - 1606000000) / 31557600000);
    81→    return `${wd} — ${toPersianDigits(String(jalaliYear))}`;
    82→  }
    83→}

    89→const TehranClock = memo(function TehranClock() {

    91→  useEffect(() => {

    95→    const id = setInterval(tick, 5 * 60_000);
    96→    return () => clearInterval(id);
    97→  }, []);
    98→  return <span className="font-stat truncate">{now || "..."}</span>;

   101→/** Estimate calories burned during a workout using MET × weight × hours */
   102→function estimateCaloriesBurned(
   103→  minutes: number,
   104→  userWeightKg: number,
   105→  metValue = 6.5
   106→): number {
   107→  const hours = minutes / 60;
   108→  return Math.round(metValue * userWeightKg * hours);

  1442→/** Circular progress indicator for body fat percentage */
  1443→function BodyFatRing({ percent }: { percent: number }) {
  1444→  const size = 56;
  1445→  const stroke = 5;
  1446→  const radius = (size - stroke) / 2;
  1447→  const circumference = 2 * Math.PI * radius;
  1448→  // درصد چربی را به فرکانس بصری (5-35) تبدیل می‌کنیم تا نمودار معنادار باشد
  1449→  const visualFraction = Math.min(1, Math.max(0.05, (percent - 3) / 35));
  1450→  const offset = circumference * (1 - visualFraction);
  1451→  const color = percent < 14 ? "#06b6d4" : percent < 21 ? "#10b981" : percent < 26 ? "#f59e0b" : "#ef4444";
  1452→
  1453→  return (
  1454→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
  1455→      <svg width={size} height={size} className="-rotate-90">
  1456→        <circle
  1457→          cx={size / 2}
  1458→          cy={size / 2}
  1459→          r={radius}
  1460→          fill="none"
  1461→          stroke="#f1f5f9"
  1462→          strokeWidth={stroke}
  1463→        />
  1464→        <motion.circle
  1465→          cx={size / 2}
  1466→          cy={size / 2}
  1467→          r={radius}
  1468→          fill="none"
  1469→          stroke={color}
  1470→          strokeWidth={stroke}
  1471→          strokeLinecap="round"
  1472→          strokeDasharray={circumference}
  1473→          initial={{ strokeDashoffset: circumference }}
  1474→          animate={{ strokeDashoffset: offset }}
  1475→          transition={{ duration: 1, ease: "easeOut" }}
  1476→        />
  1477→      </svg>
  1478→      <div className="absolute inset-0 flex items-center justify-center">
  1479→        <span className="text-base font-black font-stat text-slate-900 leading-none">
  1480→          {toPersianDigits(percent.toFixed(1))}
  1481→        </span>
  1482→      </div>
  1483→    </div>
  1484→  );
  1485→}
