    33→    33→ * اگر هزینه‌ای گزارش نشد undefined برمی‌گردد (لاگر خودش نرخ پیش‌فرض می‌گذارد).

    35→    35→function extractSmsCost(data: unknown): number | undefined {
    36→    36→  if (!data || typeof data !== "object") return undefined;
    37→    37→  const d = data as Record<string, unknown>;
    38→    38→  const candidates: unknown[] = [d.cost];
    39→    39→  const inner = d.data;
    40→    40→  if (inner && typeof inner === "object") {
    41→    41→    if (Array.isArray(inner)) {
    42→    42→      const first = inner[0];
    43→    43→      if (first && typeof first === "object") {
    44→    44→        candidates.push((first as Record<string, unknown>).cost);
    45→    45→      }
    46→    46→    } else {
    47→    47→      candidates.push((inner as Record<string, unknown>).cost);
    48→    48→    }
    49→    49→  }
    50→    50→  for (const c of candidates) {
    51→    51→    if (typeof c === "number" && Number.isFinite(c) && c > 0) return c;
    52→    52→    if (typeof c === "string" && c.trim() !== "" && Number.isFinite(Number(c))) return Number(c);
    53→    53→  }
    54→    54→  return undefined;
    55→    55→}

   101→   101→/**
   102→   102→ * Normalize an Iranian mobile number to the format SMS.ir expects:
   103→   103→ *   "09123456789"  → "9123456789"
   104→   104→ *   "989123456789" → "9123456789"
   105→   105→ *   "+989123456789"→ "9123456789"
   106→   106→ *   "00989123456789"→ "9123456789"
   107→   107→ */
   108→   108→export function normalizeMobileForSmsIr(mobile: string): string {
   109→   109→  let m = mobile.replace(/\s/g, "").replace(/[+\-()]/g, "");
   110→   110→  // Strip country code variants
   111→   111→  if (m.startsWith("0098")) m = m.slice(4);
   112→   112→  else if (m.startsWith("98")) m = m.slice(2);
   113→   113→  // Strip leading 0
   114→   114→  if (m.startsWith("0")) m = m.slice(1);
   115→   115→  return m;
   116→   116→}

   166→   166→ *     فراخوانی و timeout ۸ ثانیه‌ای. بدون bulk، بدون fallback، بدون ۱۰۱.
   167→   167→ *     (WebOTP فقط بهبود UX است؛ تحویل سریع کد شرط اصلی است.)

   169→   169→ *     تنظیم شده باشد و SMSIR_USE_RAW_SEND=true → اول bulk با فرمت WebOTP
   170→   170→ *     (درج خودکار کد در اندروید) و فقط در شکستش یک‌بار قالب. بدون
   171→   171→ *     SMSIR_LINE هیچ‌وقت bulk امتحان نمی‌شود.

   176→   176→ * OTP فقط وقتی با فرمت WebOTP/هش اپ ارسال می‌شود که SMSIR_LINE در env باشد؛
   177→   177→ * بدون خط، مسیر قالبی می‌رود که متنش ثابت است و نه @دامنه #کد دارد و نه
   178→   178→ * هش اپ → نه WebOTP کروم و نه SMS Retriever اپ کاری ندارند.
   179→   179→ * حالا خط را می‌توان از «تنظیمات سایت → sms_line» هم گذاشت (بدون SSH).

   181→   181→async function getSmsLineConfig(): Promise<{ line: string; fromSetting: boolean }> {
   182→   182→  const env = process.env.SMSIR_LINE?.trim();
   183→   183→  if (env) return { line: env, fromSetting: false };
   184→   184→  try {
   185→   185→    const { db } = await import("@/lib/db");
   186→   186→    const row = await db.siteSetting.findUnique({
   187→   187→      where: { key: "sms_line" },
   188→   188→      select: { value: true },
   189→   189→    });
   190→   190→    const v = row?.value?.trim() || "";
   191→   191→    return { line: v, fromSetting: !!v };
   192→   192→  } catch {
   193→   193→    return { line: "", fromSetting: false };
   194→   194→  }
   195→   195→}

   197→   197→export async function sendOtpSms(
   198→   198→  mobile: string,
   199→   199→  code: string,

   201→   201→  opts?: { userId?: string | null }

   207→   207→      success: false,
   208→   208→      error: "SMSIR_API_KEY تنظیم نشده است.",
   209→   209→    };
   210→   210→  }

   212→   212→  const normalized = normalizeMobileForSmsIr(mobile);
   213→   213→  if (!/^9\d{9}$/.test(normalized)) {

   224→   224→

   229→   229→    await logSmsResultCost({ normalizedMobile: normalized, scenario: "otp", templateId: process.env.SMSIR_TEMPLATE_ID ?? null, result, userId: opts?.userId });
   230→   230→    return result;
   231→   231→  }
   232→   232→
   233→   233→  // حالت اختیاری WebOTP — فقط با line معتبر (env یا تنظیمات سایت).
   234→   234→  const raw = await sendBulkSms(normalized, code, apiKey, line);
   235→   235→  if (raw.success) {
   236→   236→    await logSmsResultCost({ normalizedMobile: normalized, scenario: "otp", templateId: null, result: raw, userId: opts?.userId });
   237→   237→    return raw;
   238→   238→  }
   239→   239→  console.warn(
   240→   240→    `[smsir] bulk (WebOTP) شکست خورد — OTP از مسیر قالب می‌رود. علت: ${raw.error ?? "نامشخص"}`

   243→   243→  await logSmsResultCost({ normalizedMobile: normalized, scenario: "otp", templateId: process.env.SMSIR_TEMPLATE_ID ?? null, result: fallback, userId: opts?.userId });
   244→   244→  return fallback;
   245→   245→}

   247→   247→/**
   248→   248→ * ارسال پیامک با قالب عمومی verify (استفاده داخلی).
   249→   249→ * ساختار پاسخ sms.ir: { status: 1, message: "موفق", data: {...} }
   250→   250→ */
   251→   251→async function postVerify(
   252→   252→  normalizedMobile: string,
   253→   253→  templateId: number,
   254→   254→  parameters: Array<{ name: string; value: string }>,
   255→   255→  apiKey: string,
   256→   256→  label: string,
   257→   257→  timeoutMs = 15_000
   258→   258→): Promise<SmsIrResult> {
   259→   259→  try {
   260→   260→    const res = await fetch(SMSIR_API_URL, {
   261→   261→      method: "POST",
   262→   262→      headers: {
   263→   263→        "Content-Type": "application/json",
   264→   264→        Accept: "application/json",
   265→   265→        // CRITICAL: sms.ir uses `x-api-key`, NOT `Authorization: Bearer`.
   266→   266→        "x-api-key": apiKey,
   267→   267→      },
   268→   268→      body: JSON.stringify({ mobile: normalizedMobile, templateId, parameters }),
   269→   269→      cache: "no-store",
   270→   270→      signal: AbortSignal.timeout(timeoutMs),
   271→   271→    });
   272→   272→
   273→   273→    let data: unknown = null;
   274→   274→    try {
   275→   275→      data = await res.json();
   276→   276→    } catch {
   277→   277→      // Non-JSON response — keep data null.
   278→   278→    }
   279→   279→
   280→   280→    if (!res.ok) {
   281→   281→      // 🩹 v45: کد وضعیت و پیام sms.ir (مثل «۱۰۱ — شماره خط نامعتبر میباشد»)
   282→   282→      // هم در status/error بیاید تا لاگ pm2 دقیقاً مثل پنل خوانا باشد
   283→   283→      // (هم‌تراز با رفتار bulk از v35).
   284→   284→      const bodyMsg =
   285→   285→        data && typeof data === "object" && "message" in data
   286→   286→          ? String((data as { message: unknown }).message ?? "")
   287→   287→          : "";
   288→   288→      const bodyStatus =
   289→   289→        data && typeof data === "object" && "status" in data
   290→   290→          ? (data as { status: unknown }).status
   291→   291→          : undefined;
   292→   292→      return {
   293→   293→        success: false,
   294→   294→        status: typeof bodyStatus === "number" ? bodyStatus : res.status,
   295→   295→        raw: data,
   296→   296→        error: `sms.ir ${label} خطای HTTP ${res.status}${
   297→   297→          bodyStatus !== undefined ? ` (کد وضعیت ${String(bodyStatus)})` : ""
   298→   298→        }${bodyMsg ? ` — ${bodyMsg}` : ""}`,
   299→   299→      };
   300→   300→    }
   301→   301→
   302→   302→    const statusField =
   303→   303→      data && typeof data === "object" && "status" in data
   304→   304→        ? (data as { status: unknown }).status
   305→   305→        : undefined;
   306→   306→
   307→   307→    if (statusField !== undefined && statusField !== 1) {
   308→   308→      const messageField =
   309→   309→        data && typeof data === "object" && "message" in data
   310→   310→          ? (data as { message: unknown }).message
   311→   311→          : undefined;
   312→   312→      return {
   313→   313→        success: false,
   314→   314→        status: typeof statusField === "number" ? statusField : res.status,
   315→   315→        raw: data,
   316→   316→        error: `sms.ir ${label} خطای وضعیت ${String(statusField)}${
   317→   317→          messageField ? ` — ${String(messageField)}` : ""
   318→   318→        }`,
   319→   319→      };
   320→   320→    }
   321→   321→
   322→   322→    return { success: true, status: res.status, raw: data, cost: extractSmsCost(data) };
   323→   323→  } catch (e) {
   324→   324→    return {
   325→   325→      success: false,
   326→   326→      error: e instanceof Error ? e.message : `خطای ناشناخته در اتصال به sms.ir (${label})`,
   327→   327→    };
   328→   328→  }
   329→   329→}

   348→   348→  const templateIdRaw = process.env.SMSIR_TICKET_TEMPLATE_ID || "942763";

   612→   612→  const apiKey = process.env.SMSIR_API_KEY;
   613→   613→
   614→   614→  // v53 — شناسهٔ صریح (templateIdOverride) بر env ارجحیت دارد
   615→   615→  const overrideRaw =
   616→   616→    opts.templateIdOverride != null ? String(opts.templateIdOverride).trim() : "";
   617→   617→  const templateIdRaw = overrideRaw
   618→   618→    ? overrideRaw
   619→   619→    : opts.templateEnvKey
   620→   620→      ? process.env[SMS_TEMPLATE_ENV_KEYS[opts.templateEnvKey as SmsTemplateKey]]
   621→   621→      : undefined;
   622→   622→
   623→   623→  const normalized = normalizeMobileForSmsIr(opts.mobile);
   624→   624→  if (!/^9\d{9}$/.test(normalized)) {
   625→   625→    return { sent: false, skipped: "invalid_mobile", error: `شماره نامعتبر: ${normalized}` };
   626→   626→  }
   627→   627→
   628→   628→  // ─── دداپ ابدی ───
   629→   629→  if (!opts.force) {
   630→   630→    const existing = await db.smsLog.findUnique({
   631→   631→      where: { mobile_key: { mobile: normalized, key: opts.key } },
   632→   632→      select: { status: true },
   633→   633→    });
   634→   634→    if (existing?.status === "sent") {
   635→   635→      return { sent: false, skipped: "already_sent" };
   636→   636→    }
   637→   637→  }
   638→   638→
   639→   639→  if (!apiKey) {
   640→   640→    // بدون کلید API — سناریو بی‌صدا رد می‌شود (سیستم پیامک اختیاری است)
   641→   641→    return { sent: false, skipped: "no_template", error: "SMSIR_API_KEY تنظیم نشده است." };
   642→   642→  }
   643→   643→  if (!templateIdRaw) {
   644→   644→    return { sent: false, skipped: "no_template", error: "شناسه قالب تنظیم نشده است (env یا override)." };
   645→   645→  }
   646→   646→  const templateId = Number(templateIdRaw);
   647→   647→  if (!Number.isFinite(templateId) || templateId <= 0) {
   648→   648→    return { sent: false, skipped: "no_template", error: "شناسه قالب نامعتبر است." };
   649→   649→  }
   650→   650→
   651→   651→  let params = opts.params;
   652→   652→
   653→   653→  // ─── v48 گارد طول پارامتر — فیکس ریشه‌ای خطای 114 sms.ir ───
   654→   654→  // «طول رشته مقدار پارامتر، بیش از حد مجاز (۲۵ کاراکتر) میباشد»
   655→   655→  //  • NAME طولانی → به ۲۵ کاراکتر کوتاه می‌شود (پیامک می‌رود)
   656→   656→  //  • هر پارامتر دیگر (LINK و…) اگر بعد از این گارد هنوز بلند باشد → حذف
   657→   657→  //    می‌شود (مثل نردبان خودترمیمِ بدون لینک) تا هرگز 400/114 نخوریم.
   658→   658→  // لینک‌های تمدید حالا از sms-short-link.ts می‌آیند (r/XXXXXXXX = ۱۰ کاراکتر)
   659→   659→  // و عملاً هرگز به این شاخه نمی‌خورند — این گارد ایمنیِ ساختاری است.
   660→   660→  {
   661→   661→    const MAX = 25;
   662→   662→    const clamped: Array<{ name: string; value: string }> = [];
   663→   663→    for (const p of params) {
   664→   664→      const v = String(p.value ?? "");
   665→   665→      if (v.length > MAX) {
   666→   666→        if (p.name === "NAME") {
   667→   667→          clamped.push({ name: p.name, value: v.slice(0, MAX) });
   668→   668→          console.warn(`[sendTemplateSms] (${opts.key}) NAME > ${MAX} chars → truncated to ${MAX}`);
   669→   669→        } else {
   670→   670→          console.warn(
   671→   671→            `[sendTemplateSms] (${opts.key}) پارامتر ${p.name} طولش ${v.length} بود (سقف ${MAX}) → حذف شد (ضد خطای 114)`
   672→   672→          );
   673→   673→          // پارامتر حذف می‌شود
   674→   674→        }
   675→   675→      } else {
   676→   676→        clamped.push(p);
   677→   677→      }
   678→   678→    }
   679→   679→    params = clamped;
   680→   680→  }
   681→   681→
   682→   682→  // ─── v42 گارد نام خالی — دیریکتیو مالک ───
   683→   683→  // «در هیچ‌کدام از پیام‌ها نباید به‌جای نام کاربر «کاربر» نوشته شود.»
   684→   684→  // اگر پارامتر NAME خالی است (کاربر هنوز نامش را ثبت نکرده — مثل کاربرانی که
   685→   685→  // فقط با OTP وارد شده‌اند و آنبوردینگ را تمام نکرده‌اند)، پیامک با نامِ
   686→   686→  // جعلی «کاربر» نمی‌رود.
   687→   687→  //
   688→   688→  // 🩹 v43 — اصلاح دیریکتیو مالک (قالب 669068):
   689→   689→  // «همون برای کاربری که انبوردینگ تکمیل نکرده و اسمش نیست بجای اینکه متغیر
   690→   690→  //  نیم رو خالی بذاری یا پیامک نفرستی. پیامک بفرست و بنویس ورزشکار.»
   691→   691→  // → قالب onboarding_nudge استثناست: NAME خالی خودکار با «ورزشکار» پر و
   692→   692→  //   پیامک ارسال می‌شود. بقیهٔ قالب‌ها همان گارد سخت v42 را دارند (skip
   693→   693→  //   با no_name — پیامک بدون نام یا با placeholder هرگز نمی‌رود).
   694→   694→  const fallbackName =
   695→   695→    NAME_FALLBACK_BY_TEMPLATE[opts.templateEnvKey as SmsTemplateKey];
   696→   696→  if (opts.params.some((p) => p.name === "NAME" && !String(p.value ?? "").trim())) {
   697→   697→    if (fallbackName) {
   698→   698→      params = opts.params.map((p) =>
   699→   699→        p.name === "NAME" ? { ...p, value: fallbackName } : p
   700→   700→      );
   701→   701→      console.info(
   702→   702→        `[sendTemplateSms] (${opts.key}) NAME خالی بود → «${fallbackName}» (دیریکتیو مالک v43 — قالب 669068)`
   703→   703→      );
   704→   704→    } else {
   705→   705→      console.warn(
   706→   706→        `[sendTemplateSms] (${opts.key}) skipped: empty NAME param — user has no registered name yet`
   707→   707→      );
   708→   708→      return {
   709→   709→        sent: false,
   710→   710→        skipped: "no_name",
   711→   711→        error: "نام کاربر ثبت نشده است؛ پیامک بدون نام ارسال نمی‌شود.",
   712→   712→      };
   713→   713→    }
   714→   714→  }
   715→   715→
   716→   716→  const result = await postVerify(normalized, templateId, params, apiKey, `(${opts.key})`);
   717→   717→
   718→   718→  // ─── v52 حسابداری — لاگ هزینهٔ ارسال (best-effort، مستقل از داپ) ───
   719→   719→  await logSmsResultCost({
   720→   720→    normalizedMobile: normalized,
   721→   721→    scenario: opts.key,
   722→   722→    templateId: String(templateId),
   723→   723→    result,
   724→   724→    userId: opts.userId ?? null,
   725→   725→  });
   726→   726→
   727→   727→  // ─── ثبت در SmsLog (upsert — رکورد failed قبلی به sent ارتقا می‌یابد) ───
   728→   728→  try {
   729→   729→    await db.smsLog.upsert({
   730→   730→      where: { mobile_key: { mobile: normalized, key: opts.key } },
   731→   731→      create: {
   732→   732→        mobile: normalized,
   733→   733→        key: opts.key,
   734→   734→        templateId: String(templateId),
   735→   735→        userId: opts.userId ?? null,
   736→   736→        status: result.success ? "sent" : "failed",
   737→   737→        error: result.success ? null : (result.error ?? null),
   738→   738→      },
   739→   739→      update: {
   740→   740→        templateId: String(templateId),
   741→   741→        userId: opts.userId ?? undefined,
   742→   742→        status: result.success ? "sent" : "failed",
   743→   743→        error: result.success ? null : (result.error ?? null),
   744→   744→        createdAt: result.success ? new Date() : undefined,
   745→   745→      },
   746→   746→    });
   747→   747→  } catch (e) {
   748→   748→    // دداپ هرگز نباید نتیجهٔ ارسال را خراب کند
   749→   749→    console.warn("[sendTemplateSms] smsLog upsert failed:", e);
   750→   750→  }
   751→   751→
   752→   752→  if (!result.success) {
   753→   753→    return { sent: false, error: result.error };
   754→   754→  }
   755→   755→  return { sent: true };
   756→   756→}

   758→   758→/**
   759→   759→ * v53 — ارسال پیامک با «متن آزاد» (endpoint bulk) برای سناریوهایی که قالب
   760→   760→ * ثبت‌شدهٔ sms.ir ندارند (مثلاً ترک درگاه پرداخت تا ادمین قالب را در پنل ثبت
   761→   761→ * نکرده). برخلاف sendBulkSms داخلی (WebOTP)، متن کاملاً آزاد است.
   762→   762→ *
   763→   763→ * دداپ ابدی مانند sendTemplateSms: هر (mobile, key) فقط یک بار «sent» می‌شود —
   764→   764→ * پس کلید باید معنادار و مخصوص همان سناریو/رکورد باشد (مثلاً
   765→   765→ * abandoned_cart_{paymentId}). شکست → رکورد failed (تلاش بعدی ممکن است).
   766→   766→ * خط ارسال (SMSIR_LINE) در صورت تنظیم، در بدنه می‌آید (الگوی v35).
   767→   767→ */
   768→   768→export async function sendFreeTextSms(
   769→   769→  mobile: string,
   770→   770→  message: string,
   771→   771→  opts: { key: string; userId?: string | null }
   772→   772→): Promise<SmsIrResult> {
   773→   773→  const apiKey = process.env.SMSIR_API_KEY;
   774→   774→  const normalized = normalizeMobileForSmsIr(mobile);
   775→   775→
   776→   776→  if (!/^9\d{9}$/.test(normalized)) {
   777→   777→    return { success: false, error: `شماره نامعتبر: ${normalized}` };
   778→   778→  }
   779→   779→
   780→   780→  // ─── دداپ ابدی (همان کلید سناریو — پیامک تکراری هرگز) ───
   781→   781→  try {
   782→   782→    const existing = await db.smsLog.findUnique({
   783→   783→      where: { mobile_key: { mobile: normalized, key: opts.key } },
   784→   784→      select: { status: true },
   785→   785→    });
   786→   786→    if (existing?.status === "sent") {
   787→   787→      return { success: false, error: "already_sent (dedupe)" };
   788→   788→    }
   789→   789→  } catch {
   790→   790→    // خطای خواندن دداپ نباید ارسال را ببندد
   791→   791→  }
   792→   792→
   793→   793→  if (!apiKey) {
   794→   794→    return { success: false, error: "SMSIR_API_KEY تنظیم نشده است." };
   795→   795→  }
   796→   796→
   797→   797→  const body: Record<string, unknown> = {
   798→   798→    mobiles: [normalized],
   799→   799→    messageText: message,
   800→   800→  };
   801→   801→  const line = process.env.SMSIR_LINE?.trim();
   802→   802→  if (line) body.line = line;
   803→   803→
   804→   804→  let result: SmsIrResult;
   805→   805→  try {
   806→   806→    const res = await fetch(SMSIR_BULK_URL, {
   807→   807→      method: "POST",
   808→   808→      headers: {
   809→   809→        "Content-Type": "application/json",
   810→   810→        Accept: "application/json",
   811→   811→        "x-api-key": apiKey,
   812→   812→      },
   813→   813→      body: JSON.stringify(body),
   814→   814→      cache: "no-store",
   815→   815→      signal: AbortSignal.timeout(15_000),
   816→   816→    });
   817→   817→
   818→   818→    let data: unknown = null;
   819→   819→    try {
   820→   820→      data = await res.json();
   821→   821→    } catch {}
   822→   822→
   823→   823→    if (!res.ok) {
   824→   824→      const httpMsg =
   825→   825→        data && typeof data === "object" && "message" in data
   826→   826→          ? String((data as { message: unknown }).message ?? "")
   827→   827→          : "";
   828→   828→      const httpStatus =
   829→   829→        data && typeof data === "object" && "status" in data
   830→   830→          ? ` (کد وضعیت ${String((data as { status: unknown }).status)})`
   831→   831→          : "";
   832→   832→      result = {
   833→   833→        success: false,
   834→   834→        status: res.status,
   835→   835→        raw: data,
   836→   836→        error: `sms.ir (bulk) خطای HTTP ${res.status}${httpStatus}${httpMsg ? ` — ${httpMsg}` : ""}`,
   837→   837→      };
   838→   838→    } else {
   839→   839→      const statusField =
   840→   840→        data && typeof data === "object" && "status" in data
   841→   841→          ? (data as { status: unknown }).status
   842→   842→          : undefined;
   843→   843→      if (statusField !== undefined && statusField !== 1) {
   844→   844→        const messageField =
   845→   845→          data && typeof data === "object" && "message" in data
   846→   846→            ? (data as { message: unknown }).message
   847→   847→            : undefined;
   848→   848→        result = {
   849→   849→          success: false,
   850→   850→          status: typeof statusField === "number" ? statusField : undefined,
   851→   851→          raw: data,
   852→   852→          error: `sms.ir (bulk) status=${String(statusField)}${messageField ? ` — ${String(messageField)}` : ""}`,
   853→   853→        };
   854→   854→      } else {
   855→   855→        result = { success: true, raw: data, ...extractSmsCostFields(data) };
   856→   856→      }
   857→   857→    }
   858→   858→  } catch (err: any) {
   859→   859→    result = { success: false, error: err?.message || "network error" };
   860→   860→  }
   861→   861→
   862→   862→  // ─── حسابداری + دداپ (الگوی sendTemplateSms) ───
   863→   863→  await logSmsResultCost({
   864→   864→    normalizedMobile: normalized,
   865→   865→    scenario: opts.key,
   866→   866→    templateId: "bulk",
   867→   867→    result,
   868→   868→    userId: opts.userId ?? null,
   869→   869→  });
   870→   870→
   871→   871→  try {
   872→   872→    await db.smsLog.upsert({
   873→   873→      where: { mobile_key: { mobile: normalized, key: opts.key } },
   874→   874→      create: {
   875→   875→        mobile: normalized,
   876→   876→        key: opts.key,
   877→   877→        templateId: "bulk",
   878→   878→        userId: opts.userId ?? null,
   879→   879→        status: result.success ? "sent" : "failed",
   880→   880→        error: result.success ? null : (result.error ?? null),
   881→   881→      },
   882→   882→      update: {
   883→   883→        userId: opts.userId ?? undefined,
   884→   884→        status: result.success ? "sent" : "failed",
   885→   885→        error: result.success ? null : (result.error ?? null),
   886→   886→        createdAt: result.success ? new Date() : undefined,
   887→   887→      },
   888→   888→    });
   889→   889→  } catch (e) {
   890→   890→    console.warn("[sendFreeTextSms] smsLog upsert failed:", e);
   891→   891→  }
   892→   892→
   893→   893→  return result;
   894→   894→}

   896→   896→/** استخراج هزینه از ساختار پاسخ bulk (همان الگوی extractSmsCost) */
   897→   897→function extractSmsCostFields(data: unknown): { cost?: number } {
   898→   898→  const cost = extractSmsCost(data);
   899→   899→  return cost != null ? { cost } : {};
   900→   900→}
