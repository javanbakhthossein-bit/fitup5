    58→async function loadMediaDataUrls(m: {
    59→  mediaUrl: string | null;
    60→  mediaType: string | null;
    61→  mediaFrames: string | null;
    62→}): Promise<string[]> {
    63→  try {
    64→    if (!m.mediaUrl) return [];
    65→    if (m.mediaType === "image") {
    66→      const buf = await readFile(absolutePathForUploadUrl(m.mediaUrl));
    67→      if (buf.length === 0) return [];
    68→      const mime = m.mediaUrl.endsWith(".png") ? "image/png" : "image/webp";
    69→      return [`data:${mime};base64,${buf.toString("base64")}`];
    70→    }
    71→    // ویدیو — اولویت: فریم‌های ذخیره‌شده (URL فایل) → استخراج دوباره از فایل
    72→    let frameFileUrls: string[] = [];
    73→    if (m.mediaFrames) {
    74→      try {
    75→        const parsedFrames = JSON.parse(m.mediaFrames);
    76→        if (Array.isArray(parsedFrames)) frameFileUrls = parsedFrames.filter((u: unknown): u is string => typeof u === "string");
    77→      } catch {
    78→        // JSON خراب → فال‌بک به استخراج دوباره
    79→      }
    80→    }
    81→    if (frameFileUrls.length === 0) {
    82→      // extractVideoFramesAsDataUrls خودش data URL برمی‌گرداند
    83→      return await extractVideoFramesAsDataUrls(absolutePathForUploadUrl(m.mediaUrl), 6);
    84→    }
    85→    const dataUrls: string[] = [];
    86→    for (const u of frameFileUrls.slice(0, MAX_CLIENT_FRAMES)) {
    87→      try {
    88→        const buf = await readFile(absolutePathForUploadUrl(u));
    89→        if (buf.length === 0) continue;
    90→        const mime = u.endsWith(".png") ? "image/png" : u.endsWith(".webp") ? "image/webp" : "image/jpeg";
    91→        dataUrls.push(`data:${mime};base64,${buf.toString("base64")}`);
    92→      } catch {
    93→        // فریم حذف‌شده از دیسک — رد شو
    94→      }
    95→    }
    96→    return dataUrls;
    97→  } catch (err) {
    98→    console.error("[chat] loadMediaDataUrls failed:", err);
    99→    return [];
   100→  }

   121→async function collectHistoryMediaAnalyses(
   122→  userId: string,
   123→  history: {
   124→    id: string;
   125→    content: string;
   126→    mediaUrl: string | null;
   127→    mediaType: string | null;
   128→    mediaFrames: string | null;

   130→  }[]

   132→  try {
   133→    const recent = history.slice(-MEDIA_SYNC_WINDOW);
   134→    const mediaMsgs = recent.filter((m) => m.mediaUrl && m.mediaType);
   135→    if (mediaMsgs.length === 0) return "";
   136→
   137→    const parts: string[] = [];
   138→    let onDemandBudget = 1;
   139→    for (const m of mediaMsgs) {
   140→      if (m.mediaAnalysis?.trim()) {
   141→        parts.push(formatMediaAnalysisPart(m, m.mediaAnalysis.trim()));

   143→      }

   145→      onDemandBudget--;
   146→      try {
   147→        const dataUrls = await loadMediaDataUrls(m);
   148→        if (dataUrls.length === 0) continue;

   150→          m.mediaType === "video" ? "video-frames" : "image",

   159→        } catch (cacheErr) {
   160→          console.error("[chat] mediaAnalysis cache update failed:", cacheErr);

   162→      } catch (anErr) {
   163→        console.error("[chat] on-demand media analysis failed (skipping):", anErr);

   165→    }

   167→    return `\n\n[تحلیل رسانه‌های اخیر کاربر در همین گفتگو — تولید مدل بینایی فیتاپ — برای مرجع تو:\n${parts.join("\n")}\nاگر پرسش فعلی دربارهٔ همین رسانه‌هاست، پاسخت را بر اساس این تحلیل‌ها بده؛ رسانه‌ها خودشان پیوست نشده‌اند و لازم نیست بگویی «نمی‌توانم ببینم».]`;
   168→  } catch (err) {
   169→    console.error("[chat] collectHistoryMediaAnalyses failed (continuing without sync):", err);
   170→    return "";
   171→  }

   280→  const fileName = `chat-image-${Date.now()}-${Math.random().toString(36).slice(2, 8)}.webp`;
   281→  const { url } = await savePrivateMediaFile("chat", fileName, processed);
   282→
   283→  const optimizedBase64 = processed.toString("base64");
   284→  return {

   286→    mime: "image/webp",
   287→    size: processed.length,

   559→/**
   560→ * v73.4 — کار پس‌زمینهٔ پیام مدیا‌دار (fire-and-forget):
   561→ *   فشرده‌سازی/فریم‌سازی ویدیو (در صورت نیاز) → تحلیل رسانه با مدل بینایی

   564→ * catch: placeholder پیام خطای دوستانه می‌گیرد؛ ردیف کاربر هرگز حذف نمی‌شود.
   565→ * حالت retry: اگر dataUrls آماده نبود (تلاش مجدد با همان clientId)، مدیا از
   566→ * ردیف ذخیره‌شدهٔ کاربر بارگذاری می‌شود (loadMediaDataUrls).
   567→ */
   568→async function runMediaAnalysisJob(ctx: {
   569→  user: AuthUser;
   570→  userMsgId: string;
   571→  placeholderId: string;
   572→  message: string;
   573→  kind: "image" | "video-frames" | null;
   574→  dataUrls: string[] | null;
   575→  /** فقط مسیر base64 بار اول — فشرده‌سازی + استخراج فریم اینجا انجام می‌شود */
   576→  needsVideoProcessing: boolean;
   577→  videoUrl: string | null;
   578→}): Promise<void> {
   579→  let kind = ctx.kind;
   580→  let dataUrls: string[] = ctx.dataUrls ?? [];
   581→  let imageAnalysisNote = "";
   582→  let videoAnalysisNote = "";
   583→  try {
   584→    // ۱) پس‌پردازش ویدیوی مسیر base64 — قبلاً sync بود و گیت‌وی را می‌بُرید
   585→    if (ctx.needsVideoProcessing && ctx.videoUrl) {
   586→      // فشرده‌سازی درجا (قانون مالک: همیشه نسخهٔ کم‌حجم نگه داشته شود) — هرگز throw نمی‌کند
   587→      try {
   588→        const cmp = await compressVideoFileInPlace(absolutePathForUploadUrl(ctx.videoUrl));
   589→        if (cmp.compressed) {
   590→          console.log(`[chat] video compressed (background): ${formatBytes(cmp.sizeBefore)} → ${formatBytes(cmp.sizeAfter)}`);
   591→        }
   592→      } catch (cmpErr) {
   593→        console.error("[chat] compression step failed (keeping original):", cmpErr);
   594→      }
   595→      // استخراج چند فریم کلیدی با ffmpeg + ذخیره روی دیسک (سینک رسانه پیام‌های بعدی)
   596→      try {
   597→        const frameDataUrls = await extractVideoFramesAsDataUrls(absolutePathForUploadUrl(ctx.videoUrl), 6);
   598→        if (frameDataUrls.length > 0) {
   599→          kind = "video-frames";
   600→          dataUrls = frameDataUrls;
   601→          const frameUrls = await saveFramesToDisk(frameDataUrls);
   602→          await db.chatMessage.update({
   603→            where: { id: ctx.userMsgId },
   604→            data: { mediaFrames: frameUrls.length > 0 ? JSON.stringify(frameUrls) : null },
   605→          });
   606→          // مصرف سهمیه — استخراج فریم موفق بود (قبلاً بعد از extraction بود)

   609→          // v75 — استخراج فریم حتی بعد از fallback ریموکس ناموفق بود؛ پیام صادقانه
   610→          // (دیگر مدل عذرخواهیِ «ویدیو همراهش نبود» نمی‌کند — صریح می‌گوییم ویدیو
   611→          // رسیده ولی پردازشش نشده و باید دوباره بفرستد)
   612→          videoAnalysisNote = `\n\n[ویدیوی کاربر دریافت شده اما استخراج فریم از آن ممکن نشد (فرمت/کدک پشتیبانی‌نشده یا فایل خراب). صادقانه به کاربر بگو ویدیوش پردازش نشد و از او بخواه با فرمت mp4/H.264 دوباره ضبط و ارسال کند یا حرکت را توضیح بدهد. هرگز نگو ویدیویی ارسال نشده است.]`;
   613→        }
   614→      } catch (err) {
   615→        console.error("[chat] video frame extraction failed (background):", err);
   616→        videoAnalysisNote = `\n\n[ویدیوی کاربر دریافت شده اما استخراج فریم از آن ممکن نشد (خطای پردازش). صادقانه به کاربر بگو ویدیوش پردازش نشد و از او بخواه دوباره بفرستد یا حرکت را توضیح بدهد. هرگز نگو ویدیویی ارسال نشده است.]`;
   617→      }
   618→    }
   619→
   620→    // ۲) حالت retry (تلاش مجدد با همان clientId بدون dataUrls آماده) —
   621→    //    مدیا از ردیف ذخیره‌شده بارگذاری می‌شود
   622→    if (dataUrls.length === 0 && !ctx.needsVideoProcessing) {
   623→      const row = await db.chatMessage.findUnique({ where: { id: ctx.userMsgId } });
   624→      if (row?.mediaUrl && row.mediaType) {

   630→    // ۳) ─── v71 مرحلهٔ ۱: تحلیل رسانهٔ همین پیام توسط مدل بینایی (همان کد قبلی) ───
   631→    // خروجی در mediaAnalysis کش می‌شود و به مرحلهٔ ۲ (دیپ‌سیک) تزریق می‌شود.
   632→    // v75 — یک retry داخلی هم اضافه شد: خطای گذرای گیت‌وی/مدل دیگر مستقیماً به
   633→    // «نتونستم ببینم» نمی‌رسد؛ اول یک‌بار دوباره تلاش می‌شود (درخواست مالک:
   634→    // «ویدیو همراهش نبود» دیگر در هیچ‌جا نباید رخ بدهد).
   635→    let currentMediaAnalysis = "";
   636→    if (dataUrls.length > 0 && kind) {
   637→      let anErrLast: unknown = null;
   638→      for (let attempt = 1; attempt <= 2; attempt++) {
   639→        try {
   640→          currentMediaAnalysis = await analyzeChatMedia(kind, dataUrls, ctx.message, ctx.user.id);
   641→          anErrLast = null;

   645→          console.error(`[chat] analyzeChatMedia failed (attempt ${attempt}/2):`, anErr);
   646→          if (attempt < 2) await new Promise((r) => setTimeout(r, 2000));
   647→        }
   648→      }
   649→      if (anErrLast == null && currentMediaAnalysis) {
   650→        try {
   651→          await db.chatMessage.update({ where: { id: ctx.userMsgId }, data: { mediaAnalysis: currentMediaAnalysis } });
   652→        } catch (cacheErr) {
   653→          console.error("[chat] current media analysis cache failed:", cacheErr);
   654→        }
   655→      }
   656→      if (anErrLast != null) {
   657→        // هر دو تلاش شکست خورد — نکتهٔ صادقانه و کاربردی (بدون جملاتی که مدل را
   658→        // به انکار دیدن رسانه وادار کند):

   661→        } else {
   662→          imageAnalysisNote = `\n\n[عکس کاربر دریافت شده است اما پردازش خودکار تصویر در این لحظه ناموفق بود (خطای موقت سرویس). صادقانه به کاربر بگو پردازش عکس الان ممکن نشد و از او بخواه عکس را دوباره بفرستد. هرگز نگو عکسی دریافت نشده است.]`;
   663→        }
   664→      }
   665→    }
   666→
   667→    // ۴) ساخت محتوای نهایی پیام کاربر (شامل یادداشت‌های شکست تحلیل در صورت وجود)
   668→    let finalMessage = ctx.message;
   669→    if (imageAnalysisNote) {
   670→      finalMessage = `${ctx.message || "(بدون متن)"}${imageAnalysisNote}`;
   671→    }
   672→    if (videoAnalysisNote) {
   673→      finalMessage = `${finalMessage}${ctx.message || imageAnalysisNote ? "\n\n" : ""}${videoAnalysisNote}`;
   674→    }
   675→
   676→    // ۴-پ) v77 — تشخیص و اجرای درخواست تغییر برنامه از متنِ همراه رسانه
   677→    // (مثلاً عکس غذا + «اینو توی برنامه‌ام جایگزین کن») — یادداشت سیستمیِ
   678→    // وضعیت بازتولید به مدل پاسخ‌نویس تزریق می‌شود
   679→    let planChangeNote = "";
   680→    if (ctx.message && ctx.message.trim()) {
   681→      try {
   682→        const pcr = await applyPlanChangeRequest(ctx.user.id, ctx.message);

   710→    } catch (upErr) {

   712→    }
   713→  }
