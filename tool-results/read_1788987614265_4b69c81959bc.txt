   185→    /**
   186→     * v1.5.5 — لینک عمیق ورودی (App Link یا اسکیم fitup:// یا اعلان) یا null:
   187→     *  ۱) extra «fitup_link» از NotificationSync — مسیر/کوئری نسبی مثل
   188→     *     "?tab=progress&section=checkup" یا "/panel" → SITE_URL + آن
   189→     *  ۲) dataِ https روی دامنهٔ خودمان (fittup.ir) → همان URL — پارامترهای
   190→     *     ?tab= ?ref= ?renewal= ?screen= ?article= دست‌نخورده می‌مانند
   191→     *  ۳) v1.5.5 — اسکیم اختصاصی fitup://open?url=<encoded> از صفحهٔ /go سایت
   192→     *     (لینک‌های هوشمند پیامک — فقط URL دامنهٔ خودمان پذیرفته می‌شود)
   193→     * (null-امن — هرگز به‌خاطر لینک خراب کرش نمی‌کند)
   194→     */
   195→    private fun deepLinkUrl(from: Intent?): String? {
   196→        if (from == null) return null
   197→        try {
   198→            val extra = from.getStringExtra("fitup_link")
   199→            if (!extra.isNullOrBlank() && (extra.startsWith("/") || extra.startsWith("?"))) {
   200→                return BuildConfig.SITE_URL.trimEnd('/') + extra
   201→            }
   202→            val data = from.data ?: return null
   203→            val scheme = data.scheme?.lowercase()
   204→            // ─── v1.5.5: fitup://open?url=<encoded> — از لینک هوشمند /go سایت ───
   205→            if (scheme == "fitup") {
   206→                val target = data.getQueryParameter("url") ?: return null
   207→                if (target.startsWith("/")) {
   208→                    // مسیر نسبی روی دامنهٔ خودمان
   209→                    return BuildConfig.SITE_URL.trimEnd('/') + target
   210→                }
   211→                val t = android.net.Uri.parse(target)
   212→                val tScheme = t.scheme?.lowercase()
   213→                val tHost = t.host?.lowercase() ?: return null
   214→                if (isAllowedHost(tHost) && tScheme == "https") return target
   215→                return null
   216→            }
   217→            if (scheme != "https") return null
   218→            val host = data.host?.lowercase() ?: return null
   219→            // فقط دامنه‌های مجاز — همان منطق isAllowedHost (فیتاپ و زیردامنه‌ها)
   220→            if (isAllowedHost(host)) return data.toString()
   221→        } catch (_: Exception) {
   222→        }
   223→        return null
   224→    }

   547→    /* ───────────── پرداخت درون‌برنامه‌ای بازار (پولکی) ───────────── */
   548→
   549→    private fun setupPayment() {
   550→        // کلید RSA از پیشخان توسعه‌دهندگان بازار (تب «پرداخت درون‌برنامه‌ای» برنامه)
   551→        val configuredKey = BuildConfig.BAZAAR_RSA_PUBLIC_KEY.trim()
   552→        rsaKeyValid = configuredKey.isNotEmpty() && configuredKey != "PASTE_YOUR_RSA_KEY_HERE"
   553→        if (!rsaKeyValid) {
   554→            Log.w("FitUp", "⚠️ BAZAAR_RSA_PUBLIC_KEY تنظیم نشده — خرید fail-closed می‌شود (امنیت). کلید را از پیشخان بازار بگیرید و در app/build.gradle.kts قرار دهید.")
   555→        }
   556→        val securityCheck = if (rsaKeyValid) {
   557→            SecurityCheck.Enable(rsaPublicKey = configuredKey)
   558→        } else {
   559→            // اتصال برای query/restore باز می‌ماند؛ خود خرید fail-closed است (startBazaarPurchase)
   560→            SecurityCheck.Disable
   561→        }
   562→
   563→        val config = PaymentConfiguration(localSecurityCheck = securityCheck)
   564→        payment = Payment(context = this, config = config)
   565→
   566→        // اتصال به سرویس پرداخت بازار — DSL با receiver
   567→        connection = payment?.connect {
   568→            connectionSucceed {
   569→                paymentReady = true
   570→                // بازیابی خریدهای consume-نشده (کرش بین پرداخت و فعال‌سازی سرور)
   571→                queryUnconsumedPurchases()
   572→            }
   573→            connectionFailed { _ ->
   574→                paymentReady = false
   575→            }
   576→            disconnected {
   577→                paymentReady = false
   578→            }
   579→        }
   580→    }

   582→    /** بازیابی خریدها — فقط وقتی صفحه JS آماده است (پس از onPageFinished صدا زده می‌شود) */
   583→    private var jsReadyForRestore = false

   589→    /** خریدهای consume-نشده → به سایت (idempotent) → سایت بعد از فعال‌سازی consume می‌کند */
   590→    private fun queryUnconsumedPurchases() {
   591→        if (!jsReadyForRestore) return
   592→        try {
   593→            payment?.getPurchasedProducts {
   594→                querySucceed { purchaseList ->
   595→                    val items = purchaseList.filter { !it.purchaseToken.isNullOrBlank() }
   596→                    if (items.isEmpty()) return@querySucceed
   597→                    val arr = JSONArray()
   598→                    for (p in items) {
   599→                        arr.put(
   600→                            JSONObject()
   601→                                .put("productId", p.productId ?: "")
   602→                                .put("purchaseToken", p.purchaseToken ?: "")
   603→                                .put("orderId", p.orderId ?: "")
   604→                                // v1.5.7 — امضای خرید برای مسیر پشتیبان RSA سرور
   605→                                .put("dataJson", p.originalJson ?: "")
   606→                                .put("signature", p.dataSignature ?: "")
   607→                        )
   608→                    }
   609→                    runOnUiThread {
   610→                        webView.evaluateJavascript(
   611→                            "window.__fitupBazaarRestore && window.__fitupBazaarRestore($arr);",
   612→                            null
   613→                        )
   614→                    }
   615→                }
   616→                queryFailed { _ ->
   617→                    // بازار قدیمی/نصب‌نشده — نادیده بگیر
   618→                }
   619→            }
   620→        } catch (e: Exception) {
   621→            Log.w("FitUp", "queryPurchases failed: ${e.message}")
   622→        }
   623→    }
   624→
   625→    /** شروع خرید از بازار — نتیجه به وب برمی‌گردد. fail-closed بدون کلید RSA.
   626→     *  dynamicPriceToken: شناسه قیمت پویا (تخفیف/اعتبار ارتقا) — خالی = قیمت پایه SKU */
   627→    private fun startBazaarPurchase(sku: String, payloadJson: String, callbackId: String, dynamicPriceToken: String) {
   628→        // پاسخ به وب: window.__bazaarPaymentResult(id, resultObject)
   629→        fun respond(json: JSONObject) {
   630→            runOnUiThread {
   631→                webView.evaluateJavascript(
   632→                    "window.__bazaarPaymentResult && window.__bazaarPaymentResult('$callbackId', $json);",
   633→                    null
   634→                )
   635→            }
   636→        }
   637→
   638→        if (!rsaKeyValid) {
   639→            // fail-closed (ممیزی 2-d باگ #1): بدون کلید RSA، خرید هرگز انجام نمی‌شود —
   640→            // نه SecurityCheck.Disable با «خرید بدون وریفای محلی».
   641→            respond(
   642→                JSONObject()
   643→                    .put("ok", false)
   644→                    .put("error", "پرداخت درون‌برنامه‌ای پیکربندی نشده است (کلید RSA). لطفاً از نسخه وب سایت خرید کنید یا اپ را به‌روزرسانی کنید.")
   645→            )
   646→            return
   647→        }
   648→        if (!paymentReady) {
   649→            respond(
   650→                JSONObject()
   651→                    .put("ok", false)
   652→                    .put("error", "اتصال به پرداخت بازار برقرار نیست. برنامه بازار را باز کنید، وارد حساب شوید و دوباره تلاش کنید.")
   653→            )
   654→            return
   655→        }
   656→        if (activePaymentCallbackId != null) {
   657→            respond(JSONObject().put("ok", false).put("error", "یک پرداخت دیگر در حال انجام است."))
   658→            return
   659→        }
   660→        activePaymentCallbackId = callbackId
   661→
   662→        val finish: (JSONObject) -> Unit = { json ->
   663→            activePaymentCallbackId = null
   664→            respond(json)
   665→        }
   666→
   667→        val request = PurchaseRequest(
   668→            productId = sku,
   669→            payload = payloadJson,
   670→            dynamicPriceToken = dynamicPriceToken.ifBlank { null } // قیمت پویا (تخفیف) اگر سایت ثبت کرده باشد
   671→        )
   672→        payment?.purchaseProduct(
   673→            registry = activityResultRegistry,
   674→            request = request,
   675→        ) {
   676→            purchaseSucceed { purchaseInfo ->
   677→                val json = JSONObject()
   678→                    .put("ok", true)
   679→                    .put("purchaseToken", purchaseInfo.purchaseToken ?: "")
   680→                    .put("orderId", purchaseInfo.orderId ?: "")
   681→                    .put("productId", purchaseInfo.productId ?: sku)
   682→                    .put("payload", purchaseInfo.payload ?: payloadJson)
   683→                    .put("purchaseTime", purchaseInfo.purchaseTime ?: 0L)
   684→                    // v1.5.7 — امضای خرید (originalJson + signature) به سایت می‌رود تا
   685→                    // سرور بتواند در نبود API بازار، امضا را با کلید RSA پنل
   686→                    // راستی‌آزمایی کند (رفع ایراد بازرسی بازار درباره پیغام
   687→                    // «اعتبارنامه‌های API بازار تنظیم نیست» بعد از خرید موفق)
   688→                    .put("dataJson", purchaseInfo.originalJson ?: "")
   689→                    .put("signature", purchaseInfo.dataSignature ?: "")
   690→                // سایت خودش purchaseToken را برای فعال‌سازی به /api/payment/bazaar/purchase
   691→                // می‌فرستد و بعد از موفقیت consumePurchase را صدا می‌زند (تمدید ممکن می‌شود)
   692→                finish(json)
   693→            }
   694→            purchaseFailed { throwable ->
   695→                finish(
   696→                    JSONObject()
   697→                        .put("ok", false)
   698→                        .put("error", "پرداخت ناموفق: ${throwable?.message ?: "خطای نامشخص"}")
   699→                )
   700→            }
   701→            purchaseCanceled {
   702→                finish(
   703→                    JSONObject()
   704→                        .put("ok", false)
   705→                        .put("canceled", true)
   706→                        .put("error", "پرداخت توسط شما لغو شد.")
   707→                )
   708→            }
   709→            purchaseFlowBegan {
   710→                // جریان خرید شروع شد — نیازی به عمل نیست
   711→            }
   712→            failedToBeginFlow { throwable ->
   713→                // شروع جریان شکست خورد (بازار قدیمی/نصب‌نشده) — گارد single-flight آزاد می‌شود
   714→                finish(
   715→                    JSONObject()
   716→                        .put("ok", false)
   717→                        .put("error", "امکان شروع پرداخت نبود: ${throwable?.message ?: "برنامه بازار به‌روز نیست یا نصب نیست"}")
   718→                )
   719→            }
   720→        }
   721→    }

   741→    /** ذخیره data URL (PNG/PDF) در Downloads اندروید */
   742→    @SuppressLint("InlinedApi")
   743→    private fun downloadDataUrl(filename: String, dataUrl: String) {
   744→        try {
   745→            val safeName = filename.ifBlank { "fitup-${System.currentTimeMillis()}" }
   746→                .replace(Regex("[^A-Za-z0-9._-\\u0600-\\u06FF ]"), "_")
   747→            val comma = dataUrl.indexOf(',')
   748→            if (comma < 0 || !dataUrl.startsWith("data:", ignoreCase = true)) {
   749→                runOnUiThread { toast("فایل قابل ذخیره نیست") }
   750→                return
   751→            }
   752→            val meta = dataUrl.substring(5, comma)
   753→            val mime = meta.substringBefore(";").ifBlank { "application/octet-stream" }
   754→            val isBase64 = meta.contains("base64", ignoreCase = true)
   755→            val bytes = if (isBase64) {
   756→                Base64.decode(dataUrl.substring(comma + 1), Base64.DEFAULT)
   757→            } else {
   758→                dataUrl.substring(comma + 1).toByteArray(Charsets.UTF_8)
   759→            }
   760→
   761→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   762→                // اندروید ۱۰+ — MediaStore (بدون نیاز به مجوز)
   763→                val values = ContentValues().apply {
   764→                    put(MediaStore.Downloads.DISPLAY_NAME, safeName)
   765→                    put(MediaStore.Downloads.MIME_TYPE, mime)
   766→                    put(MediaStore.Downloads.IS_PENDING, 1)
   767→                }
   768→                val uri = contentResolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, values)
   769→                if (uri == null) {
   770→                    runOnUiThread { toast("خطا در ذخیره فایل") }
   771→                    return
   772→                }
   773→                contentResolver.openOutputStream(uri)?.use { it.write(bytes) }
   774→                values.clear()
   775→                values.put(MediaStore.Downloads.IS_PENDING, 0)
   776→                contentResolver.update(uri, values, null, null)
   777→            } else {
   778→                // اندروید ۹ و پایین‌تر — پوشه اپ (بدون مجوز نوشتن)
   779→                val dir = getExternalFilesDir(Environment.DIRECTORY_DOCUMENTS) ?: filesDir
   780→                val file = File(dir, safeName)
   781→                FileOutputStream(file).use { it.write(bytes) }
   782→            }
   783→            runOnUiThread { toast("«$safeName» ذخیره شد ✓") }
   784→        } catch (e: Exception) {
   785→            Log.e("FitUp", "download failed: ${e.message}")
   786→            runOnUiThread { toast("خطا در ذخیره فایل") }
   787→        }
   788→    }

   839→    /**
   840→     * مجوز runtime دوربین/میکروفون — دقیقاً وقتی سایت ضبط صدا/ویدیو می‌خواهد
   841→     * (WebChromeClient.onPermissionRequest، بعد از دیالوگ اجازه فارسی).
   842→     * اگر همه مجوزها از قبل داده شده باشند مستقیم grant می‌شود؛ وگرنه
   843→     * دیالوگ سیستمی اندروید.
   844→     */
   845→    private fun requestMediaRuntimePermissions(webRequest: PermissionRequest) {
   846→        // درخواست معلق قبلی را رد کن (درخواست همزمان/تکراری از صفحه)
   847→        pendingWebPermissionRequest?.let { if (it !== webRequest) it.deny() }
   848→        pendingWebPermissionRequest = webRequest
   849→        val perms = mutableListOf<String>()
   850→        if (webRequest.resources.contains(PermissionRequest.RESOURCE_VIDEO_CAPTURE)) {
   851→            perms.add(android.Manifest.permission.CAMERA)
   852→        }
   853→        if (webRequest.resources.contains(PermissionRequest.RESOURCE_AUDIO_CAPTURE)) {
   854→            perms.add(android.Manifest.permission.RECORD_AUDIO)
   855→        }
   856→        val alreadyGranted = perms.all {
   857→            ContextCompat.checkSelfPermission(this, it) == PackageManager.PERMISSION_GRANTED
   858→        }
   859→        if (perms.isEmpty() || alreadyGranted) {
   860→            pendingWebPermissionRequest = null
   861→            runOnUiThread { webRequest.grant(webRequest.resources) }
   862→        } else {
   863→            try {
   864→                mediaPermissionLauncher.launch(perms.toTypedArray())
   865→            } catch (_: Exception) {
   866→                // دیالوگ سیستم باز نشد (مثلاً گوشی خاص) — fail بسته
   867→                pendingWebPermissionRequest = null
   868→                runOnUiThread { webRequest.deny() }
   869→            }
   870→        }
   871→    }

   918→    /** کد OTP از کلیپ‌بورد — v1.5.7: هم کدِ خالص و هم «متن کامل پیامک» کپی‌شده.
   919→     *  برای جلوگیری از درج اشتباه عدد ۴-رقمی نامرتبط (مثل سال «1404»)، استخراج
   920→     *  گروه ۴-رقمی فقط وقتی انجام می‌شود که متن شامل «فیتاپ» یا fittup باشد یا
   921→     *  خود کلیپ‌بورد دقیقاً کد ۴-رقمی باشد. */
   922→    private fun maybeDispatchClipboardOtp() {
   923→        try {
   924→            val cm = getSystemService(Context.CLIPBOARD_SERVICE) as? android.content.ClipboardManager ?: return
   925→            val text = cm.primaryClip?.getItemAt(0)?.coerceToText(this)?.toString()?.trim() ?: return
   926→            if (text == lastClipboardDispatched) return
   927→            val code = when {
   928→                Regex("^\\d{4}$").matches(text) -> text
   929→                text.contains("فیتاپ") || text.contains("fittup", ignoreCase = true) ->
   930→                    Regex("(?<!\\d)\\d{4}(?!\\d)").findAll(text).map { it.value }.lastOrNull()
   931→                else -> null
   932→            }
   933→            if (code != null) {
   934→                lastClipboardDispatched = text
   935→                dispatchOtpCode(code)
   936→            }
   937→        } catch (_: Exception) {}
   938→    }
