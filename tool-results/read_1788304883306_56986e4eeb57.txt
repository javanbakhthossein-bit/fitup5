    63→/** ساعت فعلی تهران (۰-۲۳) — با Intl و timeZone Asia/Tehran، نه timezone مرورگر */
    64→function getTehranHour(): number {
    65→  try {
    66→    const h = new Intl.DateTimeFormat("en-US", {
    67→      timeZone: "Asia/Tehran",
    68→      hour: "numeric",
    69→      hour12: false,
    70→    }).format(new Date());
    71→    const parsed = parseInt(h, 10);
    72→    if (Number.isFinite(parsed)) return parsed % 24;
    73→  } catch {
    74→    // Intl در دسترس نیست — fallback به ساعت محلی
    75→  }
    76→  return new Date().getHours();
    77→}

    79→/** Format a JS Date as a Persian (Jalali) string with day-of-week + date + Tehran time */
    80→function formatPersianDateTime(d: Date): string {
    81→  try {
    82→    const weekdayNames = ["یکشنبه", "دوشنبه", "سه‌شنبه", "چهارشنبه", "پنجشنبه", "جمعه", "شنبه"];
    83→    // Intl supports islamic-civil but for Persian (Jalali) use fa-IR with persian calendar
    84→    const parts = new Intl.DateTimeFormat("fa-IR-u-ca-persian", {
    85→      weekday: "long",
    86→      day: "numeric",
    87→      month: "long",
    88→      year: "numeric",
    89→      timeZone: "Asia/Tehran",
    90→    }).formatToParts(d);
    91→    // Extract parts
    92→    const get = (t: string) => parts.find((p) => p.type === t)?.value ?? "";
    93→    const day = get("weekday");
    94→    const date = `${get("day")} ${get("month")} ${get("year")}`;
    95→    // Time — بر اساس ساعت تهران (نه timezone مرورگر)
    96→    const timeParts = new Intl.DateTimeFormat("en-US", {
    97→      timeZone: "Asia/Tehran",
    98→      hour: "2-digit",
    99→      minute: "2-digit",
   100→      hour12: false,
   101→    }).formatToParts(d);
   102→    const hh = timeParts.find((p) => p.type === "hour")?.value ?? "00";
   103→    const mm = timeParts.find((p) => p.type === "minute")?.value ?? "00";
   104→    return `${day}، ${date} - ${toPersianDigits(hh)}:${toPersianDigits(mm)}`;
   105→  } catch {
   106→    // Fallback if Intl calendar is unavailable
   107→    const dayIdx = (d.getDay() + 1) % 7;
   108→    const wd = PERSIAN_WEEKDAYS[dayIdx];
   109→    const hh = String(d.getHours()).padStart(2, "0");
   110→    const mm = String(d.getMinutes()).padStart(2, "0");
   111→    return `${wd} - ${toPersianDigits(hh)}:${toPersianDigits(mm)}`;
   112→  }
   113→}

   115→/**

   117→ * interval یک‌ثانیه‌ای فقط این کامپوننت کوچک را re-render می‌کند، نه کل

   120→const TehranClock = memo(function TehranClock() {
   121→  const [now, setNow] = useState<string>("");
   122→  useEffect(() => {
   123→    const tick = () => setNow(formatPersianDateTime(new Date()));
   124→    tick();
   125→    const id = setInterval(tick, 1000);
   126→    return () => clearInterval(id);
   127→  }, []);
   128→  return <span className="font-stat truncate">{now || "در حال بارگذاری..."}</span>;
   129→});

   131→/** Estimate calories burned during a workout using MET × weight × hours */
   132→function estimateCaloriesBurned(
   133→  minutes: number,
   134→  userWeightKg: number,
   135→  metValue = 6.5
   136→): number {
   137→  const hours = minutes / 60;
   138→  return Math.round(metValue * userWeightKg * hours);
   139→}

   772→/* ============ ACTIVITY RINGS (Apple-style SVG) ============ */
   773→function ActivityRings({
   774→  workout,
   775→  calories,
   776→  water,
   777→}: {
   778→  workout: number;
   779→  calories: number;
   780→  water: number;
   781→}) {
   782→  const rings = [
   783→    { progress: workout, color: "#F4C542", radius: 52, label: "تمرین" },
   784→    { progress: calories, color: "#10b981", radius: 40, label: "کالری" },
   785→    { progress: water, color: "#06b6d4", radius: 28, label: "آب" },
   786→  ];

   788→    <div className="relative w-44 h-44 shrink-0">
   789→      <svg viewBox="0 0 120 120" className="w-full h-full -rotate-90">
   790→        {rings.map((ring, i) => {
   791→          const c = 2 * Math.PI * ring.radius;
   792→          return (
   793→            <g key={i}>
   794→              <circle
   795→                cx="60"
   796→                cy="60"
   797→                r={ring.radius}
   798→                fill="none"
   799→                stroke={ring.color}
   800→                strokeWidth="9"
   801→                strokeOpacity="0.15"

   803→              <motion.circle
   804→                cx="60"
   805→                cy="60"
   806→                r={ring.radius}
   807→                fill="none"
   808→                stroke={ring.color}
   809→                strokeWidth="9"
   810→                strokeLinecap="round"
   811→                strokeDasharray={c}
   812→                initial={{ strokeDashoffset: c }}
   813→                animate={{ strokeDashoffset: c * (1 - ring.progress) }}
   814→                transition={{ duration: 1.2, delay: i * 0.15, ease: "easeOut" }}
   815→                style={{ filter: `drop-shadow(0 0 4px ${ring.color}80)` }}
   816→              />
   817→            </g>

   819→        })}
   820→      </svg>
   821→      <div className="absolute inset-0 flex items-center justify-center">

   826→      </div>
   827→    </div>

  1458→/** Circular progress indicator for body fat percentage */
  1459→function BodyFatRing({ percent }: { percent: number }) {
  1460→  const size = 56;
  1461→  const stroke = 5;
  1462→  const radius = (size - stroke) / 2;
  1463→  const circumference = 2 * Math.PI * radius;
  1464→  // درصد چربی را به فرکانس بصری (5-35) تبدیل می‌کنیم تا نمودار معنادار باشد
  1465→  const visualFraction = Math.min(1, Math.max(0.05, (percent - 3) / 35));
  1466→  const offset = circumference * (1 - visualFraction);
  1467→  const color = percent < 14 ? "#06b6d4" : percent < 21 ? "#10b981" : percent < 26 ? "#f59e0b" : "#ef4444";
  1468→
  1469→  return (
  1470→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
  1471→      <svg width={size} height={size} className="-rotate-90">
  1472→        <circle
  1473→          cx={size / 2}
  1474→          cy={size / 2}
  1475→          r={radius}
  1476→          fill="none"
  1477→          stroke="#f1f5f9"
  1478→          strokeWidth={stroke}
  1479→        />
  1480→        <motion.circle
  1481→          cx={size / 2}
  1482→          cy={size / 2}
  1483→          r={radius}
  1484→          fill="none"
  1485→          stroke={color}
  1486→          strokeWidth={stroke}
  1487→          strokeLinecap="round"
  1488→          strokeDasharray={circumference}
  1489→          initial={{ strokeDashoffset: circumference }}
  1490→          animate={{ strokeDashoffset: offset }}
  1491→          transition={{ duration: 1, ease: "easeOut" }}
  1492→        />
  1493→      </svg>
  1494→      <div className="absolute inset-0 flex items-center justify-center">
  1495→        <span className="text-base font-black font-stat text-slate-900 leading-none">
  1496→          {toPersianDigits(percent.toFixed(1))}
  1497→        </span>
  1498→      </div>
  1499→    </div>
  1500→  );
  1501→}
