   266→  async function submit() {
   267→    if (photos.length === 0) {
   268→      toast.error("حداقل یک عکس بدن ارسال کنید");
   269→      return;
   270→    }
   271→    // ویدیو اختیاری است — حتی برای پلن Ultimate — نیازی به بررسی الزامی بودن نیست

   273→    setSubmitError("");

   276→    // مرحله «در حال ساخت برنامه» حذف شد چون تولید برنامه در پس‌زمینه انجام
   277→    // می‌شود و این درخواست دیگر برای تولید برنامه منتظر نمی‌ماند.
   278→    const stageTimers: ReturnType<typeof setTimeout>[] = [];
   279→    stageTimers.push(setTimeout(() => setSubmitStage("analyzing"), 1500));
   280→    try {
   281→      const fd = new FormData();
   282→
   283→      // FIX v37: فشرده‌سازی/تبدیل سمت کلاینت قبل از آپلود —
   284→      // ① HEIC آیفون (اگر مرورگر بتواند decode کند) → JPEG/WebP تبدیل می‌شود و
   285→      //    دیگر به خطای libheif سمت سرور نمی‌خورَد؛ ② حجم آپلود (در IAB با اینترنت
   286→      //    ضعیف) چند برابر سریع‌تر می‌شود. downscaleImage در هر خطایی فایل اصلی
   287→      //    برمی‌گرداند، پس این گام هرگز آپلود را نمی‌شکند.
   288→      try {
   289→        const { downscaleImage } = await import("@/lib/fitness/client-image");
   290→        for (const p of photos) {
   291→          const processed = await downscaleImage(p, 1600, 0.85);
   292→          fd.append("bodyPhotos", processed);
   293→        }
   294→      } catch {
   295→        photos.forEach((p) => fd.append("bodyPhotos", p));
   296→      }
   297→      if (video) fd.append("bodyVideo", video);
   298→
   299→      // fetchJson: پاسخ HTML (صفحه خطای گیت‌وی/سرور) → خطای فارسی دوستانه
   300→      // به‌جای «Unexpected token '<'»
   301→      const { res, data } = await fetchJson<any>("/api/coach/submit-body-analysis", {
   302→        method: "POST",
   303→        body: fd,
   304→        // توجه: keepalive نگذارید — کروم بدنه‌های keepalive بزرگتر از 64KB را رد می‌کند
   305→        // و عکس/ویدیوهای واقعی هرگز به سرور نمی‌رسند.
   306→      });
   307→      if (!res.ok) {
   308→        if (data?.code === "IMAGE_DECODE_FAILED") {
   309→          throw new Error(
   310→            "عکس ارسالی قابل پردازش نیست (فرمت HEIC آیفون). لطفاً از تنظیمات دوربین، فرمت را روی «سازگارتر» بگذارید یا عکس JPG/PNG بفرستید."
   311→          );
   312→        }
   313→        throw new Error(data?.error || "خطا در ارسال مدیا");
   314→      }
   315→
   316→      // ─── قرارداد جدید (async): این endpoint دیگر برنامه را سینکرون نمی‌سازد ───
   317→      // پاسخ سریع برمی‌گردد: {programStarted, pendingStatus, awaitingDecision, message}
   318→      //  - programStarted=true / pendingStatus="generating" → تولید در پس‌زمینه شروع شد
   319→      //  - awaitingDecision=true / pendingStatus="pending_body_photo" → پیش‌نیاز دیگری مانده
   320→      const started =
   321→        Boolean(data?.programStarted) || data?.pendingStatus === "generating";
   322→      const blocked =
   323→        Boolean(data?.awaitingDecision) ||
   324→        data?.pendingStatus === "pending_body_photo";
   325→      const hasPlan = !started && !blocked && Boolean(data?.hasWorkoutPlan);
   326→
   327→      setSubmitStage("done");
   328→      // پاک کردن فرم
   329→      setPhotos([]);
   330→      setVideo(null);
   331→      setOpen(false);
   332→
   333→      // به‌روزرسانی state بر اساس پاسخ واقعی سرور (نه هاردکد true/ready)
   334→      setState({
   335→        needsBodyPhoto: state?.needsBodyPhoto ?? false,
   336→        canSubmitVideo: state?.canSubmitVideo ?? false,
   337→        pendingStatus: started
   338→          ? "generating"
   339→          : data?.pendingStatus ?? (hasPlan ? "ready" : null),
   340→        hasWorkoutPlan: hasPlan,
   341→        awaitingMedia: started ? false : Boolean(data?.awaitingMedia),
   342→      });
   343→
   344→      if (started) {
   345→        // تولید برنامه در پس‌زمینه شروع شد — خبر فوری و صادقانه به کاربر
   346→        toast.success(
   347→          "عکس‌های بدن شما ذخیره و تحلیل شد ✅ برنامه شما در حال آماده‌سازی است — پس از آماده‌سازی به شما اطلاع می‌دهیم."
   348→        );
   349→      } else if (blocked) {
   350→        // برنامه ساخته نشد — دلیل بلاک‌کننده سرور را نشان بده
   351→        // (مثلاً «تعیین تکلیف ویدیو یا آزمایش خون»)
   352→        toast.info(data?.message || "پیش‌نیازهای باقی‌مانده را تعیین تکلیف کنید.");
   353→      } else if (hasPlan) {
   354→        toast.success(data?.message || "برنامه شما ساخته شد! 🎯");
   355→        // بارگذاری برنامه جدید — فقط وقتی واقعاً برنامه ساخته شده
   356→        const { res: planRes, data: planData } = await fetchJson<any>(
   357→          "/api/coach/plan",
   358→          { cache: "no-store" }
   359→        );
   360→        if (planRes.ok) {
   361→          if (planData?.workout) setWorkoutPlan(planData.workout);
   362→          if (planData?.meal) setMealPlan(planData.meal);
   363→        }
   364→      } else {
   365→        toast.info(
   366→          data?.message || "عکس‌های بدن شما ذخیره و تحلیل شد. وضعیت برنامه خود را از بخش «برنامه‌ها» بررسی کنید."
   367→        );
   368→      }
   369→
   370→      // وضعیت پیش‌نیازها سمت سرور تغییر کرده — بنر پیش‌نیازها/داشبورد را refresh کن
   371→      // (مثلاً حالت «در حال آماده‌سازی» در PrerequisitesBanner ظاهر شود)
   372→      if (typeof window !== "undefined") window.dispatchEvent(new Event("prereq-updated"));
   373→    } catch (e) {
   374→      const msg = e instanceof Error ? e.message : "خطا در ارسال مدیا";
   375→      setSubmitError(msg);
   376→      setSubmitStage("error");
   377→      toast.error(msg);
   378→    } finally {
   379→      for (const t of stageTimers) clearTimeout(t);
   380→      setSubmitting(false);
   381→      // ریست مرحله بعد از ۳ ثانیه (مگر خطا)
   382→      setTimeout(() => {
   383→        setSubmitStage((s) => (s === "error" ? s : "idle"));
   384→      }, 3000);
   385→    }
   386→  }
   387→
   388→  const measurementFields: { key: keyof MeasurementsForm; label: string; required?: boolean }[] = [
   389→    { key: "waist", label: "دور کمر", required: true },
   390→    { key: "neck", label: "دور گردن", required: true },
   391→    { key: "hip", label: "دور باسن" },
   392→    { key: "chest", label: "دور سینه" },

   396→    { key: "calf", label: "دور ساق پا" },
   397→  ];

   399→  return (
   400→    <>
   401→      {/* بنر قدیمی حذف شد — فقط مودال آپلود عکس بدن */}
   402→      <AnimatePresence>
   403→        {open && (
   404→          <motion.div
   405→            initial={{ opacity: 0 }}
   406→            animate={{ opacity: 1 }}
