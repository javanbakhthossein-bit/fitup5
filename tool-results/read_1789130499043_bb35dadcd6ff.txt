    29→  # deploy.sh خودش از unzip مستثنی است: بازنویسی اسکریپتِ در حالِ اجرا باعث

    31→  # v72.2: مقایسهٔ نسخه‌دار — قبلاً فقط cmp بود و هر تفاوتی یعنی زیپ جدیدتر!

    40→    if [ -z "$ZIPVER" ] && [ -z "$CURVER" ]; then
    41→      REPLACE=1
    42→    else
    43→      NEWEST=$(printf '%s\n%s\n' "${ZIPVER:-0}" "${CURVER:-0}" | sort -V | tail -1)
    44→      if [ "$NEWEST" = "${ZIPVER:-0}" ] && [ -n "$ZIPVER" ]; then
    45→        REPLACE=1
    46→      fi

    53→        cp deploy.sh.zipnew deploy.sh && rm -f deploy.sh.zipnew
    54→        export DEPLOY_REEXEC=1
    55→        exec bash deploy.sh
    56→      fi
    57→    else
    58→      echo "  ⛔ deploy.sh داخل زیپ (نسخهٔ ${ZIPVER:-بدون‌نشان/قدیمی}) جدیدتر از نسخهٔ در حال اجرا (v${CURVER:-?}) نیست — فایل فعلی حفظ شد"
    59→    fi
    60→  fi
    61→  rm -f deploy.sh.zipnew 2>/dev/null || true
    62→fi
    63→if [ ! -f "prisma/schema.prisma" ] || [ ! -d "src" ]; then

    79→# ۰-ب. گارد هویت PWA (v29 — باگ دیالوگ «مرور به‌روزرسانی» کروم)
    80→# کروم برای WebAPKهای نصب‌شده هر بار که «نام یا آیکون» وب‌اپ تغییر کند یک
    81→# دیالوگ تأیید هویت (با دکمهٔ حذف نصب!) نشان می‌دهد — حتی اگر تغییر عمدی نباشد
    82→# (مثلاً re-export تصادفی لوگو با بایت‌های متفاوت). این گارد هویت PWA را هش
    83→# می‌کند و اگر بین دیپلوی‌ها تغییر کرده باشد با هشدار بزرگ اطلاع می‌دهد تا
    84→# «هیچ‌وقت» تصادفی تغییر نکند. دیپلوی fail نمی‌شود — فقط هشدار صادقانه.

   192→# v22: فهرست فایل‌های مجاز = محتوای جدیدترین زیپ (unzip -Z1) ∪ مانیفست.
   193→# هیچ فایلی که در زیپ هست حذف نمی‌شود — حتی اگر مانیفست غلط/قدیمی باشد.

   196→  echo "🧹 پاک‌سازی فایل‌های قدیمی (stale)..."
   197→  ALLOWED_FILES=$(
   198→    {
   199→      if [ -n "$LATEST_ZIP" ]; then
   200→        unzip -Z1 "$LATEST_ZIP" 2>/dev/null | grep -v '/$' || true
   201→      fi
   202→      if [ -f ".deploy-manifest.txt" ]; then
   203→        tr ' ' '\n' < .deploy-manifest.txt || true
   204→      fi
   205→    } | grep -v '^$' | LC_ALL=C sort -u
   206→  )
   207→  STALE_COUNT=0
   208→  if [ -n "$ALLOWED_FILES" ]; then

   211→        src/*|prisma/*|scripts/*)
   212→          if [ -f "$stale_file" ]; then
   213→            rm -f "$stale_file"
   214→            STALE_COUNT=$((STALE_COUNT + 1))
   215→            echo "  🗑 حذف stale: $stale_file"
   216→          fi
   217→          ;;
   218→      esac
   219→    done < <(LC_ALL=C comm -23 <(find src prisma scripts -type f 2>/dev/null | LC_ALL=C sort) <(printf '%s\n' "$ALLOWED_FILES" | LC_ALL=C sort -u))
   220→  fi
   221→  if [ "$STALE_COUNT" -eq 0 ]; then
   222→    echo "  ✓ فایل stale ای وجود ندارد"
   223→  else
   224→    echo "  ✓ $STALE_COUNT فایل قدیمی حذف شد"
   225→  fi
   226→else
