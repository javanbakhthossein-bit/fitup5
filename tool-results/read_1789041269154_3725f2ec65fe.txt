    81→    81→function MacroRing({
    82→    82→  label,
    83→    83→  grams,
    84→    84→  calories,
    85→    85→  color,
    86→    86→  icon,
    87→    87→}: {
    88→    88→  label: string;
    89→    89→  grams: number;

    91→    91→  color: string;

    94→    94→  const size = 84;
    95→    95→  const stroke = 7;
    96→    96→  const radius = (size - stroke) / 2;
    97→    97→  const circumference = 2 * Math.PI * radius;
    98→    98→  // Each macro is shown as a fraction of its calorie contribution vs the
    99→    99→  // other two. This gives a sense of proportion rather than a fixed max.
   100→   100→  // We just render a fully-stroked ring with the grams label inside.
   101→   101→  const fillFraction = 0.78; // visual fill — purely aesthetic
   102→   102→
   103→   103→  return (
   104→   104→    <div className="flex flex-col items-center gap-1.5">
   105→   105→      <div className="relative" style={{ width: size, height: size }}>
   106→   106→        <svg width={size} height={size} className="-rotate-90">
   107→   107→          <circle
   108→   108→            cx={size / 2}
   109→   109→            cy={size / 2}
   110→   110→            r={radius}
   111→   111→            fill="none"
   112→   112→            stroke="#f1f5f9"
   113→   113→            strokeWidth={stroke}
   114→   114→          />
   115→   115→          <motion.circle
   116→   116→            cx={size / 2}
   117→   117→            cy={size / 2}
   118→   118→            r={radius}
   119→   119→            fill="none"
   120→   120→            stroke={color}
   121→   121→            strokeWidth={stroke}
   122→   122→            strokeLinecap="round"
   123→   123→            strokeDasharray={circumference}
   124→   124→            initial={{ strokeDashoffset: circumference }}
   125→   125→            animate={{ strokeDashoffset: circumference * (1 - fillFraction) }}
   126→   126→            transition={{ duration: 1, ease: "easeOut" }}
   127→   127→          />
   128→   128→        </svg>
   129→   129→        <div className="absolute inset-0 flex flex-col items-center justify-center">

   139→   139→    </div>
   140→   140→  );
   141→   141→}

   328→   328→  // دقیقه صبر) — سمت سرور هم همهٔ درخواست‌های هم‌زمان به «یک» تولید در-جریان

   334→   334→  async function fetchAnalysisOnce(url: string): Promise<{ ok: boolean; permanentFail: boolean; json: any }> {
   335→   335→    const controller = new AbortController();
   336→   336→    const timer = setTimeout(() => controller.abort(), ANALYSIS_ATTEMPT_TIMEOUT_MS);
   337→   337→    try {
   338→   338→      const res = await fetch(url, { signal: controller.signal });
   339→   339→      const json = await res.json().catch(() => null);
   340→   340→      if (res.ok && json?.analysis) return { ok: true, permanentFail: false, json };
   341→   341→      // خطاهای دائمی — retry بی‌فایده است (پروفایل نبود / سشن نیست)
   342→   342→      if (res.status === 400 || res.status === 401) {
   343→   343→        return { ok: false, permanentFail: true, json };
   344→   344→      }
   345→   345→      return { ok: false, permanentFail: false, json };
   346→   346→    } catch {
   347→   347→      // abort / network — قابل‌retry
   348→   348→      return { ok: false, permanentFail: false, json: null };
   349→   349→    } finally {
   350→   350→      clearTimeout(timer);
   351→   351→    }
   352→   352→  }

   354→   354→  async function loadAnalysis(force = false) {
   355→   355→    if (force) setRefreshing(true);

   357→   357→      const url = force ? "/api/onboarding/analysis?force=1" : "/api/onboarding/analysis";
   358→   358→      for (let attempt = 1; attempt <= ANALYSIS_MAX_ATTEMPTS; attempt++) {
   359→   359→        const { ok, permanentFail, json } = await fetchAnalysisOnce(url);
   360→   360→        if (ok) {
   361→   361→          setData(json as AnalysisResponse);
   362→   362→          return;
   363→   363→        }
   364→   364→        if (permanentFail) break;
   365→   365→        // مکث بین تلاش‌ها (backoff ملایم) — در طول retryها صفحهٔ «در حال تحلیل»
   366→   366→        // جذاب نشان داده می‌شود، نه کارت خطا
   367→   367→        if (attempt < ANALYSIS_MAX_ATTEMPTS) {
   368→   368→          await new Promise((r) => setTimeout(r, attempt <= 4 ? 2_500 : 6_000));
   369→   369→        }
   370→   370→      }
   371→   371→      if (force) toast.error("خطا در بازخوانی تحلیل");
   372→   372→    } finally {
   373→   373→      setLoading(false);
   374→   374→      setRefreshing(false);
   375→   375→    }
   376→   376→  }

   391→   391→    : null;
   392→   392→
   393→   393→  const bmi = data?.bmi ?? 0;
   394→   394→  const bmiCategory = bmi < 18.5 ? "کم‌وزن" : bmi < 25 ? "وزن نرمال" : bmi < 30 ? "اضافه‌وزن" : "چاق";
   395→   395→  const bmiColor = bmi < 18.5 ? "text-cyan-500" : bmi < 25 ? "text-emerald-500" : bmi < 30 ? "text-amber-500" : "text-red-500";
   396→   396→
   397→   397→  function openPurchase(planId: Plan) {

   406→   406→

   410→   410→

   412→   412→  // قبلاً اگر fetch fail می‌شد یا analysis وجود نداشت، data=null ولی رندر
   413→   413→  // ادامه می‌یافت و BMI=0 به‌عنوان «کم‌وزن» با توصیه‌های غلط نمایش داده می‌شد.
   414→   414→  if (!data) {
