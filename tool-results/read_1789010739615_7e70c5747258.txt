   161→  async function validateUserDiscount(code: string) {
   162→    if (!code.trim()) {
   163→      toast.error("کد تخفیف اختصاصی موجود نیست");
   164→      return;
   165→    }
   166→    if (!isLoggedIn) {

   168→      onNeedLogin();
   169→      return;
   170→    }
   171→    setValidating(true);
   172→    try {
   173→      // Fetch the user's actual per-user discount info to display the correct
   174→      // percentage. v47: reason-aware — تمدید (renewal) یا هدیهٔ خرید اول.

   176→      let kindLabel = "تخفیف تمدید";

   178→        const infoRes = await fetch("/api/user-discount-code");
   179→        if (infoRes.ok) {
   180→          const info = await infoRes.json();
   181→          if (info?.value && info?.value > 0) percent = info.value;
   182→          if (info?.reason && info.reason !== "renewal_loyalty") kindLabel = "تخفیف اختصاصی";
   183→        }
   184→      } catch {
   185→        /* keep default 15% */
   186→      }
   187→      const estDiscount = Math.round((originalAmount * percent) / 100);
   188→      const estFinal = Math.max(0, originalAmount - estDiscount);
   189→      setDiscountCode(code);
   190→      setUserDiscountCode(code);
   191→      setDiscountInfo({
   192→        valid: true,
   193→        discountValue: estDiscount,
   194→        finalAmount: estFinal,
   195→        label: `کد اختصاصی ${code} (${toPersianDigits(percent)}٪ ${kindLabel})`,
   196→      });
   197→      toast.success(`کد اختصاصی اعمال شد! ${toPersianDigits(percent)}٪ ${kindLabel}`);
   198→    } finally {
   199→      setValidating(false);
   200→    }
   201→  }

   244→  async function startCheckout() {

   257→      setOverlay(null);
   258→      return;

   260→    // ─── T11: اپ کافه‌بازار — پرداخت از طریق درگاه درون‌برنامه‌ای بازار ───
   261→    // طبق قانون انتشار بازار، فروش اشتراک دیجیتال در اپ بازار فقط از IAB بازار مجاز است.
   262→    // پل JS (window.fitupBazaarPurchase) در MainActivity اپ اندروید تعریف شده است.

   292→          setOverlay(null);
   293→          return;

   299→        authority: data.authority,
   300→        gatewayUrl: data.gatewayUrl,
   301→        simulated: data.simulated === true,
   302→        finalAmount: data.finalAmount,
   303→      });
   304→      if (paymentMethod === "wallet") {
   305→        // FE-C2: مقادیر fresh را مستقیم پاس بده — setPaymentData در همین tick
   306→        // هنوز در closure فعلی قابل مشاهده نیست و بدون override،
   307→        // completePayment با paymentData===null بی‌صدا return می‌کرد.
   308→        await completePayment("OK", data.paymentId, data.authority);
   309→      } else {
   310→        setStep("gateway");
   311→      }
   312→    } catch (e) {
   313→      toast.error(e instanceof Error ? e.message : "خطا در ایجاد پرداخت");

   319→  /**
   320→   * T11: خرید از طریق پرداخت درون‌برنامه‌ای کافه‌بازار (پولکی).
   321→   *
   322→   * اپ اندروید (WebView) پل window.fitupBazaarPurchase(sku, payload, dynamicPriceToken)
   323→   * را فراهم می‌کند:
   324→   *  ۱. اگر تخفیف/اعتبار ارتقا هست، ابتدا قیمت پویا در بازار ثبت می‌شود
   325→   *     (/api/payment/bazaar/dynamic-price) و شناسه‌اش به اپ پاس می‌شود
   326→   *  ۲. درگاه پرداخت بازار با همان مبلغ دقیق باز می‌شود و کاربر پرداخت می‌کند
   327→   *  ۳. نتیجه (purchaseToken + orderId) به همین تابع برمی‌گردد
   328→   *  ۴. برای فعال‌سازی اشتراک به /api/payment/bazaar/purchase ارسال می‌شود
   329→   *     (سرور خرید را با Developer API بازار راستی‌آزمایی می‌کند)
   330→   *
   331→   * SKU محصولات بازار: fitup_{planId} — در پیشخان بازار با همین شناسه تعریف می‌شوند.
   332→   */
   333→  async function startBazaarCheckout() {
   334→    const sku = `fitup_${plan.id}`;
   335→    setLoading(true);
   336→    try {
   337→      // ─── قیمت پویا (تخفیف/اعتبار ارتقا) — اگر هست، مبلغ دقیق در بازار ثبت می‌شود ───
   338→      let dynamicPriceId: string | null = null;
   339→      try {
   340→        const dpRes = await fetch("/api/payment/bazaar/dynamic-price", {
   341→          method: "POST",
   342→          headers: { "Content-Type": "application/json" },
   343→          body: JSON.stringify({
   344→            planId: plan.id,
   345→            discountCode: discountInfo?.valid && !userDiscountCode ? discountCode : undefined,
   346→            userDiscountCode: discountInfo?.valid && userDiscountCode ? userDiscountCode : undefined,
   347→          }),
   348→        });
   349→        const dpData = await dpRes.json().catch(() => ({}));
   350→        if (dpRes.ok && typeof dpData?.dynamicPriceId === "string" && dpData.dynamicPriceId) {
   351→          dynamicPriceId = dpData.dynamicPriceId;
   352→        }
   353→        // dynamicPriceId=null → قیمت پایه SKU (تخفیفی نبود یا سرویس تنظیم نیست)
   354→      } catch {
   355→        // خطا → خرید با قیمت پایه SKU ادامه می‌یابد
   356→      }
   357→
   358→      const purchase = await (window as any).fitupBazaarPurchase(
   359→        sku,
   360→        { planId: plan.id, userId: user?.id || null },
   361→        dynamicPriceId
   362→      );
   363→      if (!purchase?.ok) {
   364→        if (purchase?.canceled) {
   365→          toast.info(purchase.error || "پرداخت لغو شد");
   366→        } else {
   367→          toast.error(purchase?.error || "پرداخت درون‌برنامه‌ای بازار ناموفق بود");
   368→        }
   369→        return;
   370→      }
   371→      // فعال‌سازی اشتراک در سرور — v55: امضای خرید (originalJson+signature) هم
   372→      // همراه می‌رود تا اگر API بازار در دسترس نبود، سرور با کلید RSA پنل،
   373→      // امضای خرید را محلی راستی‌آزمایی کند (رفع خطای «اعتبارنامه API بازار»
   374→      // بعد از خرید موفق — ایراد بازرسی بازار).
   375→      const res = await fetch("/api/payment/bazaar/purchase", {
   376→        method: "POST",
   377→        headers: { "Content-Type": "application/json" },
   378→        body: JSON.stringify({
   379→          planId: plan.id,
   380→          productId: purchase.productId || sku,
   381→          purchaseToken: purchase.purchaseToken,
   382→          orderId: purchase.orderId,
   383→          dataJson: purchase.dataJson,
   384→          signature: purchase.signature,
   385→          // برای ثبت مبلغ واقعی/کد تخفیف در حسابداری (سرور مستقل دوباره محاسبه می‌کند)
   386→          discountCode: discountInfo?.valid && !userDiscountCode ? discountCode : undefined,
   387→          userDiscountCode: discountInfo?.valid && userDiscountCode ? userDiscountCode : undefined,
   388→        }),
   389→      });
   390→      const data = await res.json();
   391→      if (!res.ok) {
   392→        if (data.code === "ONBOARDING_REQUIRED") {
   393→          toast.info(data.error);
   394→          setScreen("onboarding");
   395→          setOverlay(null);
   396→          return;
   397→        }
   398→        throw new Error(data.error);
   399→      }
   400→      // ─── Consume محصول مصرفی بازار ───
   401→      // محصولات consumable تا وقتی consume نشوند قابل خرید مجدد نیستند (تمدید).
   402→      // پس از فعال‌سازی موفق روی سرور، به اپ می‌گوییم خرید را consume کند —
   403→      // اگر فعال‌سازی سرور شکست می‌خورد، خرید unconsume می‌ماند و اپ در
   404→      // اجرای بعدی با «بازیابی خریدها» (restore) همان توکن را دوباره می‌فرستد.
   405→      try {
   406→        (window as any).FitUpNative?.consumePurchase?.(purchase.purchaseToken);
   407→      } catch {}
   408→      toast.success(data.message || "اشتراک شما فعال شد 🎉");
   409→      // رفرش اطلاعات کاربر (باز شدن قابلیت‌های پلن) — الگوی همان completePayment
   410→      try {
   411→        const meRes = await fetch("/api/auth/me", { cache: "no-store" });
   412→        const meData = await meRes.json();
   413→        if (meData?.user) {
   414→          setUser(meData.user);
   415→        }
   416→      } catch {
   417→        // رفرش اختیاری است
   418→      }
   419→      // نمایش رسید موفق (بدون درگاه — خرید بازار درون‌برنامه‌ای بود)
   420→      setPaymentData({
   421→        paymentId: data.paymentId,
   422→        authority: purchase.purchaseToken,
   423→        gatewayUrl: null as unknown as string,
   424→        simulated: false,
   425→        finalAmount: plan.price,
   426→      });
   427→      setReceipt({
   428→        success: true,
   429→        refId: purchase.orderId || purchase.purchaseToken.slice(0, 20),
   430→        message: data.message || "پرداخت درون‌برنامه‌ای کافه‌بازار موفق بود",
   431→        user: undefined,
   432→      });
   433→      setStep("receipt");
   434→    } catch (e) {
   435→      toast.error(e instanceof Error ? e.message : "خطا در پرداخت بازار");
   436→    } finally {
   437→      setLoading(false);
   438→    }
   439→  }

   441→  /**
   442→   * باز کردن درگاه پرداخت زرین‌پال.
   443→   *
   444→   * - حالت واقعی (real Zarinpal): در همان تب فعلی به gatewayUrl هدایت می‌شود.
   445→   *   زرین‌پال پس از پرداخت به callback_url (?payment_verify=1) redirect می‌کند
   446→   *   و PaymentVerifyHandler به‌صورت خودکار verify را صدا می‌زند. این کار از
   447→   *   ایجاد تب اضافی و گیج‌شدن کاربر جلوگیری می‌کند.
   448→   * - حالت شبیه‌سازی: در یک تب جدید باز می‌شود تا کاربر بتواند بدون ترک صفحه،
   449→   *   دکمه‌های موفق/ناموفق را امتحان کند.
   450→   */
   451→  function openZarinpalGateway() {
   452→    if (redirecting) return; // گارد دابل‌کلیک — navigation در جریان است
   453→    if (!paymentData?.gatewayUrl) {
   454→      toast.error("لینک درگاه پرداخت در دسترس نیست");
   455→      return;
   456→    }
   457→    setRedirecting(true);
   458→    // واقعی: هدایت در همان تب — زرین‌پال به ?payment_verify=1 برمی‌گرداند
   459→    toast.info("در حال انتقال به درگاه زرین‌پال... پس از پرداخت، به‌صورت خودکار بازمی‌گردید.");
   460→    // 🩹 v45 (باگ «هر چقدر روی دکمهٔ درگاه پرداخت میزنم وارد نمیشه»):
   461→    // قبلاً redirecting یک‌طرفه بود — اگر ناوبری به هر دلیلی انجام نمی‌شد
   462→    // (شبکه ضعیف، بلاک‌شدن ناوبری در WebView، mixed content)، دکمه برای همیشه
   463→    // با اسپینر «در حال انتقال…» مرده می‌ماند. حالا بعد از ۱۲ ثانیه قفل
   464→    // خودکار آزاد می‌شود (بازگشت زودتر با pageshow/visibilitychange پوشش داده شده)

   470→   * تأیید پرداخت.
   471→   *
   472→   * FE-C2: paymentId/authority را می‌توان مستقیم پاس داد (override) — وقتی
   473→   * completePayment بلافاصله بعد از setPaymentData در همان handler صدا زده
   474→   * می‌شود، state هنوز در closure رندر قبلی است؛ بدون override بی‌صدا return
   475→   * می‌شد و در تلاش دوم پرداختِ قبلی (stale) verify می‌شد.
   476→   */
   477→  async function completePayment(
   478→    status: "OK" | "NOK" | "CANCELLED",
   479→    paymentIdOverride?: string,
   480→    authorityOverride?: string,
   481→  ) {
   482→    const paymentId = paymentIdOverride ?? paymentData?.paymentId;
   483→    const authority = authorityOverride ?? paymentData?.authority;
   484→    if (!paymentId) return;
   485→    setStep("processing");
   486→    try {
   487→      const res = await fetch("/api/payment/verify", {
   488→        method: "POST",
   489→        headers: { "Content-Type": "application/json" },
   490→        body: JSON.stringify({
   491→          paymentId,
   492→          status,
   493→          authority,
   494→        }),
   495→      });
   496→      const data = await res.json();
   497→      // FIX: پاسخ verifying یک message دارد نه error — قبلاً toast خالی می‌داد
   498→      if (!res.ok) throw new Error(data.error || data.message);
   499→      setReceipt(data);
   500→      if (data.success && data.user) {
   501→        setUser(data.user);
   502→        toast.success("پلن شما فعال شد! در حال ساخت برنامه... 🎉");
   503→      }
   504→      setStep("receipt");
   505→    } catch (e) {
   506→      toast.error(e instanceof Error ? e.message : "خطا در تایید پرداخت");
   507→      setStep("form");
   508→    }
   509→  }
