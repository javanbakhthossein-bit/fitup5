   300→     * true  → بیرون از WebView هندل شد (مرورگر/دیالر)

   303→    private fun routeUrl(url: Uri): Boolean {
   304→        val host = url.host?.lowercase() ?: return false
   305→        val siteHost = try { URI(BuildConfig.SITE_URL).host?.lowercase() } catch (_: Exception) { null }
   306→
   307→        // سایت خودمان — همیشه داخل WebView
   308→        if (host == "fittup.ir" || host.endsWith(".fittup.ir")) return false
   309→        if (siteHost != null && (host == siteHost || host.endsWith(".$siteHost"))) return false
   310→
   311→        // درگاه پرداخت سایت (زرین‌پال) و بانک‌ها (شاپراک) — داخل WebView تا کاربر
   312→        // بعد از پرداخت به پنل برگردد (payment_verify در همان WebView اجرا می‌شود)
   313→        if (host == "zarinpal.com" || host.endsWith(".zarinpal.com")) return false
   314→        if (host == "zarin.link" || host.endsWith(".zarin.link")) return false
   315→        if (host == "shaparak.ir" || host.endsWith(".shaparak.ir")) return false
   316→
   317→        // شبکه‌های اجتماعی/استورها/سایت‌های دیگر → مرورگر بیرونی
   318→        openExternal(url)
   319→        return true
   320→    }

   322→    /** اسکیم‌های غیر وب — tel/mailto/intent و غیره */
   323→    private fun handleExternalScheme(uri: Uri): Boolean {
   324→        try {
   325→            when (uri.scheme?.lowercase()) {
   326→                "tel" -> startActivity(Intent(Intent.ACTION_DIAL, uri))
   327→                "mailto" -> startActivity(Intent(Intent.ACTION_SENDTO, uri))
   328→                "sms" -> startActivity(Intent(Intent.ACTION_SENDTO, uri))
   329→                "intent" -> {
   330→                    val intent = Intent.parseUri(uri.toString(), Intent.URI_INTENT_SCHEME)
   331→                    startActivity(intent)
   332→                }
   333→                else -> startActivity(Intent(Intent.ACTION_VIEW, uri))
   334→            }
   335→        } catch (_: Exception) {
   336→            // اپی برای این لینک نیست — نادیده بگیر
   337→        }
   338→        return true
   339→    }

   366→    private fun injectBridgeHelper() {
   367→        val js = """
   368→            (function() {
   369→              if (window.__fitupOwnAppInjected) return;
   370→              window.__fitupOwnAppInjected = true;
   371→              // آیا داخل اپ اختصاصی فیتاپ هستیم؟

   375→              };
   376→              // کوکی pwa_standalone — سرور برای URL «/» مستقیم صفحهٔ auth را

   380→              try {
   381→                if (document.cookie.indexOf('pwa_standalone=1') === -1) {
   382→                  document.cookie = 'pwa_standalone=1; path=/; max-age=31536000; samesite=lax';
   383→                }
   384→              } catch (e) {}
   385→            })();
   386→        """.trimIndent()
   387→        runOnUiThread {
   388→            webView.evaluateJavascript(js, null)

   419→        } catch (e: Exception) {

   425→    /** شروع دانلود با DownloadManager → پوشه Download اپ (بدون مجوز نوشتن) */
   426→    private fun startNativeDownload(url: String, fileName: String, isApk: Boolean) {
   427→        val safeName = fileName.replace(Regex("[^A-Za-z0-9._-]"), "_").ifBlank {
   428→            "fitup-${System.currentTimeMillis()}" + if (isApk) ".apk" else ""
   429→        }
   430→        val request = DownloadManager.Request(Uri.parse(url))
   431→            .setTitle(if (isApk) "نسخه جدید فیتاپ" else safeName)
   432→            .setDescription(if (isApk) "دانلود مستقیم از فیتاپ" else "دانلود فایل")
   433→            .setNotificationVisibility(DownloadManager.Request.VISIBILITY_VISIBLE_NOTIFY_COMPLETED)
   434→            .setDestinationInExternalFilesDir(this, Environment.DIRECTORY_DOWNLOADS, safeName)
   435→            .setAllowedOverMetered(true)
   436→            .setAllowedOverRoaming(true)
   437→        try {
   438→            val dm = getSystemService(Context.DOWNLOAD_SERVICE) as DownloadManager
   439→            lastDownloadId = dm.enqueue(request)
   440→            lastApkFile = File(
   441→                getExternalFilesDir(Environment.DIRECTORY_DOWNLOADS) ?: filesDir,
   442→                safeName
   443→            )
   444→        } catch (e: Exception) {
   445→            Log.e("FitUpApp", "download enqueue failed: ${e.message}")
   446→            toast("دانلود ممکن نشد")
   447→        }
   448→    }

   450→    /** گیرندهٔ پایان دانلود → اگر APK بود، دیالوگ نصب */
   451→    private fun setupDownloadCompleteReceiver() {
   452→        val receiver = object : BroadcastReceiver() {
   453→            override fun onReceive(context: Context?, intent: Intent?) {
   454→                if (intent?.action != DownloadManager.ACTION_DOWNLOAD_COMPLETE) return
   455→                try {
   456→                    val id = intent.getLongExtra(DownloadManager.EXTRA_DOWNLOAD_ID, -1L)
   457→                    if (id != lastDownloadId) return
   458→                    val file = lastApkFile
   459→                    if (file == null || !file.name.endsWith(".apk", true)) return
   460→                    if (!file.exists() || file.length() < 10_000) {
   461→                        // فایل ناقص/خراب — دانلود منیجر خودش خطا نشان داده
   462→                        lastApkFile = null
   463→                        return
   464→                    }
   465→                    showInstallDialog(file)
   466→                } catch (_: Exception) {}
   467→            }
   468→        }
   469→        try {
   470→            if (Build.VERSION.SDK_INT >= 33) {
   471→                registerReceiver(receiver, IntentFilter(DownloadManager.ACTION_DOWNLOAD_COMPLETE), Context.RECEIVER_NOT_EXPORTED)
   472→            } else {
   473→                registerReceiver(receiver, IntentFilter(DownloadManager.ACTION_DOWNLOAD_COMPLETE))
   474→            }
   475→        } catch (_: Exception) {}
   476→    }

   526→    /** نام فایل از Content-Disposition یا URL */
   527→    private fun guessFileName(contentDisposition: String?, mimeType: String?, url: String): String {
   528→        try {
   529→            if (!contentDisposition.isNullOrBlank()) {
   530→                val m = Regex("filename\\*=UTF-8''([^;]+)|filename=\"?([^\";]+)\"?", RegexOption.IGNORE_CASE)
   531→                    .find(contentDisposition)
   532→                if (m != null) {
   533→                    val name = (m.groupValues[1].ifBlank { m.groupValues[2] })
   534→                        .replace("+", " ")
   535→                    if (name.isNotBlank()) return java.net.URLDecoder.decode(name, "UTF-8")
   536→                }
   537→            }
   538→            if (!mimeType.isNullOrBlank() && mimeType.equals("application/vnd.android.package-archive", true)) {
   539→                return "fitup-${System.currentTimeMillis()}.apk"
   540→            }
   541→            val path = URI(url).path ?: return "fitup-file"
   542→            return path.substringAfterLast('/').ifBlank { "fitup-file" }
   543→        } catch (_: Exception) {
   544→            return "fitup-file"
   545→        }
   546→    }

   548→    /* ───────────── ذخیره data URL (PNG/PDF) ───────────── */
   549→
   550→    @SuppressLint("InlinedApi")
   551→    private fun downloadDataUrl(filename: String, dataUrl: String) {
   552→        try {
   553→            val safeName = filename.ifBlank { "fitup-${System.currentTimeMillis()}" }
   554→                .replace(Regex("[^A-Za-z0-9._-\\u0600-\\u06FF ]"), "_")
   555→            val comma = dataUrl.indexOf(',')
   556→            if (comma < 0 || !dataUrl.startsWith("data:", ignoreCase = true)) {
   557→                runOnUiThread { toast("فایل قابل ذخیره نیست") }
   558→                return
   559→            }
   560→            val meta = dataUrl.substring(5, comma)
   561→            val mime = meta.substringBefore(";").ifBlank { "application/octet-stream" }
   562→            val isBase64 = meta.contains("base64", ignoreCase = true)
   563→            val bytes = if (isBase64) {
   564→                Base64.decode(dataUrl.substring(comma + 1), Base64.DEFAULT)
   565→            } else {
   566→                dataUrl.substring(comma + 1).toByteArray(Charsets.UTF_8)
   567→            }
   568→
   569→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   570→                val values = ContentValues().apply {
   571→                    put(MediaStore.Downloads.DISPLAY_NAME, safeName)
   572→                    put(MediaStore.Downloads.MIME_TYPE, mime)
   573→                    put(MediaStore.Downloads.IS_PENDING, 1)
   574→                }
   575→                val uri = contentResolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, values)
   576→                if (uri == null) {
   577→                    runOnUiThread { toast("خطا در ذخیره فایل") }
   578→                    return
   579→                }
   580→                contentResolver.openOutputStream(uri)?.use { it.write(bytes) }
   581→                values.clear()
   582→                values.put(MediaStore.Downloads.IS_PENDING, 0)
   583→                contentResolver.update(uri, values, null, null)
   584→            } else {
   585→                val dir = getExternalFilesDir(Environment.DIRECTORY_DOCUMENTS) ?: filesDir
   586→                val file = File(dir, safeName)
   587→                FileOutputStream(file).use { it.write(bytes) }
   588→            }
   589→            runOnUiThread { toast("«$safeName» ذخیره شد ✓") }
   590→        } catch (e: Exception) {
   591→            Log.e("FitUpApp", "download failed: ${e.message}")
   592→            runOnUiThread { toast("خطا در ذخیره فایل") }
   593→        }
   594→    }

   669→        pendingWebPermissionRequest?.let { if (it !== webRequest) it.deny() }

   689→                runOnUiThread { webRequest.deny() }

   735→    private fun registerSmsReceiver() {
   736→        if (smsReceiver != null) return
   737→        val receiver = object : BroadcastReceiver() {
   738→            override fun onReceive(context: Context?, intent: Intent?) {
   739→                if (intent?.action != "android.provider.Telephony.SMS_RECEIVED") return
   740→                try {
   741→                    val msgs = android.provider.Telephony.Sms.Intents.getMessagesFromIntent(intent) ?: return
   742→                    for (m in msgs) {
   743→                        val body = m?.displayMessageBody?.toString() ?: continue
   744→                        if (!body.contains("فیتاپ") && !body.contains("FitUp", true)) continue
   745→                        val match = Regex("(^|\\D)(\\d{4})(\\D|$)").find(body) ?: continue
   746→                        dispatchOtpCode(match.groupValues[2])
   747→                        break
   748→                    }
   749→                } catch (_: Exception) {}
   750→            }
   751→        }
   752→        try {
   753→            registerReceiver(receiver, android.content.IntentFilter("android.provider.Telephony.SMS_RECEIVED"))
   754→            smsReceiver = receiver
   755→        } catch (_: Exception) {}
   756→    }

   779→    /**
   780→     * در هر اجرا /api/app/own/latest را می‌خواند.
   781→     * مودال زیبای آپدیت را خود سایت (AppUpdateModal) نشان می‌دهد؛ این چک نیتیو
   782→     * فقط safety-net است: اگر صفحه سایت بالا نیامده باشد (خطای شبکه) و نسخهٔ
   783→     * جدید «اجباری» بود، دیالوگ نیتیو با دانلود مستقیم نشان می‌دهیم.
   784→     */
   785→    private fun checkAppVersion() {
   786→        thread {
   787→            try {
   788→                val url = URL(BuildConfig.SITE_URL.trimEnd('/') + "/api/app/own/latest")
   789→                val conn = url.openConnection() as HttpURLConnection
   790→                conn.connectTimeout = 10_000
   791→                conn.readTimeout = 10_000
   792→                conn.instanceFollowRedirects = true
   793→                val body = conn.inputStream.bufferedReader().use { it.readText() }
   794→                conn.disconnect()
   795→                val json = JSONObject(body)
   796→                if (!json.optBoolean("available", false)) return@thread
   797→                val latest = json.optInt("latestVersionCode", 1)
   798→                val forced = json.optBoolean("forceUpdate", false)
   799→                val apkUrl = BuildConfig.SITE_URL.trimEnd('/') + "/api/app/own/download"
   800→                if (BuildConfig.VERSION_CODE < latest && forced) {
   801→                    runOnUiThread {
   802→                        // فقط وقتی سایت بالا نیامده (وگرنه مودال سایت خودش را نشان می‌دهد)
   803→                        if (errorView.visibility == View.VISIBLE && !forceDialogShown) {
   804→                            forceDialogShown = true
   805→                            showForceUpdateDialog(apkUrl)
   806→                        }
   807→                    }
   808→                }
   809→            } catch (e: Exception) {
   810→                // آفلاین/خطا — نادیده بگیر؛ در اجرای بعدی دوباره تلاش می‌شود
   811→            }
   812→        }
   813→    }
