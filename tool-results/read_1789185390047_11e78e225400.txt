   291→    /**
   292→     * v1.2.6 — لینک عمیق ورودی (App Link یا اسکیم fitup:// یا اعلان) یا null:
   293→     *  ۱) extra «fitup_link» از NotificationSync — مسیر/کوئری نسبی مثل
   294→     *     "?tab=progress&section=checkup" یا "/panel" → SITE_URL + آن
   295→     *  ۲) dataِ https روی دامنهٔ خودمان (fittup.ir) → همان URL — پارامترهای
   296→     *     ?tab= ?ref= ?renewal= ?screen= ?article= دست‌نخورده می‌مانند
   297→     *  ۳) v1.2.6 — اسکیم اختصاصی fitup://open?url=<encoded> از صفحهٔ /go سایت
   298→     *     (لینک‌های هوشمند پیامک — فقط URL دامنهٔ خودمان پذیرفته می‌شود)
   299→     * (null-امن — هرگز به‌خاطر لینک خراب کرش نمی‌کند)
   300→     */
   301→    private fun deepLinkUrl(from: Intent?): String? {
   302→        if (from == null) return null
   303→        try {
   304→            val extra = from.getStringExtra("fitup_link")
   305→            if (!extra.isNullOrBlank() && (extra.startsWith("/") || extra.startsWith("?"))) {
   306→                return BuildConfig.SITE_URL.trimEnd('/') + extra
   307→            }
   308→            val data = from.data ?: return null
   309→            val scheme = data.scheme?.lowercase()
   310→            // ─── v1.2.6: fitup://open?url=<encoded> — از لینک هوشمند /go سایت ───
   311→            if (scheme == "fitup") {
   312→                val target = data.getQueryParameter("url") ?: return null
   313→                if (target.startsWith("/")) {
   314→                    // مسیر نسبی روی دامنهٔ خودمان
   315→                    return BuildConfig.SITE_URL.trimEnd('/') + target
   316→                }
   317→                val t = android.net.Uri.parse(target)
   318→                val tScheme = t.scheme?.lowercase()
   319→                val tHost = t.host?.lowercase() ?: return null
   320→                val isOurHost = tHost == "fittup.ir" || tHost.endsWith(".fittup.ir")
   321→                if (isOurHost && (tScheme == "https")) return target
   322→                return null
   323→            }
   324→            if (scheme != "https") return null
   325→            val host = data.host?.lowercase() ?: return null
   326→            // فقط دامنهٔ خودمان — همان منطق routeUrl (fittup.ir و زیردامنه‌ها)
   327→            if (host == "fittup.ir" || host.endsWith(".fittup.ir")) return data.toString()
   328→        } catch (_: Exception) {
   329→        }
   330→        return null
   331→    }

   537→    private fun handleExternalScheme(uri: Uri): Boolean {
   538→        try {
   539→            when (uri.scheme?.lowercase()) {
   540→                "tel" -> startActivity(Intent(Intent.ACTION_DIAL, uri))
   541→                "mailto" -> startActivity(Intent(Intent.ACTION_SENDTO, uri))
   542→                "sms" -> startActivity(Intent(Intent.ACTION_SENDTO, uri))
   543→                "intent" -> {
   544→                    val intent = Intent.parseUri(uri.toString(), Intent.URI_INTENT_SCHEME)
   545→                    // سخت‌سازی امنیتی (ضد intent-redirection): فقط ACTION_VIEW عمومی —

   548→                    intent.selector = null
   549→                    intent.setPackage(null)
   550→                    startActivity(intent)

   553→            }
   554→        } catch (_: Exception) {
   555→            // اپی برای این لینک نیست — نادیده بگیر
   556→        }
   557→        return true
   558→    }

   687→    private fun startNativeDownload(url: String, fileName: String): Boolean {
   688→        val safeName = fileName.replace(Regex("[^A-Za-z0-9._-]"), "_").ifBlank {
   689→            "fitup-${System.currentTimeMillis()}"

   692→            val request = DownloadManager.Request(Uri.parse(url))
   693→                .setTitle(safeName)
   694→                .setDescription("دانلود فایل")
   695→                .setNotificationVisibility(DownloadManager.Request.VISIBILITY_VISIBLE_NOTIFY_COMPLETED)

   709→    /** جلوی دوبار اجرای پشت‌سرهم (برادکست + onResume هم‌زمان) */
   710→    private var handlingDownloadFinish = false

   712→    /** پایان یک دانلود: فقط پاک‌سازی وضعیت pending.
   713→     *  فایل‌های معمولی نوتیف خود DownloadManager را دارند؛ APK دیگر اصلاً از
   714→     *  این مسیر نمی‌گذرد (v1.2.8 → مرورگر بیرونی). */
   715→    private fun handleDownloadFinished(id: Long) {
   716→        if (id <= 0 || id != pendingDownloadId()) return
   717→        if (handlingDownloadFinish) return
   718→        handlingDownloadFinish = true
   719→        try {
   720→            clearPendingDownload()
   721→        } finally {
   722→            handlingDownloadFinish = false
   723→        }
   724→    }

   727→     *  بازسازی اکتیویتی/بستن اپ) — فقط پاک‌سازی وضعیت pending (v1.2.8). */
   728→    private fun checkPendingDownload() {
   729→        val id = pendingDownloadId()
   730→        if (id > 0) handleDownloadFinished(id)
   731→    }

   733→    /** گیرندهٔ پایان دانلود (فایل‌های معمولی) → پاک‌سازی وضعیت pending.
   734→     *  ⚠ RECEIVER_EXPORTED (نه NOT_EXPORTED): برادکست ACTION_DOWNLOAD_COMPLETE
   735→     *  «protected» است و فقط دانلودمنیجر/سیستم می‌تواند بفرستد؛ در اندروید ۱۴+
   736→     *  گیرندهٔ NOT_EXPORTED این برادکست را نمی‌گیرد (فرستنده DownloadProvider است). */
   737→    private fun setupDownloadCompleteReceiver() {
   738→        val receiver = object : BroadcastReceiver() {
   739→            override fun onReceive(context: Context?, intent: Intent?) {
   740→                if (intent?.action != DownloadManager.ACTION_DOWNLOAD_COMPLETE) return
   741→                val id = intent.getLongExtra(DownloadManager.EXTRA_DOWNLOAD_ID, -1L)
   742→                handleDownloadFinished(id)
   743→            }
   744→        }
   745→        try {
   746→            if (Build.VERSION.SDK_INT >= 33) {
   747→                registerReceiver(
   748→                    receiver,
   749→                    IntentFilter(DownloadManager.ACTION_DOWNLOAD_COMPLETE),
   750→                    Context.RECEIVER_EXPORTED
   751→                )
   752→            } else {
   753→                registerReceiver(receiver, IntentFilter(DownloadManager.ACTION_DOWNLOAD_COMPLETE))
   754→            }
   755→        } catch (_: Exception) {}
   756→    }

   758→    /** نام فایل از Content-Disposition یا URL */
   759→    private fun guessFileName(contentDisposition: String?, mimeType: String?, url: String): String {
   760→        try {
   761→            if (!contentDisposition.isNullOrBlank()) {
   762→                val m = Regex("filename\\*=UTF-8''([^;]+)|filename=\"?([^\";]+)\"?", RegexOption.IGNORE_CASE)
   763→                    .find(contentDisposition)
   764→                if (m != null) {
   765→                    val name = (m.groupValues[1].ifBlank { m.groupValues[2] })
   766→                        .replace("+", " ")
   767→                    if (name.isNotBlank()) return java.net.URLDecoder.decode(name, "UTF-8")
   768→                }
   769→            }
   770→            if (!mimeType.isNullOrBlank() && mimeType.equals("application/vnd.android.package-archive", true)) {
   771→                return "fitup-${System.currentTimeMillis()}.apk"
   772→            }
   773→            val path = URI(url).path ?: return "fitup-file"
   774→            return path.substringAfterLast('/').ifBlank { "fitup-file" }
   775→        } catch (_: Exception) {
   776→            return "fitup-file"
   777→        }
   778→    }

   783→    private fun downloadDataUrl(filename: String, dataUrl: String) {
   784→        try {
   785→            val safeName = filename.ifBlank { "fitup-${System.currentTimeMillis()}" }
   786→                .replace(Regex("[^A-Za-z0-9._-\\u0600-\\u06FF ]"), "_")
   787→            val comma = dataUrl.indexOf(',')
   788→            if (comma < 0 || !dataUrl.startsWith("data:", ignoreCase = true)) {
   789→                runOnUiThread { toast("فایل قابل ذخیره نیست") }
   790→                return
   791→            }
   792→            val meta = dataUrl.substring(5, comma)
   793→            val mime = meta.substringBefore(";").ifBlank { "application/octet-stream" }
   794→            val isBase64 = meta.contains("base64", ignoreCase = true)
   795→            val bytes = if (isBase64) {
   796→                Base64.decode(dataUrl.substring(comma + 1), Base64.DEFAULT)
   797→            } else {
   798→                dataUrl.substring(comma + 1).toByteArray(Charsets.UTF_8)
   799→            }
   800→
   801→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   802→                val values = ContentValues().apply {
   803→                    put(MediaStore.Downloads.DISPLAY_NAME, safeName)
   804→                    put(MediaStore.Downloads.MIME_TYPE, mime)
   805→                    put(MediaStore.Downloads.IS_PENDING, 1)
   806→                }
   807→                val uri = contentResolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, values)
   808→                if (uri == null) {
   809→                    runOnUiThread { toast("خطا در ذخیره فایل") }
   810→                    return
   811→                }
   812→                contentResolver.openOutputStream(uri)?.use { it.write(bytes) }
   813→                values.clear()
   814→                values.put(MediaStore.Downloads.IS_PENDING, 0)
   815→                contentResolver.update(uri, values, null, null)
   816→            } else {
   817→                val dir = getExternalFilesDir(Environment.DIRECTORY_DOCUMENTS) ?: filesDir
   818→                val file = File(dir, safeName)
   819→                FileOutputStream(file).use { it.write(bytes) }
   820→            }
   821→            runOnUiThread { toast("«$safeName» ذخیره شد ✓") }
   822→        } catch (e: Exception) {
   823→            Log.e("FitUpApp", "download failed: ${e.message}")
   824→            runOnUiThread { toast("خطا در ذخیره فایل") }
   825→        }
   826→    }

  1003→                    Regex("(?<!\\d)\\d{4}(?!\\d)").findAll(text).map { it.value }.lastOrNull()

  1005→            }
  1006→            if (code != null) {
  1007→                lastClipboardDispatched = text
  1008→                dispatchOtpCode(code)

  1010→        } catch (_: Exception) {}

  1087→                val url = URL(BuildConfig.SITE_URL.trimEnd('/') + "/api/app/own/latest")

  1089→                conn.connectTimeout = 10_000
  1090→                conn.readTimeout = 10_000
  1091→                conn.instanceFollowRedirects = true
  1092→                val body = conn.inputStream.bufferedReader().use { it.readText() }
  1093→                conn.disconnect()
  1094→                val json = JSONObject(body)
  1095→                if (!json.optBoolean("available", false)) return@thread
  1096→                val latest = json.optInt("latestVersionCode", 1)
  1097→                val forced = json.optBoolean("forceUpdate", false)
  1098→                val apkUrl = BuildConfig.SITE_URL.trimEnd('/') + "/api/app/own/download"
  1099→                if (BuildConfig.VERSION_CODE < latest && forced) {
  1100→                    runOnUiThread {

  1102→                        if (errorView.visibility == View.VISIBLE && !forceDialogShown) {

  1106→                    }
  1107→                }
  1108→            } catch (e: Exception) {
  1109→                // آفلاین/خطا — نادیده بگیر؛ در اجرای بعدی دوباره تلاش می‌شود
  1110→            }
  1111→        }
  1112→    }
