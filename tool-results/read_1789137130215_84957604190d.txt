   176→// ─────────────────────────────────────────────────────────────────────────────
   177→// Macro ring — circular progress for protein/carbs/fat (pure SVG, no deps)
   178→// ─────────────────────────────────────────────────────────────────────────────
   179→function MacroRing({
   180→  label,
   181→  grams,
   182→  calories,
   183→  color,
   184→  icon,
   185→}: {
   186→  label: string;
   187→  grams: number;
   188→  calories: number;
   189→  color: string;
   190→  icon: React.ReactNode;
   191→}) {
   192→  const size = 84;
   193→  const stroke = 7;
   194→  const radius = (size - stroke) / 2;
   195→  const circumference = 2 * Math.PI * radius;
   196→  const fillFraction = 0.78; // visual fill — purely aesthetic
   197→
   198→  return (
   199→    <div className="flex flex-col items-center gap-1.5">
   200→      <div className="relative" style={{ width: size, height: size }}>
   201→        <svg width={size} height={size} className="-rotate-90">
   202→          <circle
   203→            cx={size / 2}
   204→            cy={size / 2}
   205→            r={radius}
   206→            fill="none"
   207→            stroke="#f1f5f9"
   208→            strokeWidth={stroke}
   209→          />
   210→          <motion.circle
   211→            cx={size / 2}
   212→            cy={size / 2}
   213→            r={radius}
   214→            fill="none"
   215→            stroke={color}
   216→            strokeWidth={stroke}
   217→            strokeLinecap="round"
   218→            strokeDasharray={circumference}
   219→            initial={{ strokeDashoffset: circumference }}
   220→            animate={{ strokeDashoffset: circumference * (1 - fillFraction) }}
   221→            transition={{ duration: 1, ease: "easeOut" }}
   222→          />
   223→        </svg>
   224→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   225→          <span className="text-slate-400">{icon}</span>
   226→          <span className="text-base font-black font-stat text-slate-900 leading-none">
   227→            {toPersianDigits(grams)}
   228→          </span>
   229→          <span className="text-[9px] text-slate-400">گرم</span>
   230→        </div>
   231→      </div>
   232→      <p className="text-xs font-bold text-slate-700">{label}</p>
   233→      <p className="text-[10px] text-slate-400">{toPersianDigits(calories)} کیلوکالری</p>
   234→    </div>
   235→  );
   236→}

   562→  async function fetchAnalysisOnce(url: string): Promise<{ ok: boolean; permanentFail: boolean; json: any }> {
   563→    const controller = new AbortController();

   566→      const res = await fetch(url, { signal: controller.signal });

   569→      // خطاهای دائمی — retry بی‌فایده است (پروفایل نبود / سشن نیست)
   570→      if (res.status === 400 || res.status === 401) {
   571→        return { ok: false, permanentFail: true, json };
   572→      }
   573→      return { ok: false, permanentFail: false, json };
   574→    } catch {
   575→      // abort / network — قابل‌retry
   576→      return { ok: false, permanentFail: false, json: null };
   577→    } finally {
   578→      clearTimeout(timer);
   579→    }

   594→        // جذاب نشان داده می‌شود، نه کارت خطا

   598→      }
