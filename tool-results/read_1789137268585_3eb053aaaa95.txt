    32→async function saveFramesToDisk(dataUrls: string[]): Promise<string[]> {
    33→  const urls: string[] = [];
    34→  const stamp = Date.now();
    35→  for (let i = 0; i < dataUrls.length; i++) {
    36→    try {
    37→      const parsed = parseDataUrl(dataUrls[i], "image/jpeg");
    38→      const ext = parsed.mime === "image/png" ? "png" : parsed.mime === "image/webp" ? "webp" : "jpg";
    39→      const buffer = Buffer.from(parsed.base64, "base64");
    40→      if (buffer.length === 0) continue;
    41→      const fileName = `chat-vframe-${stamp}-${Math.random().toString(36).slice(2, 8)}-${i}.${ext}`;
    42→      const { url } = await savePrivateMediaFile("chat", fileName, buffer);
    43→      urls.push(url);
    44→    } catch (frameErr) {
    45→      console.error("[chat] frame save failed (skipping):", frameErr);
    46→    }
    47→  }
    48→  return urls;
    49→}

    52→ * v71 — بارگذاری مدیای یک پیام تاریخی به‌صورت data URL (برای تحلیل روی‌تقاضا).
    53→ * عکس: فایل بهینه‌شده روی دیسک. ویدیو: فریم‌های ذخیره‌شده (v70) یا استخراج دوباره ffmpeg.

    56→async function loadMediaDataUrls(m: {
    57→  mediaUrl: string | null;
    58→  mediaType: string | null;
    59→  mediaFrames: string | null;
    60→}): Promise<string[]> {
    61→  try {
    62→    if (!m.mediaUrl) return [];
    63→    if (m.mediaType === "image") {
    64→      const buf = await readFile(absolutePathForUploadUrl(m.mediaUrl));
    65→      if (buf.length === 0) return [];
    66→      const mime = m.mediaUrl.endsWith(".png") ? "image/png" : "image/webp";
    67→      return [`data:${mime};base64,${buf.toString("base64")}`];
    68→    }
    69→    // ویدیو — اولویت: فریم‌های ذخیره‌شده (URL فایل) → استخراج دوباره از فایل
    70→    let frameFileUrls: string[] = [];
    71→    if (m.mediaFrames) {
    72→      try {
    73→        const parsedFrames = JSON.parse(m.mediaFrames);
    74→        if (Array.isArray(parsedFrames)) frameFileUrls = parsedFrames.filter((u: unknown): u is string => typeof u === "string");
    75→      } catch {
    76→        // JSON خراب → فال‌بک به استخراج دوباره
    77→      }
    78→    }
    79→    if (frameFileUrls.length === 0) {
    80→      // extractVideoFramesAsDataUrls خودش data URL برمی‌گرداند
    81→      return await extractVideoFramesAsDataUrls(absolutePathForUploadUrl(m.mediaUrl), 6);
    82→    }
    83→    const dataUrls: string[] = [];
    84→    for (const u of frameFileUrls.slice(0, MAX_CLIENT_FRAMES)) {
    85→      try {
    86→        const buf = await readFile(absolutePathForUploadUrl(u));
    87→        if (buf.length === 0) continue;
    88→        const mime = u.endsWith(".png") ? "image/png" : u.endsWith(".webp") ? "image/webp" : "image/jpeg";
    89→        dataUrls.push(`data:${mime};base64,${buf.toString("base64")}`);
    90→      } catch {
    91→        // فریم حذف‌شده از دیسک — رد شو
    92→      }
    93→    }
    94→    return dataUrls;
    95→  } catch (err) {
    96→    console.error("[chat] loadMediaDataUrls failed:", err);
    97→    return [];
    98→  }
    99→}

   119→async function collectHistoryMediaAnalyses(
   120→  userId: string,
   121→  history: {
   122→    id: string;
   123→    content: string;
   124→    mediaUrl: string | null;
   125→    mediaType: string | null;
   126→    mediaFrames: string | null;
   127→    mediaAnalysis: string | null;
   128→  }[]
   129→): Promise<string> {
   130→  try {
   131→    const recent = history.slice(-MEDIA_SYNC_WINDOW);
   132→    const mediaMsgs = recent.filter((m) => m.mediaUrl && m.mediaType);
   133→    if (mediaMsgs.length === 0) return "";
   134→
   135→    const parts: string[] = [];
   136→    let onDemandBudget = 1;
   137→    for (const m of mediaMsgs) {
   138→      if (m.mediaAnalysis?.trim()) {
   139→        parts.push(formatMediaAnalysisPart(m, m.mediaAnalysis.trim()));
   140→        continue;
   141→      }
   142→      if (onDemandBudget <= 0) continue;
   143→      onDemandBudget--;
   144→      try {
   145→        const dataUrls = await loadMediaDataUrls(m);
   146→        if (dataUrls.length === 0) continue;
   147→        const text = await analyzeChatMedia(
   148→          m.mediaType === "video" ? "video-frames" : "image",
   149→          dataUrls,
   150→          m.content,
   151→          userId
   152→        );
   153→        parts.push(formatMediaAnalysisPart(m, text));
   154→        // کش روی همان پیام تاریخی — دفعات بعد بدون کال ویژن
   155→        try {
   156→          await db.chatMessage.update({ where: { id: m.id }, data: { mediaAnalysis: text } });
   157→        } catch (cacheErr) {
   158→          console.error("[chat] mediaAnalysis cache update failed:", cacheErr);
   159→        }
   160→      } catch (anErr) {
   161→        console.error("[chat] on-demand media analysis failed (skipping):", anErr);
   162→      }
   163→    }
   164→    if (parts.length === 0) return "";
   165→    return `\n\n[تحلیل رسانه‌های اخیر کاربر در همین گفتگو — تولید مدل بینایی فیتاپ — برای مرجع تو:\n${parts.join("\n")}\nاگر پرسش فعلی دربارهٔ همین رسانه‌هاست، پاسخت را بر اساس این تحلیل‌ها بده؛ رسانه‌ها خودشان پیوست نشده‌اند و لازم نیست بگویی «نمی‌توانم ببینم».]`;
   166→  } catch (err) {
   167→    console.error("[chat] collectHistoryMediaAnalyses failed (continuing without sync):", err);
   168→    return "";
   169→  }
   170→}

   176→async function validateChatVideoUrl(userId: string, url: string): Promise<string | null> {
   177→  if (!url.startsWith("/uploads/chat/")) return null;
   178→  if (url.includes("..")) return null;

   180→  const fileName = url.split("/").pop() || "";
   181→  if (!fileName.startsWith(`chat-video-${userId}-`)) return null;
   182→  const filePath = absolutePathForUploadUrl(url);
   183→  try {
   184→    const st = await stat(filePath);
   185→    if (!st.isFile() || st.size === 0) return null;
   186→  } catch {
   187→    return null;
   188→  }
   189→  return url;
   190→}

   205→/**

   208→ *  - "data:image/jpeg;base64,...."

   211→function parseDataUrl(value: string, fallbackMime = "image/jpeg"): { mime: string; base64: string } {
   212→  const match = value.match(/^data:([^;]+);base64,(.+)$/);
   213→  if (match) {
   214→    return { mime: match[1], base64: match[2] };
   215→  }
   216→  return { mime: fallbackMime, base64: value };
   217→}

   244→/**
   245→ * ذخیره عکس با کاهش حجم خودکار:
   246→ *  - resize به حداکثر 1600px

   254→const MAX_IMAGE_INPUT_PIXELS = 60_000_000;

   256→async function saveOptimizedImage(
   257→  base64: string,
   258→  mime: string
   259→): Promise<{ url: string; mime: string; size: number; optimizedBase64: string } | null> {
   260→  const sharp = (await import("sharp")).default;
   261→  const buffer = Buffer.from(base64, "base64");
   262→
   263→  // کاهش حجم + resize + WebP (با محدودیت پیکسل — fail-safe)
   264→  let processed: Buffer;
   265→  try {
   266→    processed = await sharp(buffer, { limitInputPixels: MAX_IMAGE_INPUT_PIXELS })
   267→      .resize(1600, 1600, {
   268→        fit: "inside",
   269→        withoutEnlargement: true,
   270→      })
   271→      .webp({ quality: 75 })
   272→      .toBuffer();
   273→  } catch (err) {
   274→    console.error("[chat] image optimization failed (too large/corrupt):", err);
   275→    return null;
   276→  }
   277→
   278→  const fileName = `chat-image-${Date.now()}-${Math.random().toString(36).slice(2, 8)}.webp`;
   279→  const { url } = await savePrivateMediaFile("chat", fileName, processed);
   280→
   281→  const optimizedBase64 = processed.toString("base64");
   282→  return {
   283→    url,
   284→    mime: "image/webp",
   285→    size: processed.length,
   286→    optimizedBase64,
   287→  };
   288→}

   368→export async function POST(req: NextRequest) {
   369→  try {
   370→    // گیت پلن: چت با مربی هوشمند فقط برای پلن پیشرفته و حرفه‌ای
   371→    const { userId } = await requirePlanCapability("aiChat");

   373→    const rl = rateLimit(`coach-chat:${userId}`, 30, 60_000);
   374→    if (!rl.ok) {
   375→      return rateLimitResponse(rl.retryAfterSec);
   376→    }
   377→    const user = await requireAuth();
   378→    const body = await req.json();
   379→    const message: string = typeof body.message === "string" ? body.message : "";
   380→    const imageBase64: string | undefined = typeof body.imageBase64 === "string" ? body.imageBase64 : undefined;
   381→    const videoBase64: string | undefined = typeof body.videoBase64 === "string" ? body.videoBase64 : undefined;
   382→    // ─── مسیر multipart جدید (ویدیوهای بدون محدودیت حجم — Task 4-a) ───
   383→    // کلاینت ابتدا POST /api/coach/chat/upload را صدا می‌زند (فایل ذخیره + فشرده‌سازی
   384→    // + استخراج فریم + مصرف سهمیه) و بعد اینجا videoUrl/videoFrames را می‌فرستد.
   385→    const clientVideoUrl: string | undefined = typeof body.videoUrl === "string" ? body.videoUrl : undefined;

   696→      console.error("[chat] buildSportsProfileContext failed (continuing without it):", ctxErr);
   697→    }
   698→    if (sportsFileCtx) {
   699→      athleticProfile.push(sportsFileCtx);
   700→    }
   701→
   702→    // Load recent history (last 15 messages for faster AI response)
   703→    // ─── H3: قبلاً orderBy asc + take 15 «قدیمی‌ترین» ۱۵ پیام را می‌آورد (حافظه چت خراب بود).
   704→    // الگوی درست: desc + take → آخرین پیام‌ها، سپس reverse برای ترتیب زمانی.
   705→    // پیام جاری (userMsg) حذف می‌شود چون جداگانه به‌عنوان پیام آخر به AI پاس داده می‌شود.
   706→    const recentHistory = await db.chatMessage.findMany({
   707→      where: { userId: user.id, id: { not: userMsg.id } },
   708→      orderBy: { createdAt: "desc" },
   709→      take: 15,
   710→    });
   711→    const history = recentHistory.reverse();
   712→
   713→    // ─── v71 سینک رسانه — دیریکتیو مالک: «متن‌ها با دیپ‌سیک پاسخ داده بشه و
   714→    // رسانه‌ها با جمنای تحلیل بشن و این تنوع با هم سینک باشه» ───
   715→    // تحلیل‌های متنیِ کش‌شدهٔ جمنای (mediaAnalysis) از ۶ پیام آخر جمع می‌شود و
   716→    // به کال متنیِ دیپ‌سیک تزریق می‌شود — بدون هیچ کال ویژن تکراری. مدیای اخیر
   717→    // بدون کش (پیام قدیمی v70) یک‌بار روی‌تقاضا تحلیل و کش می‌شود (بودجه: ۱).
   718→    const historyMediaAnalyses = await collectHistoryMediaAnalyses(
   719→      user.id,
   720→      history.map((h) => ({
   721→        id: h.id,
   722→        content: h.content,
   723→        mediaUrl: h.mediaUrl,
   724→        mediaType: h.mediaType,
   725→        mediaFrames: h.mediaFrames,
   726→        mediaAnalysis: h.mediaAnalysis,
   727→      }))
   728→    );
   729→
   730→    // Get AI response — pass user's plan tier + full athletic profile for comprehensive guidance
   731→    // v71: پاسخ نهایی همیشه دیپ‌سیک (aiChat) — تحلیل جمنایِ مدیای همین پیام +
   732→    // تحلیل‌های کش‌شدهٔ رسانه‌های قبلی، به‌صورت متن تزریق می‌شوند.
   733→    const userPlan = (user.planName as any) ?? null;
   734→    const fullContext = athleticProfile.join("\n");
   735→    const mediaAnalysisForAi = [
   736→      historyMediaAnalyses,
   737→      currentMediaAnalysis
   738→        ? `\n\n[تحلیل ${attachment?.kind === "image" ? "عکس" : "ویدیو"}ی که کاربر در همین پیام فرستاده است — تولید مدل بینایی فیتاپ:\n${currentMediaAnalysis}]`
   739→        : "",
   740→    ].filter(Boolean).join("");
   741→    let aiResponse: string;
   742→    try {
   743→      aiResponse = await aiChat(
   744→        onboarding,
   745→        history.map((h) => ({ role: h.role, content: h.content })),
   746→        finalMessage + (fullContext ? `\n\n[سیستم - پرونده ورزشی کاربر]:\n${fullContext}` : ""),
   747→        userPlan,
   748→        mediaAnalysisForAi || null,
   749→        user.id
   750→      );
   751→    } catch (err) {
   752→      // ─── حذف ردیف پیام کاربر تا یتیم نماند (باگ ممیزی 2-b) ───
   753→      // پیام کاربر قبل از کال AI ذخیره شده؛ اگر AI شکست بخورد و ردیف بماند،
   754→      // بعد از رفرش پیام بی‌جواب ظاهر می‌شود و ارسال مجدد ردیف تکراری می‌سازد.
   755→      // حذف best-effort — خطای حذف نباید خطای اصلی را بپوشاند.
   756→      try {
   757→        await db.chatMessage.delete({ where: { id: userMsg.id } });
   758→      } catch (delErr) {
   759→        console.error("[chat] failed to delete orphaned user message:", delErr);
   760→      }
   761→      throw err;
   762→    }
   763→
   764→    // Save AI response
   765→    // ─── Task 4-a: تولید TTS حذف شد — پاسخ فقط متن است ───
   766→    const aiMsg = await db.chatMessage.create({
   767→      data: { userId: user.id, role: "assistant", content: aiResponse },
   768→    });
   769→
   770→    return Response.json({
   771→      userMessage: {
   772→        id: userMsg.id,
   773→        role: userMsg.role,
   774→        content: userMsg.content,
   775→        mediaUrl: userMsg.mediaUrl ?? null,
   776→        mediaType: userMsg.mediaType ?? null,
   777→        createdAt: userMsg.createdAt.toISOString(),
   778→      },
   779→      aiMessage: {
   780→        id: aiMsg.id,
   781→        role: aiMsg.role,
   782→        content: aiMsg.content,
   783→        mediaUrl: null,
   784→        mediaType: null,
   785→        createdAt: aiMsg.createdAt.toISOString(),
   786→      },
   787→    });
   788→  } catch (e) {
   789→    return apiError(e);
   790→  }
