    24→export function smsFirstName(name: string | null | undefined): string {
    25→  const first = String(name || "")
    26→    .replace(/[\r\n\t]/g, " ")
    27→    .trim()
    28→    .split(/\s+/)[0] || "";
    29→  return first.slice(0, 25);
    30→}

   661→export interface AbandonedCartSmsOptions {

   666→  link: string;
   667→  /** v61 — با حذف مسیر متن آزاد دیگر استفاده نمی‌شود (برای سازگاری call-site) */

   673→export interface AbandonedCartSmsResult {
   674→  sent: boolean;
   675→  /** کدام مسیر تحویل داد (v61: فقط قالب — bulk حذف شد) */
   676→  via?: "template";
   677→  /** already_sent = دداپ قبلی (هرگز دوباره نفرست) */
   678→  skipped?: "already_sent" | "invalid_mobile";
   679→  error?: string;
   680→}
   681→
   682→/** رکورد failed در SmsLog (best-effort — هرگز نتیجهٔ ارسال را تغییر نمی‌دهد) */
   683→async function recordAbandonedCartFailure(
   684→  normalizedMobile: string,
   685→  dedupeKey: string,
   686→  templateId: string,
   687→  userId: string | null | undefined,
   688→  error: string
   689→): Promise<void> {
   690→  try {
   691→    const { db } = await import("@/lib/db");
   692→    await db.smsLog.upsert({
   693→      where: { mobile_key: { mobile: normalizedMobile, key: dedupeKey } },
   694→      create: {
   695→        mobile: normalizedMobile,
   696→        key: dedupeKey,
   697→        templateId: templateId || "template",
   698→        userId: userId ?? null,
   699→        status: "failed",
   700→        error,
   701→      },
   702→      update: {
   703→        status: "failed",
   704→        error,
   705→      },
   706→    });
   707→  } catch (e) {
   708→    console.warn("[sendAbandonedCartSms] failed-record upsert error:", e);
   709→  }
   710→}
   711→
   712→/**
   713→ * ارسال پیامک ترک درگاه — فقط با قالب sms.ir (نردبان خودترمیم پارامترها).
   714→ * v61: مسیر bulk/متن آزاد حذف شد (دیریکتیو مالک + خط خدماتی که bulk را
   715→ * پشتیبانی نمی‌کند). شکست قالب = رکورد failed و تلاشِ sweep بعدی.
   716→ */
   717→export async function sendAbandonedCartSms(
   718→  opts: AbandonedCartSmsOptions
   719→): Promise<AbandonedCartSmsResult> {
   720→  const { user, dedupeKey, link, templateId } = opts;
   721→
   722→  const normalized = normalizeMobileForSmsIr(user.mobile);
   723→  if (!/^9\d{9}$/.test(normalized)) {
   724→    return { sent: false, skipped: "invalid_mobile", error: `شماره نامعتبر: ${normalized}` };
   725→  }
   726→
   727→  const name = smsFirstName(user.name);
   728→
   729→  // ─── مسیر واحد — قالب sms.ir (v61: قالبی الزامی؛ bulk حذف شد) ───
   730→  const templateIdSafe = templateId?.trim() || "";
   731→  if (!templateIdSafe) {
   732→    const noTpl =
   733→      "شناسه قالب ترک خرید تنظیم نیست (SMSIR_ABANDONED_CART_TEMPLATE_ID یا SiteSetting: abandoned_cart_sms_template_id) — پیامک قالبی ارسال نشد.";
   734→    console.warn(`[sendAbandonedCartSms] (key=${dedupeKey}) ${noTpl}`);
   735→    await recordAbandonedCartFailure(normalized, dedupeKey, "", user.id, noTpl);
   736→    return { sent: false, error: noTpl };
   737→  }
   738→
   739→  // نردبان خودترمیم پارامترهای قالب: [NAME+LINK] → [NAME] → []
   740→  const attempts: Array<Array<{ name: string; value: string }>> = [
   741→    [
   742→      { name: "NAME", value: name },
   743→      { name: "LINK", value: link },
   744→    ],
   745→    [{ name: "NAME", value: name }],
   746→    [],
   747→  ];
   748→  let last: SendTemplateSmsResult = { sent: false, error: "no attempt" };
   749→  for (const params of attempts) {
   750→    last = await sendTemplateSms({
   751→      key: dedupeKey,
   752→      templateEnvKey: undefined,
   753→      templateIdOverride: templateIdSafe,
   754→      mobile: user.mobile,
   755→      params,
   756→      userId: user.id,
   757→    });
   758→    if (last.sent) return { sent: true, via: "template" };
   759→    if (last.skipped === "already_sent" || last.skipped === "invalid_mobile") {
   760→      return {
   761→        sent: false,
   762→        skipped: last.skipped === "invalid_mobile" ? "invalid_mobile" : "already_sent",
   763→        error: last.error,
   764→      };
   765→    }
   766→    // no_template / no_name / خطای درگاه → رانگ بعدی نردبان
   767→  }
   768→
   769→  // ─── شکست نهایی قالب — رکورد failed برای دیده‌شدن؛ sweep بعدی دوباره تلاش می‌کند ───
   770→  // (sendTemplateSms خودش در هر تلاش رکورد ثبت می‌کند؛ این upsert فقط گارانتی
   771→  //  می‌کند که حتماً یک رکورد failed با آخرین خطا موجود باشد.)
   772→  await recordAbandonedCartFailure(
   773→    normalized,
   774→    dedupeKey,
   775→    templateIdSafe,
   776→    user.id,
   777→    last.error ?? "abandoned_cart sms failed"
   778→  );
   779→
   780→  return { sent: false, error: last.error ?? "abandoned_cart sms failed" };
   781→}
