    28→/** پارس لیست ذخیره‌شده به‌صورت JSON یا CSV */
    29→function safeParseList(raw: string | null | undefined): string[] {
    30→  if (!raw) return [];
    31→  const t = raw.trim();
    32→  if (!t) return [];
    33→  try {
    34→    const p = JSON.parse(t);
    35→    if (Array.isArray(p)) return p.map((x) => String(x));
    36→    if (typeof p === "string") return p.split(",").map((s) => s.trim()).filter(Boolean);
    37→    return [];
    38→  } catch {
    39→    return t.split(",").map((s) => s.trim()).filter(Boolean);
    40→  }
    41→}

   287→/**
   288→ * شروع تولید برنامه در پس‌زمینه برای کاربر.
   289→ *
   290→ * پیش‌شرط‌ها:
   291→ *  - اشتراک فعال یا pending (پلن مؤثر)
   292→ *  - پروفایل آنبوردینگ موجود
   293→ *  - همه پیش‌نیازها تعیین تکلیف شده باشند
   294→ *  - هیچ تولید در حال اجرای تازه‌ای وجود نداشته باشد (جلوگیری از دوباره‌کاری)
   295→ *
   296→ * خروجی فوری برمی‌گرداند؛ تولید در پس‌زمینه ادامه می‌یابد.
   297→ */
   298→export async function startProgramGenerationInBackground(userId: string): Promise<StartGenerationResult> {
   299→  // پلن مؤثر (active یا pending داخل پنجره)
   300→  const dto = await buildUserDto(userId);
   301→  const effectivePlan = (dto?.planName ?? null) as Plan | null;
   302→  if (!effectivePlan) {
   303→    return { started: false, reason: "no_plan" };
   304→  }
   305→
   306→  const profile = await db.onboardingProfile.findUnique({ where: { userId } });
   307→  if (!profile) {
   308→    return { started: false, reason: "no_profile" };
   309→  }
   310→
   311→  // پیش‌نیازها باید تعیین تکلیف شده باشند
   312→  const prereqCheck = await checkPrerequisites(userId, effectivePlan);
   313→  if (!prereqCheck.canGenerateProgram) {
   314→    return { started: false, reason: "prerequisites_incomplete", blockingReason: prereqCheck.blockingReason };
   315→  }
   316→
   317→  // ─── 🩹 v45 (باگ بحرانی مالک + نشتی اعتبار AI) — دو گارد سخت ───
   318→  // ① سقف روزانه: حداکثر ۵ شروع تولید در ۲۴ ساعت برای هر کاربر — هر منبعی
   319→  // (پرداخت، تحلیل ویدیو/خون، watchdog، سوئیپ، ادمین) از این سقف عبور
   320→  // نمی‌کند؛ حلقه‌های بی‌پایان دیگر نمی‌توانند اعتبار AI را بسوزانند.
   321→  // ② گارد برنامهٔ تازه: اگر آخرین درخواست ready است و برنامهٔ فعالِ
   322→  // ساخته‌شده بعد از آن وجود دارد، تولید دوباره بی‌معنی است (قبلاً reason
   323→  // «already_has_fresh_plan» در interface بود ولی هرگز برگردانده نمی‌شد و
   324→  // همین حلقهٔ «هر ۳۰ دقیقه نوتیف برنامهٔ شما آماده شد» را ساخته بود).
   325→  const DAY_MS = 24 * 60 * 60 * 1000;
   326→  const dayAgo = new Date(Date.now() - DAY_MS);
   327→  const recentStarts = await db.programRequest.count({
   328→    where: { userId, createdAt: { gte: dayAgo } },
   329→  });
   330→  if (recentStarts >= 5) {
   331→    console.warn(
   332→      `[program-generation] daily budget exhausted for ${userId} (${recentStarts} starts/24h) — refusing`
   333→    );
   334→    return { started: false, reason: "daily_budget" };
   335→  }
   336→
   337→  // جلوگیری از تولید همزمان: اگر آخرین درخواست «generating» و تازه است، دوباره شروع نکن
   338→  // (پنجره ۲۰ دقیقه‌ای هم‌راستا با بازیابی درخواست‌های گیرکرده — M3)
   339→  const latestReq = await db.programRequest.findFirst({
   340→    where: { userId },
   341→    orderBy: { createdAt: "desc" },
   342→  });
   343→  const stuckWindowAgo = new Date(Date.now() - STUCK_GENERATION_WINDOW_MS);
   344→
   345→  // 🩹 v45 — گارد برنامهٔ تازه (بخش ②): آخرین درخواست ready + برنامهٔ فعال
   346→  // بعد از آن = کاربر برنامه را دارد؛ تولید مجدد فقط هزینهٔ AI + نوتیف تکراری
   347→  if (latestReq && latestReq.status === "ready") {
   348→    const freshPlan = await db.workoutPlan.findFirst({
   349→      where: {
   350→        userId,
   351→        active: true,
   352→        createdAt: { gte: latestReq.updatedAt },
   353→      },
   354→      select: { id: true },
   355→    });
   356→    if (freshPlan) {
   357→      return { started: false, reason: "already_has_fresh_plan" };
   358→    }
   359→  }
   360→
   361→  // ─── Claim اتمیک (رفع TOCTOU) ───
   362→  // قبلاً «خواندن وضعیت» و «نوشتن generating» دو مرحله جدا بودند؛ دو فراخوانی
   363→  // همزمان (مثلاً verify + submit-body-analysis) هر دو از چک رد می‌شدند و

   432→  // ─── تولید در پس‌زمینه (fire-and-forget) — v38 تاب‌آور ───

   496→
   497→      if (workoutOk && mealOk) {
   498→        if (reqId) {

   529→        notifyProgramReadySms(userId, reqId ?? undefined);
   530→        console.log(`[program-generation] completed for ${userId} in ${Math.round((Date.now() - startedAt) / 1000)}s`);
   531→        return;

   533→
   534→      // سمتِ جاافتاده را دقیق گزارش کن
   535→      const missing: string[] = [];

   537→      if (!mealOk) missing.push("برنامه غذایی");
   538→      throw new Error(
   539→        `تولید ${missing.join(" و ")} پس از ۲ تلاش ناموفق بود${lastErrMsg ? ` — ${lastErrMsg}` : ""}`
   540→      );
   541→    } catch (err) {
   542→      const errMsg = String((err as Error)?.message || err).slice(0, 480) || "خطای ناشناخته";
   543→      console.error("[program-generation] background generation failed:", err);
   544→      try {
   545→        if (reqId) {
   546→          await db.programRequest.update({
   547→            where: { id: reqId },
   548→            data: { status: "failed", lastError: errMsg },
   549→          });
   550→        }
   551→      } catch {}
   552→      try {
   553→        await createNotification(
   554→          userId,
   555→          "system",
   556→          "خطا در تولید برنامه — از تب برنامه‌ها دوباره تلاش کنید ⚠️",

   558→          "?tab=programs",
   559→          { from: "program-generation", action: "plan_generation_failed" }
   560→        );
   561→      } catch {}
   562→    }
   563→  })();
   564→
   565→  return { started: true };
   566→}

   580→const STUCK_GENERATION_WINDOW_MS = 50 * 60 * 1000;

   582→/**
   583→ * Watchdog خودترمیم — بازیابی درخواست‌های «generating» گیرکرده (یتیم‌شده).
   584→ *
   585→ * چرا: تولید برنامه fire-and-forget است. اگر پروسه سرور وسط تولید restart شود
   586→ * (dev-mode memory restart، crash، دیپلوی)، پروسه پس‌زمینه می‌میرد و
   587→ * ProgramRequest برای همیشه «generating» می‌ماند.
   588→ *
   589→ * رفتار:
   590→ *  - درخواست گیرکرده (>۲۰ دقیقه بدون به‌روزرسانی) پیدا می‌شود
   591→ *  - اگر برنامه‌ای بعد از شروع همین چرخه تولید ذخیره شده باشد → ready
   592→ *    (تولید کامل شده ولی آپدیت status به‌خاطر مرگ پروسه نرسیده)
   593→ *  - اگر برنامه‌ای نیست و autoRetryCount < 1:
   594→ *      → یک‌بار به‌صورت خودکار تولید دوباره شروع می‌شود (بدون دخالت کاربر)
   595→ *  - وگرنه: failed (کاربر از تب برنامه‌ها دکمه retry را می‌زند)
   596→ *
   597→ * این تابع از endpointهایی صدا زده می‌شود که فرانت‌اند موقع generating به آن‌ها
   598→ * poll می‌کند (coach/plan GET و program-history GET) — یعنی watchdog همان‌جا
   599→ * فعال می‌شود که کاربر منتظر برنامه است.
   600→ */
   601→export async function recoverStuckGenerations(userId: string): Promise<void> {
   602→  try {
   603→    const stuckWindowAgo = new Date(Date.now() - STUCK_GENERATION_WINDOW_MS);
   604→    const stuck = await db.programRequest.findFirst({
   605→      where: { userId, status: "generating", updatedAt: { lt: stuckWindowAgo } },
   606→      orderBy: { createdAt: "desc" },
   607→    });
   608→    if (!stuck) return;
   609→
   610→    // آیا برنامه‌ای در این مدت ذخیره شده؟ (تولید ممکن است کامل شده باشد ولی
   611→    // آپدیت status به‌خاطر مرگ پروسه نرسیده باشد)
   612→    // ─── M2: فقط برنامه‌ای که «بعد از شروع همین چرخه تولید» ساخته شده دلیل
   613→    // آماده‌بودن است — updatedAt همان لحظه‌ای است که وضعیت generating شده؛
   614→    // برنامه دوره قبلی نباید تولید گیرکردهِ کاربر تمدیدی را ready کند.
   615→    const hasPlan = await db.workoutPlan.findFirst({
   616→      where: { userId, createdAt: { gte: stuck.updatedAt } },
   617→      orderBy: { createdAt: "desc" },
   618→      select: { id: true },
   619→    });
   620→    if (hasPlan) {
   621→      // برنامه موجود است — وضعیت را ready کن (نه failed!)
   622→      await db.programRequest.update({
   623→        where: { id: stuck.id },
   624→        data: { status: "ready" },
   625→      });
   626→      // v37: نوتیف همزمان (قبلاً فقط پیامک می‌رفت — درخواست مالک: پیامک ↔ نوتیف جفت باشند)
   627→      // v63 — dedupe عنوان‌محورِ ۲۴ساعته (هم‌راستا با مسیرهای دیگر) + متنِ یکسان با
   628→      // مسیر تولید عادی. قبلاً متنِ متفاوت بود و dedupeِ (عنوان+متن) آن را رد نمی‌کرد
   629→      // → کاربر هم نوتیف «ساخته شد» و هم «آماده است» می‌گرفت (گزارش نوتیف ۵تایی).
   630→      const alreadyNotified = await db.notification.findFirst({
   631→        where: {
   632→          userId,
   633→          type: "achievement",
   634→          title: "برنامه شما آماده شد! 🎯",
   635→          createdAt: { gte: new Date(Date.now() - 24 * 60 * 60 * 1000) },
   636→        },
   637→        select: { id: true },
   638→      });
   639→      if (!alreadyNotified) {
   640→        await createNotification(
   641→          userId,
   642→          "achievement",
   643→          "برنامه شما آماده شد! 🎯",
   644→          "برنامه تمرینی و غذایی شخصی‌سازی‌شده شما توسط فیتاپ هوشمند ساخته شد. از بخش «تمرینات» و «تغذیه» مشاهده کنید.",
   645→          "?tab=programs"
   646→        );
   647→      }
   648→      // v32: پیامک «برنامه آماده شد» (قالب 663678)
   649→      // v66: دداپ بر چرخهٔ بازیابی‌شده (stuck.id) — دورهٔ بعدی هم پیامک می‌گیرد
   650→      notifyProgramReadySms(userId, stuck.id);
   651→      console.log("[watchdog] recovered orphaned generating → ready (plan exists):", stuck.id);
   652→      return;
   653→    }
   654→
   655→    // برنامه‌ای نیست — اگر تابه‌حال خودترمیم نشده، یک‌بار دوباره شروع کن
   656→    if (stuck.autoRetryCount < 1) {
   657→      await db.programRequest.update({
   658→        where: { id: stuck.id },
   659→        data: { status: "failed", autoRetryCount: { increment: 1 } },
   660→      });
   661→      console.log("[watchdog] auto-retrying orphaned generation:", stuck.id);
   662→      const result = await startProgramGenerationInBackground(userId);

   670→

   672→    await db.programRequest.update({
   673→      where: { id: stuck.id },
   674→      data: { status: "failed" },
   675→    });
   676→    console.log("[watchdog] marked stuck generation as failed (retry budget exhausted):", stuck.id);
   677→  } catch (e) {
   678→    // watchdog هرگز نباید مسیر caller را بشکند
   679→    console.error("[watchdog] recoverStuckGenerations error:", e);
   680→  }

   683→/**
   684→ * v38 — سوئیپ خودکار درخواست‌های «ناموفق» (درخواست مالک: «به هیچ وجه نباید
   685→ * کاربر بدون برنامه بماند»).

   687→ * قبلاً درخواست failed برای همیشه failed می‌ماند مگر کاربر/ادمین دستی retry

   698→const FAILED_RETRY_GAP_MS = 15 * 60 * 1000;

   701→// حتی با همهٔ گاردهای v45، اگر روزی دیتای خراب جدید یا مسیر دورزدنی پیدا شود،
   702→// این بودجه در-حافظه حداکثر سوئیپ‌های مجاز در شبانه‌روز را محدود می‌کند.
   703→// هر تولید برنامهٔ کامل ~۵۰هزار توکن گران است؛ سقف ۶ در ۲۴h یعنی بدترین-case
   704→// نشتی از مسیر سوئیپ ≈ ۳۰۰K توکن/روز — محدود و قابل‌قبول، نه ابدی.
   705→const SWEEP_DAILY_REQUEUE_MAX = 6;
   706→const sweepBudget = { date: "", count: 0 };
   707→
   708→function tryConsumeSweepBudget(): boolean {
   709→  const today = new Date().toISOString().slice(0, 10);
   710→  if (sweepBudget.date !== today) {
   711→    sweepBudget.date = today;
   712→    sweepBudget.count = 0;
   713→  }
   714→  if (sweepBudget.count >= SWEEP_DAILY_REQUEUE_MAX) {
   715→    console.warn(
   716→      `[failed-recovery] daily sweep budget exhausted (${SWEEP_DAILY_REQUEUE_MAX}/24h) — skipping requeue`
   717→    );
   718→    return false;
   719→  }
   720→  sweepBudget.count++;
   721→  return true;
   722→}

   724→export async function recoverFailedGenerations(limit = 3): Promise<number> {
   725→  try {
   726→    const gapAgo = new Date(Date.now() - FAILED_RETRY_GAP_MS);
   727→    const failedRequests = await db.programRequest.findMany({
   728→      where: {
   729→        status: "failed",
   730→        attempts: { lt: FAILED_AUTO_RETRY_MAX },
   731→        updatedAt: { lt: gapAgo },
   732→      },
   733→      orderBy: { updatedAt: "asc" },
   734→      take: limit,
   735→      select: { id: true, userId: true, plan: true, createdAt: true, updatedAt: true },
   736→    });
   737→
   738→    let requeued = 0;
   739→    for (const req of failedRequests) {
   740→      // کاربر نبلاک و دارای پروفایل باشد (خطای دائمی ورودی را دور بزن)
   741→      const user = await db.user.findUnique({
   742→        where: { id: req.userId },
   743→        select: { isBlocked: true },
   744→      });
   745→      if (!user || user.isBlocked) continue;
   746→
   747→      // ─── 🩹 v45 — ریشه باگ بحرانی مالک (نوتیف هر ۳۰ دقیقه + سوختن AI) ───
   748→      // قبلاً این سوئیپ رکورد قدیمی failed را انتخاب می‌کرد ولی
   749→      // startProgramGenerationInBackground روی «آخرین» درخواست کاربر عمل
   750→      // می‌کرد (که معمولاً ready بود) → هر ۳۰ دقیقه: claim برنامهٔ سالم →
   751→      // تولید کامل AI (۴ فراخوانی مدل گران) → نوتیف تکراری → رکورد قدیمی
   752→      // همچنان failed/attempts=0 می‌ماند → حلقهٔ ابدی.
   753→      // حالا:
   754→      //   ① فقط اگر همین رکورد «آخرین» درخواست کاربر باشد صف‌مجدد می‌شود؛
   755→      //   ② رکوردهای قدیمی‌تر terminal می‌شوند (attempts=MAX) تا دیگر
   756→      //      هرگز انتخاب نشوند — دیتای خرابِ موجود هم خودترمیم می‌شود؛
   757→      //   ③ اگر بعد از ساخت همین رکوردِ failed برنامهٔ فعال ذخیره شده
   758→      //      باشد، کاربر برنامه را دارد → تولید تکرار ممنوع، terminal.
   759→      const latest = await db.programRequest.findFirst({
   760→        where: { userId: req.userId },
   761→        orderBy: { createdAt: "desc" },
   762→        select: { id: true },
   763→      });
   764→      if (!latest || latest.id !== req.id) {
   765→        await db.programRequest
   766→          .update({
   767→            where: { id: req.id },
   768→            data: {
   769→              attempts: FAILED_AUTO_RETRY_MAX,
   770→              lastError: "v45 sweep: superseded by a newer request — terminal",
   771→            },
   772→          })
   773→          .catch(() => {});
   774→        console.log(
   775→          `[failed-recovery] ${req.id} is stale (latest=${latest?.id ?? "none"}) → terminalized, will not requeue`
   776→        );
   777→        continue;
   778→      }
   779→      const planAfterFailure = await db.workoutPlan.findFirst({
   780→        where: {
   781→          userId: req.userId,
   782→          active: true,
   783→          createdAt: { gte: req.createdAt },
   784→        },
   785→        select: { id: true },
   786→      });
   787→      if (planAfterFailure) {
   788→        await db.programRequest
   789→          .update({
   790→            where: { id: req.id },
   791→            data: {
   792→              attempts: FAILED_AUTO_RETRY_MAX,
   793→              status: "ready",
   794→              lastError: null,
   795→            },
   796→          })
   797→          .catch(() => {});
   798→        console.log(
   799→          `[failed-recovery] ${req.id} already has a plan created after it → marked ready (no duplicate AI generation)`
   800→        );
   801→        continue;
   802→      }
   803→
   804→      const result: StartGenerationResult = tryConsumeSweepBudget()
   805→        ? await startProgramGenerationInBackground(req.userId)
   806→        : { started: false, reason: "daily_budget" };
   807→      if (result.started) {
   808→        requeued++;
   809→        console.log(
   810→          `[failed-recovery] requeued failed generation for user ${req.userId} (request ${req.id})`
   811→        );
   812→      } else {
   813→        console.log(
   814→          `[failed-recovery] could not requeue ${req.id}: ${result.reason}${result.blockingReason ? ` — ${result.blockingReason}` : ""}`
   815→        );
   816→      }
   817→    }
   818→    return requeued;
   819→  } catch (e) {
   820→    console.error("[failed-recovery] recoverFailedGenerations error:", e);
   821→    return 0;
   822→  }
   823→}
