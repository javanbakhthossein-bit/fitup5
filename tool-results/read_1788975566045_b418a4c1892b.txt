    53→ * - گرد کردن به حداکثر یک رقم اعشار (۱۲.۳۴۵۶۷ → ۱۲.۳)
    54→ * - حذف صفرهای بی‌مورد انتهایی (۱۲.۰ → ۱۲)
    55→ * - تبدیل به ارقام فارسی
    56→ *
    57→ * باگ قبلی: مقادیر خام AI (مثل ۳۱.۵۴۹۹۹۹) یا جمع‌های floating-point
    58→ * (مثل ۱۲۵.۴۰۰۰۰۰۰۰۰۰۰۰۰۱) بدون گرد شدن نمایش داده می‌شدند.
    59→ */
    60→function fmtNutrient(n: number | null | undefined): string {
    61→  const v = Number(n) || 0;
    62→  const r = Math.round(v * 10) / 10;
    63→  return toPersianDigits(r % 1 === 0 ? String(Math.round(r)) : String(r));
    64→}

    66→/** گرد کردن مقدار عددی در لحظه ثبت/دریافت — ریشه‌ای‌ترین جای رفع باگ اعشار */
    67→function roundNutrient<T extends number | null | undefined>(v: T): number {
    68→  const n = Number(v) || 0;
    69→  return Math.round(n * 10) / 10;
    70→}

  1143→function MacroStat({
  1144→  label,

  1146→  color,
  1147→}: {
  1148→  label: string;
  1149→  value: string;
  1150→  color: "orange" | "yellow" | "cyan" | "purple";
  1151→}) {
  1152→  const colorMap = {
  1153→    orange: "bg-orange-50 text-orange-600",
  1154→    yellow: "bg-yellow-50 text-yellow-700",
  1155→    cyan: "bg-cyan-50 text-cyan-700",

  1158→  return (

  1160→      <p className="text-sm font-black leading-none">{value}</p>

  1166→// ════════════════════════════════════════════════════════════════════════════
  1167→// Macro Donut
  1168→// ════════════════════════════════════════════════════════════════════════════
  1169→function MacroDonut({ protein, carbs, fat }: { protein: number; carbs: number; fat: number }) {
  1170→  const total = protein + carbs + fat || 1;
  1171→  const pPct = (protein / total) * 100;
  1172→  const cPct = (carbs / total) * 100;
  1173→  const fPct = (fat / total) * 100;
  1174→
  1175→  const segments = [
  1176→    { pct: pPct, color: "#F4C542", label: "پروتئین", value: protein, icon: Beef },
  1177→    { pct: cPct, color: "#06b6d4", label: "کربوهیدرات", value: carbs, icon: Wheat },
  1178→    { pct: fPct, color: "#a855f7", label: "چربی", value: fat, icon: Droplet },
  1179→  ];
  1180→
  1181→  let offset = 0;
  1182→  const c = 2 * Math.PI * 40;
  1183→
  1184→  return (
  1185→    <div className="flex items-center gap-4">
  1186→      <div className="relative w-32 h-32 shrink-0">
  1187→        <svg viewBox="0 0 100 100" className="w-full h-full -rotate-90">
  1188→          <circle
  1189→            cx="50"
  1190→            cy="50"
  1191→            r="40"
  1192→            fill="none"
  1193→            stroke="currentColor"
  1194→            strokeWidth="10"
  1195→            className="text-muted/30"
  1196→          />
  1197→          {segments.map((seg, i) => {
  1198→            const len = (seg.pct / 100) * c;
  1199→            const dasharray = `${len} ${c - len}`;
  1200→            const el = (
  1201→              <motion.circle
  1202→                key={i}
  1203→                cx="50"
  1204→                cy="50"
  1205→                r="40"
  1206→                fill="none"
  1207→                stroke={seg.color}
  1208→                strokeWidth="10"
  1209→                strokeDasharray={dasharray}
  1210→                strokeDashoffset={-offset}
  1211→                initial={{ opacity: 0 }}
  1212→                animate={{ opacity: 1 }}
  1213→                transition={{ delay: i * 0.2 }}
  1214→                style={{ filter: `drop-shadow(0 0 3px ${seg.color}60)` }}
  1215→              />
  1216→            );
  1217→            offset += len;
  1218→            return el;
  1219→          })}
  1220→        </svg>
  1221→        <div className="absolute inset-0 flex flex-col items-center justify-center">
  1222→          <span className="text-lg font-black">{toPersianDigits(Math.round(total))}</span>
  1223→          <span className="text-[10px] text-muted-foreground">گرم کل</span>
  1224→        </div>
  1225→      </div>
  1226→      <div className="flex-1 space-y-2">
  1227→        {segments.map((seg, i) => (
  1228→          <div key={i} className="flex items-center gap-2">
  1229→            <div className="w-3 h-3 rounded-full" style={{ background: seg.color }} />
  1230→            <div className="flex-1">
  1231→              <p className="text-xs text-muted-foreground">{seg.label}</p>
  1232→              <p className="text-sm font-bold">
  1233→                {toPersianDigits(Math.round(seg.value))}g{" "}
  1234→                <span className="text-[10px] text-muted-foreground font-normal">
  1235→                  ({toPersianDigits(Math.round(seg.pct))}٪)
  1236→                </span>
  1237→              </p>
  1238→            </div>
  1239→          </div>
  1240→        ))}
  1241→      </div>
  1242→    </div>
  1243→  );
  1244→}
