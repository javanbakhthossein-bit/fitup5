    87→// ===================== ERROR LOGS (مدیریت خطاها) =====================
    88→// ذخیره خطاهای client-side (window.onerror)، API route‌ها و server components

    90→model ErrorLog {
    91→  id          String   @id @default(cuid())
    92→  // نوع منبع خطا: "client" (window.onerror) | "api" (API routes) | "server" (server components)
    93→  source      String   // "client" | "api" | "server"
    94→  // کد وضعیت HTTP (برای خطاهای API) یا 0 برای خطاهای client
    95→  statusCode  Int      @default(0)
    96→  // پیام خطا (رشته واقعی خطا)
    97→  message     String
    98→  // Stack trace (در صورت موجود بودن)
    99→  stack       String?
   100→  // URL محل وقوع خطا (page URL یا API path)
   101→  url         String   @default("")
   102→  // HTTP method (برای خطاهای API)
   103→  method      String?  // "GET" | "POST" | ...
   104→  // شناسه کاربر در صورت لاگین بودن (nullable برای خطاهای ناشناس)
   105→  userId      String?
   106→  user        User?    @relation(fields: [userId], references: [id], onDelete: SetNull)
   107→  // اطلاعات مرورگر / User Agent
   108→  userAgent   String?  @default("")
   109→  // زمینه اضافی (JSON string — request body، query params و غیره)
   110→  context     String?  @default("{}")

   112→  reviewed    Boolean  @default(false)
   113→  createdAt   DateTime @default(now())
   114→
   115→  @@index([source])
   116→  @@index([createdAt])
   117→  @@index([userId])
   118→  @@index([reviewed])

   254→// ===================== SUBSCRIPTION & PAYMENT =====================
   255→model Subscription {
   256→  id            String   @id @default(cuid())
   257→  userId        String
   258→  user          User     @relation(fields: [userId], references: [id], onDelete: Cascade)
   259→  plan          String   // basic | standard | advanced | ultimate
   260→  // status: active (در حال استفاده) | pending (خرید شده ولی پیش‌نیازها تکمیل نشده) | expired | cancelled
   261→  // برای پلن‌های advanced/ultimate، وقتی کاربر خرید می‌کند اما هنوز عکس بدن/ویدیو/آزمایش خون را
   262→  // آپلود نکرده، اشتراک با status="pending" و startDate/endDate=null ساخته می‌شود.
   263→  // وقتی پیش‌نیازها تکمیل شد (ساخت برنامه در submit-body-analysis)، status به "active" تغییر
   264→  // می‌کند و startDate=now، endDate=now+45day تنظیم می‌شود.
   265→  // ۴۵ روز از زمان تکمیل پیش‌نیازها شروع می‌شود، نه از زمان خرید.
   266→  status        String   @default("active") // active | pending | expired | cancelled
   267→  startDate     DateTime? // null وقتی status=pending
   268→  endDate       DateTime? // null وقتی status=pending
   269→  // تاریخ لغو توسط ادمین — وقتی ادمین اشتراک را لغو می‌کند، این فیلد set می‌شود
   270→  // و endDate اصلی حفظ می‌شود (دیگر overwrite نمی‌شود). برای نمایش بازه واقعی
   271→  // برنامه به کاربر: از startDate تا cancelledAt (اگر لغو شده) یا endDate (اگر منقضی شده).
   272→  cancelledAt   DateTime?
   273→  durationDays  Int      @default(45) // طول دوره
   274→  pricePaid     Int      // مبلغ پرداخت‌شده واقعی (با تخفیف)
   275→  discountCode  String?  // کد تخفیف استفاده‌شده
   276→  paymentId     String?  // شناسه Payment ایجادکننده این اشتراک (برای استرداد دقیق ادمین)
   277→  createdAt     DateTime @default(now())
   278→
   279→  @@index([userId])
   280→  @@index([status])
   281→  @@index([cancelledAt])
   282→}

   863→// ===================== OWN ANDROID APP RELEASES (اپ اختصاصی فیتاپ) =====================
   864→// نسخه‌های اپ اندروید «اختصاصی» فیتاپ (WebView پنل کاربری) که فقط از خود سایت
   865→// دانلود می‌شود (نه کافه‌بازار). اپ در هر اجرا /api/app/own/latest را صدا می‌زند؛
   866→// اگر versionCode اپ < آخرین نسخه → مودال زیبای آپدیت (changelog + دکمه دانلود).
   867→// فایل APK در uploads/apk/ ذخیره و از /api/app/own/download سرو می‌شود.
   868→model OwnAppRelease {
   869→  id          String   @id @default(cuid())
   870→  versionName String            // نام نسخه نمایشی — مثلاً "1.0.0"
   871→  versionCode Int               // کد نسخه صحیح صعودی — مثلاً 1
   872→  changelog   String   @default("") // تغییرات نسخه (متن فارسی، چندخطی)
   873→  fileName    String            // نام فایل ذخیره‌شده در uploads/apk
   874→  fileSize    Int      @default(0) // حجم فایل (بایت)
   875→  downloads   Int      @default(0) // تعداد دانلود (آمار داخلی)
   876→  forceUpdate Boolean  @default(false) // آپدیت اجباری برای نسخه‌های قدیمی‌تر؟
   877→  isActive    Boolean  @default(true)  // قابل عرضه است؟
   878→  createdAt   DateTime @default(now())
   879→
   880→  @@index([versionCode])
   881→  @@index([createdAt])
   882→}
