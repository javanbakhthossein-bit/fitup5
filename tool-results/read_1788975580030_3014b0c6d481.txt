    22→ * دیریکتیو مالک: اگر نام ثبت نشده باشد رشتهٔ خالی برمی‌گردد (نه «کاربر»)
    23→ * و گارد sendTemplateSms ارسال را مدیریت می‌کند تا هیچ پیامکی با نام جعلی نرود.

    25→export function smsFirstName(name: string | null | undefined): string {
    26→  const first = String(name || "")
    27→    .replace(/[\r\n\t]/g, " ")
    28→    .trim()
    29→    .split(/\s+/)[0] || "";
    30→  return first.slice(0, 25);
    31→}

   684→export interface AbandonedCartSmsOptions {
   685→  user: SmsUserTarget;
   686→  /** کلید دداپ — abandoned_cart_{paymentId} */
   687→  dedupeKey: string;
   688→  /** لینک کوتاه نسبی مثل r/XXXXXXXX (خروجی createSmsShortLink) */
   689→  link: string;
   690→  /** درصد تخفیف برای متن (از SiteSetting) */
   691→  percent: number;
   692→  /** شناسه قالب sms.ir از SiteSetting — خالی = مستقیم bulk */
   693→  templateId: string | null;
   694→}
   695→
   696→export interface AbandonedCartSmsResult {
   697→  sent: boolean;
   698→  /** کدام مسیر تحویل داد */
   699→  via?: "template" | "bulk";
   700→  /** already_sent = دداپ قبلی (هرگز دوباره نفرست) */
   701→  skipped?: "already_sent" | "invalid_mobile";
   702→  error?: string;
   703→}
   704→
   705→/**
   706→ * ارسال پیامک ترک درگاه با نردبان قالب → bulk.
   707→ * در هر حالتی که پیامک نرود و هم داپِ قبلی هم نباشد، یک رکورد failed در
   708→ * SmsLog برای کلید ثبت می‌شود (قابل مشاهده در گزارش‌ها + تلاش مجدد sweep بعدی).
   709→ */
   710→export async function sendAbandonedCartSms(
   711→  opts: AbandonedCartSmsOptions
   712→): Promise<AbandonedCartSmsResult> {
   713→  const { user, dedupeKey, link, percent, templateId } = opts;
   714→
   715→  const normalized = normalizeMobileForSmsIr(user.mobile);
   716→  if (!/^9\d{9}$/.test(normalized)) {
   717→    return { sent: false, skipped: "invalid_mobile", error: `شماره نامعتبر: ${normalized}` };
   718→  }
   719→
   720→  const name = smsFirstName(user.name);
   721→
   722→  // ─── مسیر ۱ — قالب sms.ir (اگر شناسه در SiteSetting ثبت شده) ───
   723→  if (templateId && templateId.trim()) {
   724→    const attempts: Array<Array<{ name: string; value: string }>> = [
   725→      [
   726→        { name: "NAME", value: name },
   727→        { name: "LINK", value: link },
   728→      ],
   729→      [{ name: "NAME", value: name }],
   730→      [],
   731→    ];
   732→    let last: SendTemplateSmsResult = { sent: false, error: "no attempt" };
   733→    for (const params of attempts) {
   734→      last = await sendTemplateSms({
   735→        key: dedupeKey,
   736→        templateEnvKey: undefined,
   737→        templateIdOverride: templateId,
   738→        mobile: user.mobile,
   739→        params,
   740→        userId: user.id,
   741→      });
   742→      if (last.sent) return { sent: true, via: "template" };
   743→      if (last.skipped === "already_sent" || last.skipped === "invalid_mobile") {
   744→        return {
   745→          sent: false,
   746→          skipped: last.skipped === "invalid_mobile" ? "invalid_mobile" : "already_sent",
   747→          error: last.error,
   748→        };
   749→      }
   750→      // no_template / no_name / خطای درگاه → رانگ بعدی نردبان؛ در انتها bulk
   751→    }
   752→    console.warn(
   753→      `[sendAbandonedCartSms] template path failed (key=${dedupeKey}) — fallback to bulk. last:`,
   754→      last.error ?? last.skipped
   755→    );
   756→  }
   757→
   758→  // ─── مسیر ۲ — متن آزاد (bulk) ───
   759→  // نام ثبت‌نشده → «ورزشکار» (دیریکتیو مالک v43 — هیچ پیامکی با نام جعلی/خالی نمی‌رود)
   760→  const text = buildAbandonedCartSmsText(name || "ورزشکار", link, percent);
   761→  const bulk = await sendFreeTextSms(user.mobile, text, {
   762→    key: dedupeKey,
   763→    userId: user.id,
   764→  });
   765→  if (bulk.success) {
   766→    return { sent: true, via: "bulk" };
   767→  }
   768→  if (bulk.error?.includes("already_sent")) {
   769→    return { sent: false, skipped: "already_sent" };
   770→  }
   771→
   772→  // ─── شکست نهایی (مثلاً نبود کلید API در sandbox) — رکورد failed برای دیده‌شدن ───
   773→  // (اگر مسیر قالب/bulk خودشان رکورد failed گذاشته باشند، همین upsert همان را
   774→  //  به‌روز می‌کند — رکورد تکراری ساخته نمی‌شود.)
   775→  try {
   776→    const { db } = await import("@/lib/db");
   777→    await db.smsLog.upsert({
   778→      where: { mobile_key: { mobile: normalized, key: dedupeKey } },
   779→      create: {
   780→        mobile: normalized,
   781→        key: dedupeKey,
   782→        templateId: templateId?.trim() || "bulk",
   783→        userId: user.id,
   784→        status: "failed",
   785→        error: bulk.error ?? "abandoned_cart sms failed",
   786→      },
   787→      update: {
   788→        status: "failed",
   789→        error: bulk.error ?? "abandoned_cart sms failed",
   790→      },
   791→    });
   792→  } catch (e) {
   793→    console.warn("[sendAbandonedCartSms] failed-record upsert error:", e);
   794→  }
   795→
   796→  return { sent: false, error: bulk.error ?? "abandoned_cart sms failed" };
   797→}
